## 1. Q学习算法 - 实际应用示例

### 1.1 Taxi-v3 完整训练示例

```python
"""
Q学习算法 - Taxi-v3环境完整训练
这个环境模拟了出租车接乘客的任务
"""

import numpy as np
import gymnasium as gym
from collections import defaultdict
import matplotlib.pyplot as plt
import time

class QLearningTaxi:
    """Q学习出租车智能体"""
    
    def __init__(self, n_states, n_actions, learning_rate=0.1, 
                 gamma=0.99, epsilon=1.0, epsilon_decay=0.995, 
                 epsilon_min=0.01):
        
        self.n_states = n_states
        self.n_actions = n_actions
        self.alpha = learning_rate   # 学习率
        self.gamma = gamma         # 折扣因子
        self.epsilon = epsilon     # 探索率
        self.epsilon_decay = epsilon_decay  # 探索率衰减
        self.epsilon_min = epsilon_min     # 最小探索率
        
        # Q表：使用字典更高效（稀疏状态）
        self.Q = defaultdict(lambda: np.zeros(n_actions))
        self.episode_count = 0
    
    def choose_action(self, state):
        """ε-greedy动作选择"""
        if np.random.random() < self.epsilon:
            return np.random.randint(self.n_actions)
        else:
            return np.argmax(self.Q[state])
    
    def update(self, state, action, reward, next_state, done):
        """Q学习核心更新"""
        # 当前Q值
        current_q = self.Q[state][action]
        
        # 最优下一状态Q值
        best_next_q = np.max(self.Q[next_state])
        
        # TD目标
        if done:
            target = reward  # 终止状态无未来奖励
        else:
            target = reward + self.gamma * best_next_q
        
        # TD误差
        td_error = target - current_q
        
        # 更新Q值
        self.Q[state][action] = current_q + self.alpha * td_error
        
        # 更新探索率
        self.epsilon = max(self.epsilon_min, self.epsilon * self.epsilon_decay)
        self.episode_count += 1
    
    def get_policy(self):
        """提取确定性策略"""
        return {s: np.argmax(a) for s, a in self.Q.items()}

def train_q_learning():
    """训练Q学习智能体"""
    # 创建环境
    env = gym.make('Taxi-v3', render_mode=None)
    
    n_states = env.observation_space.n
    n_actions = env.action_space.n
    
    print(f"状态数: {n_states}, 动作数: {n_actions}")
    
    # 创建智能体
    agent = QLearningTaxi(
        n_states=n_states,
        n_actions=n_actions,
        learning_rate=0.1,
        gamma=0.99,
        epsilon=1.0,
        epsilon_decay=0.995,
        epsilon_min=0.01
    )
    
    # 训练参数
    n_episodes = 10000
    max_steps = 200
    
    # 记录训练曲线
    episode_rewards = []
    episode_success = []
    
    success_count = 0
    
    for episode in range(n_episodes):
        state, _ = env.reset()
        total_reward = 0
        done = False
        
        for step in range(max_steps):
            # 选择动作
            action = agent.choose_action(state)
            
            # 执行动作
            next_state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            
            # 判断是否成功（正奖励表示成功接乘客）
            is_success = (reward > 0)
            
            # 更新Q表
            agent.update(state, action, reward, next_state, done)
            
            total_reward += reward
            state = next_state
            
            if done:
                if is_success:
                    success_count += 1
                break
        
        episode_rewards.append(total_reward)
        episode_success.append(1 if is_success else 0)
        
        # 打印进度
        if (episode + 1) % 1000 == 0:
            avg_reward = np.mean(episode_rewards[-1000:])
            success_rate = np.mean(episode_success[-1000:]) * 100
            print(f"Episode {episode+1}/{n_episodes}: "
                  f"Avg Reward: {avg_reward:.1f}, "
                  f"Success Rate: {success_rate:.1f}%, "
                  f"Epsilon: {agent.epsilon:.3f}")
    
    env.close()
    
    # 测试训练结果
    print("\n测试10回合:")
    test_success = 0
    for _ in range(10):
        state, _ = env.reset()
        done = False
        
        for step in range(200):
            action = np.argmax(agent.Q[state])
            state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            
            if done:
                if reward > 0:
                    test_success += 1
                break
    
    print(f"测试成功率: {test_success}/10 = {test_success*10}%")
    
    return agent, episode_rewards

if __name__ == "__main__":
    agent, rewards = train_q_learning()
```

### 1.2 参数调优指南

| 参数 | 默认值 | 调整范围 | 影响 |
|------|---------|----------|------|
| α (学习率) | 0.1 | 0.01-0.5 | 过大不稳定，过慢收敛慢 |
| γ (折扣因子) | 0.99 | 0.9-0.999 | 越小越贪婪，越大越远视 |
| ε (探索率) | 1.0 | - | 开始为1，逐渐衰减 |
| ε_decay | 0.995 | 0.99-0.999 | 衰减越快探索越少 |

### 1.3 问题诊断

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| Q值不收敛 | 学习率过大 | 减小α到0.01-0.05 |
| 卡在局部最优 | 探索不足 | 提高ε_decay或ε_min |
| 训练不稳定 | 未归一化状态 | 使用StandardScaler |
| 收敛太慢 | 状态空间大 | 使用神经网络(DQN) |

## 2. PPO算法 - 连续控制实战

### 2.1 双机器人平衡

```python
"""
PPO算法 - 双机器人平衡任务
实现两个倒立摆在平衡杆上的协调控制
"""

import torch
import torch.nn as nn
import torch.optim as optim
from torch.distributions import Normal
import numpy as np
import gymnasium as gym

class ActorCritic(nn.Module):
    """Actor-Critic网络"""
    
    def __init__(self, state_dim, action_dim, hidden_dim=64):
        super().__init__()
        
        # 共享特征层
        self.feature = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.Tanh(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.Tanh()
        )
        
        # Actor: 策略均值和标准差
        self.actor_mean = nn.Linear(hidden_dim, action_dim)
        self.actor_log_std = nn.Parameter(torch.zeros(action_dim))
        
        # Critic: 状态价值
        self.critic = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.Tanh(),
            nn.Linear(hidden_dim, 1)
        )
    
    def forward(self, x):
        features = self.feature(x)
        
        # Actor输出
        mean = self.actor_mean(features)
        std = torch.exp(self.actor_log_std)
        
        # Critic输出
        value = self.critic(features)
        
        return mean, std, value

class PPOAgent:
    """PPO智能体"""
    
    def __init__(self, state_dim, action_dim, 
                 lr=3e-4, gamma=0.99, lam=0.95,
                 clip_eps=0.2, value_coef=0.5, entropy_coef=0.01):
        
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.gamma = gamma
        self.lam = lam
        self.clip_eps = clip_eps
        self.value_coef = value_coef
        self.entropy_coef = entropy_coef
        
        # 网络
        self.policy = ActorCritic(state_dim, action_dim)
        self.optimizer = optim.Adam(self.policy.parameters(), lr=lr)
    
    def get_action(self, state):
        """采样动作"""
        mean, std, value = self.policy(torch.FloatTensor(state))
        
        dist = Normal(mean, std)
        action = dist.sample()
        log_prob = dist.log_prob(action).sum(dim=-1)
        
        return action.numpy(), log_prob.item(), value.item()
    
    def evaluate_actions(self, states, actions):
        """评估动作"""
        mean, std, values = self.policy(states)
        
        dist = Normal(mean, std)
        log_probs = dist.log_prob(actions).sum(dim=-1)
        entropy = dist.entropy().sum(dim=-1).mean()
        
        return log_probs, values, entropy
    
    def update(self, states, actions, rewards, next_states, dones):
        """PPO更新"""
        # 转为张量
        states = torch.FloatTensor(states)
        actions = torch.FloatTensor(actions)
        rewards = torch.FloatTensor(rewards)
        next_states = torch.FloatTensor(next_states)
        dones = torch.FloatTensor(dones)
        
        # 计算GAE
        with torch.no_grad():
            _, _, next_values = self.policy(next_states)
            values = self.policy(states)[2]
            
            deltas = rewards + self.gamma * (1 - dones) * next_values.squeeze() - values.squeeze()
            advantages = torch.zeros_like(rewards)
            gae = 0
            for t in reversed(range(len(deltas))):
                gae = deltas[t] + self.gamma * self.lam * gae
                advantages[t] = gau
        
        advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)
        
        # 计算策略损失
        log_probs, values, entropy = self.evaluate_actions(states, actions)
        
        # PPO裁剪
        ratio = torch.exp(log_probs)
        surr1 = ratio * advantages
        surr2 = torch.clamp(ratio, 1 - self.clip_eps, 1 + self.clip_eps) * advantages
        policy_loss = -torch.min(surr1, surr2).mean()
        
        # 值函数损失
        values = values.squeeze()
        value_loss = nn.MSELoss()(values, rewards + self.gamma * (1 - dones) * next_values.squeeze())
        
        # 熵损失
        entropy_loss = -entropy
        
        # 总损失
        loss = policy_loss + self.value_coef * value_loss + self.entropy_coef * entropy_loss
        
        # 更新
        self.optimizer.zero_grad()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(self.policy.parameters(), 0.5)
        self.optimizer.step()
        
        return loss.item()

def train_ppo():
    """训练PPO"""
    env = gym.make('BipedalWalker-v3')
    
    state_dim = env.observation_space.shape[0]
    action_dim = env.action_space.shape[0]
    
    print(f"状态维度: {state_dim}, 动作维度: {action_dim}")
    
    agent = PPOAgent(state_dim, action_dim)
    
    n_episodes = 1000
    n_steps = 2048  # 每个episode的步数
    
    rewards_history = []
    
    for episode in range(n_episodes):
        state, _ = env.reset()
        episode_reward = 0
        
        # 存储轨迹
        states, actions, rewards, next_states, dones = [], [], [], [], []
        
        for step in range(n_steps):
            action, log_prob, value = agent.get_action(state)
            
            next_state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            
            states.append(state)
            actions.append(action)
            rewards.append(reward)
            next_states.append(next_state)
            dones.append(float(done))
            
            state = next_state
            episode_reward += reward
            
            if done:
                state, _ = env.reset()
                episode_reward = 0
        
        # 更新
        agent.update(states, actions, rewards, next_states, dones)
        
        rewards_history.append(episode_reward)
        
        if episode % 100 == 0:
            print(f"Episode {episode}: Reward = {episode_reward:.1f}")
        
        if episode % 500 == 0:
            avg_reward = np.mean(rewards_history[-500:])
            print(f"  平均500回合奖励: {avg_reward:.1f}")
    
    env.close()
    return agent

if __name__ == "__main__":
    agent = train_ppo()
```

## 3. 常见问题快速诊断表

| 现象 | 可能原因 | 快速检查 |
|------|----------|----------|
| 损失=Nan | 梯度爆炸 | 检查梯度裁剪 |
| Reward不增加 | 探索过度 | 降低ε_decay |
| 提前终止 | 惩罚过大 | 调整奖励函数 |
| 策略退化 | 价值估计错误 | 检查Critic损失 |
| 训练卡住 | 学习率过低 | 增大lr |
