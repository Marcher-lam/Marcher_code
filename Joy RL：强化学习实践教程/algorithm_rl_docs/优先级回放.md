# 优先级回放（Prioritized Experience Replay）学习文档

> 根据TD误差优先级采样，提高样本效率的DQN改进

---

## 1. 算法基础认知

**一句话定义**：优先级回放（Prioritized Experience Replay）根据样本的TD误差大小决定采样优先级，让DQN优先学习"意外"的样本，从而提高样本效率。

**直觉类比**：普通的经验回放是"随机抽题"，优先级回放是"错题优先"——你越不会的题目越要经常做，这样才能快速进步。

**历史背景**：由Schaul等人在2015年提出，是对DQN的重要改进。

**算法定位**：
- 类型：DQN改进 → 样本效率优化
- 输出：优化采样策略
- 模型类型：优先级采样

---

## 2. 核心原理

### 2.1 核心思想

TD误差越大 = 样本越"意外" = 越需要学习

**优先级定义**：
$$p_i = |TD\_error_i|^\alpha + \epsilon$$

### 2.2 采样概率

$$P(i) = \frac{p_i}{\sum_j p_j}$$

### 2.3 重要性采样权重

为了补偿非均匀采样带来的偏差：
$$w_i = (\frac{1}{N \cdot P(i)})^\beta$$

最终权重：$w_i \cdot TD\_error$

---

## 3. 数据结构：SumTree

### 3.1 为什么用SumTree？

普通列表：O(N)采样 → 太慢
SumTree：O(log N)采样 → 快

### 3.2 SumTree实现

```python
class SumTree:
    """二叉树存储优先级"""
    def __init__(self, capacity):
        self.capacity = capacity
        self.tree = np.zeros(2 * capacity - 1)
        self.data_idx = capacity - 1
    
    def add(self, priority, data):
        """添加新样本"""
        tree_idx = self.data_idx
        self.data_idx = (self.data_idx + 1) % self.capacity
        
        self.tree[tree_idx] = priority
        self._propagate(tree_idx, priority - self.tree[tree_idx])
    
    def sample(self, seg_num):
        """优先级采样"""
        # 随机选择段落
        idx = np.random.randint(0, seg_num)
        # 在SumTree中查找
        return self._find(0, idx)
    
    def _propagate(self, idx, change):
        # 向上传播变化
        parent = (idx - 1) // 2
        self.tree[parent] += change
        if parent != 0:
            self._propagate(parent, change)
    
    def _find(self, idx, target):
        # 二分查找
        if idx >= self.capacity - 1:
            return self.tree[idx], idx - (self.capacity - 1)
        
        left = 2 * idx + 1
        if target < self.tree[left]:
            return self._find(left, target)
        return self._find(left + 1, target - self.tree[left])
```

## 4. 调库实现

```python
import numpy as np
import torch
import torch.nn as nn

class PrioritizedReplayBuffer:
    """优先级回放缓冲区"""
    
    def __init__(self, capacity, alpha=0.6, beta=0.4):
        self.capacity = capacity
        self.alpha = alpha  # 优先级指数
        self.beta = beta    # 重要性采样指数
        self.pos = 0
        self.size = 0
        
        # 特征存储
        self.states = np.zeros((capacity, 4), dtype=np.float32)
        self.actions = np.zeros(capacity, dtype=np.int64)
        self.rewards = np.zeros(capacity, dtype=np.float32)
        self.next_states = np.zeros((capacity, 4), dtype=np.float32)
        self.dones = np.zeros(capacity, dtype=np.float32)
        
        # 优先级树
        self.tree = np.zeros(2 * capacity - 1, dtype=np.float32)
        self.max_priority = 1.0
    
    def add(self, state, action, reward, next_state, done):
        # 存储样本
        self.states[self.pos] = state
        self.actions[self.pos] = action
        self.rewards[self.pos] = reward
        self.next_states[self.pos] = next_state
        self.dones[self.pos] = done
        
        # 设置最大优先级
        self.tree[self.pos + self.capacity - 1] = self.max_priority
        self._propagate(self.pos + self.capacity - 1, self.max_priority)
        
        self.pos = (self.pos + 1) % self.capacity
        self.size = min(self.size + 1, self.capacity)
    
    def sample(self, batch_size):
        """优先级采样"""
        # 计算段落大小
        segment = self.tree[0] / batch_size
        
        # 采样
        indices = []
        priorities = []
        for _ in range(batch_size):
            # 随机位置
            pos = np.random.uniform(segment * np.random.randint(0, batch_size),
                              segment * (np.random.randint(0, batch_size) + 1))
            idx = self._find(0, pos)
            priorities.append(self.tree[idx])
            indices.append(idx)
        
        # 重要性采样权重
        probs = np.array(priorities) / self.tree[0]
        weights = (self.size * probs) ** (-self.beta)
        weights = weights / weights.max()
        
        # 提取样本
        batch = (
            self.states[indices],
            self.actions[indices],
            self.rewards[indices],
            self.next_states[indices],
            self.dones[indices]
        )
        
        return batch, indices, weights
    
    def update_priorities(self, indices, td_errors):
        """更新优先级"""
        for idx, error in zip(indices, td_errors):
            priority = (abs(error) + 0.01) ** self.alpha
            self.tree[idx + self.capacity - 1] = priority
            self._propagate(idx + self.capacity - 1, priority - self.tree[idx + self.capacity - 1])
            self.max_priority = max(self.max_priority, priority)
    
    def _propagate(self, idx, change):
        parent = (idx - 1) // 2
        self.tree[parent] += change
        if parent != 0:
            self._propagate(parent, change)
    
    def _find(self, idx, target):
        if idx >= self.capacity - 1:
            return idx - self.capacity + 1, self.tree[idx]
        
        left = 2 * idx + 1
        if target <= self.tree[left]:
            return self._find(left, target)
        return self._find(left + 1, target - self.tree[left])

# ===============================
# PER-DQN整合
# ===============================
class PERDQN:
    """优先级回放DQN"""
    
    def __init__(self, state_dim, action_dim):
        self.replay = PrioritizedReplayBuffer(100000, alpha=0.6)
        self.q_net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, action_dim)
        )
        self.optimizer = torch.optim.Adam(self.q_net.parameters(), lr=1e-3)
    
    def update(self, batch, indices, weights, gamma=0.99):
        states, actions, rewards, next_states, dones = batch
        
        # 计算Q值
        q_values = self.q_net(states).gather(1, actions.unsqueeze(1))
        
        # 目标Q值
        with torch.no_grad():
            next_q = self.q_net(next_states).max(1)[0]
            target = rewards + gamma * (1 - dones) * next_q
        
        # 带权重的TD误差
        weights = torch.FloatTensor(weights).unsqueeze(1)
        td_errors = target.unsqueeze(1) - q_values
        loss = (weights * td_errors ** 2).mean()
        
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        
        # 更新优先级
        self.replay.update_priorities(indices, td_errors.detach().numpy())
        
        return loss.item()
```

|------|---------|----------|
