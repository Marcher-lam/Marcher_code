# 双线性插值学习文档

## 1. 算法基础认知

双线性插值（Bilinear Interpolation）是一种在二维网格上进行插值的方法，通过对四个相邻像素点进行线性组合来计算目标位置的值。这种方法广泛应用于图像处理、计算机图形学、深度学习中的特征图resize操作等场景。

双线性插值相比于最近邻插值（Nearest Neighbor）能够产生更平滑的结果，相比于双三次插值（Bicubic）计算量更小，因此在实际应用中被广泛使用。

核心思想：
- 在x方向进行两次线性插值
- 在y方向进行一次线性插值
- 利用四个相邻像素点的加权平均值

应用场景：
- 图像缩放
- 特征图上采样/下采样
- 坐标变换
- 图像配准

## 2. 核心原理

### 2.1 数学公式

双线性插值的数学定义为：

设四个相邻像素点的值分别为：
- Q11 = (x1, y1), 值 f(Q11)
- Q21 = (x2, y1), 值 f(Q21)
- Q12 = (x1, y2), 值 f(Q12)
- Q22 = (x2, y2), 值 f(Q22)

目标点P = (x, y) 位于这四个点构成的矩形内。

首先在x方向进行线性插值：

$$R_1 = \frac{x_2 - x}{x_2 - x_1} \cdot f(Q_{11}) + \frac{x - x_1}{x_2 - x_1} \cdot f(Q_{21})$$

$$R_2 = \frac{x_2 - x}{x_2 - x_1} \cdot f(Q_{12}) + \frac{x - x_1}{x_2 - x_1} \cdot f(Q_{22})$$

然后在y方向进行线性插值：

$$P = \frac{y_2 - y}{y_2 - y_1} \cdot R_1 + \frac{y - y_1}{y_2 - y_1} \cdot R_2$$

```python
import numpy as np

def bilinear_interpolation_formula():
    """双线性插值公式"""
    # 简化的单位正方形版本
    # 假设 x1=0, y1=0, x2=1, y2=1
    
    # 目标点
    x, y = 0.4, 0.6
    
    # 四个角点的值
    f00, f10 = 100, 200  # y=0 的两个点
    f01, f11 = 150, 250  # y=1 的两个点
    
    # 在x方向插值
    R1 = (1 - x) * f00 + x * f10  # R1 = (1-x)*f(0,0) + x*f(1,0)
    R2 = (1 - x) * f01 + x * f11  # R2 = (1-x)*f(0,1) + x*f(1,1)
    
    # 在y方向插值
    P = (1 - y) * R1 + y * R2
    
    return P
```

## 3. PyTorch实现

### 3.1 F.grid_sample 实现

PyTorch中使用 `torch.nn.functional.grid_sample` 实现双线性插值，这是深度学习中最重要的上采样函数之一。

```python
import torch
import torch.nn.functional as F
import numpy as np

def grid_sample_basic():
    """grid_sample基本使用"""
    
    # 输入图像 [B, C, H, W]
    input_tensor = torch.randn(1, 3, 4, 4)
    
    # 生成网格坐标 [-1, 1] 范围
    # 这里的坐标是归一化的，(-1,-1)表示左上角，(1,1)表示右下角
    grid = torch.tensor([
        [[0.0, 0.0], [0.0, 0.5], [0.0, -0.5]],
        [[0.5, 0.0], [0.5, 0.5], [0.5, -0.5]],
        [[-0.5, 0.0], [-0.5, 0.5], [-0.5, -0.5]]
    ]).unsqueeze(0)
    
    # 进行双线性插值
    output = F.grid_sample(input_tensor, grid, mode='bilinear', padding_mode='zeros')
    
    print(f"Input shape: {input_tensor.shape}")
    print(f"Grid shape: {grid.shape}")
    print(f"Output shape: {output.shape}")


def create_normalized_grid(height, width):
    """
    创建归一化网格
    
    返回:
        grid: [H, W, 2] 坐标在[-1, 1]范围
    """
    
    # 创建网格
    x = torch.linspace(-1, 1, width)
    y = torch.linspace(-1, 1, height)
    
    # 生成网格
    grid_y, grid_x = torch.meshgrid(y, x, indexing='ij')
    
    # 堆叠
    grid = torch.stack([grid_x, grid_y], dim=-1)
    
    return grid


def resize_with_grid_sample(input_tensor, target_size):
    """
    使用grid_sample进行图像缩放
    
    参数:
        input_tensor: [B, C, H, W]
        target_size: (H, W)
        
    返回:
        output: [B, C, H', W']
    """
    B, C, H, W = input_tensor.shape
    target_H, target_W = target_size
    
    # 创建原始网格
    grid = create_normalized_grid(H, W)
    grid = grid.unsqueeze(0).expand(B, -1, -1, -1).to(input_tensor.device)
    
    # 创建目标网格（用于上采样）
    target_grid = create_normalized_grid(target_H, target_W)
    target_grid = target_grid.unsqueeze(0).expand(B, -1, -1, -1).to(input_tensor.device)
    
    # 缩放系数
    scale_x = (W - 1) / (target_W - 1)
    scale_y = (H - 1) / (target_H - 1)
    
    # 调整网格坐标
    target_grid[..., 0] = target_grid[..., 0] * scale_x
    target_grid[..., 1] = target_grid[..., 1] * scale_y
    
    # 采样
    output = F.grid_sample(input_tensor, target_grid, mode='bilinear', padding_mode='zeros', align_corners=True)
    
    return output
```

```python
def manual_bilinear_interpolation():
    """手动实现双线性插值"""
    
    def bilinear_interpolate(image, x, y):
        """
        对单个点进行双线性插值
        
        参数:
            image: 2D图像 [H, W]
            x, y: 插值坐标（浮点数）
            
        返回:
            value: 插值结果
        """
        H, W = image.shape
        
        # 获取四周的整数坐标
        x0 = int(np.floor(x))
        x1 = x0 + 1
        y0 = int(np.floor(y))
        y1 = y0 + 1
        
        # 边界检查
        if x0 < 0 or x1 >= W or y0 < 0 or y1 >= H:
            # 边界情况，返回最近邻
            x = np.clip(x, 0, W - 1)
            y = np.clip(y, 0, H - 1)
            return image[int(y), int(x)]
        
        # 计算权重
        wx1 = x - x0
        wy1 = y - y0
        wx0 = 1 - wx1
        wy0 = 1 - wy1
        
        # 四个点的值
        f00 = image[y0, x0]
        f01 = image[y0, x1]
        f10 = image[y1, x0]
        f11 = image[y1, x1]
        
        # 双线性插值
        R1 = wx0 * f00 + wx1 * f01
        R2 = wx0 * f10 + wx1 * f11
        P = wy0 * R1 + wy1 * R2
        
        return P
    
    # 测试
    image = np.array([
        [100, 150, 200, 250],
        [120, 170, 220, 270],
        [140, 190, 240, 290],
        [160, 210, 260, 310]
    ], dtype=np.float32)
    
    # 插值
    result = bilinear_interpolate(image, 1.5, 1.5)
    print(f"Result: {result}")
    
    return result


def resize_image_manual(image, target_size):
    """
    手动实现图像缩放
    
    参数:
        image: [H, W] 或 [H, W, C]
        target_size: (H, W)
        
    返回:
        resized: [H', W'] 或 [H', W', C]
    """
    H, W = image.shape[:2]
    target_H, target_W = target_size
    
    # 计算缩放比例
    scale_x = (W - 1) / (target_W - 1)
    scale_y = (H - 1) / (target_H - 1)
    
    if len(image.shape) == 2:
        # 灰度图
        resized = np.zeros((target_H, target_W))
        
        for j in range(target_H):
            for i in range(target_W):
                x = i * scale_x
                y = j * scale_y
                resized[j, i] = bilinear_interpolate(image, x, y)
    else:
        # 彩色图
        C = image.shape[2]
        resized = np.zeros((target_H, target_W, C))
        
        for j in range(target_H):
            for i in range(target_W):
                x = i * scale_x
                y = j * scale_y
                for c in range(C):
                    resized[j, i, c] = bilinear_interpolate(image[:, :, c], x, y)
    
    return resized
```

## 4. 数学公式与推导

### 4.1 详细推导

双线性插值可以分解为以下步骤：

设目标点P(x,y)，四个相邻点为：
- Q11 = (x1, y1)
- Q21 = (x2, y1)
- Q12 = (x1, y2)
- Q22 = (x2, y2)

第一步：计算x方向的插值

$$R_1 = f(Q_{11}) + \frac{x - x_1}{x_2 - x_1}(f(Q_{21}) - f(Q_{11}))$$

$$R_2 = f(Q_{12}) + \frac{x - x_1}{x_2 - x_1}(f(Q_{22}) - f(Q_{12}))$$

第二步：计算y方向的插值

$$P = f(P) = R_1 + \frac{y - y_1}{y_2 - y_1}(R_2 - R_1)$$

```python

    # 左下角 Q12 = (0, 1), f(Q12) = f01
    # 右下角 Q22 = (1, 1), f(Q22) = f11
    
    # 目标点 P = (x, y)
    
    # 第一步：x方向插值
    R1 = (1-x) * f00 + x * f10
    R2 = (1-x) * f01 + x * f11
```

```python
def compute_weights():

    def get_bilinear_weights(x, y):
        """
        计算双线性插值的四个权重
        
        参数:
            x, y: 插值坐标
            
        返回:
            weights: [4] 分别是 f00, f10, f01, f11 的权重

            wx0 * wy1,  # f(Q12)
```

```python
def spatial_transformer_network():
    """STN空间变换网络"""
    
    class STN(nn.Module):
        """空间变换网络"""
        
        def __init__(self):
            super().__init__()
            
            # 定位网络
            self.localization = nn.Sequential(
                nn.Conv2d(1, 32, kernel_size=3, padding=1),
                nn.ReLU(),
                nn.MaxPool2d(2),
                nn.Conv2d(32, 64, kernel_size=3, padding=1),
                nn.ReLU(),
                nn.MaxPool2d(2)
            )
            
            # 回归变换参数
            self.fc = nn.Sequential(
                nn.Linear(64 * 16 * 16, 256),
                nn.ReLU(),
                nn.Linear(256, 6)  # 仿射变换的6个参数
            )
        
        def forward(self, x):
            # 获取变换参数
            xs = self.localization(x)
            xs = xs.view(xs.size(0), -1)
            theta = self.fc(xs)
            theta = theta.view(-1, 2, 3)
            
            # 创建网格
            grid = F.affine_grid(theta, x.size(), align_corners=True)
            
            # 采样
            output = F.grid_sample(x, grid, mode='bilinear', padding_mode='zeros', align_corners=True)
            
            return output
    
    # 测试

    x = torch.randn(4, 1, 64, 64)
    output = stn(x)
```

## 6. 优缺点分析

### 6.1 优点

1. **计算效率高**：只涉及简单的算术运算
2. **平滑性好**：比最近邻插值更平滑
3. **可微**：梯度可以回传
4. **通用性强**：适用于各种数据

### 6.2 缺点

1. **精度有限**：是线性近似，不是非线性
2. **可能有伪影**：在高对比度边缘可能产生模糊
3. **不是旋转不变**

### 6.3 改进方法

1. **Bicubic插值**：使用16个点，更平滑
2. ** Lanczos插值**：使用sinc函数，减少伪影
3. **内容感知缩放**：保留重要结构

```python
import numpy as np
from typing import Tuple, Union


class BilinearInterpolator:
    """双线性插值器"""
    
    def __init__(self, image: np.ndarray):
        """
        初始化
        
        参数:
            image: 输入图像 [H, W] 或 [H, W, C]
        """
        self.image = image.astype(np.float32)
        self.H, self.W = image.shape[:2]
    
    def interpolate_single(self, x: float, y: float) -> Union[float, np.ndarray]:
        """单点插值"""
        
        # 获取四个角点
        x0 = int(np.floor(x))
        x1 = x0 + 1
        y0 = int(np.floor(y))
        y1 = y0 + 1
        
        # 边界检查
        if x0 < 0 or x1 >= self.W or y0 < 0 or y1 >= self.H:
            x = np.clip(x, 0, self.W - 1)
            y = np.clip(y, 0, self.H - 1)
            return self.image[int(y), int(x)]
        
        # 计算权重
        wx1 = x - x0
        wy1 = y - y0
        wx0 = 1 - wx1
        wy0 = 1 - wy1
        
        # 获取四个角点的值
        if len(self.image.shape) == 2:
            f00 = self.image[y0, x0]
            f01 = self.image[y0, x1]
            f10 = self.image[y1, x0]
            f11 = self.image[y1, x1]
            
            # 插值
            R1 = wx0 * f00 + wx1 * f01
            R2 = wx0 * f10 + wx1 * f11
            return wy0 * R1 + wy1 * R2
        else:
            C = self.image.shape[2]
            result = np.zeros(C)
            
            for c in range(C):
                f00 = self.image[y0, x0, c]
                f01 = self.image[y0, x1, c]
                f10 = self.image[y1, x0, c]
                f11 = self.image[y1, x1, c]
                
                R1 = wx0 * f00 + wx1 * f01
                R2 = wx0 * f10 + wx1 * f11
                result[c] = wy0 * R1 + wy1 * R2
            
            return result
    
    def resize(self, target_size: Tuple[int, int]) -> np.ndarray:
        """图像缩放"""
        
        target_H, target_W = target_size
        
        # 计算缩放比例
        scale_x = (self.W - 1) / (target_W - 1) if target_W > 1 else 0
        scale_y = (self.H - 1) / (target_H - 1) if target_H > 1 else 0
        
        # 创建输出图像
        if len(self.image.shape) == 2:
            output = np.zeros((target_H, target_W), dtype=np.float32)
        else:
            C = self.image.shape[2]
            output = np.zeros((target_H, target_W, C), dtype=np.float32)
        
        # 遍历每个像素
        for j in range(target_H):
            for i in range(target_W):
                x = i * scale_x
                y = j * scale_y
                output[j, i] = self.interpolate_single(x, y)
        
        return output


class GridSamplePyTorch:
    """模拟torch的grid_sample"""
    
    @staticmethod
    def forward(input_tensor: torch.Tensor, grid: torch.Tensor, 
               mode: str = 'bilinear', padding_mode: str = 'zeros',
               align_corners: bool = False) -> torch.Tensor:
        """
        手动实现grid_sample
        
        参数:
            input_tensor: [B, C, H, W]
            grid: [B, H_out, W_out, 2] 值在 [-1, 1]
            mode: 'bilinear' or 'nearest'
            padding_mode: 'zeros', 'border', 'reflection'
            align_corners: 是否对齐角点
        """
        B, C, H, W = input_tensor.shape
        _, H_out, W_out, _ = grid.shape
        
        # 转换网格坐标到像素坐标
        if align_corners:
            grid_x = (grid[..., 0] + 1) / 2 * (W - 1)
            grid_y = (grid[..., 1] + 1) / 2 * (H - 1)
        else:
            grid_x = (grid[..., 0] + 1) / 2 * W
            grid_y = (grid[..., 1] + 1) / 2 * H
        
        # 批量和通道维度
        output = torch.zeros(B, C, H_out, W_out, device=input_tensor.device)
        
        for b in range(B):
            for c in range(C):
                for j in range(H_out):
                    for i in range(W_out):
                        x = grid_x[b, j, i].item()
                        y = grid_y[b, j, i].item()
                        
                        # 双线性插值
                        output[b, c, j, i] = bilinear_sample(
                            input_tensor[b, c], x, y, padding_mode
                        )
        
        return output


def bilinear_sample(image: torch.Tensor, x: float, y: float, 
                  padding_mode: str = 'zeros') -> float:
    """双线性采样单个值"""
    
    H, W = image.shape
    
    x0 = int(np.floor(x))
    x1 = x0 + 1
    y0 = int(np.floor(y))
    y1 = y0 + 1
    
    # 边界处理
    if padding_mode == 'zeros':
        if x0 < 0 or x1 >= W or y0 < 0 or y1 >= H:
            return 0.0
    elif padding_mode == 'border':
        x0 = max(0, min(x0, W - 1))
        x1 = max(0, min(x1, W - 1))
        y0 = max(0, min(y0, H - 1))
        y1 = max(0, min(y1, H - 1))
    
    # 计算权重
    wx1 = x - x0
    wy1 = y - y0
    wx0 = 1 - wx1
    wy0 = 1 - wy1
    
    # 获取值
    f00 = image[y0, x0].item()
    f01 = image[y0, x1].item()
    f10 = image[y1, x0].item()
    f11 = image[y1, x1].item()
    
    # 插值
    R1 = wx0 * f00 + wx1 * f01
    R2 = wx0 * f10 + wx1 * f11
    return wy0 * R1 + wy1 * R2
```

```python
    """评估插值质量"""
    
    # 创建测试图像
    original = np.random.rand(64, 64)

    methods = ['nearest', 'bilinear']
    results = {}
    
    for method in methods:
        interpolator = BilinearInterpolator(original)
        output = interpolator.resize((128, 128)) if method == 'bilinear' else None
        
        # 评估指标
        if output is not None:
            results[method] = {
                'PSNR': compute_psnr(original, output),
                'SSIM': compute_ssim(original, output)
            }
    
    return results


def compute_psnr(img1, img2):
    """计算PSNR"""
    mse = np.mean((img1 - img2) ** 2)
    if mse == 0:
        return float('inf')
    return 20 * np.log10(1.0 / np.sqrt(mse))


def compute_ssim(img1, img2):
    """计算SSIM（简化版）"""
    # 简化实现
    return np.mean(np.abs(img1 - img2))
```
