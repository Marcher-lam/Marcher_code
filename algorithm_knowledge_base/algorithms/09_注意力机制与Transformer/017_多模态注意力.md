# 多模态注意力学习文档

## 1. 算法基础认知

多模态注意力（Multimodal Attention）是指在处理来自不同模态（如图像、文本、音频等）的信息时，通过注意力机制来动态地融合和选择性地关注不同模态的信息。这种机制在视觉问答、图像字幕生成、多模态翻译、语音识别等任务中发挥着重要作用。

多模态注意力的核心思想源于人类的多模态感知能力——人类在理解场景时会自然地将注意力分配到不同的感官信息上。例如，当听到一段声音时，我们可能会看向声音来源的方向；当看到一张图片时，我们可能会有选择地关注其中的关键物体。

在深度学习中，多模态注意力机制允许模型：
1. 动态地决定关注哪个模态的信息
2. 在不同模态之间建立对应关系
3. 处理模态之间的对齐和关联
4. 提高模型的解释性和可解释性

多模态注意力的发展历程：
1. 早期：简单的特征拼接或加权平均
2. 中期：共同注意力（Co-Attention）、循环注意力
3. 近期：Transformer-based 多模态注意力、跨模态Transformer

## 2. 核心原理

### 2.1 多模态融合的基本框架

多模态注意力融合的一般框架可以表示为：

$$\alpha_t = \text{Softmax}(W_a \cdot \tanh(W_v V + W_t T))$$

其中：
- $V$ 是视觉特征
- $T$ 是文本特征
- $\alpha_t$ 是文本到视觉的注意力权重
- $W_a, W_v, W_t$ 是可学习的参数

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class MultimodalAttentionBasic(nn.Module):
    """基础多模态注意力"""
    
    def __init__(self, vision_dim, text_dim, hidden_dim):
        super().__init__()
        
        # 视觉和文本的投影矩阵
        self.W_v = nn.Linear(vision_dim, hidden_dim)
        self.W_t = nn.Linear(text_dim, hidden_dim)
        self.W_a = nn.Linear(hidden_dim, 1)
        
    def forward(self, visual_features, text_features):
        """
        参数:
            visual_features: [B, N_v, D_v] 视觉特征
            text_features: [B, N_t, D_t] 文本特征
            
        返回:
            attended_visual: [B, N_v, D_v] 加权后的视觉特征
            attended_text: [B, N_t, D_t] 加权后的文本特征
        """
        B, N_v, D_v = visual_features.shape
        N_t, D_t = text_features.shape[1], text_features.shape[2]
        
        # 投影
        V_proj = self.W_v(visual_features)  # [B, N_v, H]
        T_proj = self.W_t(text_features)  # [B, N_t, H]
        
        # 计算注意力权重
        # 视觉到文本的注意力
        attention_vt = torch.matmul(V_proj, T_proj.transpose(1, 2)) / math.sqrt(V_proj.shape[-1])
        alpha_vt = F.softmax(attention_vt, dim=-1)  # [B, N_v, N_t]
        
        # 文本到视觉的注意力
        alpha_tv = F.softmax(attention_vt.transpose(1, 2), dim=-1)  # [B, N_t, N_v]
        
        # 应用注意力
        attended_visual = torch.matmul(alpha_tv, visual_features)  # [B, N_v, D_v]
        attended_text = torch.matmul(alpha_vt, text_features)  # [B, N_t, D_t]
        
        return attended_visual, attended_text
```

### 2.2 共同注意力机制

共同注意力（Co-Attention）是指同时计算视觉和文本的注意力，使两者能够相互关注：

```python
class CoAttention(nn.Module):
    """共同注意力机制"""
    
    def __init__(self, vision_dim, text_dim, hidden_dim):
        super().__init__()
        
        self.vision_dim = vision_dim
        self.text_dim = text_dim
        self.hidden_dim = hidden_dim
        
        # 视觉注意力
        self.v_proj = nn.Linear(vision_dim, hidden_dim)
        self.v_att = nn.Linear(hidden_dim, 1)
        
        # 文本注意力
        self.t_proj = nn.Linear(text_dim, hidden_dim)
        self.t_att = nn.Linear(hidden_dim, 1)
        
        # 共同注意力
        self.fusion = nn.Linear(hidden_dim * 2, hidden_dim)
        
    def forward(self, visual_features, text_features, mask=None):
        """
        参数:
            visual_features: [B, N_v, D_v] 视觉特征
            text_features: [B, N_t, D_t] 文本特征
            mask: [B, N_t] 可选的mask
            
        返回:
            visual_context: [B, D] 视觉上下文
            text_context: [B, D] 文本上下文
        """
        B = visual_features.shape[0]
        
        # 投影
        V = self.v_proj(visual_features)  # [B, N_v, H]
        T = self.t_proj(text_features)  # [B, N_t, H]
        
        # ========== 视觉注意力（关注文本）==========
        # 计算相似度矩阵
        similarity = torch.matmul(V, T.transpose(1, 2)) / math.sqrt(self.hidden_dim)
        
        # 计算文本注意力
        text_attention = self.t_att(torch.tanh(V.unsqueeze(2) + T.unsqueeze(1))).squeeze(-1)  # [B, N_v, N_t]
        
        if mask is not None:
            text_attention = text_attention.masked_fill(mask.unsqueeze(1) == 0, float('-inf'))
        
        alpha_text = F.softmax(text_attention, dim=-1)
        
        # 应用注意力得到视觉上下文
        visual_context = torch.matmul(alpha_text, T)  # [B, N_v, H]
        
        # ========== 文本注意力（关注视觉）==========
        visual_attention = self.v_att(torch.tanh(T.unsqueeze(2) + V.unsqueeze(1))).squeeze(-1)  # [B, N_t, N_v]
        alpha_visual = F.softmax(visual_attention, dim=-1)
        
        # 应用注意力得到文本上下文
        text_context = torch.matmul(alpha_visual, V)  # [B, N_t, H]
        
        # 融合
        visual_global = visual_context.mean(dim=1)  # [B, H]
        text_global = text_context.mean(dim=1)  # [B, H]
        
        fusion = torch.cat([visual_global, text_global], dim=-1)
        output = self.fusion(fusion)
        
        return output, alpha_visual, alpha_text
```

### 2.3 注意力头的扩展

多模态注意力可以扩展为多头注意力：

$$MultiHead(Q, K, V) = \text{Concat}(head_1, ..., head_h)W^O$$

其中 $head_i = \text{Attention}(QW_i^Q, KW_i^K, VW_i^V)$

```python
class MultimodalMultiHeadAttention(nn.Module):
    """多模态多头注意力"""
    
    def __init__(self, vision_dim, text_dim, hidden_dim, num_heads=8):
        super().__init__()
        
        self.num_heads = num_heads
        self.hidden_dim = hidden_dim
        self.head_dim = hidden_dim // num_heads
        
        assert hidden_dim % num_heads == 0
        
        # 视觉投影
        self.v_proj = nn.Linear(vision_dim, hidden_dim)
        self.v_k = nn.Linear(vision_dim, hidden_dim)
        self.v_v = nn.Linear(vision_dim, hidden_dim)
        
        # 文本投影
        self.t_proj = nn.Linear(text_dim, hidden_dim)
        self.t_k = nn.Linear(text_dim, hidden_dim)
        self.t_v = nn.Linear(text_dim, hidden_dim)
        
        # 输出投影
        self.out_proj = nn.Linear(hidden_dim, hidden_dim)
        
    def forward(self, visual_features, text_features):
        """
        返回:
            output: [B, N_t, D] 或 [B, N_v, D]
        """
        B = visual_features.shape[0]
        
        # 投影并reshape为多头
        V_q = self.v_proj(visual_features).view(B, -1, self.num_heads, self.head_dim).transpose(1, 2)
        V_k = self.v_k(visual_features).view(B, -1, self.num_heads, self.head_dim).transpose(1, 2)
        V_v = self.v_v(visual_features).view(B, -1, self.num_heads, self.head_dim).transpose(1, 2)
        
        T_q = self.t_proj(text_features).view(B, -1, self.num_heads, self.head_dim).transpose(1, 2)
        T_k = self.t_k(text_features).view(B, -1, self.num_heads, self.head_dim).transpose(1, 2)
        T_v = self.t_v(text_features).view(B, -1, self.num_heads, self.head_dim).transpose(1, 2)
        
        # 跨模态注意力：文本Query关注视觉
        attention = torch.matmul(T_q, V_k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        attention = F.softmax(attention, dim=-1)
        
        # 应用注意力
        text_attended = torch.matmul(attention, V_v)  # [B, H, N_t, D_h]
        
        # reshape回原始形状
        text_attended = text_attended.transpose(1, 2).contiguous().view(B, -1, self.hidden_dim)
        
        # 输出投影
        output = self.out_proj(text_attended)
        
        return output
```

## 3. 数学公式与推导

### 3.1 跨模态注意力机制

跨模态注意力的核心是计算不同模态特征之间的相关性：

$$e_{ij} = \frac{v_i^T t_j}{\sqrt{d_k}}$$

$$\alpha_j = \frac{\exp(e_{ij})}{\sum_k \exp(e_{ik})}$$

$$\text{output}_j = \sum_i \alpha_j v_i$$

其中 $v_i$ 是视觉特征，$t_j$ 是文本特征。

```python
def cross_attention_math():
    """跨模态注意力的数学推导"""
    
    # 假设有N个视觉特征和M个文本特征
    # V = [N, d_k] 视觉特征矩阵
    # T = [M, d_k] 文本特征矩阵
    
    # 1. 计算相似度矩阵
    # E = V @ T^T = [N, M] 相似度矩阵
    E = torch.matmul(V, T.transpose(0, 1)) / math.sqrt(d_k)
    
    # 2. Softmax归一化
    # alpha = softmax(E, dim=-1) = [N, M]
    alpha = F.softmax(E, dim=-1)
    
    # 3. 加权求和
    # output = alpha @ V = [M, d_k]
    output = torch.matmul(alpha, V)
    
    return output
```

### 3.2 自注意力与跨模态注意力的结合

```python
class MultimodalSelfCrossAttention(nn.Module):
    """结合自注意力和跨模态注意力"""
    
    def __init__(self, dim, num_heads=8):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        
        # 自注意力
        self.self_attn = nn.MultiheadAttention(dim, num_heads)
        
        # 跨模态交叉注意力
        self.cross_attn = nn.MultiheadAttention(dim, num_heads)
        
        # 层归一化
        self.norm1 = nn.LayerNorm(dim)
        self.norm2 = nn.LayerNorm(dim)
        
    def forward(self, multimodal_features, mode='cross'):
        """
        参数:
            multimodal_features: [N, B, D] 多模态特征序列
            mode: 'self', 'cross', 或 'both'
        """
        if mode == 'self':
            attn_output, _ = self.self_attn(multimodal_features, multimodal_features, multimodal_features)
            output = self.norm1(multimodal_features + attn_output)
        elif mode == 'cross':
            # 假设前N_v个是视觉，后N_t个是文本
            N_v = ...
            visual = multimodal_features[:N_v]
            text = multimodal_features[N_v:]
            
            # 文本关注视觉
            cross_output, _ = self.cross_attn(text, visual, visual)
            output = self.norm2(text + cross_output)
        
        return output
```

## 4. PyTorch实现

### 4.1 基础多模态Transformer

```python
class VisionTextTransformer(nn.Module):
    """视觉-文本Transformer"""
    
    def __init__(self, vision_dim=2048, text_dim=768, hidden_dim=512, num_heads=8, num_layers=6, dropout=0.1):
        super().__init__()
        
        self.vision_proj = nn.Linear(vision_dim, hidden_dim)
        self.text_proj = nn.Linear(text_dim, hidden_dim)
        
        # 位置编码
        self.vision_pos = PositionalEncoding(hidden_dim, dropout)
        self.text_pos = PositionalEncoding(hidden_dim, dropout)
        
        # Transformer编码器
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim,
            nhead=num_heads,
            dim_feedforward=hidden_dim * 4,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)
        
        # 多模态融合
        self.fusion = nn.Linear(hidden_dim * 2, hidden_dim)
        
    def forward(self, vision_features, text_features):
        """
        参数:
            vision_features: [B, N_v, D_v]
            text_features: [B, N_t, D_t]
            
        返回:
            fused: [B, hidden_dim]
        """
        # 投影
        vision_emb = self.vision_proj(vision_features)
        text_emb = self.text_proj(text_features)
        
        # 位置编码
        vision_emb = self.vision_pos(vision_emb)
        text_emb = self.text_pos(text_emb)
        
        # 拼接
        multimodal = torch.cat([vision_emb, text_emb], dim=1)
        
        # Transformer编码
        encoded = self.transformer(multimodal)
        
        # 全局池化
        pooled = encoded.mean(dim=1)
        
        # 融合
        output = self.fusion(pooled)
        
        return output


class PositionalEncoding(nn.Module):
    """位置编码"""
    
    def __init__(self, d_model, max_len=5000, dropout=0.1):
        super().__init__()
        self.dropout = nn.Dropout(p=dropout)
        
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0)
        self.register_buffer('pe', pe)
        
    def forward(self, x):
        x = x + self.pe[:, :x.size(1)]
        return self.dropout(x)
```

### 4.2 跨模态注意力融合

```python
class CrossModalAttentionFusion(nn.Module):
    """跨模态注意力融合"""
    
    def __init__(self, vision_dim, text_dim, output_dim):
        super().__init__()
        
        # 维度映射
        self.vision_mapping = nn.Linear(vision_dim, output_dim)
        self.text_mapping = nn.Linear(text_dim, output_dim)
        
        # 跨模态注意力层
        self.cross_attention = nn.MultiheadAttention(output_dim, num_heads=8)
        
        # 融合层
        self.vision_gate = nn.Sequential(
            nn.Linear(output_dim, output_dim),
            nn.Sigmoid()
        )
        self.text_gate = nn.Sequential(
            nn.Linear(output_dim, output_dim),
            nn.Sigmoid()
        )
        
        # 输出层
        self.output = nn.Linear(output_dim * 2, output_dim)
        
    def forward(self, vision_features, text_features):
        """
        参数:
            vision_features: [B, N_v, D_v]
            text_features: [B, N_t, D_t]
            
        返回:
            fused: [B, D]
        """
        # 维度映射
        V = self.vision_mapping(vision_features)
        T = self.text_mapping(text_features)
        
        # 跨模态注意力
        # 文本关注视觉
        text_to_vision, attn_weights = self.cross_attention(
            T, V, V
        )
        
        # 门控机制
        v_gate = self.vision_gate(V.mean(dim=1))
        t_gate = self.text_gate(T.mean(dim=1))
        
        # 加权
        v_weighted = V * v_gate.unsqueeze(1)
        t_weighted = T * t_gate.unsqueeze(1)
        
        # 融合
        concat = torch.cat([v_weighted, t_weighted], dim=-1)
        output = self.output(concat)
        
        return output
```

### 4.3 完整的视觉问答模型

```python
class VQAModel(nn.Module):
    """视觉问答模型"""
    
    def __init__(self, num_answers):
        super().__init__()
        
        # 图像编码器
        self.image_encoder = nn.Sequential(
            nn.Linear(2048, 1024),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(1024, 512)
        )
        
        # 文本编码器
        self.text_encoder = nn.GRU(
            input_size=300,
            hidden_size=512,
            num_layers=2,
            batch_first=True
        )
        
        # 多模态融合
        self.multimodal_fusion = CrossModalAttentionFusion(
            vision_dim=512,
            text_dim=512,
            output_dim=512
        )
        
        # 答案分类
        self.classifier = nn.Sequential(
            nn.Linear(512, 1024),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(1024, num_answers)
        )
        
    def forward(self, image_features, question_tokens, question_lengths):
        """
        参数:
            image_features: [B, 2048] 或 [B, N, 2048]
            question_tokens: [B, T] 词索引
            question_lengths: [B] 序列长度
        """
        # 如果是单张图像，重复N次
        if len(image_features.shape) == 2:
            image_features = image_features.unsqueeze(1)
        
        # 图像编码
        image_encoded = self.image_encoder(image_features)
        
        # 文本编码
        text_encoded, hidden = self.text_encoder(question_tokens)
        
        # 移除padding
        mask = torch.arange(text_encoded.size(1), device=text_encoded.device).unsqueeze(0) < question_lengths.unsqueeze(1)
        text_encoded = text_encoded * mask.unsqueeze(-1)
        
        # 多模态融合
        fused = self.multimodal_fusion(image_encoded, text_encoded)
        
        # 分类
        logits = self.classifier(fused)
        
        return logits
```

### 5.2 图像字幕生成

```python
class ImageCaptioningModel(nn.Module):
    """图像字幕生成模型"""
    
    def __init__(self, vocab_size, embed_dim=512, hidden_dim=512):
        super().__init__()
        
        # 图像编码
        self.image_encoder = nn.Linear(2048, embed_dim)
        
        # 文本解码器
        self.embedding = nn.Embedding(vocab_size, embed_dim)
        self.decoder = nn.LSTM(
            input_size=embed_dim,
            hidden_size=hidden_dim,
            num_layers=2,
            batch_first=True
        )
        
        # 多模态注意力
        self.attention = nn.MultiheadAttention(hidden_dim, num_heads=8)
        
        # 输出层
        self.output = nn.Linear(hidden_dim, vocab_size)
        
    def forward(self, image_features, captions, teacher_forcing_ratio=0.5):
        """
        参数:
            image_features: [B, 2048]
            captions: [B, T] 目标序列
        """
        batch_size = image_features.shape[0]
        max_len = captions.shape[1]
        
        # 编码图像
        image_encoded = self.image_encoder(image_features)  # [B, D]
        image_encoded = image_encoded.unsqueeze(1)  # [B, 1, D]
        
        # 解码
        outputs = []
        hidden = None
        
        for t in range(max_len):
            if t == 0:
                # 使用图像作为第一个输入
                decoder_input = image_encoded
            else:
                # 嵌入
                word_embed = self.embedding(captions[:, t-1])
                decoder_input = word_embed.unsqueeze(1)
            
            # 解码
            output, hidden = self.decoder(decoder_input, hidden)
            
            # 图像注意力
            attended, _ = self.attention(output.transpose(0, 1), image_encoded.transpose(0, 1), image_encoded.transpose(0, 1))
            attended = attended.transpose(0, 1)
            
            # 预测
            pred = self.output(attended.squeeze(1))
            outputs.append(pred)
        
        outputs = torch.stack(outputs, dim=1)
        
        return outputs
```

### 5.3 多模态情感分析

```python
class MultimodalSentimentModel(nn.Module):
    """多模态情感分析"""
    
    def __init__(self, num_classes):
        super().__init__()
        
        # 文本编码
        self.text_encoder = nn.Sequential(
            nn.Embedding(50000, 512),
            nn.LSTM(512, 256, batch_first=True)
        )
        
        # 视觉编码
        self.video_encoder = nn.Sequential(
            nn.Linear(2048, 1024),
            nn.ReLU(),
            nn.Linear(1024, 512)
        )
        
        # 音频编码
        self.audio_encoder = nn.Sequential(
            nn.Linear(128, 256),
            nn.ReLU(),
            nn.Linear(256, 512)
        )
        
        # 多模态注意力融合
        self.fusion_attention = MultimodalAttentionFusion(512, 512, 512, modality_count=3)
        
        # 分类
        self.classifier = nn.Sequential(
            nn.Linear(512, 256),
            nn.ReLU(),
            nn.Linear(256, num_classes)
        )
        
    def forward(self, text_features, video_features, audio_features):
        """
        参数:
            text_features: [B, T, D_text]
            video_features: [B, D_video]
            audio_features: [B, D_audio]
        """
        # 编码
        text_encoded, _ = self.text_encoder(text_features)
        text_encoded = text_encoded.mean(dim=1)
        
        video_encoded = self.video_encoder(video_features)
        audio_encoded = self.audio_encoder(audio_features)
        
        # 融合
        fused = self.fusion_attention([text_encoded, video_encoded, audio_encoded])
        
        # 分类
        logits = self.classifier(fused)
        
        return logits
```

## 6. 优缺点分析

### 6.1 优点

1. **灵活性**：可以动态地关注不同模态的相关信息
2. **可解释性**：注意力权重可以可视化，展示模型关注的位置
3. **强表达能力**：能够捕捉复杂的跨模态关系
4. **模块化**：可以方便地与其他模型组合

### 6.2 缺点

1. **计算复杂度**：注意力计算是O(N×M)的复杂度
2. **内存占用**：需要存储注意力权重矩阵
3. **训练困难**：需要更多的数据和调参
4. **可能过度关注**：可能导致模型过度关注噪声

### 6.3 改进方向

1. **稀疏注意力**：只计算部分位置的注意力
2. **分层注意力**：先粗后细的注意力
3. **记忆机制**：加入外部记忆模块

```python
class EfficientAttention(nn.Module):
    """高效注意力实现"""
    
    def __init__(self, dim, num_heads=8, reduction=4):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.reduction = reduction
        
        # 降维
        self.query = nn.Linear(dim, dim)
        self.key = nn.Linear(dim, dim // reduction)
        self.value = nn.Linear(dim, dim // reduction)
        
        # 输出
        self.out = nn.Linear(dim // reduction, dim)
        
    def forward(self, x):
        B, N, D = x.shape
        
        # 投影
        q = self.query(x).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.key(x).view(B, N, self.num_heads, self.head_dim // self.reduction).transpose(1, 2)
        v = self.value(x).view(B, N, self.num_heads, self.head_dim // self.reduction).transpose(1, 2)
        
        # 注意力
        attention = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(self.head_dim // self.reduction)
        attention = F.softmax(attention, dim=-1)
        
        # 应用
        out = torch.matmul(attention, v)
        out = out.transpose(1, 2).contiguous().view(B, N, -1)
        
        return self.out(out)
```

## 8. 手工代码实现

```python
class ManualMultimodalAttention:
    """手动实现多模态注意力"""
    
    @staticmethod
    def create_attention_matrix(visual_features, text_features, temperature=1.0):
        """
        创建跨模态注意力矩阵
        
        参数:
            visual_features: [N_v, D]
            text_features: [N_t, D]
            temperature: 温度参数
            
        返回:
            attention: [N_t, N_v]
        """
        # 计算相似度
        similarity = torch.matmul(text_features, visual_features.t())
        
        # 温度缩放
        similarity = similarity / temperature
        
        # Softmax
        attention = F.softmax(similarity, dim=-1)
        
        return attention
    
    @staticmethod
    def apply_attention(features, attention):
        """
        应用注意力
        
        参数:
            features: [N_v, D]
            attention: [N_t, N_v]
            
        返回:
            output: [N_t, D]
        """
        output = torch.matmul(attention, features)
        return output
    
    @staticmethod
    def bidirectional_attention(visual_features, text_features, temperature=1.0):
        """
        双向注意力
        
        返回:
            v2t_attention: [N_v, N_t] 视觉到文本的注意力
            t2v_attention: [N_t, N_v] 文本到视觉的注意力
        """
        v2t = torch.matmul(visual_features, text_features.t()) / temperature
        v2t_att = F.softmax(v2t, dim=-1)
        
        t2v = torch.matmul(text_features, visual_features.t()) / temperature
        t2v_att = F.softmax(t2v, dim=-1)
        
        return v2t_att, t2v_att
```

```python
def compute_attention_diversity(attention):
    """计算注意力的多样性"""
    # 使用熵来衡量
    entropy = -torch.sum(attention * torch.log(attention + 1e-10), dim=-1)
    return entropy.mean().item()


def compute_cross_modal_correlation(visual_features, text_features):
    """计算跨模态相关性"""
    # 使用典型相关分析（简化版）
    v_mean = visual_features.mean(dim=0)
    t_mean = text_features.mean(dim=0)
    
    v_centered = visual_features - v_mean
    t_centered = text_features - t_mean
    
    correlation = torch.matmul(v_centered, t_centered.t())
    
    return correlation.abs().mean().item()
```
