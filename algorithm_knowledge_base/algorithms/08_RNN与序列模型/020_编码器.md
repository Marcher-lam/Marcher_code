## 1. 算法基础认知

### 1.1 一句话定义

编码器（Encoder）是深度学习和自然语言处理中的核心概念，指的是将输入数据转换为一种更紧凑、更具表示能力的潜在表示的神经网络组件。编码器的核心功能是"编码"——将原始的高维输入（如文本、图像、音频）映射到一个低维的语义空间中，提取关键特征并去除冗余信息。

### 1.2 直觉类比

想象你在看一部电影：
- **原始输入**：2小时的电影（视频+音频）
- **编码器**：提取关键情节、人物、情感
- **输出**：一句话剧情简介

这就是编码器做的事情——信息压缩与提取！

## 2. 核心原理

### 2.1 Seq2Seq编码器

基于RNN的编码器是Seq2Seq架构的核心组件：

$$h_t = \text{RNN}(x_t, h_{t-1})$$

最终隐藏状态：
$$c = h_T$$

### 2.2 Transformer编码器

完全基于注意力机制：

**多头自注意力**：
$$\text{MultiHead}(Q, K, V) = \text{Concat}(\text{head}_1, ..., \text{head}_h)W^O$$

**缩放点积注意力**：
$$\text{Attention}(Q, K, V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right)V$$

**前馈网络**：
$$\text{FFN}(x) = \text{ReLU}(xW_1 + b_1)W_2 + b_2$$

### 2.3 位置编码

注入位置信息：

**正弦位置编码**：
$$PE_{(pos, 2i)} = \sin\left(\frac{pos}{10000^{2i/d_{model}}}\right)$$
$$PE_{(pos, 2i+1)} = \cos\left(\frac{pos}{10000^{2i/d_{model}}}\right)$$

## 3. 数学公式与推导

### 3.1 BERT编码器架构

多层Transformer Encoder的堆叠：

$$\text{EncoderLayer}(x) = \text{LayerNorm}(x + \text{FFN}(x + \text{Attention}(x))$$

每层包含：
1. 多头自注意力层
2. Add & LayerNorm
3. 前馈神经网络
4. Add & LayerNorm

### 3.2 残差连接与层归一化

残差连接（Residual Connection）：
$$y = F(x) + x$$

层归一化（Layer Normalization）：
$$\text{LayerNorm}(x) = \gamma \odot \frac{x - \mu}{\sqrt{\sigma^2 + \epsilon}} + beta$$

其中：
$$\mu = \frac{1}{d}\sum_{i=1}^{d} x_i$$
$$\sigma^2 = \frac{1}{d}\sum_{i=1}^{d} (x_i - \mu)^2$$

### 3.3 遮掩语言建模（MLM）

BERT的预训练任务：

1. 随机选择15%的token位置
2. 80%替换为[MASK]
3. 10%替换为随机token
4. 10%保持不变

预测目标：
$$L_{MLM} = -\sum_{i \in M} \log P(x_i | x_{\setminus M})$$

### 3.4 下一句预测（NSP）

判断句子B是否紧跟句子A：
$$L_{NSP} = -\log P(\text{IsNext}) - \log P(\text{NotNext})$$

## 4. PyTorch实现

### 4.1 Transformer编码器

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math


class PositionalEncoding(nn.Module):
    """正弦位置编码"""
    def __init__(self, d_model, max_len=5000, dropout=0.1):
        super(PositionalEncoding, self).__init__()
        self.dropout = nn.Dropout(p=dropout)
        
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() * 
                    (-math.log(10000.0) / d_model))
        
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        
        pe = pe.unsqueeze(0)
        self.register_buffer('pe', pe)
    
    def forward(self, x):
        x = x + self.pe[:, :x.size(1)]
        return self.dropout(x)


class MultiHeadAttention(nn.Module):
    """多头自注意力"""
    def __init__(self, d_model, n_heads, dropout=0.1):
        super(MultiHeadAttention, self).__init__()
        assert d_model % n_heads == 0
        
        self.d_model = d_model
        self.n_heads = n_heads
        self.d_k = d_model // n_heads
        
        self.W_Q = nn.Linear(d_model, d_model)
        self.W_K = nn.Linear(d_model, d_model)
        self.W_V = nn.Linear(d_model, d_model)
        self.W_O = nn.Linear(d_model, d_model)
        
        self.dropout = nn.Dropout(p=dropout)
    
    def forward(self, Q, K, V, mask=None):
        batch_size = Q.size(0)
        
        # 线性投影并分头
        Q = self.W_Q(Q).view(batch_size, -1, self.n_heads, self.d_k).transpose(1, 2)
        K = self.W_K(K).view(batch_size, -1, self.n_heads, self.d_k).transpose(1, 2)
        V = self.W_V(V).view(batch_size, -1, self.n_heads, self.d_k).transpose(1, 2)
        
        # 缩放点积注意力
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        
        if mask is not None:
            scores = scores.masked_fill(mask == 0, float('-inf'))
        
        attn_weights = F.softmax(scores, dim=-1)
        attn_weights = self.dropout(attn_weights)
        
        # 加权求和
        context = torch.matmul(attn_weights, V)
        context = context.transpose(1, 2).contiguous().view(
            batch_size, -1, self.d_model)
        
        output = self.W_O(context)
        
        return output, attn_weights


class TransformerEncoderLayer(nn.Module):
    """Transformer编码器层"""
    def __init__(self, d_model, n_heads, d_ff, dropout=0.1):
        super(TransformerEncoderLayer, self).__init__()
        
        self.self_attn = MultiHeadAttention(d_model, n_heads, dropout)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        
        self.ffn = nn.Sequential(
            nn.Linear(d_model, d_ff),
            nn.ReLU(),
            nn.Dropout(p=dropout),
            nn.Linear(d_ff, d_model),
            nn.Dropout(p=dropout)
        )
        
        self.dropout = nn.Dropout(p=dropout)
    
    def forward(self, x, mask=None):
        # 自注意力 + 残差
        attn_output, _ = self.self_attn(x, x, x, mask)
        x = self.norm1(x + self.dropout(attn_output))
        
        # 前馈网络 + 残差
        ffn_output = self.ffn(x)
        x = self.norm2(x + self.dropout(ffn_output))
        
        return x


class TransformerEncoder(nn.Module):
    """Transformer编码器"""
    def __init__(self, vocab_size, d_model=512, n_heads=8, n_layers=6,
                 d_ff=2048, dropout=0.1, max_len=5000):
        super(TransformerEncoder, self).__init__()
        
        self.d_model = d_model
        
        # 词嵌入
        self.embedding = nn.Embedding(vocab_size, d_model)
        self.pos_encoding = PositionalEncoding(d_model, max_len, dropout)
        
        # 多层编码器
        self.layers = nn.ModuleList([
            TransformerEncoderLayer(d_model, n_heads, d_ff, dropout)
            for _ in range(n_layers)
        ])
        
        self.norm = nn.LayerNorm(d_model)
    
    def forward(self, x, mask=None):
        # 词嵌入 + 位置编码
        x = self.embedding(x) * math.sqrt(self.d_model)
        x = self.pos_encoding(x)
        
        # 逐层通过编码器
        for layer in self.layers:
            x = layer(x, mask)
        
        return self.norm(x)
```

### 4.2 BERT编码器实现

```python
class BERTEncoder(nn.Module):
    """BERT编码器（简化版）"""
    def __init__(self, vocab_size, d_model=768, n_heads=12, n_layers=12,
                 d_ff=3072, dropout=0.1, max_len=512):
        super(BERTEncoder, self).__init__()
        
        self.embedding = nn.Embedding(vocab_size, d_model)
        self.pos_embedding = nn.Embedding(max_len, d_model)
        self.layer_norm = nn.LayerNorm(d_model)
        
        self.layers = nn.ModuleList([
            TransformerEncoderLayer(d_model, n_heads, d_ff, dropout)
            for _ in range(n_layers)
        ])
        
        self.dropout = nn.Dropout(p=dropout)
    
    def forward(self, input_ids, attention_mask=None):
        # 嵌入
        batch_size, seq_len = input_ids.size()
        
        token_emb = self.embedding(input_ids)
        
        # 位置编码
        position_ids = torch.arange(seq_len, dtype=torch.long, device=input_ids.device)
        position_ids = position_ids.unsqueeze(0).expand_as(input_ids)
        pos_emb = self.pos_embedding(position_ids)
        
        x = self.layer_norm(token_emb + pos_emb)
        x = self.dropout(x)
        
        # 掩码处理
        if attention_mask is not None:
            mask = attention_mask.unsqueeze(1).unsqueeze(2)
            mask = mask.to(x.dtype)
            mask = (1.0 - mask) * -10000.0
        else:
            mask = None
        
        # 通过各层
        for layer in self.layers:
            x = layer(x, mask)
        
        return x
```

## 5. 代码示例

### 5.1 文本编码

```python
class TextEncoder:
    """文本编码器"""
    
    def __init__(self, model_name='bert-base-uncased'):
        from transformers import BertModel, BertTokenizer
        
        self.tokenizer = BertTokenizer.from_pretrained(model_name)
        self.model = BertModel.from_pretrained(model_name)
        self.model.eval()
    
    def encode(self, text, pooling='cls'):
        """编码文本"""
        encoding = self.tokenizer(
            text,
            padding=True,
            truncation=True,
            max_length=512,
            return_tensors='pt'
        )
        
        with torch.no_grad():
            outputs = self.model(**encoding)
        
        if pooling == 'cls':
            return outputs.pooler_output
        elif pooling == 'mean':
            return outputs.last_hidden_state.mean(dim=1)
        elif pooling == 'max':
            return outputs.last_hidden_state.max(dim=1)[0]
    
    def encode_batch(self, texts, pooling='cls'):
        """批量编码"""
        encoding = self.tokenizer(
            texts,
            padding=True,
            truncation=True,
            max_length=512,
            return_tensors='pt'
        )
        
        with torch.no_grad():
            outputs = self.model(**encoding)
        
        if pooling == 'cls':
            return outputs.pooler_output
        elif pooling == 'mean':
            return outputs.last_hidden_state[:, :, :].mean(dim=1)
    
    def similarity(self, text1, text2):
        """计算相似度"""
        emb1 = self.encode(text1)
        emb2 = self.encode(text2)
        
        cos = nn.CosineSimilarity()
        return cos(emb1, emb2).item()
```

```python
class ImageEncoder(nn.Module):
    """图像编码器"""
    
    def __init__(self, backbone='resnet50', pretrained=True):
        super(ImageEncoder, self).__init__()
        
        if backbone == 'resnet50':
            import torchvision.models as models
            backbone = models.resnet50(pretrained=pretrained)
            self.model = nn.Sequential(*list(backbone.children())[:-1])
        elif backbone == 'vit':
            from transformers import ViTModel
            self.model = ViTModel.from_pretrained('google/vit-base-patch16-224')
    
    def forward(self, x):
        features = self.model(x)
        if isinstance(features, dict):
            features = features['pooler_output']
        return features


def encode_images(images):
    """编码图像"""
    encoder = ImageEncoder()
    features = encoder(images)
    return features
```

### 5.3 多模态编码

```python
class MultimodalEncoder(nn.Module):
    """多模态编码器"""
    
    def __init__(self):
        super().__init__()
        
        self.text_encoder = BERTEncoder()
        self.image_encoder = ImageEncoder()
        
        # 投影层
        self.text_projection = nn.Linear(768, 512)
        self.image_projection = nn.Linear(2048, 512)
    
    def forward(self, text, image):
        # 文本编码
        text_features = self.text_encoder(text)
        text_features = self.text_projection(text_features)
        
        # 图像编码
        image_features = self.image_encoder(image)
        image_features = self.image_projection(image_features)
        
        # 融合
        combined = torch.cat([text_features, image_features], dim=1)
        
        return combined
```

```python
class SentenceMatcher(nn.Module):
    """句子匹配"""
    
    def __init__(self, encoder):
        super().__init__()
        self.encoder = encoder
        self.classifier = nn.Linear(encoder.d_model * 3, 2)
    
    def forward(self, input_ids1, input_ids2):
        emb1 = self.encoder(input_ids1)
        emb2 = self.encoder(input_ids2)
        
        # 交互特征
        combined = torch.cat([emb1, emb2, emb1 * emb2], dim=1)
        
        return self.classifier(combined)
```

```python
        super().__init__()
        self.encoder = encoder
    
    def forward(self, question_ids, context_ids):
        q_emb = self.encoder(question_ids)
```

## 8. 手工代码实现（核心算法手写 + 注释）

```python
import torch, torch.nn as nn, torch.nn.functional as F

class 编码器Net(nn.Module):
    def __init__(self, dim_in=20, dim_h=64, dim_out=2):
        super().__init__()
        self.fc1, self.fc2 = nn.Linear(dim_in, dim_h), nn.Linear(dim_h, dim_out)
    def forward(self, x): return self.fc2(F.relu(self.fc1(x)))

def train(model, X, y, epochs=100, lr=0.001):
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    for ep in range(epochs):
        opt.zero_grad()
        loss = nn.CrossEntropyLoss()(model(X), y)
        loss.backward(); opt.step()
        if (ep+1)%20==0: print(f"Ep{ep+1} loss={loss.item():.4f}")

m = 编码器Net()
train(m, torch.randn(500,20), torch.randint(0,2,(500,)))
```
