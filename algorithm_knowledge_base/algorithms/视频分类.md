# 视频分类（Video Classification）学习文档

## 1. 算法基础认知

视频分类是计算机视觉领域的重要任务，旨在识别视频中的内容并进行分类。与静态图像分类不同，视频分类需要同时处理空间信息（单帧图像内容）和时序信息（帧之间的运动和变化）。

视频分类面临的主要挑战包括：
1. **高维数据**：视频是三维数据（时间x高度x宽度），数据量远大于图像
2. **时序建模**：需要捕捉帧之间的运动和变化
3. **计算效率**：处理长视频需要大量计算资源

常用的视频分类方法包括：
- **双流网络**（Two-Stream CNN）：分别处理空间流和光流
- **3D CNN**：使用3D卷积同时处理时空信息
- **RNN/LSTM**：在CNN特征基础上进行时序建模
- **Video Transformer**：使用Transformer处理视频

## 2. 核心原理

### 2.1 视频表示

视频可以表示为帧序列：$V = (I_1, I_2, ..., I_T)$，其中$T$是帧数。

**采样策略**：
- 均匀采样：从$T$帧中均匀采样$n$帧
- 随机采样：随机选择$n$帧
- 密集采样：使用连续$n$帧

### 2.2 双流网络

**空间流**：处理单帧RGB图像，捕捉外观信息。
$$S = CNN_{spatial}(I_t)$$

**光流**：处理帧间运动信息，捕捉运动特征。
$$O = CNN_{optical}(\text{OpticalFlow}(I_t, I_{t+1}))$$

**融合**：
$$P(y|V) = \text{softmax}(W \cdot [S; O] + b)$$

### 2.3 3D卷积

3D卷积核在时间维度上也有感受野：
$$Y_{c}[d, i, j] = \sum_{k} \sum_{m} \sum_{n} I[d+k, i+m, j+n] \cdot W[c, k, m, n]$$

## 3. 数学公式与推导

### 3.1 C3D卷积

对于输入视频$X \in \mathbb{R}^{C \times T \times H \times W}$和3D卷积核$W \in \mathbb{R}^{C \times k_t \times k_h \times k_w}$：
$$Y = X * W$$

### 3.2 I3D (Inflated 3D ConvNet)

将2D卷积膨胀为3D：
$$W_{3D}(t, i, j) = W_{2D}(i, j) \cdot \alpha(t)$$

其中$\alpha(t)$是时间维度的插值。

### 3.3 时序建模

**LSTM**：
$$h_t = \text{LSTM}(h_{t-1}, x_t)$$

**Transformer**：
$$\text{Attention}(Q, K, V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d}}\right)V$$

import torch
from torch.utils.data import Dataset, DataLoader
import cv2
import numpy as np

class VideoDataset(Dataset):
    def __init__(self, video_paths, labels, num_frames=16, transform=None):
        self.video_paths = video_paths
        self.labels = labels
        self.num_frames = num_frames
        self.transform = transform
    
    def __len__(self):
        return len(self.video_paths)
    
    def _load_video(self, path):
        cap = cv2.VideoCapture(path)
        frames = []
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            frames.append(frame)
        
        cap.release()
        
        if len(frames) == 0:
            return np.zeros((self.num_frames, 224, 224, 3))
        
        indices = np.linspace(0, len(frames)-1, self.num_frames).astype(int)
        frames = [frames[i] for i in indices]
        
        return np.array(frames)
    
    def __getitem__(self, idx):
        frames = self._load_video(self.video_paths[idx])
        label = self.labels[idx]
        
        if self.transform:
            frames = self.transform(frames)
        
        frames = torch.FloatTensor(frames).permute(0, 3, 1, 2) / 255.0
        
        mean = torch.tensor([0.485, 0.456, 0.406]).view(3, 1, 1, 1)
        std = torch.tensor([0.229, 0.224, 0.225]).view(3, 1, 1, 1)
        frames = (frames - mean) / std
        
        return frames, label


class VideoTransform:
    def __call__(self, frames):
        transformed = []
        for frame in frames:
            frame = cv2.resize(frame, (224, 224))
            transformed.append(frame)
        return np.array(transformed)
```

import torch.nn as nn

class C3D(nn.Module):
    """C3D视频分类模型"""
    def __init__(self, num_classes=400):
        super(C3D, self).__init__()
        
        self.conv1 = nn.Conv3d(3, 64, kernel_size=3, padding=1)
        self.pool1 = nn.MaxPool3d(kernel_size=(1, 2, 2), stride=(1, 2, 2))
        
        self.conv2 = nn.Conv3d(64, 128, kernel_size=3, padding=1)
        self.pool2 = nn.MaxPool3d(kernel_size=(2, 2, 2), stride=(2, 2, 2))
        
        self.conv3 = nn.Conv3d(128, 256, kernel_size=3, padding=1)
        self.conv4 = nn.Conv3d(256, 256, kernel_size=3, padding=1)
        self.pool3 = nn.MaxPool3d(kernel_size=(2, 2, 2), stride=(2, 2, 2))
        
        self.conv5 = nn.Conv3d(256, 512, kernel_size=3, padding=1)
        self.conv6 = nn.Conv3d(512, 512, kernel_size=3, padding=1)
        self.pool4 = nn.MaxPool3d(kernel_size=(2, 2, 2), stride=(2, 2, 2))
        
        self.fc1 = nn.Linear(8192, 4096)
        self.fc2 = nn.Linear(4096, 4096)
        self.fc3 = nn.Linear(4096, num_classes)
        
        self.dropout = nn.Dropout(0.5)
    
    def forward(self, x):
        x = self.pool1(torch.relu(self.conv1(x)))
        x = self.pool2(torch.relu(self.conv2(x)))
        x = self.pool3(torch.relu(self.conv4(torch.relu(self.conv3(x)))))
        x = self.pool4(torch.relu(self.conv6(torch.relu(self.conv5(x)))))
        
        x = x.view(x.size(0), -1)
        x = torch.relu(self.fc1(x))
        x = self.dropout(x)
        x = torch.relu(self.fc2(x))
        x = self.dropout(x)
        x = self.fc3(x)
        
        return x


class I3D(nn.Module):
    """Inflated 3D Network"""
    def __init__(self, num_classes=400):
        super(I3D, self).__init__()
        
        self.features = self._build_features()
        
        self.avg_pool = nn.AdaptiveAvgPool3d((1, 1, 1))
        self.dropout = nn.Dropout(0.5)
        self.fc = nn.Linear(1024, num_classes)
    
    def _build_features(self):
        layers = []
        
        conv1 = nn.Conv3d(3, 64, kernel_size=(5, 7, 7), stride=(2, 2, 2), padding=(2, 3, 3))
        layers.append(conv1)
        layers.append(nn.ReLU())
        layers.append(nn.MaxPool3d(kernel_size=(2, 3, 3), stride=(2, 2, 2)))
        
        for in_channels, out_channels in [(64, 64), (64, 128), (128, 256), (256, 512)]:
            for _ in range(2):
                layers.append(nn.Conv3d(in_channels, out_channels, kernel_size=3, padding=1))
                layers.append(nn.ReLU())
            if in_channels != 64:
                layers.append(nn.MaxPool3d(kernel_size=2, stride=2))
        
        return nn.Sequential(*layers)
    
    def forward(self, x):
        x = self.features(x)
        x = self.avg_pool(x)
        x = x.view(x.size(0), -1)
        x = self.dropout(x)
        x = self.fc(x)
        return x
```

### 4.3 双流网络

```python
class TwoStreamCNN(nn.Module):
    """双流视频分类网络"""
    def __init__(self, num_classes):
        super(TwoStreamCNN, self).__init__()
        
        self.spatial_stream = self._build_stream()
        self.temporal_stream = self._build_stream()
        
        self.fc_fusion = nn.Linear(4096 + 4096, num_classes)
    
    def _build_stream(self):
        return nn.Sequential(
            nn.Conv2d(3, 96, kernel_size=7, stride=2, padding=3),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=3, stride=2, padding=1),
            
            nn.Conv2d(96, 256, kernel_size=5, padding=2),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=3, stride=2, padding=1),
            
            nn.Conv2d(256, 512, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.Conv2d(512, 512, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=3, stride=2, padding=1),
            
            nn.Conv2d(512, 4096, kernel_size=7, padding=0),
            nn.ReLU(),
            nn.AdaptiveAvgPool2d((1, 1))
        )
    
    def forward(self, spatial_input, temporal_input):
        spatial_feat = self.spatial_stream(spatial_input)
        temporal_feat = self.temporal_stream(temporal_input)
        
        fused = torch.cat([spatial_feat, temporal_feat], dim=1)
        output = self.fc_fusion(fused)
        
        return output


def compute_optical_flow(video_path):
    """计算光流"""
    cap = cv2.VideoCapture(video_path)
    ret, prev_frame = cap.read()
    prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
    
    flows = []
    
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        
        curr_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        flow = cv2.calcOpticalFlowFarneback(prev_gray, curr_gray, None, 0.5, 3, 15, 3, 5, 1.2, 0)
        flows.append(flow)
        prev_gray = curr_gray
    
    cap.release()
    return np.array(flows)
```

### 4.4 训练与推理

```python
def train_video_model(model, train_loader, val_loader, device, num_epochs=50):
    model = model.to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.AdamW(model.parameters(), lr=1e-4)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=num_epochs)
    
    best_acc = 0
    
    for epoch in range(num_epochs):
        model.train()
        total_loss = 0
        correct = 0
        total = 0
        
        for videos, labels in train_loader:
            videos = videos.to(device)
            labels = labels.to(device)
            
            optimizer.zero_grad()
            outputs = model(videos)
            loss = criterion(outputs, labels)
            
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
            _, predicted = outputs.max(1)
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()
        
        train_acc = 100. * correct / total
        
        val_acc = evaluate_video_model(model, val_loader, device)
        
        scheduler.step()
        
        print(f'Epoch {epoch+1}, Loss: {total_loss/len(train_loader):.4f}, '
              f'Train Acc: {train_acc:.2f}%, Val Acc: {val_acc:.2f}%')
        
        if val_acc > best_acc:
            best_acc = val_acc
            torch.save(model.state_dict(), 'best_video_model.pth')
    
    return model


def evaluate_video_model(model, dataloader, device):
    model.eval()
    correct = 0
    total = 0
    
    with torch.no_grad():
        for videos, labels in dataloader:
            videos = videos.to(device)
            labels = labels.to(device)
            
            outputs = model(videos)
            _, predicted = outputs.max(1)
            
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()
    
    return 100. * correct / total
```
