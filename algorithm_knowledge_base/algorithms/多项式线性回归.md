## 补充材料：多项式线性回归变体与扩展

### A1. 分段多项式回归

分段多项式（Piecewise Polynomial）在不同区间使用不同的多项式：

```python
class PiecewisePolynomialRegression:
    """分段多项式回归"""
    
    def __init__(self, n_segments=4, degree=3):
        self.n_segments = n_segments
        self.degree = degree
        self.models = []
        self.boundaries = []
    
    def fit(self, X, y):
        X = np.array(X).flatten()
        y = np.array(y)
        
        # 计算分段边界
        self.boundaries = np.percentile(X, 
            np.linspace(0, 100, self.n_segments + 1))
        
        # 对每段拟合多项式
        for i in range(self.n_segments):
            mask = (X >= self.boundaries[i]) & (X < self.boundaries[i + 1])
            if mask.sum() > self.degree:
                X_seg = X[mask]
                y_seg = y[mask]
                
                # 添加多项式特征
                X_poly = self._poly_features(X_seg, self.degree)
                
                # 最小二乘拟合
                coefs = np.linalg.lstsq(X_poly, y_seg, rcond=None)[0]
                self.models.append(coefs)
            else:
                self.models.append(np.array([y[mask].mean()]))
        
        return self
    
    def _poly_features(self, X, degree):
        return np.vstack([X ** d for d in range(degree + 1)]).T
    
    def predict(self, X):
        X = np.array(X).flatten()
        y_pred = np.zeros(len(X))
        
        for i, x in enumerate(X):
            # 找到对应段
            seg_idx = np.searchsorted(self.boundaries[1:], x)
            seg_idx = min(seg_idx, self.n_segments - 1)
            
            # 预测
            x_poly = self._poly_features(np.array([x]), self.degree)
            y_pred[i] = x_poly @ self.models[seg_idx]
        
        return y_pred
```

### A2. 样条回归（Spline Regression）

样条回归使用B样条基函数：

```python
class BSplineRegression:
    """B样条回归"""
    
    def __init__(self, degree=3, n_knots=10):
        self.degree = degree
        self.n_knots = n_knots
        self.knots = None
        self.coefs = None
    
    def _basis_functions(self, x, knots, degree):
        """B样条基函数"""
        if degree == 0:
            return ((x >= knots[:-1]) & (x < knots[1:])).astype(float)
        
        # 递归计算
        N = np.zeros((len(x), len(knots) - 1))
        
        if degree == 0:
            for i in range(len(knots) - 1):
                N[:, i] = ((x >= knots[i]) & (x < knots[i + 1]))
        else:
            left = self._basis_functions(x, knots[:-(degree + 1)], degree - 1)
            right = self._basis_functions(x, knots[1:], degree - 1)
            
            left_term = (x - knots[:-degree - 1]) / (knots[degree:-1] - knots[:-degree - 1] + 1e-10)
            right_term = (knots[degree + 1:] - x) / (knots[degree + 1:] - knots[1:-1] + 1e-10)
            
            N = left * left_term[:, :left.shape[1]] + right * right_term[:, :right.shape[1]]
        
        return N
    
    def fit(self, X, y):
        X = np.array(X).flatten()
        y = np.array(y)
        
        # 等间距knots
        self.knots = np.percentile(X, np.linspace(0, 100, self.n_knots))
        
        # B样条基
        Phi = self._basis_functions(X, self.knots, self.degree)
        
        # 最小二乘
        self.coefs = np.linalg.lstsq(Phi, y, rcond=None)[0]
        
        return self
    
    def predict(self, X):
        X = np.array(X).flatten()
        Phi = self._basis_functions(X, self.knots, self.degree)
        return Phi @ self.coefs
```

### A3. 岭回归与多项式结合

```python
class PolynomialRidgeRegression:
    """多项式岭回归"""
    
    def __init__(self, degree=3, alpha=1.0):
        self.degree = degree
        self.alpha = alpha
        self.coefs = None
        self.mean = None
        self.std = None
    
    def fit(self, X, y):
        X = np.array(X)
        y = np.array(y)
        
        # 多项式特征
        X_poly = self._poly_features(X, self.degree)
        
        # 标准化
        self.mean = X_poly.mean(axis=0)
        self.std = X_poly.std(axis=0)
        self.std[self.std < 1e-10] = 1
        X_poly = (X_poly - self.mean) / self.std
        
        # 岭回归
        n_samples, n_features = X_poly.shape
        I = np.eye(n_features)
        I[0, 0] = 0  # 不正则化截距
        
        XTX = X_poly.T @ X_poly + self.alpha * I
        XTy = X_poly.T @ y
        self.coefs = np.linalg.solve(XTX, XTy)
        
        return self
    
    def _poly_features(self, X, degree):
        return np.vstack([X ** d for d in range(degree + 1)]).T
    
    def predict(self, X):
        X = np.array(X)
        X_poly = self._poly_features(X, self.degree)
        X_poly = (X_poly - self.mean) / self.std
        return X_poly @ self.coefs
```

### A4. 多项式回归可视化进阶

```python
import numpy as np
import matplotlib.pyplot as plt

def visualize_polynomial_degree_selection():
    """可视化多项式次数选择"""
    np.random.seed(42)
    
    # 真实函数：y = sin(2πx) + 噪声
    n_samples = 50
    X = np.linspace(0, 1, n_samples)
    y_true = np.sin(2 * np.pi * X)
    y = y_true + np.random.randn(n_samples) * 0.3
    
    degrees = [1, 3, 5, 7, 9, 15]
    
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))
    axes = axes.flatten()
    
    X_test = np.linspace(0, 1, 200)
    train_errors = []
    test_errors = []
    
    for i, degree in enumerate(degrees):
        # 拟合
        coefs = np.polyfit(X, y, degree)
        
        # 训练误差
        y_train_pred = np.polyval(coefs, X)
        train_mse = np.mean((y - y_train_pred) ** 2)
        train_errors.append(train_mse)
        
        # 测试误差（真实函数）
        y_test_pred = np.polyval(coefs, X_test)
        test_mse = np.mean((y_true - y_test_pred) ** 2)
        test_errors.append(test_mse)
        
        # 可视化
        ax = axes[i]
        ax.scatter(X, y, alpha=0.5, s=30, label='Data')
        ax.plot(X_test, y_true, 'k-', linewidth=2, label='True')
        ax.plot(X_test, y_test_pred, 'r--', linewidth=2, label='Predict')
        ax.set_title(f'Degree={degree}')
        ax.set_ylim(-2, 2)
        ax.legend()
        ax.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.savefig('polynomial_degree.png', dpi=150)
    plt.show()
    
    # 误差曲线
    plt.figure(figsize=(10, 6))
    plt.semilogy(degrees, train_errors, 'o-', label='Train MSE')
    plt.semilogy(degrees, test_errors, 's-', label='Test MSE')
    plt.xlabel('Polynomial Degree')
    plt.ylabel('MSE (log scale)')
    plt.title('Bias-Variance Tradeoff')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.savefig('polynomial_bias_variance.png', dpi=150)
    plt.show()


def visualize_polynomial_types():
    """可视化不同类型的多项式"""
    np.random.seed(42)
    
    X = np.linspace(-3, 3, 100)
    
    polynomial_types = {
        'Linear': [0, 1],
        'Quadratic': [0, 1, 2],
        'Cubic': [0, 1, 2, 3],
        'Quartic': [0, 1, 2, 3, 4],
    }
    
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    axes = axes.flatten()
    
    for idx, (name, degree) in enumerate(polynomial_types.items()):
        ax = axes[idx]
        
        # 不同次数的项
        for d in range(degree[0], degree[1] + 1):
            y = X ** d
            ax.plot(X, y / y.max(), linewidth=1.5, 
                   label=f'x^{d}', alpha=0.7)
        
        ax.axhline(y=0, color='k', linestyle='--', alpha=0.3)
        ax.set_title(f'{name} Basis')
        ax.set_xlabel('x')
        ax.set_ylabel('Normalized Value')
        ax.legend()
        ax.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.savefig('polynomial_types.png', dpi=150)
    plt.show()


def compare_regularization():
    """比较不同正则化"""
    np.random.seed(42)
    
    # 生成数据
    n_samples = 30
    X = np.linspace(0, 1, n_samples)
    y = np.sin(2 * np.pi * X) + np.random.randn(n_samples) * 0.2
    
    degrees = [9]
    alphas = [0, 0.001, 0.01, 0.1, 1.0]
    
    X_test = np.linspace(0, 1, 200)
    y_true = np.sin(2 * np.pi * X_test)
    
    fig, axes = plt.subplots(1, len(alphas), figsize=(15, 3))
    
    for i, alpha in enumerate(alphas):
        # 岭回归
        X_poly = np.vstack([X ** d for d in range(degrees[0] + 1)]).T
        X_test_poly = np.vstack([X_test ** d for d in range(degrees[0] + 1)]).T
        
        # 标准化
        X_mean = X_poly.mean(axis=0)
        X_std = X_poly.std(axis=0)
        X_std[X_std < 1e-10] = 1
        
        X_poly_norm = (X_poly - X_mean) / X_std
        X_test_poly_norm = (X_test_poly - X_mean) / X_std
        
        # 岭回归
        I = np.eye(degrees[0] + 1)
        I[0, 0] = 0
        coefs = np.linalg.lstsq(
            X_poly_norm.T @ X_poly_norm + alpha * I,
            X_poly_norm.T @ y,
            rcond=None
        )[0]
        
        y_pred = X_test_poly_norm @ coefs
        
        axes[i].scatter(X, y, alpha=0.5, s=20)
        axes[i].plot(X_test, y_true, 'k-', linewidth=1.5, label='True')
        axes[i].plot(X_test, y_pred, 'r-', linewidth=1.5, label='Predict')
        axes[i].set_title(f'α={alpha}')
        axes[i].set_ylim(-1.5, 1.5)
        axes[i].legend()
    
    plt.tight_layout()
    plt.savefig('regularization_comparison.png', dpi=150)
    plt.show()


def show_fit_under_different_noise():
    """展示不同噪声水平下的拟合"""
    np.random.seed(42)
    
    noise_levels = [0.01, 0.05, 0.1, 0.2, 0.5]
    
    X = np.linspace(0, 1, 30)
    y_true = np.sin(2 * np.pi * X)
    
    fig, axes = plt.subplots(1, len(noise_levels), figsize=(15, 3))
    
    for i, noise in enumerate(noise_levels):
        y = y_true + np.random.randn(len(X)) * noise
        
        # 拟合
        coefs = np.polyfit(X, y, 5)
        
        X_test = np.linspace(0, 1, 100)
        y_pred = np.polyval(coefs, X_test)
        
        axes[i].scatter(X, y, alpha=0.5, s=20)
        axes[i].plot(X_test, y_true, 'k-', linewidth=1, label='True')
        axes[i].plot(X_test, y_pred, 'r-', linewidth=1.5, label='Fit')
        axes[i].set_title(f'Noise={noise}')
        axes[i].set_ylim(-1.5, 1.5)
        axes[i].legend()
    
    plt.tight_layout()
    plt.savefig('noise_levels.png', dpi=150)
    plt.show()
```
