## 2. 核心原理与Python

```python
import numpy as np

def medical_decision_sim(n_patients=200, n_treatments=3, horizon=10):
    """医疗决策模拟：在线学习最优治疗方案"""
    # 真实效果（未知）
    true_effects = np.array([0.3, 0.7, 0.5])
    beliefs = np.ones(n_treatments) * 0.5
    counts = np.zeros(n_treatments)
    total_health = 0

    for p in range(n_patients):
        for t in range(horizon):
            # Thompson采样选治疗
            samples = np.random.beta(1 + beliefs*10, 1 + (1-beliefs)*10)
            treatment = np.argmax(samples)
            # 观察效果
            outcome = np.random.rand() < true_effects[treatment]
            total_health += outcome
            counts[treatment] += 1
            beliefs[treatment] += (outcome - beliefs[treatment]) / counts[treatment]

    print(f"真实效果: {true_effects}")
    print(f"学习估计: {np.round(beliefs, 2)}")
    print(f"选择次数: {counts.astype(int)}")
    print(f"总健康收益: {total_health}/{n_patients*horizon}")

np.random.seed(42)
medical_decision_sim()
```
