## 1. 基础认知

**一句话定义**：竞价问题决定在广告拍卖中出多少价；定价问题决定产品价格以最大化收入——两者都是学习+优化的结合。

## 2. 核心原理与Python

```python
import numpy as np

def ad_bidding_sim(n_rounds=500, budget=1000):
    """广告竞价模拟"""
    true_values = np.random.uniform(0.5, 2.0, 10)  # 每个广告位的价值
    bids = np.ones(10) * 1.0
    total_value = 0; remaining = budget

    for r in range(n_rounds):
        if remaining <= 0: break
        # 选择出价的广告位
        slot = np.argmax(bids * true_values - bids)
        bid = bids[slot]
        # 竞价（简化：赢概率与出价正相关）
        win = np.random.rand() < bid / (bid + 1)
        if win:
            cost = bid * 0.5
            if cost <= remaining:
                remaining -= cost
                total_value += true_values[slot]
                # 更新价值估计
        # 调整出价
        bids[slot] *= 0.99

    print(f"真实价值: {np.round(true_values, 2)}")
    print(f"总收益: {total_value:.1f}, 剩余预算: {remaining:.0f}")

np.random.seed(42)
ad_bidding_sim()
```
