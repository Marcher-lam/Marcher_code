## 2. 核心原理与Python

```python
import numpy as np

def newsvendor_final(c=5, p=10, h=1, n_days=1000):
    """报童问题-最终奖励版"""
    demands = np.random.poisson(50, n_days)
    best_q = int(np.quantile(demands, (p-c)/(p+h)))
    profits = [p*min(d, best_q) - c*best_q - h*max(0, best_q-d) for d in demands]
    return best_q, np.mean(profits)

def newsvendor_cumulative(c=5, p=10, h=1, T=100, alpha=0.1):
    """报童问题-累积奖励版（在线学习）"""
    q = 40
    total_profit = 0
    for t in range(T):
        demand = np.random.poisson(50)
        sales = min(demand, q)
        profit = p*sales - c*q - h*max(0, q-demand)
        total_profit += profit
        # 梯度更新
        if demand > q:
            q += alpha * (p - c)
        else:
            q -= alpha * (c + h)
        q = max(1, int(q))
    return total_profit / T

def inventory_with_lags(c=5, p=12, h=1, lead_time=3, T=100):
    """带滞后(提前期)的库存问题"""
    inventory = 50; pending = []; total = 0
    for t in range(T):
        # 到货
        arrived = sum(q for tt, q in pending if tt <= t)
        pending = [(tt, q) for tt, q in pending if tt > t]
        inventory += arrived
        demand = np.random.poisson(30)
        sales = min(demand, inventory)
        inventory -= sales
        total += p * sales - h * inventory
        # 订货（提前期lead_time天后到）
        order = max(0, 80 - inventory - len(pending)*20)
        pending.append((t + lead_time, order))
    return total
```
