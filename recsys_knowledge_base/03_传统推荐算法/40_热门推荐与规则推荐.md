# 热门推荐与规则推荐 学习文档

## 1. 算法基础认知

### 1.1 什么是热门推荐？

热门推荐是最简单、最直观的推荐策略，基于"从众心理"，向所有用户推荐最受欢迎的物品。

### 1.2 什么是规则推荐？

规则推荐基于预设的业务规则进行推荐，如"买了A的人也买了B"、"新用户推荐新品"等。

### 1.3 为什么需要这些简单方法？

| 优势 | 说明 |
|------|------|
| 冷启动 | 新用户无行为数据时可用 |
| 高效 | 计算简单，响应极快 |
| 可解释 | 推荐理由清晰明确 |
| 基线 | 作为复杂模型的对比基准 |

## 2. 热门推荐原理

### 2.1 基本热门度

```
热门度 = 交互次数 / 时间衰减因子
```

### 2.2 时间衰减热门度

```
热门度(t) = Σ exp(-λ * (t - t_i))

其中:
- t: 当前时间
- t_i: 第i次交互的时间
- λ: 衰减系数
```

### 2.3 多维度热门度

```python
import numpy as np
from typing import List, Dict, Tuple
from datetime import datetime, timedelta
from collections import defaultdict


class PopularRecommender:
    """
    热门推荐器
    """

    def __init__(self, time_decay: float = 0.01, top_k: int = 100):
        """
        参数:
            time_decay: 时间衰减系数
            top_k: 保留的热门物品数量
        """
        self.time_decay = time_decay
        self.top_k = top_k
        self.item_scores = {}
        self.item_interactions = defaultdict(list)

    def add_interaction(self, item_id: str, timestamp: float):
        """添加交互记录"""
        self.item_interactions[item_id].append(timestamp)

    def compute_scores(self, current_time: float):
        """计算热门分数"""
        self.item_scores = {}

        for item_id, timestamps in self.item_interactions.items():
            # 时间衰减分数
            score = sum(
                np.exp(-self.time_decay * (current_time - t))
                for t in timestamps
            )
            self.item_scores[item_id] = score

    def recommend(self, n: int = 10, exclude_items: List[str] = None) -> List[Tuple[str, float]]:
        """
        推荐热门物品

        参数:
            n: 推荐数量
            exclude_items: 要排除的物品

        返回:
            [(item_id, score), ...]
        """
        exclude_set = set(exclude_items or [])

        # 排序
        sorted_items = sorted(
            self.item_scores.items(),
            key=lambda x: x[1],
            reverse=True
        )

        # 过滤并返回
        result = [
            (item, score)
            for item, score in sorted_items
            if item not in exclude_set
        ][:n]

        return result


class CategoryPopularRecommender:
    """
    分品类热门推荐器
    """

    def __init__(self, time_decay: float = 0.01):
        self.time_decay = time_decay
        self.category_items = defaultdict(dict)  # {category: {item: [timestamps]}}

    def add_interaction(self, item_id: str, category: str, timestamp: float):
        """添加交互"""
        if item_id not in self.category_items[category]:
            self.category_items[category][item_id] = []
        self.category_items[category][item_id].append(timestamp)

    def recommend_by_category(self, category: str, current_time: float,
                             n: int = 10) -> List[Tuple[str, float]]:
        """按品类推荐"""
        if category not in self.category_items:
            return []

        scores = {}
        for item_id, timestamps in self.category_items[category].items():
            score = sum(
                np.exp(-self.time_decay * (current_time - t))
                for t in timestamps
            )
            scores[item_id] = score

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)[:n]
```

## 3. 规则推荐原理

### 3.1 常见规则类型

```
1. 关联规则: 买了A -> 推荐B
2. 时序规则: 新用户 -> 推荐新品
3. 属性规则: 喜欢科幻 -> 推荐科幻类
4. 组合规则: 条件A AND 条件B -> 推荐C
```

### 3.2 规则推荐实现

```python
from typing import List, Dict, Tuple, Set, Optional
from dataclasses import dataclass
from enum import Enum


class RuleType(Enum):
    """规则类型"""
    ASSOCIATION = "association"  # 关联规则
    SEQUENCE = "sequence"        # 时序规则
    ATTRIBUTE = "attribute"      # 属性规则
    COMPOSITE = "composite"      # 组合规则


@dataclass
class RecommendationRule:
    """推荐规则"""
    rule_id: str
    rule_type: RuleType
    conditions: Dict  # 触发条件
    recommendations: List[str]  # 推荐物品
    priority: int = 0  # 优先级
    enabled: bool = True


class RuleBasedRecommender:
    """
    规则推荐器
    """

    def __init__(self):
        self.rules: List[RecommendationRule] = []
        self.association_rules = {}  # {item: [related_items]}

    def add_rule(self, rule: RecommendationRule):
        """添加规则"""
        self.rules.append(rule)
        self.rules.sort(key=lambda x: x.priority, reverse=True)

    def add_association_rule(self, trigger_item: str,
                            related_items: List[Tuple[str, float]]):
        """
        添加关联规则

        参数:
            trigger_item: 触发物品
            related_items: [(related_item, confidence), ...]
        """
        self.association_rules[trigger_item] = related_items

    def recommend_by_association(self, interacted_items: List[str],
                                 n: int = 10) -> List[Tuple[str, float]]:
        """
        基于关联规则推荐

        参数:
            interacted_items: 用户交互过的物品
            n: 推荐数量

        返回:
            [(item_id, confidence), ...]
        """
        scores = defaultdict(float)

        for item in interacted_items:
            if item in self.association_rules:
                for related_item, confidence in self.association_rules[item]:
                    scores[related_item] = max(scores[related_item], confidence)

        # 排除已交互物品
        interacted_set = set(interacted_items)
        result = [
            (item, score)
            for item, score in sorted(scores.items(),
                                     key=lambda x: x[1],
                                     reverse=True)
            if item not in interacted_set
        ][:n]

        return result

    def recommend_by_rules(self, user_profile: Dict,
                          context: Dict,
                          n: int = 10) -> List[Tuple[str, str]]:
        """
        基于规则推荐

        参数:
            user_profile: 用户画像
            context: 上下文信息
            n: 推荐数量

        返回:
            [(item_id, rule_id), ...]
        """
        recommendations = []
        matched_rules = set()

        for rule in self.rules:
            if not rule.enabled:
                continue

            if self._match_rule(rule, user_profile, context):
                for item in rule.recommendations:
                    if item not in [r[0] for r in recommendations]:
                        recommendations.append((item, rule.rule_id))
                        matched_rules.add(rule.rule_id)

                if len(recommendations) >= n:
                    break

        return recommendations[:n]

    def _match_rule(self, rule: RecommendationRule,
                   user_profile: Dict, context: Dict) -> bool:
        """检查规则是否匹配"""
        conditions = rule.conditions

        if rule.rule_type == RuleType.SEQUENCE:
            # 新用户规则
            if conditions.get("is_new_user"):
                return user_profile.get("registration_days", 999) < 7

        elif rule.rule_type == RuleType.ATTRIBUTE:
            # 属性规则
            required_attrs = conditions.get("required_attributes", {})
            for attr, values in required_attrs.items():
                user_value = user_profile.get(attr)
                if user_value not in values:
                    return False
            return True

        elif rule.rule_type == RuleType.COMPOSITE:
            # 组合规则
            return all(
                self._check_condition(cond, user_profile, context)
                for cond in conditions.get("conditions", [])
            )

        return False

    def _check_condition(self, condition: Dict,
                        user_profile: Dict, context: Dict) -> bool:
        """检查单个条件"""
        field = condition.get("field")
        operator = condition.get("operator")
        value = condition.get("value")

        actual_value = user_profile.get(field) or context.get(field)

        if actual_value is None:
            return False

        if operator == "eq":
            return actual_value == value
        elif operator == "in":
            return actual_value in value
        elif operator == "gt":
            return actual_value > value
        elif operator == "lt":
            return actual_value < value
        elif operator == "gte":
            return actual_value >= value
        elif operator == "lte":
            return actual_value <= value

        return False


class CompositeRecommender:
    """
    组合推荐器 - 热门 + 规则
    """

    def __init__(self, popular_weight: float = 0.3,
                 rule_weight: float = 0.7):
        self.popular_recommender = PopularRecommender()
        self.rule_recommender = RuleBasedRecommender()
        self.popular_weight = popular_weight
        self.rule_weight = rule_weight

    def recommend(self, user_profile: Dict, context: Dict,
                 n: int = 10) -> List[Tuple[str, float, str]]:
        """
        组合推荐

        返回:
            [(item_id, score, source), ...]
        """
        recommendations = {}

        # 热门推荐
        popular_recs = self.popular_recommender.recommend(n * 2)
        for item, score in popular_recs:
            recommendations[item] = {
                'score': score * self.popular_weight,
                'source': 'popular'
            }

        # 规则推荐
        rule_recs = self.rule_recommender.recommend_by_rules(
            user_profile, context, n * 2
        )
        for item, rule_id in rule_recs:
            if item in recommendations:
                recommendations[item]['score'] += 1.0 * self.rule_weight
                recommendations[item]['source'] = 'both'
            else:
                recommendations[item] = {
                    'score': 1.0 * self.rule_weight,
                    'source': 'rule'
                }

        # 排序返回
        sorted_recs = sorted(
            recommendations.items(),
            key=lambda x: x[1]['score'],
            reverse=True
        )[:n]

        return [
            (item, data['score'], data['source'])
            for item, data in sorted_recs
        ]
```

## 4. 关联规则挖掘

### 4.1 Apriori 算法

```python
from itertools import combinations
from typing import List, Set, Dict, Tuple


class AprioriMiner:
    """
    Apriori 关联规则挖掘
    """

    def __init__(self, min_support: float = 0.01,
                 min_confidence: float = 0.5):
        """
        参数:
            min_support: 最小支持度
            min_confidence: 最小置信度
        """
        self.min_support = min_support
        self.min_confidence = min_confidence

    def fit(self, transactions: List[List[str]]) -> Dict:
        """
        挖掘关联规则

        参数:
            transactions: 交易列表

        返回:
            {antecedent: [(consequent, confidence), ...]}
        """
        n_transactions = len(transactions)

        # 生成频繁项集
        frequent_itemsets = self._find_frequent_itemsets(transactions)

        # 生成关联规则
        rules = defaultdict(list)

        for itemset, support in frequent_itemsets.items():
            if len(itemset) < 2:
                continue

            # 对每个非空子集生成规则
            for size in range(1, len(itemset)):
                for antecedent in combinations(itemset, size):
                    antecedent = frozenset(antecedent)
                    consequent = itemset - antecedent

                    # 计算置信度
                    antecedent_support = frequent_itemsets.get(antecedent, 0)
                    if antecedent_support > 0:
                        confidence = support / antecedent_support

                        if confidence >= self.min_confidence:
                            rules[tuple(antecedent)].append(
                                (tuple(consequent), confidence)
                            )

        return dict(rules)

    def _find_frequent_itemsets(self,
                               transactions: List[List[str]]) -> Dict:
        """发现频繁项集"""
        n_transactions = len(transactions)

        # 单项频繁集
        item_counts = defaultdict(int)
        for trans in transactions:
            for item in set(trans):
                item_counts[item] += 1

        min_count = self.min_support * n_transactions
        frequent_items = {
            frozenset([item]): count
            for item, count in item_counts.items()
            if count >= min_count
        }

        all_frequent = frequent_items.copy()
        current_itemsets = set(frequent_items.keys())
        k = 2

        while current_itemsets:
            # 生成候选
            candidates = self._generate_candidates(current_itemsets, k)

            # 计数
            candidate_counts = defaultdict(int)
            for trans in transactions:
                trans_set = set(trans)
                for candidate in candidates:
                    if candidate.issubset(trans_set):
                        candidate_counts[candidate] += 1

            # 筛选频繁集
            current_itemsets = {
                candidate
                for candidate, count in candidate_counts.items()
                if count >= min_count
            }

            for itemset in current_itemsets:
                all_frequent[itemset] = candidate_counts[itemset]

            k += 1

        return all_frequent

    def _generate_candidates(self, itemsets: Set[frozenset],
                            k: int) -> Set[frozenset]:
        """生成候选集"""
        candidates = set()
        itemsets_list = list(itemsets)

        for i in range(len(itemsets_list)):
            for j in range(i + 1, len(itemsets_list)):
                # 合并两个项集
                union = itemsets_list[i] | itemsets_list[j]
                if len(union) == k:
                    # 检查所有子集是否频繁
                    all_subsets_frequent = all(
                        frozenset(s) in itemsets
                        for s in combinations(union, k - 1)
                    )
                    if all_subsets_frequent:
                        candidates.add(union)

        return candidates


def demo_association_rules():
    """关联规则示例"""
    # 模拟交易数据
    transactions = [
        ['牛奶', '面包', '黄油'],
        ['牛奶', '面包'],
        ['牛奶', '鸡蛋'],
        ['面包', '黄油'],
        ['牛奶', '面包', '鸡蛋'],
        ['牛奶', '鸡蛋', '黄油'],
        ['面包', '鸡蛋'],
        ['牛奶', '面包', '黄油', '鸡蛋'],
    ]

    # 挖掘规则
    miner = AprioriMiner(min_support=0.25, min_confidence=0.5)
    rules = miner.fit(transactions)

    print("关联规则:")
    for antecedent, consequents in rules.items():
        print(f"  {set(antecedent)} -> ", end="")
        for consequent, conf in consequents:
            print(f"{set(consequent)} (置信度: {conf:.2f}) ", end="")
        print()


if __name__ == "__main__":
    demo_association_rules()
```

### 5.2 实时更新热门榜单

```python
class RealtimePopularTracker:
    """
    实时热门追踪器
    """

    def __init__(self, window_size: int = 3600):
        """
        参数:
            window_size: 时间窗口（秒）
        """
        self.window_size = window_size
        self.item_events = defaultdict(list)  # {item: [(timestamp, score)]}

    def add_event(self, item_id: str, timestamp: float, score: float = 1.0):
        """添加事件"""
        self.item_events[item_id].append((timestamp, score))

    def get_top_k(self, current_time: float, k: int = 10):
        """获取当前热门Top-K"""
        scores = {}

        for item_id, events in self.item_events.items():
            # 只计算时间窗口内的事件
            window_score = sum(
                score for ts, score in events
                if current_time - ts <= self.window_size
            )
            if window_score > 0:
                scores[item_id] = window_score

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)[:k]

    def cleanup(self, current_time: float):
        """清理过期数据"""
        for item_id in list(self.item_events.keys()):
            self.item_events[item_id] = [
                (ts, score) for ts, score in self.item_events[item_id]
                if current_time - ts <= self.window_size * 2
            ]
            if not self.item_events[item_id]:
                del self.item_events[item_id]
```

## 6. 学习总结

### 6.1 核心要点

1. **热门推荐**：简单高效，适合冷启动
2. **规则推荐**：可解释性强，业务可控
3. **组合策略**：结合多种方法的优势

### 6.2 适用场景

| 场景 | 推荐方法 |
|------|----------|
| 新用户 | 热门 + 规则 |
| 新物品 | 同类热门 |
| 无行为数据 | 热门 |
| 特定业务需求 | 规则 |
