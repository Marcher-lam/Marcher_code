# 双塔模型 学习文档

## 1. 算法基础认知

### 1.1 什么是双塔模型？

双塔模型（Two-Tower Model）是一种经典的深度学习推荐架构，由两个独立的神经网络塔组成：

- **用户塔（User Tower）**：将用户特征编码为用户向量
- **物品塔（Item Tower）**：将物品特征编码为物品向量

最终通过计算两个向量的相似度（通常是点积）得到推荐分数。

```
用户特征 ──→ 用户塔 ──→ 用户向量 ──┐
                                    ├──→ 点积 ──→ 分数
物品特征 ──→ 物品塔 ──→ 物品向量 ──┘
```

### 1.2 为什么使用双塔模型？

**优势：**

1. **解耦计算**：用户和物品向量可以独立预计算
2. **高效召回**：物品向量可以建立索引，支持 ANN 检索
3. **可扩展性**：新增物品只需计算物品向量
4. **工业友好**：适合大规模推荐系统

**与其他模型对比：**

| 模型 | 特征交叉 | 预计算 | 适用阶段 |
|------|----------|--------|----------|
| FM/DeepFM | 有 | 否 | 排序 |
| 双塔模型 | 最后点积 | 是 | 召回 |
| DIN | 有 | 否 | 排序 |

### 1.3 典型应用场景

- **召回阶段**：从百万物品中召回千级候选
- **相似物品**：基于物品向量找相似物品
- **跨域推荐**：用户向量与另一域物品向量匹配

## 2. 模型架构

### 2.1 基本结构

```python
import torch
import torch.nn as nn

class TwoTowerModel(nn.Module):
    """
    双塔模型基本实现
    """

    def __init__(self, user_feature_configs, item_feature_configs,
                 embed_dim=64, hidden_units=[256, 128]):
        """
        参数:
            user_feature_configs: dict, 用户特征配置
            item_feature_configs: dict, 物品特征配置
            embed_dim: int, 最终向量维度
            hidden_units: list, 隐藏层维度
        """
        super().__init__()

        # 用户塔
        self.user_tower = UserTower(
            user_feature_configs, embed_dim, hidden_units
        )

        # 物品塔
        self.item_tower = ItemTower(
            item_feature_configs, embed_dim, hidden_units
        )

    def forward(self, user_features, item_features):
        """
        前向传播

        返回:
            scores: (batch,) 预测分数
        """
        # 获取用户向量
        user_vector = self.user_tower(user_features)  # (batch, embed_dim)

        # 获取物品向量
        item_vector = self.item_tower(item_features)  # (batch, embed_dim)

        # 计算点积
        scores = torch.sum(user_vector * item_vector, dim=1)

        return scores

    def get_user_vector(self, user_features):
        """获取用户向量（用于推理）"""
        return self.user_tower(user_features)

    def get_item_vector(self, item_features):
        """获取物品向量（用于推理）"""
        return self.item_tower(item_features)


class UserTower(nn.Module):
    """用户塔"""

    def __init__(self, feature_configs, embed_dim, hidden_units):
        super().__init__()

        # 特征 Embedding
        self.embeddings = nn.ModuleDict()
        input_dim = 0

        for name, config in feature_configs.items():
            if config['type'] == 'categorical':
                self.embeddings[name] = nn.Embedding(
                    config['vocab_size'],
                    config.get('embed_dim', 16)
                )
                input_dim += config.get('embed_dim', 16)
            elif config['type'] == 'numerical':
                input_dim += 1

        # MLP
        layers = []
        for hidden in hidden_units:
            layers.append(nn.Linear(input_dim, hidden))
            layers.append(nn.ReLU())
            layers.append(nn.BatchNorm1d(hidden))
            input_dim = hidden

        layers.append(nn.Linear(input_dim, embed_dim))
        self.mlp = nn.Sequential(*layers)

    def forward(self, features):
        """前向传播"""
        emb_list = []

        for name, emb_layer in self.embeddings.items():
            if name in features:
                emb = emb_layer(features[name])
                emb_list.append(emb)

        # 添加数值特征
        for name, value in features.items():
            if name not in self.embeddings and isinstance(value, torch.Tensor):
                emb_list.append(value.unsqueeze(-1) if value.dim() == 1 else value)

        # 拼接
        concat = torch.cat(emb_list, dim=-1)

        # MLP
        output = self.mlp(concat)

        # L2 归一化
        output = torch.nn.functional.normalize(output, p=2, dim=-1)

        return output


class ItemTower(nn.Module):
    """物品塔"""

    def __init__(self, feature_configs, embed_dim, hidden_units):
        super().__init__()

        # 与用户塔类似
        self.embeddings = nn.ModuleDict()
        input_dim = 0

        for name, config in feature_configs.items():
            if config['type'] == 'categorical':
                self.embeddings[name] = nn.Embedding(
                    config['vocab_size'],
                    config.get('embed_dim', 16)
                )
                input_dim += config.get('embed_dim', 16)
            elif config['type'] == 'numerical':
                input_dim += 1

        layers = []
        for hidden in hidden_units:
            layers.append(nn.Linear(input_dim, hidden))
            layers.append(nn.ReLU())
            layers.append(nn.BatchNorm1d(hidden))
            input_dim = hidden

        layers.append(nn.Linear(input_dim, embed_dim))
        self.mlp = nn.Sequential(*layers)

    def forward(self, features):
        emb_list = []

        for name, emb_layer in self.embeddings.items():
            if name in features:
                emb = emb_layer(features[name])
                emb_list.append(emb)

        for name, value in features.items():
            if name not in self.embeddings and isinstance(value, torch.Tensor):
                emb_list.append(value.unsqueeze(-1) if value.dim() == 1 else value)

        concat = torch.cat(emb_list, dim=-1)
        output = self.mlp(concat)
        output = torch.nn.functional.normalize(output, p=2, dim=-1)

        return output
```

### 2.2 DSSM 模型

DSSM（Deep Structured Semantic Models）是微软提出的双塔模型，最初用于搜索排序。

```python
class DSSMModel(nn.Module):
    """
    DSSM 模型实现
    """

    def __init__(self, query_vocab_size, doc_vocab_size,
                 embed_dim=128, hidden_units=[256, 128, 64]):
        super().__init__()

        # Query 塔（相当于用户塔）
        self.query_embedding = nn.Embedding(query_vocab_size, embed_dim)
        self.query_mlp = self._build_mlp(embed_dim, hidden_units, embed_dim)

        # Document 塔（相当于物品塔）
        self.doc_embedding = nn.Embedding(doc_vocab_size, embed_dim)
        self.doc_mlp = self._build_mlp(embed_dim, hidden_units, embed_dim)

        # 温度参数
        self.temperature = nn.Parameter(torch.ones(1))

    def _build_mlp(self, input_dim, hidden_units, output_dim):
        layers = []
        for hidden in hidden_units:
            layers.append(nn.Linear(input_dim, hidden))
            layers.append(nn.Tanh())  # DSSM 使用 Tanh
            input_dim = hidden
        layers.append(nn.Linear(input_dim, output_dim))
        return nn.Sequential(*layers)

    def forward(self, query_ids, doc_ids, negative_doc_ids=None):
        """
        前向传播

        参数:
            query_ids: (batch,) Query ID
            doc_ids: (batch,) 正样本 Document ID
            negative_doc_ids: (batch, num_neg) 负样本 Document ID
        """
        # Query 向量
        query_emb = self.query_embedding(query_ids)
        query_vec = self.query_mlp(query_emb)
        query_vec = torch.nn.functional.normalize(query_vec, p=2, dim=-1)

        # 正样本 Document 向量
        pos_doc_emb = self.doc_embedding(doc_ids)
        pos_doc_vec = self.doc_mlp(pos_doc_emb)
        pos_doc_vec = torch.nn.functional.normalize(pos_doc_vec, p=2, dim=-1)

        # 正样本分数
        pos_scores = torch.sum(query_vec * pos_doc_vec, dim=-1) / self.temperature

        if negative_doc_ids is not None:
            # 负样本 Document 向量
            batch_size, num_neg = negative_doc_ids.shape
            neg_doc_emb = self.doc_embedding(negative_doc_ids.view(-1))
            neg_doc_vec = self.doc_mlp(neg_doc_emb)
            neg_doc_vec = torch.nn.functional.normalize(neg_doc_vec, p=2, dim=-1)
            neg_doc_vec = neg_doc_vec.view(batch_size, num_neg, -1)

            # 负样本分数
            neg_scores = torch.bmm(
                neg_doc_vec,
                query_vec.unsqueeze(-1)
            ).squeeze(-1) / self.temperature

            return pos_scores, neg_scores

        return pos_scores
```

### 3.1 负采样

双塔模型需要负样本来学习区分正负样本。

import torch
import numpy as np
import random

class NegativeSampler:
    """负采样器"""

    def __init__(self, num_items: int, num_neg: int = 4,
                 sampling_strategy: str = 'random'):
        """
        参数:
            num_items: 物品总数
            num_neg: 负样本数量
            sampling_strategy: 采样策略
        """
        self.num_items = num_items
        self.num_neg = num_neg
        self.sampling_strategy = sampling_strategy

        # 流行度分布（用于流行度采样）
        self.item_popularity = None

    def set_popularity(self, item_popularity):
        """设置物品流行度"""
        total = sum(item_popularity.values())
        self.item_popularity = {
            k: v / total for k, v in item_popularity.items()
        }

    def sample(self, positive_items: list) -> torch.Tensor:
        """
        负采样

        参数:
            positive_items: 正样本物品 ID 列表

        返回:
            negative_items: (batch, num_neg) 负样本
        """
        batch_size = len(positive_items)
        negatives = []

        for pos_item in positive_items:
            negs = []
            while len(negs) < self.num_neg:
                if self.sampling_strategy == 'random':
                    neg = random.randint(0, self.num_items - 1)
                elif self.sampling_strategy == 'popularity':
                    neg = self._sample_by_popularity()
                elif self.sampling_strategy == 'hard':
                    neg = self._sample_hard_negatives()

                # 排除正样本
                if neg != pos_item:
                    negs.append(neg)

            negatives.append(negs)

        return torch.LongTensor(negatives)

class InBatchNegativeSampler:
    """
    In-batch 负采样
    使用同一 batch 内其他样本的正样本作为负样本
    """

    def __init__(self, temperature=0.1):
        self.temperature = temperature

    def compute_loss(self, user_vectors, item_vectors):
        """
        计算 in-batch 负采样损失

        参数:
            user_vectors: (batch, dim) 用户向量
            item_vectors: (batch, dim) 物品向量
        """
        batch_size = user_vectors.size(0)

        # 计算相似度矩阵
        # (batch, batch)
        similarity = torch.mm(user_vectors, item_vectors.t()) / self.temperature

        # 对角线是正样本
        labels = torch.arange(batch_size, device=user_vectors.device)

        # 交叉熵损失
        loss = torch.nn.functional.cross_entropy(similarity, labels)

        return loss

### 3.2 损失函数

```python
class TwoTowerLoss(nn.Module):
    """双塔模型损失函数"""

    def __init__(self, loss_type='bpr', temperature=0.1):
        """
        参数:
            loss_type: 损失类型 ('bpr', 'softmax', 'contrastive')
            temperature: 温度参数
        """
        super().__init__()
        self.loss_type = loss_type
        self.temperature = temperature

    def forward(self, pos_scores, neg_scores=None, user_vectors=None, item_vectors=None):
        """计算损失"""
        if self.loss_type == 'bpr':
            return self._bpr_loss(pos_scores, neg_scores)
        elif self.loss_type == 'softmax':
            return self._softmax_loss(pos_scores, neg_scores)
        elif self.loss_type == 'contrastive':
            return self._contrastive_loss(user_vectors, item_vectors)

    def _bpr_loss(self, pos_scores, neg_scores):
        """BPR 损失"""
        # pos_scores: (batch,)
        # neg_scores: (batch, num_neg)

        pos_scores = pos_scores.unsqueeze(-1)  # (batch, 1)
        diff = pos_scores - neg_scores  # (batch, num_neg)

        # BPR: -log(sigmoid(pos - neg))
        loss = -torch.log(torch.sigmoid(diff) + 1e-10).mean()

        return loss

    def _softmax_loss(self, pos_scores, neg_scores):
        """Softmax 损失"""
        # 拼接正负样本分数
        all_scores = torch.cat([
            pos_scores.unsqueeze(-1),
            neg_scores
        ], dim=-1)  # (batch, 1 + num_neg)

        # 正样本在第 0 位
        labels = torch.zeros(all_scores.size(0), dtype=torch.long, device=all_scores.device)

        loss = torch.nn.functional.cross_entropy(all_scores / self.temperature, labels)

        return loss

    def _contrastive_loss(self, user_vectors, item_vectors):
        """对比学习损失（InfoNCE）"""
        batch_size = user_vectors.size(0)

        # 相似度矩阵
        similarity = torch.mm(user_vectors, item_vectors.t()) / self.temperature

        # 对角线是正样本
        labels = torch.arange(batch_size, device=user_vectors.device)

        loss = torch.nn.functional.cross_entropy(similarity, labels)

        return loss
```

## 4. YouTube DNN

YouTube DNN 是 Google 提出的双塔召回模型。

```python
class YouTubeDNN(nn.Module):
    """
    YouTube DNN 召回模型

    特点：
    1. 用户塔使用历史行为序列
    2. 物品塔只有 Embedding
    3. 负采样训练
    """

    def __init__(self, num_users, num_items, embed_dim=256,
                 item_features_dim=0, hidden_units=[1024, 512, 256]):
        super().__init__()

        self.num_items = num_items
        self.embed_dim = embed_dim

        # 物品 Embedding
        self.item_embedding = nn.Embedding(num_items, embed_dim)

        # 用户塔
        user_input_dim = embed_dim + item_features_dim  # 历史行为 + 用户特征

        layers = []
        input_dim = user_input_dim
        for hidden in hidden_units:
            layers.append(nn.Linear(input_dim, hidden))
            layers.append(nn.ReLU())
            layers.append(nn.BatchNorm1d(hidden))
            input_dim = hidden

        # 最后输出 embed_dim 维向量
        layers.append(nn.Linear(input_dim, embed_dim))
        self.user_mlp = nn.Sequential(*layers)

    def encode_user(self, user_history_items, user_features=None):
        """
        编码用户

        参数:
            user_history_items: (batch, seq_len) 用户历史物品 ID
            user_features: (batch, feature_dim) 用户特征（可选）
        """
        # 历史物品 Embedding
        history_emb = self.item_embedding(user_history_items)  # (batch, seq_len, embed_dim)

        # 平均池化
        user_vec = history_emb.mean(dim=1)  # (batch, embed_dim)

        # 拼接用户特征
        if user_features is not None:
            user_vec = torch.cat([user_vec, user_features], dim=-1)

        # MLP
        user_vec = self.user_mlp(user_vec)

        # L2 归一化
        user_vec = torch.nn.functional.normalize(user_vec, p=2, dim=-1)

        return user_vec

    def encode_item(self, item_ids):
        """编码物品"""
        item_vec = self.item_embedding(item_ids)
        item_vec = torch.nn.functional.normalize(item_vec, p=2, dim=-1)
        return item_vec

    def forward(self, user_history_items, pos_item_ids, neg_item_ids=None):
        """
        前向传播

        参数:
            user_history_items: (batch, seq_len) 用户历史
            pos_item_ids: (batch,) 正样本物品
            neg_item_ids: (batch, num_neg) 负样本物品
        """
        # 用户向量
        user_vec = self.encode_user(user_history_items)

        # 正样本向量
        pos_item_vec = self.encode_item(pos_item_ids)

        # 正样本分数
        pos_scores = torch.sum(user_vec * pos_item_vec, dim=-1)

        if neg_item_ids is not None:
            # 负样本向量
            batch_size, num_neg = neg_item_ids.shape
            neg_item_vec = self.encode_item(neg_item_ids.view(-1))
            neg_item_vec = neg_item_vec.view(batch_size, num_neg, -1)

            # 负样本分数
            neg_scores = torch.bmm(
                neg_item_vec,
                user_vec.unsqueeze(-1)
            ).squeeze(-1)

            return pos_scores, neg_scores

        return pos_scores

    def get_all_item_vectors(self):
        """获取所有物品向量（用于建立索引）"""
        all_items = torch.arange(self.num_items)
        return self.encode_item(all_items).detach().numpy()
```

### 5.1 Faiss 索引

```python
import faiss
import numpy as np

class VectorIndex:
    """向量索引"""

    def __init__(self, dimension: int, index_type: str = 'ivf'):
        """
        参数:
            dimension: 向量维度
            index_type: 索引类型 ('flat', 'ivf', 'hnsw')
        """
        self.dimension = dimension
        self.index_type = index_type
        self.index = None

    def build(self, vectors: np.ndarray, nlist: int = 100):
        """
        构建索引

        参数:
            vectors: (num_vectors, dimension) 向量矩阵
            nlist: IVF 聚类中心数量
        """
        num_vectors = vectors.shape[0]

        # 归一化
        faiss.normalize_L2(vectors)

        if self.index_type == 'flat':
            # 暴力搜索，精度最高
            self.index = faiss.IndexFlatIP(self.dimension)

        elif self.index_type == 'ivf':
            # IVF 索引，速度和精度的平衡
            quantizer = faiss.IndexFlatIP(self.dimension)
            self.index = faiss.IndexIVFFlat(
                quantizer, self.dimension, nlist
            )
            self.index.train(vectors)

        elif self.index_type == 'hnsw':
            # HNSW 索引，速度最快
            self.index = faiss.IndexHNSWFlat(self.dimension, 32)

        # 添加向量
        self.index.add(vectors)

        print(f"索引构建完成，共 {self.index.ntotal} 个向量")

    def search(self, query: np.ndarray, top_k: int = 10) -> tuple:
        """
        搜索

        参数:
            query: (batch, dimension) 查询向量
            top_k: 返回数量

        返回:
            scores: (batch, top_k) 相似度
            indices: (batch, top_k) 索引
        """
        # 归一化
        faiss.normalize_L2(query)

        # 搜索
        scores, indices = self.index.search(query, top_k)

        return scores, indices

    def save(self, path: str):
        """保存索引"""
        faiss.write_index(self.index, path)

    def load(self, path: str):
        """加载索引"""
        self.index = faiss.read_index(path)


class RecommendRetriever:
    """推荐检索器"""

    def __init__(self, model, vector_index):
        self.model = model
        self.vector_index = vector_index

    def retrieve(self, user_features, top_k: int = 100):
        """
        召回

        参数:
            user_features: 用户特征
            top_k: 返回数量
        """
        # 获取用户向量
        with torch.no_grad():
            user_vector = self.model.get_user_vector(user_features)
            user_vector = user_vector.cpu().numpy()

        # 向量检索
        scores, indices = self.vector_index.search(user_vector, top_k)

        return indices[0], scores[0]
```

import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader
import numpy as np

class TwoTowerDataset(Dataset):
    """双塔模型数据集"""

    def __init__(self, interactions, num_items, num_neg=4):
        """
        参数:
            interactions: list of (user_id, item_id)
            num_items: 物品总数
            num_neg: 负样本数量
        """
        self.interactions = interactions
        self.num_items = num_items
        self.num_neg = num_neg

    def __len__(self):
        return len(self.interactions)

    def __getitem__(self, idx):
        user_id, pos_item_id = self.interactions[idx]

        # 负采样
        neg_item_ids = []
        while len(neg_item_ids) < self.num_neg:
            neg_id = np.random.randint(0, self.num_items)
            if neg_id != pos_item_id:
                neg_item_ids.append(neg_id)

        return {
            'user_id': user_id,
            'pos_item_id': pos_item_id,
            'neg_item_ids': np.array(neg_item_ids)
        }

        model.train()
        total_loss = 0

        for batch in dataloader:
            user_ids = batch['user_id'].to(device)
            pos_item_ids = batch['pos_item_id'].to(device)
            neg_item_ids = batch['neg_item_ids'].to(device)

            user_features = {'user_id': user_ids}
            item_features = {'item_id': pos_item_ids}

            # 前向传播
            pos_scores = model(user_features, item_features)

            # 负样本分数
            neg_scores_list = []
            for i in range(config['num_neg']):
                neg_features = {'item_id': neg_item_ids[:, i]}
                neg_score = model(user_features, neg_features)
                neg_scores_list.append(neg_score)

            neg_scores = torch.stack(neg_scores_list, dim=1)

            # 计算损失
            loss = criterion(pos_scores, neg_scores)

            # 反向传播
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            total_loss += loss.item()

        avg_loss = total_loss / len(dataloader)
        print(f"Epoch {epoch+1}/{config['epochs']}, Loss: {avg_loss:.4f}")

    return model


if __name__ == "__main__":
    model = train_two_tower()
    print("训练完成！")
