# 排序学习 (Learning to Rank) 学习文档

## 1. 算法基础认知

### 1.1 什么是 Learning to Rank？

Learning to Rank (LTR) 是将机器学习应用于排序问题的方法，目标是学习一个排序函数，使得对查询返回的结果按相关性排序。

### 1.2 三种主要方法

| 方法 | 描述 | 代表算法 |
|------|------|----------|
| Pointwise | 单文档打分 | Regression, Classification |
| Pairwise | 文档对比较 | RankNet, LambdaRank |
| Listwise | 整体列表优化 | ListNet, LambdaMART |

### 1.3 推荐系统中的排序

```
召回 → 粗排 → 精排 → 重排
         ↑      ↑      ↑
      LTR   LTR   LTR/Diversity
```

## 2. Pointwise 方法

### 2.1 原理

将排序问题转化为回归或分类问题：

- **回归**: 预测相关性分数
- **分类**: 预测相关/不相关

### 2.2 实现

```python
import numpy as np
import torch
import torch.nn as nn
from typing import List, Dict, Tuple


class PointwiseLTRModel(nn.Module):
    """
    Pointwise 排序模型
    """

    def __init__(self, input_dim: int, hidden_dims: List[int] = [256, 128]):
        super().__init__()

        layers = []
        prev_dim = input_dim

        for hidden_dim in hidden_dims:
            layers.extend([
                nn.Linear(prev_dim, hidden_dim),
                nn.ReLU(),
                nn.Dropout(0.2)
            ])
            prev_dim = hidden_dim

        layers.append(nn.Linear(prev_dim, 1))

        self.mlp = nn.Sequential(*layers)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        参数:
            x: (batch, input_dim) 特征

        返回:
            (batch, 1) 分数
        """
        return self.mlp(x)


class PointwiseLoss(nn.Module):
    """
    Pointwise 损失函数
    """

    def __init__(self, loss_type: str = 'mse'):
        super().__init__()
        self.loss_type = loss_type

        if loss_type == 'mse':
            self.loss_fn = nn.MSELoss()
        elif loss_type == 'bce':
            self.loss_fn = nn.BCEWithLogitsLoss()
        elif loss_type == 'cross_entropy':
            self.loss_fn = nn.CrossEntropyLoss()

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        return self.loss_fn(pred.squeeze(), target.float())


def train_pointwise(model, dataloader, optimizer, loss_fn, epochs: int = 10):
    """训练 Pointwise 模型"""
    model.train()

    for epoch in range(epochs):
        total_loss = 0

        for features, labels in dataloader:
            optimizer.zero_grad()

            pred = model(features)
            loss = loss_fn(pred, labels)

            loss.backward()
            optimizer.step()

            total_loss += loss.item()

        print(f"Epoch {epoch+1}, Loss: {total_loss / len(dataloader):.4f}")


# 示例：使用回归预测评分
class RatingPredictionModel(PointwiseLTRModel):
    """
    评分预测模型（Pointwise）
    """

    def __init__(self, input_dim: int):
        super().__init__(input_dim, [256, 128, 64])

    def predict_rating(self, x: torch.Tensor) -> torch.Tensor:
        """预测评分 (1-5)"""
        score = self.forward(x)
        return torch.clamp(score, 1, 5)
```

## 3. Pairwise 方法

### 3.1 原理

比较文档对的相对顺序，而非绝对分数。

```
对于查询 q，文档 i 比 j 更相关:
- 期望: score(i) > score(j)
- 损失: 惩罚 score(i) < score(j) 的情况
```

### 3.2 RankNet

```python
class RankNet(nn.Module):
    """
    RankNet: Pairwise 排序模型

    使用概率模型比较两个文档
    """

    def __init__(self, input_dim: int, hidden_dim: int = 128):
        super().__init__()

        self.fc = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim, 1)
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """预测单个文档的分数"""
        return self.fc(x)

    def predict_pair(self, x_i: torch.Tensor, x_j: torch.Tensor) -> torch.Tensor:
        """
        预测文档 i 比 j 更好的概率

        P(i > j) = sigmoid(f(x_i) - f(x_j))
        """
        s_i = self.forward(x_i)
        s_j = self.forward(x_j)

        return torch.sigmoid(s_i - s_j)


class PairwiseLoss(nn.Module):
    """
    Pairwise 损失函数
    """

    def __init__(self, sigma: float = 1.0):
        super().__init__()
        self.sigma = sigma

    def forward(self, pred_i: torch.Tensor, pred_j: torch.Tensor,
                target: torch.Tensor) -> torch.Tensor:
        """
        参数:
            pred_i: 文档 i 的预测分数
            pred_j: 文档 j 的预测分数
            target: 1 表示 i > j, -1 表示 i < j, 0 表示相等

        返回:
            损失值
        """
        # 预测的排序概率
        pred_diff = pred_i - pred_j
        pred_prob = torch.sigmoid(self.sigma * pred_diff)

        # 目标概率
        target_prob = (target + 1) / 2  # [-1, 0, 1] -> [0, 0.5, 1]

        # 交叉熵损失
        loss = -target_prob * torch.log(pred_prob + 1e-10) - \
               (1 - target_prob) * torch.log(1 - pred_prob + 1e-10)

        return loss.mean()


class RankNetTrainer:
    """
    RankNet 训练器
    """

    def __init__(self, model: RankNet, learning_rate: float = 0.001):
        self.model = model
        self.optimizer = torch.optim.Adam(model.parameters(), lr=learning_rate)
        self.loss_fn = PairwiseLoss()

    def generate_pairs(self, features: torch.Tensor, labels: torch.Tensor,
                      num_pairs: int = 10) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """
        生成文档对

        参数:
            features: (batch, input_dim)
            labels: (batch,) 相关性标签
            num_pairs: 生成的对数

        返回:
            features_i, features_j, target
        """
        batch_size = features.size(0)

        features_i_list = []
        features_j_list = []
        targets_list = []

        for _ in range(num_pairs):
            # 随机选择两个文档
            idx_i = torch.randint(0, batch_size, (1,))
            idx_j = torch.randint(0, batch_size, (1,))

            feat_i = features[idx_i]
            feat_j = features[idx_j]

            label_i = labels[idx_i]
            label_j = labels[idx_j]

            # 确定目标
            if label_i > label_j:
                target = 1.0
            elif label_i < label_j:
                target = -1.0
            else:
                target = 0.0

            features_i_list.append(feat_i)
            features_j_list.append(feat_j)
            targets_list.append(target)

        return (
            torch.cat(features_i_list),
            torch.cat(features_j_list),
            torch.tensor(targets_list)
        )

    def train_step(self, features: torch.Tensor, labels: torch.Tensor) -> float:
        """训练一步"""
        self.model.train()
        self.optimizer.zero_grad()

        # 生成文档对
        feat_i, feat_j, targets = self.generate_pairs(features, labels)

        # 预测
        pred_i = self.model(feat_i)
        pred_j = self.model(feat_j)

        # 计算损失
        loss = self.loss_fn(pred_i, pred_j, targets)

        loss.backward()
        self.optimizer.step()

        return loss.item()


def demo_ranknet():
    """RankNet 示例"""
    input_dim = 20
    batch_size = 64

    # 创建模型
    model = RankNet(input_dim)
    trainer = RankNetTrainer(model)

    # 模拟数据
    features = torch.randn(batch_size, input_dim)
    labels = torch.randint(0, 5, (batch_size,)).float()

    # 训练
    for epoch in range(10):
        loss = trainer.train_step(features, labels)
        print(f"Epoch {epoch+1}, Loss: {loss:.4f}")


if __name__ == "__main__":
    demo_ranknet()
```

### 3.3 LambdaRank

```python
class LambdaRankLoss(nn.Module):
    """
    LambdaRank 损失函数

    在 RankNet 基础上加入 NDCG 增益作为权重
    """

    def __init__(self, sigma: float = 1.0):
        super().__init__()
        self.sigma = sigma

    def compute_lambda(self, pred_i: torch.Tensor, pred_j: torch.Tensor,
                       label_i: torch.Tensor, label_j: torch.Tensor,
                       ndcg_gain: torch.Tensor) -> torch.Tensor:
        """
        计算 Lambda 梯度

        λ_ij = |ΔNDCG| * σ * (1 - σ * (s_i - s_j))

        其中 |ΔNDCG| 是交换 i 和 j 位置导致的 NDCG 变化
        """
        # 基础梯度
        pred_diff = pred_i - pred_j
        sigmoid = torch.sigmoid(self.sigma * pred_diff)

        # 目标
        S_ij = torch.sign(label_i - label_j)

        # Lambda
        lambda_ij = self.sigma * (0.5 * (1 - S_ij) - sigmoid)

        # 加上 NDCG 增益
        lambda_ij = lambda_ij * ndcg_gain

        return lambda_ij

    def forward(self, pred_i: torch.Tensor, pred_j: torch.Tensor,
                label_i: torch.Tensor, label_j: torch.Tensor,
                ndcg_gain: torch.Tensor) -> torch.Tensor:
        """计算损失"""
        lambda_ij = self.compute_lambda(pred_i, pred_j, label_i, label_j, ndcg_gain)

        # 加权损失
        pred_diff = pred_i - pred_j
        loss = lambda_ij * pred_diff

        return loss.abs().mean()
```

## 4. Listwise 方法

### 4.1 ListNet

```python
class ListNetLoss(nn.Module):
    """
    ListNet 损失函数

    使用 Softmax 将分数转换为概率分布，然后用交叉熵
    """

    def __init__(self, temperature: float = 1.0):
        super().__init__()
        self.temperature = temperature

    def forward(self, pred: torch.Tensor, target: torch.Tensor,
                mask: torch.Tensor = None) -> torch.Tensor:
        """
        参数:
            pred: (batch, list_size) 预测分数
            target: (batch, list_size) 真实相关性
            mask: (batch, list_size) 有效位置掩码

        返回:
            损失值
        """
        # Softmax 概率
        pred_prob = torch.softmax(pred / self.temperature, dim=-1)
        target_prob = torch.softmax(target / self.temperature, dim=-1)

        # 交叉熵
        loss = -torch.sum(target_prob * torch.log(pred_prob + 1e-10), dim=-1)

        if mask is not None:
            loss = loss * mask.sum(dim=-1) / mask.size(-1)

        return loss.mean()


class ListNet(nn.Module):
    """
    ListNet 排序模型
    """

    def __init__(self, input_dim: int, hidden_dim: int = 128):
        super().__init__()

        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        self.loss_fn = ListNetLoss()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.encoder(x).squeeze(-1)

    def compute_loss(self, features: torch.Tensor, labels: torch.Tensor,
                     list_indices: torch.Tensor) -> torch.Tensor:
        """
        计算 ListNet 损失

        参数:
            features: (total_docs, input_dim)
            labels: (total_docs,)
            list_indices: (total_docs,) 每个文档所属的列表ID
        """
        # 预测
        scores = self.forward(features)

        # 按列表分组计算损失
        unique_lists = list_indices.unique()
        total_loss = 0

        for list_id in unique_lists:
            mask = (list_indices == list_id)
            list_scores = scores[mask]
            list_labels = labels[mask]

            list_loss = self.loss_fn(
                list_scores.unsqueeze(0),
                list_labels.unsqueeze(0).float()
            )
            total_loss += list_loss

        return total_loss / len(unique_lists)
```

### 4.2 ListMLE

```python
class ListMLELoss(nn.Module):
    """
    ListMLE 损失函数

    最大化正确排序的似然概率
    """

    def __init__(self, temperature: float = 1.0):
        super().__init__()
        self.temperature = temperature

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        """
        参数:
            pred: (batch, list_size) 预测分数
            target: (batch, list_size) 真实相关性（用于确定正确排序）

        返回:
            负对数似然
        """
        batch_size, list_size = pred.shape

        # 获取正确排序的索引
        sorted_indices = torch.argsort(target, dim=-1, descending=True)

        # 计算似然
        log_likelihood = torch.zeros(batch_size, device=pred.device)

        for k in range(list_size):
            # 获取第 k 个位置应该排的文档
            k_th_doc = sorted_indices[:, k]

            # 计算从剩余文档中选中的概率
            remaining_scores = pred[:, k:]  # 简化：假设按顺序
            log_prob = pred[torch.arange(batch_size), k_th_doc] - \
                      torch.logsumexp(remaining_scores, dim=-1)

            log_likelihood += log_prob

        return -log_likelihood.mean()
```

## 5. 排序评估指标

### 5.1 实现

```python
from typing import List, Dict
import numpy as np


class RankingMetrics:
    """
    排序指标计算器
    """

    @staticmethod
    def ndcg_at_k(predictions: List[float], labels: List[float],
                  k: int = 10) -> float:
        """
        计算 NDCG@K
        """
        # 按预测分数排序
        sorted_indices = np.argsort(predictions)[::-1][:k]
        sorted_labels = np.array(labels)[sorted_indices]

        # DCG
        dcg = sum(
            (2 ** label - 1) / np.log2(i + 2)
            for i, label in enumerate(sorted_labels)
        )

        # IDCG
        ideal_labels = sorted(labels, reverse=True)[:k]
        idcg = sum(
            (2 ** label - 1) / np.log2(i + 2)
            for i, label in enumerate(ideal_labels)
        )

        return dcg / idcg if idcg > 0 else 0

    @staticmethod
    def mrr(predictions: List[float], labels: List[float]) -> float:
        """
        计算 MRR (Mean Reciprocal Rank)
        """
        sorted_indices = np.argsort(predictions)[::-1]

        for rank, idx in enumerate(sorted_indices, 1):
            if labels[idx] > 0:  # 相关
                return 1.0 / rank

        return 0

    @staticmethod
    def precision_at_k(predictions: List[float], labels: List[float],
                       k: int = 10) -> float:
        """
        计算 Precision@K
        """
        sorted_indices = np.argsort(predictions)[::-1][:k]
        top_k_labels = np.array(labels)[sorted_indices]

        return np.sum(top_k_labels > 0) / k

    @staticmethod
    def map_score(all_predictions: List[List[float]],
                  all_labels: List[List[float]]) -> float:
        """
        计算 MAP (Mean Average Precision)
        """
        aps = []

        for predictions, labels in zip(all_predictions, all_labels):
            sorted_indices = np.argsort(predictions)[::-1]

            num_relevant = 0
            precision_sum = 0

            for i, idx in enumerate(sorted_indices, 1):
                if labels[idx] > 0:
                    num_relevant += 1
                    precision_sum += num_relevant / i

            total_relevant = np.sum(np.array(labels) > 0)
            if total_relevant > 0:
                aps.append(precision_sum / total_relevant)

        return np.mean(aps) if aps else 0


def evaluate_ranking(model: nn.Module, test_data: List[Tuple],
                    k_list: List[int] = [1, 5, 10]) -> Dict[str, float]:
    """
    评估排序模型

    参数:
        model: 排序模型
        test_data: 测试数据 [(query_features, labels), ...]
        k_list: 评估的 K 值列表

    返回:
        指标字典
    """
    model.eval()
    metrics = {f'ndcg@{k}': [] for k in k_list}
    metrics['mrr'] = []
    metrics['map'] = []

    all_predictions = []
    all_labels = []

    with torch.no_grad():
        for features, labels in test_data:
            features = torch.tensor(features, dtype=torch.float32)
            predictions = model(features).squeeze().numpy()

            all_predictions.append(predictions.tolist())
            all_labels.append(labels)

            for k in k_list:
                ndcg = RankingMetrics.ndcg_at_k(predictions, labels, k)
                metrics[f'ndcg@{k}'].append(ndcg)

            mrr = RankingMetrics.mrr(predictions, labels)
            metrics['mrr'].append(mrr)

    # 计算 MAP
    metrics['map'] = RankingMetrics.map_score(all_predictions, all_labels)

    # 平均
    for key in metrics:
        if key != 'map':
            metrics[key] = np.mean(metrics[key])

    return metrics
```

## 6. 方法对比

| 方法 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| Pointwise | 简单、高效 | 忽略文档间关系 | 评分预测 |
| Pairwise | 考虑相对顺序 | 训练不稳定 | 精排 |
| Listwise | 优化整体排序 | 计算复杂 | 高精度排序 |

## 7. 学习总结

### 7.1 核心要点

1. **Pointwise**: 单文档打分
2. **Pairwise**: 文档对比较
3. **Listwise**: 整体列表优化

### 7.2 选择建议

```
精排场景: Listwise (LambdaMART, ListNet)
粗排场景: Pointwise (高效)
重排场景: Pairwise + Listwise 混合
```
