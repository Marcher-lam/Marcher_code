# 重排策略 学习文档

## 1. 基础认知

### 1.1 什么是重排？

重排（Re-ranking）是在精排之后对推荐结果进行**二次排序**的过程。它考虑的业务目标不只是 CTR，还包括多样性、新颖性、公平性等。

### 1.2 为什么需要重排？

精排模型通常只优化单一目标（如 CTR），但实际业务需要：
- **多样性**：避免推荐结果过于单一
- **新颖性**：引入新鲜内容
- **公平性**：给长尾物品机会
- **业务约束**：如曝光控制、打散规则

### 1.3 重排 vs 精排

| 维度 | 精排 | 重排 |
|------|------|------|
| 输入 | 候选集（几百） | 精排结果（几十） |
| 目标 | CTR/CVR | 多目标平衡 |
| 复杂度 | 高 | 低 |
| 实时性 | 要求高 | 要求更高 |

## 2. 常见重排策略

### 2.1 多样性重排

```python
import numpy as np
from typing import List, Dict, Tuple
from collections import defaultdict


class MMRDiversity:
    """
    MMR (Maximal Marginal Relevance) 多样性重排

    平衡相关性和多样性

    公式: MMR = λ * Relevance(i) - (1-λ) * max(Sim(i, j)) for j in S
    """

    def __init__(self, lambda_param=0.7):
        """
        参数:
            lambda_param: 相关性权重 (0-1)
        """
        self.lambda_param = lambda_param

    def rerank(self, items: List[Dict], similarity_matrix: np.ndarray,
               relevance_scores: np.ndarray, top_k: int = 20) -> List[Dict]:
        """
        MMR 重排

        参数:
            items: 物品列表 [{'id': x, ...}, ...]
            similarity_matrix: (n_items, n_items) 物品相似度矩阵
            relevance_scores: (n_items,) 相关性分数
            top_k: 返回数量

        返回:
            重排后的物品列表
        """
        n_items = len(items)
        selected = []
        selected_indices = []
        remaining = list(range(n_items))

        while len(selected) < top_k and remaining:
            mmr_scores = []

            for idx in remaining:
                # 相关性部分
                relevance = relevance_scores[idx]

                # 多样性部分
                if selected_indices:
                    max_sim = max(similarity_matrix[idx, j] for j in selected_indices)
                else:
                    max_sim = 0

                # MMR 分数
                mmr = self.lambda_param * relevance - (1 - self.lambda_param) * max_sim
                mmr_scores.append((idx, mmr))

            # 选择 MMR 最大的
            best_idx, _ = max(mmr_scores, key=lambda x: x[1])
            selected.append(items[best_idx])
            selected_indices.append(best_idx)
            remaining.remove(best_idx)

        return selected

    def rerank_with_embeddings(self, items: List[Dict], embeddings: np.ndarray,
                               relevance_scores: np.ndarray, top_k: int = 20) -> List[Dict]:
        """
        使用嵌入计算相似度的 MMR 重排

        参数:
            items: 物品列表
            embeddings: (n_items, dim) 物品嵌入
            relevance_scores: 相关性分数
            top_k: 返回数量
        """
        from sklearn.metrics.pairwise import cosine_similarity

        # 计算相似度矩阵
        similarity_matrix = cosine_similarity(embeddings)

        return self.rerank(items, similarity_matrix, relevance_scores, top_k)


class DeterminantalPointProcess:
    """
    DPP (Determinantal Point Process) 多样性重排

    使用行列式来度量子集的多样性
    """

    def __init__(self, alpha=0.5):
        """
        参数:
            alpha: 多样性权重
        """
        self.alpha = alpha

    def rerank(self, items: List[Dict], embeddings: np.ndarray,
               relevance_scores: np.ndarray, top_k: int = 20) -> List[Dict]:
        """
        DPP 重排

        参数:
            items: 物品列表
            embeddings: 物品嵌入
            relevance_scores: 相关性分数
            top_k: 返回数量

        返回:
            重排后的物品列表
        """
        n_items = len(items)

        # 构建核矩阵
        # L = diag(q) * S * diag(q)
        # q: 相关性分数, S: 相似度矩阵
        q = np.sqrt(relevance_scores)
        S = self._compute_similarity(embeddings)
        L = np.diag(q) @ S @ np.diag(q)

        # 贪婪选择
        selected = []
        selected_indices = []
        remaining = list(range(n_items))

        while len(selected) < top_k and remaining:
            best_idx = None
            best_det = -np.inf

            for idx in remaining:
                # 计算添加 idx 后的行列式增量
                if not selected_indices:
                    det_gain = L[idx, idx]
                else:
                    # 使用 Schur 补
                    Y = L[np.ix_(selected_indices, selected_indices)]
                    C = L[selected_indices, idx]
                    D = L[idx, idx]

                    det_Y = np.linalg.det(Y)
                    if det_Y > 0:
                        schur = D - C @ np.linalg.solve(Y, C)
                        det_gain = det_Y * schur
                    else:
                        det_gain = D

                if det_gain > best_det:
                    best_det = det_gain
                    best_idx = idx

            if best_idx is not None:
                selected.append(items[best_idx])
                selected_indices.append(best_idx)
                remaining.remove(best_idx)

        return selected

    def _compute_similarity(self, embeddings):
        """计算相似度矩阵"""
        from sklearn.metrics.pairwise import cosine_similarity
        S = cosine_similarity(embeddings)
        return self.alpha * S + (1 - self.alpha) * np.eye(len(embeddings))


class CategoryDiversity:
    """
    基于类目的多样性重排

    确保推荐结果类目分布均匀
    """

    def __init__(self, max_same_category=2, window_size=5):
        """
        参数:
            max_same_category: 窗口内同类目最大数量
            window_size: 滑动窗口大小
        """
        self.max_same_category = max_same_category
        self.window_size = window_size

    def rerank(self, items: List[Dict], category_key='category') -> List[Dict]:
        """
        类目多样性重排

        参数:
            items: 物品列表，每个物品包含 category 字段
            category_key: 类目字段名

        返回:
            重排后的物品列表
        """
        # 按原始分数排序
        sorted_items = sorted(items, key=lambda x: x.get('score', 0), reverse=True)

        result = []
        remaining = sorted_items.copy()

        while remaining:
            # 检查窗口内的类目分布
            window = result[-self.window_size:] if len(result) >= self.window_size else result
            category_count = defaultdict(int)
            for item in window:
                category_count[item.get(category_key, 'unknown')] += 1

            # 找到可以添加的物品
            added = False
            for i, item in enumerate(remaining):
                cat = item.get(category_key, 'unknown')
                if category_count[cat] < self.max_same_category:
                    result.append(item)
                    remaining.pop(i)
                    added = True
                    break

            # 如果找不到合适的，添加第一个
            if not added and remaining:
                result.append(remaining.pop(0))

        return result
```

### 2.2 打散策略

```python
class打散策略:
    """
    打散策略

    防止同类物品连续出现
    """

    def __init__(self, min_distance=3, attribute='source'):
        """
        参数:
            min_distance: 同属性物品的最小间隔
            attribute: 要打散的属性
        """
        self.min_distance = min_distance
        self.attribute = attribute

    def rerank(self, items: List[Dict]) -> List[Dict]:
        """
        打散重排

        参数:
            items: 物品列表

        返回:
            重排后的物品列表
        """
        # 按分数排序
        sorted_items = sorted(items, key=lambda x: x.get('score', 0), reverse=True)

        result = [None] * len(sorted_items)
        remaining = sorted_items.copy()

        # 按属性分组
        attr_items = defaultdict(list)
        for item in remaining:
            attr = item.get(self.attribute, 'unknown')
            attr_items[attr].append(item)

        # 按分数排序每个组
        for attr in attr_items:
            attr_items[attr].sort(key=lambda x: x.get('score', 0), reverse=True)

        # 贪婪填充
        pos = 0
        while remaining:
            filled = False

            for attr, items_list in sorted(attr_items.items(),
                                          key=lambda x: -x[1][0].get('score', 0) if x[1] else 0):
                if not items_list:
                    continue

                # 检查间隔
                can_place = True
                for i in range(max(0, pos - self.min_distance + 1), pos):
                    if result[i] and result[i].get(self.attribute) == attr:
                        can_place = False
                        break

                if can_place:
                    item = items_list.pop(0)
                    result[pos] = item
                    remaining.remove(item)
                    pos += 1
                    filled = True
                    break

            if not filled and remaining:
                # 强制放置
                item = remaining.pop(0)
                result[pos] = item
                pos += 1

        return [r for r in result if r is not None]


class ExposureControl:
    """
    曝光控制

    控制物品的曝光频率
    """

    def __init__(self, max_exposure_per_session=2, cooldown_periods=5):
        """
        参数:
            max_exposure_per_session: 单次会话最大曝光次数
            cooldown_periods: 冷却期（请求次数）
        """
        self.max_exposure = max_exposure_per_session
        self.cooldown = cooldown_periods
        self.exposure_history = defaultdict(list)  # {item_id: [request_ids]}

    def rerank(self, items: List[Dict], request_id: str) -> List[Dict]:
        """
        曝光控制重排

        参数:
            items: 物品列表
            request_id: 当前请求 ID

        返回:
            重排后的物品列表
        """
        result = []
        exposure_count = defaultdict(int)

        for item in items:
            item_id = item.get('id')

            # 检查本次会话曝光次数
            if exposure_count[item_id] >= self.max_exposure:
                continue

            # 检查冷却期
            history = self.exposure_history[item_id]
            recent_exposures = [h for h in history if h > int(request_id) - self.cooldown]

            if len(recent_exposures) < self.max_exposure:
                result.append(item)
                exposure_count[item_id] += 1
                self.exposure_history[item_id].append(int(request_id))

        return result


class PositionBiasCorrection:
    """
    位置偏差校正

    校正用户对不同位置的注意力差异
    """

    def __init__(self, position_weights=None):
        """
        参数:
            position_weights: 位置权重 [w_1, w_2, ...]
        """
        if position_weights is None:
            # 默认指数衰减
            self.position_weights = [1.0 / (1 + 0.1 * i) for i in range(100)]
        else:
            self.position_weights = position_weights

    def rerank(self, items: List[Dict], top_k: int = 20) -> List[Dict]:
        """
        位置偏差校正重排

        将高价值物品放到更容易被注意到的位置
        """
        # 计算校正后的期望收益
        for i, item in enumerate(items):
            original_score = item.get('score', 0)
            # 期望收益 = 分数 * 位置权重
            item['expected_gain'] = original_score * self.position_weights[i]

        # 按期望收益排序
        sorted_items = sorted(items, key=lambda x: x.get('expected_gain', 0), reverse=True)

        return sorted_items[:top_k]
```

### 2.3 多目标重排

```python
class MultiObjectiveReranker:
    """
    多目标重排器

    平衡多个业务目标
    """

    def __init__(self, objectives: Dict[str, float], max_iter=100):
        """
        参数:
            objectives: {目标名: 权重}
                       如 {'ctr': 0.5, 'diversity': 0.3, 'novelty': 0.2}
            max_iter: 最大迭代次数
        """
        self.objectives = objectives
        self.max_iter = max_iter

        # 初始化各目标评估器
        self.evaluators = {
            'ctr': self._eval_ctr,
            'diversity': self._eval_diversity,
            'novelty': self._eval_novelty,
            'freshness': self._eval_freshness,
        }

    def rerank(self, items: List[Dict], embeddings: np.ndarray = None,
               top_k: int = 20) -> List[Dict]:
        """
        多目标重排

        参数:
            items: 物品列表
            embeddings: 物品嵌入（用于多样性计算）
            top_k: 返回数量

        返回:
            重排后的物品列表
        """
        n_items = len(items)
        if n_items <= top_k:
            return items

        # 初始解：按 CTR 排序
        current = sorted(items, key=lambda x: x.get('score', 0), reverse=True)[:top_k]

        best_score = self._total_score(current, embeddings)
        best = current.copy()

        # 局部搜索
        remaining = [item for item in items if item not in current]

        for _ in range(self.max_iter):
            improved = False

            # 尝试交换
            for i in range(top_k):
                for j, candidate in enumerate(remaining):
                    # 交换
                    new_list = current.copy()
                    old_item = new_list[i]
                    new_list[i] = candidate

                    new_score = self._total_score(new_list, embeddings)

                    if new_score > best_score:
                        best_score = new_score
                        best = new_list.copy()
                        remaining[j] = old_item
                        current = new_list
                        improved = True
                        break

                if improved:
                    break

            if not improved:
                break

        return best

    def _total_score(self, items: List[Dict], embeddings: np.ndarray = None) -> float:
        """计算总分数"""
        score = 0

        for obj_name, weight in self.objectives.items():
            if obj_name in self.evaluators:
                obj_score = self.evaluators[obj_name](items, embeddings)
                score += weight * obj_score

        return score

    def _eval_ctr(self, items: List[Dict], _) -> float:
        """评估 CTR 目标"""
        if not items:
            return 0
        return np.mean([item.get('score', 0) for item in items])

    def _eval_diversity(self, items: List[Dict], embeddings: np.ndarray) -> float:
        """评估多样性目标"""
        if embeddings is None or len(items) < 2:
            return 0

        # 使用平均 pairwise 距离
        from sklearn.metrics.pairwise import cosine_distances

        indices = [item.get('idx', i) for i, item in enumerate(items)]
        sub_embeddings = embeddings[indices]

        distances = cosine_distances(sub_embeddings)
        # 取上三角
        upper_tri = distances[np.triu_indices(len(items), k=1)]

        return np.mean(upper_tri)

    def _eval_novelty(self, items: List[Dict], _) -> float:
        """评估新颖性目标"""
        if not items:
            return 0
        # 使用流行度的倒数
        novelty_scores = []
        for item in items:
            popularity = item.get('popularity', 1)
            novelty_scores.append(1.0 / (1 + popularity))

        return np.mean(novelty_scores)

    def _eval_freshness(self, items: List[Dict], _) -> float:
        """评估新鲜度目标"""
        if not items:
            return 0
        import time
        current_time = time.time()

        freshness_scores = []
        for item in items:
            publish_time = item.get('publish_time', current_time)
            age_hours = (current_time - publish_time) / 3600
            freshness = 1.0 / (1 + age_hours / 24)  # 每天衰减
            freshness_scores.append(freshness)

        return np.mean(freshness_scores)
```

## 3. 完整重排流水线

```python
class RerankPipeline:
    """
    重排流水线

    组合多个重排策略
    """

    def __init__(self, strategies: List[Tuple[str, Dict]]):
        """
        参数:
            strategies: [(策略名, 参数), ...]
        """
        self.strategies = strategies
        self._init_strategies()

    def _init_strategies(self):
        """初始化策略"""
        self.strategy_instances = []

        strategy_map = {
            'mmr': MMRDiversity,
            'dpp': DeterminantalPointProcess,
            'category_diversity': CategoryDiversity,
            'scatter': 打散策略,
            'exposure_control': ExposureControl,
            'position_correction': PositionBiasCorrection,
        }

        for name, params in self.strategies:
            if name in strategy_map:
                self.strategy_instances.append((name, strategy_map[name](**params)))

    def rerank(self, items: List[Dict], context: Dict = None) -> List[Dict]:
        """
        执行重排流水线

        参数:
            items: 精排结果
            context: 上下文信息

        返回:
            重排后的结果
        """
        context = context or {}
        result = items.copy()

        for name, strategy in self.strategy_instances:
            if name == 'mmr':
                result = strategy.rerank_with_embeddings(
                    result,
                    context.get('embeddings'),
                    context.get('scores', np.ones(len(result))),
                    context.get('top_k', 20)
                )
            elif name == 'exposure_control':
                result = strategy.rerank(result, context.get('request_id', '0'))
            else:
                result = strategy.rerank(result)

        return result

if __name__ == "__main__":

    # MMR 测试
```

## 4. 学习总结

### 4.1 核心要点

1. **多样性重排**：MMR、DPP
2. **打散策略**：防止同类聚集
3. **多目标平衡**：考虑多个业务指标

### 4.2 策略选择

| 策略 | 适用场景 |
|------|----------|
| MMR | 需要平衡相关性和多样性 |
| DPP | 需要更精确的多样性控制 |
| 类目打散 | 类目明确的场景 |
| 曝光控制 | 需要公平曝光 |
