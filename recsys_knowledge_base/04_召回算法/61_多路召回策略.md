# 推荐系统召回策略 学习文档

## 1. 召回概述

### 1.1 什么是召回？

召回（Recall）是从海量物品中快速筛选出用户可能感兴趣的候选集，是推荐系统的第一阶段。

### 1.2 召回特点

```
特点:
- 候选集大（百万级）→ 千级
- 响应时间要求高（< 50ms）
- 精度要求相对较低
- 需要高召回率

目标: 宁可错召回，不可漏召回
```

### 1.3 召回方法分类

| 类型 | 代表算法 | 特点 |
|------|----------|------|
| 协同过滤 | UserCF, ItemCF | 基于行为相似度 |
| 向量召回 | 双塔, DSSM | 基于嵌入相似度 |
| 图召回 | Node2Vec, GCN | 基于图结构 |
| 内容召回 | 标签匹配 | 基于物品属性 |
| 热门召回 | 热门榜单 | 基于全局统计 |
| 规则召回 | 业务规则 | 基于先验知识 |

## 2. 多路召回策略

### 2.1 多路召回架构

```python
from typing import List, Dict, Tuple
from collections import defaultdict
import numpy as np


class MultiChannelRecaller:
    """
    多路召回器
    """

    def __init__(self):
        self.recallers = {}
        self.weights = {}

    def register_recaller(self, name: str, recaller, weight: float = 1.0):
        """注册召回器"""
        self.recallers[name] = recaller
        self.weights[name] = weight

    def recall(self, user_id: int, n: int = 1000) -> List[Tuple[int, float]]:
        """
        多路召回

        参数:
            user_id: 用户ID
            n: 每路召回数量

        返回:
            [(item_id, score), ...]
        """
        all_items = defaultdict(float)

        for name, recaller in self.recallers.items():
            items = recaller.recall(user_id, n)
            weight = self.weights[name]

            for item_id, score in items:
                all_items[item_id] += score * weight

        # 排序
        sorted_items = sorted(
            all_items.items(),
            key=lambda x: x[1],
            reverse=True
        )

        return sorted_items


class RecallPipeline:
    """
    召回管道
    """

    def __init__(self):
        self.stages = []

    def add_stage(self, name: str, recaller, n: int):
        """添加召回阶段"""
        self.stages.append({
            'name': name,
            'recaller': recaller,
            'n': n
        })

    def execute(self, user_id: int, final_n: int = 500) -> List[int]:
        """执行召回管道"""
        candidates = None

        for stage in self.stages:
            recaller = stage['recaller']
            n = stage['n']

            if candidates is None:
                # 第一阶段
                candidates = [item for item, _ in recaller.recall(user_id, n)]
            else:
                # 后续阶段：在候选集上打分
                scored_items = recaller.score(user_id, candidates)
                candidates = [item for item, _ in scored_items[:n]]

        return candidates[:final_n]
```

### 2.2 各路召回实现

```python
class ItemCFRecaller:
    """
    ItemCF 召回
    """

    def __init__(self, item_sim_matrix: Dict, user_items: Dict, top_k: int = 50):
        self.item_sim_matrix = item_sim_matrix
        self.user_items = user_items
        self.top_k = top_k

    def recall(self, user_id: int, n: int = 200) -> List[Tuple[int, float]]:
        """基于物品相似度的召回"""
        if user_id not in self.user_items:
            return []

        user_items = self.user_items[user_id]
        scores = defaultdict(float)

        for item_id in user_items:
            if item_id not in self.item_sim_matrix:
                continue

            for sim_item, sim_score in self.item_sim_matrix[item_id][:self.top_k]:
                if sim_item not in user_items:
                    scores[sim_item] += sim_score

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)[:n]


class VectorRecaller:
    """
    向量召回
    """

    def __init__(self, user_embeddings: np.ndarray,
                 item_embeddings: np.ndarray,
                 index=None):
        self.user_embeddings = user_embeddings
        self.item_embeddings = item_embeddings
        self.index = index  # Faiss 索引

    def recall(self, user_id: int, n: int = 200) -> List[Tuple[int, float]]:
        """向量召回"""
        user_vec = self.user_embeddings[user_id]

        if self.index is not None:
            # 使用 Faiss
            distances, indices = self.index.search(
                user_vec.reshape(1, -1), n
            )
            return [
                (int(idx), float(1 / (1 + dist)))
                for idx, dist in zip(indices[0], distances[0])
            ]
        else:
            # 暴力搜索
            scores = np.dot(self.item_embeddings, user_vec)
            top_indices = np.argsort(scores)[::-1][:n]
            return [(int(idx), float(scores[idx])) for idx in top_indices]


class SwingRecaller:
    """
    Swing 召回

    基于用户共现关系的召回，减少热门物品影响
    """

    def __init__(self, user_items: Dict, alpha: float = 0.5):
        self.user_items = user_items
        self.alpha = alpha
        self.swing_scores = {}

    def compute_swing(self, min_users: int = 2):
        """计算 Swing 分数"""
        # 统计物品对的用户重叠
        item_pair_users = defaultdict(set)

        for user_id, items in self.user_items.items():
            items = list(items)
            for i in range(len(items)):
                for j in range(i + 1, len(items)):
                    pair = (items[i], items[j])
                    item_pair_users[pair].add(user_id)

        # 计算 Swing 分数
        for (i, j), users in item_pair_users.items():
            if len(users) >= min_users:
                # Swing = 1 / (|U| + α)
                score = 1.0 / (len(users) + self.alpha)

                if i not in self.swing_scores:
                    self.swing_scores[i] = {}
                self.swing_scores[i][j] = score

                if j not in self.swing_scores:
                    self.swing_scores[j] = {}
                self.swing_scores[j][i] = score

    def recall(self, user_id: int, n: int = 200) -> List[Tuple[int, float]]:
        """Swing 召回"""
        if user_id not in self.user_items:
            return []

        user_items = self.user_items[user_id]
        scores = defaultdict(float)

        for item_id in user_items:
            if item_id not in self.swing_scores:
                continue

            for related_item, swing_score in self.swing_scores[item_id].items():
                if related_item not in user_items:
                    scores[related_item] += swing_score

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)[:n]


class HotRecaller:
    """
    热门召回
    """

    def __init__(self, item_popularity: Dict[int, float]):
        self.popular_items = sorted(
            item_popularity.items(),
            key=lambda x: x[1],
            reverse=True
        )

    def recall(self, user_id: int, n: int = 100) -> List[Tuple[int, float]]:
        """热门召回（与用户无关）"""
        return self.popular_items[:n]


class CategoryRecaller:
    """
    品类召回

    用户历史偏好的品类下的热门物品
    """

    def __init__(self, user_category_pref: Dict,
                 category_items: Dict):
        self.user_category_pref = user_category_pref
        self.category_items = category_items

    def recall(self, user_id: int, n: int = 100) -> List[Tuple[int, float]]:
        """品类召回"""
        if user_id not in self.user_category_pref:
            return []

        prefs = self.user_category_pref[user_id]
        items = []

        for category, pref_score in prefs.items():
            if category in self.category_items:
                for item_id, pop_score in self.category_items[category]:
                    items.append((item_id, pref_score * pop_score))

        return sorted(items, key=lambda x: x[1], reverse=True)[:n]
```

## 3. 向量召回

### 3.1 双塔模型召回

```python
import torch
import torch.nn as nn


class TwoTowerModel(nn.Module):
    """
    双塔召回模型
    """

    def __init__(self, user_feature_dim: int, item_feature_dim: int,
                 hidden_dim: int = 128, embed_dim: int = 64):
        super().__init__()

        # 用户塔
        self.user_tower = nn.Sequential(
            nn.Linear(user_feature_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

        # 物品塔
        self.item_tower = nn.Sequential(
            nn.Linear(item_feature_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

    def encode_user(self, user_features: torch.Tensor) -> torch.Tensor:
        """编码用户"""
        return torch.nn.functional.normalize(
            self.user_tower(user_features), dim=-1
        )

    def encode_item(self, item_features: torch.Tensor) -> torch.Tensor:
        """编码物品"""
        return torch.nn.functional.normalize(
            self.item_tower(item_features), dim=-1
        )

    def forward(self, user_features: torch.Tensor,
               item_features: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        user_embed = self.encode_user(user_features)
        item_embed = self.encode_item(item_features)
        return torch.sum(user_embed * item_embed, dim=-1)


class TwoTowerRecaller:
    """
    双塔召回器
    """

    def __init__(self, model: TwoTowerModel, item_embeddings: np.ndarray):
        self.model = model
        self.item_embeddings = item_embeddings
        self.index = None

    def build_index(self):
        """构建 Faiss 索引"""
        import faiss

        n_items, dim = self.item_embeddings.shape
        self.index = faiss.IndexFlatIP(dim)
        self.index.add(self.item_embeddings.astype('float32'))

    def recall(self, user_features: np.ndarray, n: int = 200) -> List[Tuple[int, float]]:
        """向量召回"""
        # 编码用户
        with torch.no_grad():
            user_tensor = torch.FloatTensor(user_features)
            user_embed = self.model.encode_user(user_tensor).numpy()

        # 搜索
        scores, indices = self.index.search(
            user_embed.reshape(1, -1).astype('float32'), n
        )

        return [(int(idx), float(score)) for idx, score in zip(indices[0], scores[0])]
```

## 4. 图召回

### 4.1 随机游走召回

```python
class RandomWalkRecaller:
    """
    基于随机游走的召回
    """

    def __init__(self, user_item_graph, walk_length: int = 3,
                 n_walks: int = 10):
        self.graph = user_item_graph
        self.walk_length = walk_length
        self.n_walks = n_walks

    def random_walk(self, start_node, walk_length: int) -> List:
        """单次随机游走"""
        walk = [start_node]
        current = start_node

        for _ in range(walk_length):
            neighbors = self.graph.get(current, [])
            if not neighbors:
                break
            current = np.random.choice(neighbors)
            walk.append(current)

        return walk

    def recall(self, user_id: int, n: int = 100) -> List[Tuple[int, float]]:
        """随机游走召回"""
        item_scores = defaultdict(float)

        for _ in range(self.n_walks):
            walk = self.random_walk(user_id, self.walk_length * 2)

            # 提取路径中的物品
            for i, node in enumerate(walk):
                if isinstance(node, int) and node != user_id:
                    # 距离衰减
                    decay = 1.0 / (1 + i)
                    item_scores[node] += decay

        return sorted(item_scores.items(), key=lambda x: x[1], reverse=True)[:n]
```

## 5. 召回效果评估

```python
class RecallEvaluator:
    """
    召回评估器
    """

    @staticmethod
    def recall_at_k(recalled: List[int], ground_truth: List[int], k: int) -> float:
        """Recall@K"""
        recalled_k = set(recalled[:k])
        truth_set = set(ground_truth)
        if len(truth_set) == 0:
            return 0
        return len(recalled_k & truth_set) / len(truth_set)

    @staticmethod
    def hit_rate(recalled: List[int], ground_truth: List[int]) -> float:
        """命中率"""
        recalled_set = set(recalled)
        truth_set = set(ground_truth)
        return 1 if recalled_set & truth_set else 0

    @staticmethod
    def coverage(all_recalled: List[List[int]], total_items: int) -> float:
        """覆盖率"""
        unique_items = set()
        for items in all_recalled:
            unique_items.update(items)
        return len(unique_items) / total_items
```

## 6. 学习总结

### 6.1 核心要点

1. **多路召回**: 不同召回器互补
2. **向量召回**: 高效、可扩展
3. **图召回**: 利用高阶关系
4. **召回评估**: 关注召回率和覆盖率

### 6.2 最佳实践

```
1. 召回数量: 500-2000
2. 各路召回独立并行
3. 粗排前快速过滤
4. 定期更新索引
```
