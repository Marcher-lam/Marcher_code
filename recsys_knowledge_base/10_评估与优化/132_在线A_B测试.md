# 在线 A/B 测试 学习文档

## 1. 基础认知

### 1.1 什么是 A/B 测试？

A/B 测试是一种对照实验方法，将用户随机分成两组（或多组），比较不同策略的效果差异。

### 1.2 为什么需要在线测试？

| 离线评估 | 在线测试 |
|----------|----------|
| 历史数据 | 实时数据 |
| 代理指标 | 业务指标 |
| 无反馈循环 | 真实用户行为 |

### 1.3 推荐系统中的 A/B 测试

```
实验组: 使用新推荐算法
对照组: 使用旧推荐算法

比较指标: 点击率、转化率、用户停留时长等
```

```python
from dataclasses import dataclass
from typing import Dict, List, Optional, Callable
from enum import Enum
import numpy as np
import random
import hashlib
from datetime import datetime, timedelta


class ExperimentStatus(Enum):
    """实验状态"""
    DRAFT = "draft"
    RUNNING = "running"
    PAUSED = "paused"
    COMPLETED = "completed"


@dataclass
class ExperimentConfig:
    """实验配置"""
    experiment_id: str
    name: str
    description: str

    # 流量分配
    control_ratio: float = 0.5  # 对照组比例
    treatment_ratio: float = 0.5  # 实验组比例

    # 时间配置
    start_time: datetime = None
    end_time: datetime = None

    # 指标配置
    primary_metric: str = "ctr"  # 主要指标
    secondary_metrics: List[str] = None

    # 统计配置
    significance_level: float = 0.05  # 显著性水平
    power: float = 0.8  # 统计功效
    mde: float = 0.01  # 最小检测效应

    def __post_init__(self):
        if self.secondary_metrics is None:
            self.secondary_metrics = []


class ABExperiment:
    """
    A/B 测试实验
    """

    def __init__(self, config: ExperimentConfig):
        self.config = config
        self.status = ExperimentStatus.DRAFT
        self.metrics = {
            'control': defaultdict(list),
            'treatment': defaultdict(list)
        }

    def assign_group(self, user_id: str) -> str:
        """
        分配用户到组

        参数:
            user_id: 用户ID

        返回:
            'control' 或 'treatment'
        """
        # 使用一致性哈希确保同一用户始终分配到同一组
        hash_value = int(hashlib.md5(user_id.encode()).hexdigest(), 16)
        bucket = (hash_value % 10000) / 10000

        if bucket < self.config.control_ratio:
            return 'control'
        else:
            return 'treatment'

    def record_metric(self, group: str, metric_name: str, value: float):
        """记录指标"""
        self.metrics[group][metric_name].append(value)

    def get_metrics_summary(self) -> Dict:
        """获取指标摘要"""
        summary = {}

        for group in ['control', 'treatment']:
            summary[group] = {}
            for metric_name, values in self.metrics[group].items():
                if values:
                    summary[group][metric_name] = {
                        'mean': np.mean(values),
                        'std': np.std(values),
                        'count': len(values),
                        'sum': np.sum(values)
                    }

        return summary


class ABTestManager:
    """
    A/B 测试管理器
    """

    def __init__(self):
        self.experiments: Dict[str, ABExperiment] = {}

    def create_experiment(self, config: ExperimentConfig) -> ABExperiment:
        """创建实验"""
        experiment = ABExperiment(config)
        self.experiments[config.experiment_id] = experiment
        return experiment

    def start_experiment(self, experiment_id: str):
        """启动实验"""
        if experiment_id in self.experiments:
            self.experiments[experiment_id].status = ExperimentStatus.RUNNING

    def get_experiment(self, experiment_id: str) -> Optional[ABExperiment]:
        """获取实验"""
        return self.experiments.get(experiment_id)

    def get_user_group(self, experiment_id: str, user_id: str) -> Optional[str]:
        """获取用户的实验组"""
        experiment = self.get_experiment(experiment_id)
        if experiment and experiment.status == ExperimentStatus.RUNNING:
            return experiment.assign_group(user_id)
        return None
```

### 2.2 样本量计算

```python
class SampleSizeCalculator:
    """
    样本量计算器
    """

    @staticmethod
    def calculate_sample_size(baseline_rate: float,
                              mde: float,
                              significance_level: float = 0.05,
                              power: float = 0.8,
                              two_tailed: bool = True) -> int:
        """
        计算所需样本量

        参数:
            baseline_rate: 基线转化率
            mde: 最小检测效应（绝对值）
            significance_level: 显著性水平 (alpha)
            power: 统计功效 (1 - beta)
            two_tailed: 是否双尾检验

        返回:
            每组所需样本量
        """
        from scipy import stats

        # Z 分数
        if two_tailed:
            z_alpha = stats.norm.ppf(1 - significance_level / 2)
        else:
            z_alpha = stats.norm.ppf(1 - significance_level)

        z_beta = stats.norm.ppf(power)

        # 比例检验样本量公式
        p1 = baseline_rate
        p2 = baseline_rate + mde

        # 平均方差
        p_avg = (p1 + p2) / 2

        # 样本量
        numerator = (z_alpha * np.sqrt(2 * p_avg * (1 - p_avg)) +
                    z_beta * np.sqrt(p1 * (1 - p1) + p2 * (1 - p2))) ** 2
        denominator = (p2 - p1) ** 2

        sample_size = int(np.ceil(numerator / denominator))

        return sample_size

    @staticmethod
    def calculate_mde(sample_size: int,
                      baseline_rate: float,
                      significance_level: float = 0.05,
                      power: float = 0.8) -> float:
        """
        计算给定样本量下的最小检测效应

        参数:
            sample_size: 每组样本量
            baseline_rate: 基线转化率
            significance_level: 显著性水平
            power: 统计功效

        返回:
            最小检测效应
        """
        # 使用二分搜索
        from scipy import stats

        z_alpha = stats.norm.ppf(1 - significance_level / 2)
        z_beta = stats.norm.ppf(power)

        p1 = baseline_rate

        # 近似公式
        se = np.sqrt(p1 * (1 - p1) / sample_size)
        mde = (z_alpha + z_beta) * se

        return mde


def demo_sample_size():
    """样本量计算示例"""
    calculator = SampleSizeCalculator()

    # 示例：基线CTR 2%，希望检测 0.5% 的提升
    baseline = 0.02
    mde = 0.005

    n = calculator.calculate_sample_size(baseline, mde)
    print(f"基线转化率: {baseline:.2%}")
    print(f"最小检测效应: {mde:.2%}")
    print(f"每组所需样本量: {n:,}")

    # 计算所需天数
    daily_traffic = 10000
    days = n / daily_traffic
    print(f"假设每日流量 {daily_traffic:,}，需要 {days:.1f} 天")


if __name__ == "__main__":
    demo_sample_size()
```

## 3. 统计分析

### 3.1 假设检验

```python
from scipy import stats
from typing import Tuple, Dict


class StatisticalAnalyzer:
    """
    统计分析器
    """

    @staticmethod
    def t_test(control: np.ndarray, treatment: np.ndarray,
               equal_var: bool = False) -> Tuple[float, float]:
        """
        独立样本 t 检验

        参数:
            control: 对照组数据
            treatment: 实验组数据
            equal_var: 是否假设方差相等

        返回:
            (t_statistic, p_value)
        """
        t_stat, p_value = stats.ttest_ind(control, treatment, equal_var=equal_var)
        return t_stat, p_value

    @staticmethod
    def proportion_test(control_success: int, control_total: int,
                       treatment_success: int, treatment_total: int) -> Tuple[float, float]:
        """
        比例检验（Z 检验）

        参数:
            control_success: 对照组成功数
            control_total: 对照组总数
            treatment_success: 实验组成功数
            treatment_total: 实验组总数

        返回:
            (z_statistic, p_value)
        """
        p1 = control_success / control_total
        p2 = treatment_success / treatment_total

        # 合并比例
        p_pooled = (control_success + treatment_success) / (control_total + treatment_total)

        # 标准误
        se = np.sqrt(p_pooled * (1 - p_pooled) * (1/control_total + 1/treatment_total))

        # Z 统计量
        z_stat = (p2 - p1) / se

        # p 值（双尾）
        p_value = 2 * (1 - stats.norm.cdf(abs(z_stat)))

        return z_stat, p_value

    @staticmethod
    def mann_whitney_u(control: np.ndarray, treatment: np.ndarray) -> Tuple[float, float]:
        """
        Mann-Whitney U 检验（非参数检验）

        适用于非正态分布的数据
        """
        u_stat, p_value = stats.mannwhitneyu(control, treatment, alternative='two-sided')
        return u_stat, p_value

    @staticmethod
    def chi_square_test(observed: np.ndarray) -> Tuple[float, float]:
        """
        卡方检验

        参数:
            observed: 观测频数矩阵 [[a, b], [c, d]]
        """
        chi2, p_value, dof, expected = stats.chi2_contingency(observed)
        return chi2, p_value


class ABTestResult:
    """
    A/B 测试结果分析
    """

    def __init__(self, control_data: Dict, treatment_data: Dict):
        self.control = control_data
        self.treatment = treatment_data
        self.analyzer = StatisticalAnalyzer()

    def analyze(self, metric_name: str,
               test_type: str = 'proportion') -> Dict:
        """
        分析单个指标

        参数:
            metric_name: 指标名称
            test_type: 检验类型 ('proportion', 'mean', 'nonparametric')

        返回:
            分析结果字典
        """
        result = {
            'metric': metric_name,
            'control_mean': self.control['means'].get(metric_name, 0),
            'treatment_mean': self.treatment['means'].get(metric_name, 0),
            'control_count': self.control['count'],
            'treatment_count': self.treatment['count']
        }

        # 计算提升
        if result['control_mean'] > 0:
            lift = (result['treatment_mean'] - result['control_mean']) / result['control_mean']
            result['lift'] = lift
            result['lift_percent'] = f"{lift:.2%}"
        else:
            result['lift'] = 0
            result['lift_percent'] = "N/A"

        # 统计检验
        if test_type == 'proportion':
            # 比例检验
            control_success = self.control['sums'].get(metric_name, 0)
            treatment_success = self.treatment['sums'].get(metric_name, 0)

            z_stat, p_value = self.analyzer.proportion_test(
                control_success, self.control['count'],
                treatment_success, self.treatment['count']
            )
            result['test_type'] = 'z_test'

        elif test_type == 'mean':
            # t 检验
            control_values = self.control['values'].get(metric_name, [])
            treatment_values = self.treatment['values'].get(metric_name, [])

            t_stat, p_value = self.analyzer.t_test(control_values, treatment_values)
            result['test_type'] = 't_test'

        elif test_type == 'nonparametric':
            # 非参数检验
            control_values = self.control['values'].get(metric_name, [])
            treatment_values = self.treatment['values'].get(metric_name, [])

            u_stat, p_value = self.analyzer.mann_whitney_u(control_values, treatment_values)
            result['test_type'] = 'mann_whitney_u'

        result['p_value'] = p_value
        result['significant'] = p_value < 0.05

        # 置信区间
        result['ci_95'] = self._calculate_ci(result)

        return result

    def _calculate_ci(self, result: Dict) -> Tuple[float, float]:
        """计算 95% 置信区间"""
        from scipy import stats

        n1 = result['control_count']
        n2 = result['treatment_count']
        p1 = result['control_mean']
        p2 = result['treatment_mean']

        # 差值的标准误
        se = np.sqrt(p1 * (1 - p1) / n1 + p2 * (1 - p2) / n2)

        diff = p2 - p1
        z = stats.norm.ppf(0.975)

        ci_low = diff - z * se
        ci_high = diff + z * se

        return (ci_low, ci_high)

    def generate_report(self, metrics: List[str]) -> str:
        """生成报告"""
        report = []
        report.append("=" * 60)
        report.append("A/B 测试结果报告")
        report.append("=" * 60)

        report.append(f"\n对照组样本量: {self.control['count']:,}")
        report.append(f"实验组样本量: {self.treatment['count']:,}")

        report.append("\n指标分析:")
        report.append("-" * 60)

        for metric in metrics:
            result = self.analyze(metric)
            report.append(f"\n{metric}:")
            report.append(f"  对照组: {result['control_mean']:.4f}")
            report.append(f"  实验组: {result['treatment_mean']:.4f}")
            report.append(f"  提升: {result['lift_percent']}")
            report.append(f"  P值: {result['p_value']:.4f}")
            report.append(f"  显著性: {'是' if result['significant'] else '否'}")

        return "\n".join(report)
```

### 3.2 多重比较校正

```python
class MultipleComparisonCorrection:
    """
    多重比较校正
    """

    @staticmethod
    def bonferroni(p_values: List[float]) -> List[float]:
        """
        Bonferroni 校正

        校正后的 p = p * n
        """
        n = len(p_values)
        return [min(p * n, 1.0) for p in p_values]

    @staticmethod
    def benjamini_hochberg(p_values: List[float]) -> List[float]:
        """
        Benjamini-Hochberg FDR 校正
        """
        n = len(p_values)
        sorted_indices = np.argsort(p_values)
        sorted_p = np.array(p_values)[sorted_indices]

        # 计算校正后的 p 值
        adjusted = np.zeros(n)
        for i in range(n):
            rank = i + 1
            adjusted[i] = sorted_p[i] * n / rank

        # 确保单调性
        for i in range(n - 2, -1, -1):
            adjusted[i] = min(adjusted[i], adjusted[i + 1])

        # 恢复原始顺序
        result = np.zeros(n)
        result[sorted_indices] = adjusted

        return result.tolist()


def demo_multiple_comparison():
    """多重比较示例"""
    p_values = [0.01, 0.03, 0.04, 0.05, 0.08]

    print("原始 p 值:", p_values)
    print("Bonferroni 校正:", MultipleComparisonCorrection.bonferroni(p_values))
    print("BH 校正:", MultipleComparisonCorrection.benjamini_hochberg(p_values))


if __name__ == "__main__":
    demo_multiple_comparison()
```

```python
class OrthogonalExperiment:
    """
    正交实验设计

    多个实验同时运行，互不影响
    """

    def __init__(self, num_layers: int = 10):
        """
        参数:
            num_layers: 分层层数
        """
        self.num_layers = num_layers
        self.layers = [{} for _ in range(num_layers)]

    def assign_layer(self, user_id: str) -> int:
        """分配用户到某一层"""
        hash_value = int(hashlib.md5(f"layer_{user_id}".encode()).hexdigest(), 16)
        return hash_value % self.num_layers

    def get_bucket(self, user_id: str, experiment_id: str, num_buckets: int = 1000) -> int:
        """获取用户在特定实验中的桶"""
        hash_value = int(hashlib.md5(f"{experiment_id}_{user_id}".encode()).hexdigest(), 16)
        return hash_value % num_buckets

    def assign_variant(self, user_id: str, experiment_id: str,
                       variants: Dict[str, float]) -> str:
        """
        分配变体

        参数:
            user_id: 用户ID
            experiment_id: 实验ID
            variants: {变体名: 流量比例}
        """
        bucket = self.get_bucket(user_id, experiment_id)

        cumulative = 0
        for variant, ratio in variants.items():
            cumulative += int(ratio * 1000)
            if bucket < cumulative:
                return variant

        return list(variants.keys())[0]
```

## 5. 实践建议

### 5.1 常见陷阱

```
1. 新奇效应 (Novelty Effect)
   - 用户对新功能的好奇导致短期效果提升
   - 解决：观察期延长

2. 辛普森悖论
   - 分组数据趋势与整体相反
   - 解决：分层分析

3. Peek 问题
   - 过早查看结果导致假阳性
   - 解决：预先确定样本量

4. 选择偏差
   - 流量分配不均匀
   - 解决：随机化验证
```

```python
class ABTestBestPractices:
    """
    A/B 测试最佳实践
    """

    @staticmethod
    def check_randomization(control: pd.DataFrame, treatment: pd.DataFrame,
                           covariates: List[str]) -> Dict:
        """
        检查随机化是否有效

        比较两组的协变量分布是否一致
        """
        results = {}

        for cov in covariates:
            if cov in control.columns and cov in treatment.columns:
                # 均值比较
                control_mean = control[cov].mean()
                treatment_mean = treatment[cov].mean()

                # t 检验
                t_stat, p_value = stats.ttest_ind(
                    control[cov].dropna(),
                    treatment[cov].dropna()
                )

                results[cov] = {
                    'control_mean': control_mean,
                    'treatment_mean': treatment_mean,
                    'p_value': p_value,
                    'balanced': p_value > 0.05
                }

        return results

    @staticmethod
    def calculate_aa_test_pvalue(data: np.ndarray, n_simulations: int = 1000) -> float:
        """
        A/A 测试：验证实验设置正确
        """
        n = len(data) // 2
        p_values = []

        for _ in range(n_simulations):
            # 随机分成两组
            shuffled = np.random.permutation(data)
            group1 = shuffled[:n]
            group2 = shuffled[n:]

            # t 检验
            _, p = stats.ttest_ind(group1, group2)
            p_values.append(p)

        # p 值应服从均匀分布
        return np.mean(np.array(p_values) < 0.05)
```

## 6. 学习总结

### 6.1 核心要点

1. **随机化**: 确保组间可比
2. **样本量**: 预先计算
3. **统计检验**: 选择合适的方法
4. **多重比较**: 注意校正

### 6.2 推荐系统特有考量

```
1. 位置偏差: 控制展示位置
2. 时间效应: 考虑时段差异
3. 用户异质性: 分层分析
4. 长期效应: 延长观察期
```
