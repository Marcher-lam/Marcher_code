# 概率校准方法 学习文档

## 1. 概率校准概述

### 1.1 什么是概率校准？

```
概率校准 (Probability Calibration):

- 将模型输出转换为真实概率
- 校准后的概率应该反映实际事件发生率
- 例如: P=0.8 的预测，实际应该有 80% 发生

为什么需要校准:
1. 模型输出不一定是真实概率
2. 深度学习模型往往过于自信
3. A/B测试、竞价需要准确概率
4. 多模型融合需要概率可比

校准评估:
- Reliability Diagram (可靠性图)
- ECE (Expected Calibration Error)
- Brier Score
```

## 2. 校准评估方法

### 2.1 可靠性图

```python
class CalibrationEvaluator:
    """
    校准评估器
    """

    @staticmethod
    def compute_reliability(y_true: np.ndarray, y_prob: np.ndarray,
                           n_bins: int = 10) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        计算可靠性图数据

        返回:
            bin_centers: 每个区间的中心概率
            bin_accuracy: 每个区间的实际准确率
            bin_counts: 每个区间的样本数
        """
        bins = np.linspace(0, 1, n_bins + 1)
        bin_centers = (bins[:-1] + bins[1:]) / 2
        bin_accuracy = np.zeros(n_bins)
        bin_counts = np.zeros(n_bins)

        for i in range(n_bins):
            mask = (y_prob >= bins[i]) & (y_prob < bins[i + 1])
            if i == n_bins - 1:  # 包含右边界
                mask = (y_prob >= bins[i]) & (y_prob <= bins[i + 1])

            bin_counts[i] = mask.sum()
            if bin_counts[i] > 0:
                bin_accuracy[i] = y_true[mask].mean()

        return bin_centers, bin_accuracy, bin_counts

    @staticmethod
    def compute_ece(y_true: np.ndarray, y_prob: np.ndarray,
                   n_bins: int = 10) -> float:
        """
        计算期望校准误差 (ECE)

        ECE = Σ |bin_accuracy - bin_confidence| * bin_weight
        """
        bin_centers, bin_accuracy, bin_counts = CalibrationEvaluator.compute_reliability(
            y_true, y_prob, n_bins
        )

        total_samples = len(y_true)
        ece = 0.0

        for i in range(n_bins):
            if bin_counts[i] > 0:
                weight = bin_counts[i] / total_samples
                ece += np.abs(bin_accuracy[i] - bin_centers[i]) * weight

        return ece

    @staticmethod
    def compute_mce(y_true: np.ndarray, y_prob: np.ndarray,
                   n_bins: int = 10) -> float:
        """
        计算最大校准误差 (MCE)
        """
        bin_centers, bin_accuracy, bin_counts = CalibrationEvaluator.compute_reliability(
            y_true, y_prob, n_bins
        )

        max_error = 0.0
        for i in range(n_bins):
            if bin_counts[i] > 0:
                error = np.abs(bin_accuracy[i] - bin_centers[i])
                max_error = max(max_error, error)

        return max_error

    @staticmethod
    def compute_brier_score(y_true: np.ndarray, y_prob: np.ndarray) -> float:
        """
        计算 Brier Score

        BS = 1/N * Σ (y_prob - y_true)²
        """
        return np.mean((y_prob - y_true) ** 2)

    @staticmethod
    def plot_reliability_diagram(y_true: np.ndarray, y_prob: np.ndarray,
                                 n_bins: int = 10, title: str = "Reliability Diagram"):
        """绘制可靠性图"""
        bin_centers, bin_accuracy, bin_counts = CalibrationEvaluator.compute_reliability(
            y_true, y_prob, n_bins
        )

        plt.figure(figsize=(8, 8))

        # 完美校准线
        plt.plot([0, 1], [0, 1], 'k--', label='Perfectly Calibrated')

        # 实际校准
        plt.bar(bin_centers, bin_accuracy, width=0.08, alpha=0.7,
               label='Model Calibration')

        plt.xlabel('Mean Predicted Probability')
        plt.ylabel('Fraction of Positives')
        plt.title(title)
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.show()
```

## 3. 校准方法实现

### 3.1 Platt Scaling

```python
class PlattScaling:
    """
    Platt Scaling (逻辑回归校准)

    将模型输出通过 sigmoid 变换:
    P(y=1|f) = 1 / (1 + exp(A * f + B))

    参数 A, B 通过在验证集上训练得到
    """

    def __init__(self):
        self.A = 1.0
        self.B = 0.0

    def fit(self, scores: np.ndarray, labels: np.ndarray):
        """
        训练 Platt Scaling

        使用带正则化的逻辑回归
        """
        from sklearn.linear_model import LogisticRegression

        # Platt 建议使用 target: (N_+ + 1) / (N_+ + 2) 和 1 / (N_- + 2)
        n_pos = labels.sum()
        n_neg = len(labels) - n_pos

        # 平滑标签
        smoothed_labels = np.where(
            labels == 1,
            (n_pos + 1) / (n_pos + 2),
            1 / (n_neg + 2)
        )

        # 逻辑回归拟合
        lr = LogisticRegression(fit_intercept=True, C=1e10)  # 无正则化
        lr.fit(scores.reshape(-1, 1), labels)

        self.A = lr.coef_[0, 0]
        self.B = lr.intercept_[0]

        return self

    def transform(self, scores: np.ndarray) -> np.ndarray:
        """校准"""
        return 1 / (1 + np.exp(-(self.A * scores + self.B)))

    def fit_transform(self, scores: np.ndarray, labels: np.ndarray) -> np.ndarray:
        """拟合并转换"""
        self.fit(scores, labels)
        return self.transform(scores)


class TemperatureScaling:
    """
    Temperature Scaling (温度缩放)

    最简单的校准方法:
    P = sigmoid(logits / T)

    T > 1: 概率分布更平滑 (更不确定)
    T < 1: 概率分布更尖锐 (更确定)
    """

    def __init__(self):
        self.temperature = 1.0

    def fit(self, logits: np.ndarray, labels: np.ndarray):
        """
        训练温度缩放

        通过最小化 NLL 找最优温度
        """
        from scipy.optimize import minimize_scalar

        def nll_loss(t):
            if t <= 0:
                return float('inf')

            probs = self._sigmoid(logits / t)
            # 避免 log(0)
            probs = np.clip(probs, 1e-10, 1 - 1e-10)

            # 负对数似然
            nll = -np.mean(
                labels * np.log(probs) + (1 - labels) * np.log(1 - probs)
            )
            return nll

        # 搜索最优温度
        result = minimize_scalar(
            nll_loss,
            bounds=(0.1, 10.0),
            method='bounded'
        )

        self.temperature = result.x
        print(f"Optimal temperature: {self.temperature:.4f}")

        return self

    def _sigmoid(self, x: np.ndarray) -> np.ndarray:
        """Sigmoid 函数"""
        return 1 / (1 + np.exp(-x))

    def transform(self, logits: np.ndarray) -> np.ndarray:
        """校准"""
        return self._sigmoid(logits / self.temperature)

    def fit_transform(self, logits: np.ndarray, labels: np.ndarray) -> np.ndarray:
        """拟合并转换"""
        self.fit(logits, labels)
        return self.transform(logits)
```

class IsotonicCalibration:
    """
    保序回归校准

    非参数方法，学习一个单调递增的映射函数
    适合校准曲线不规则的情况
    """

    def __init__(self):
        self.isotonic = IsotonicRegression(out_of_bounds='clip')

    def fit(self, scores: np.ndarray, labels: np.ndarray):
        """训练"""
        self.isotonic.fit(scores, labels)
        return self

    def transform(self, scores: np.ndarray) -> np.ndarray:
        """校准"""
        return self.isotonic.transform(scores)

    def fit_transform(self, scores: np.ndarray, labels: np.ndarray) -> np.ndarray:
        """拟合并转换"""
        self.fit(scores, labels)
        return self.transform(scores)


class HistogramBinning:
    """
    直方图分箱校准

    将预测概率分成若干区间，每个区间用实际频率校准
    """

    def __init__(self, n_bins: int = 10):
        self.n_bins = n_bins
        self.bin_boundaries = None
        self.bin_calibrations = None

    def fit(self, scores: np.ndarray, labels: np.ndarray):
        """训练"""
        # 创建分箱边界
        self.bin_boundaries = np.linspace(0, 1, self.n_bins + 1)

        # 计算每个分箱的校准值
        self.bin_calibrations = np.zeros(self.n_bins)

        for i in range(self.n_bins):
            mask = (scores >= self.bin_boundaries[i]) & (
                scores < self.bin_boundaries[i + 1]
            )
            if i == self.n_bins - 1:
                mask = (scores >= self.bin_boundaries[i]) & (
                    scores <= self.bin_boundaries[i + 1]
                )

            if mask.sum() > 0:
                self.bin_calibrations[i] = labels[mask].mean()
            else:
                self.bin_calibrations[i] = (self.bin_boundaries[i] +
                                           self.bin_boundaries[i + 1]) / 2

        return self

    def transform(self, scores: np.ndarray) -> np.ndarray:
        """校准"""
        calibrated = np.zeros_like(scores)

        for i in range(self.n_bins):
            mask = (scores >= self.bin_boundaries[i]) & (
                scores < self.bin_boundaries[i + 1]
            )
            if i == self.n_bins - 1:
                mask = (scores >= self.bin_boundaries[i]) & (
                    scores <= self.bin_boundaries[i + 1]
                )

            calibrated[mask] = self.bin_calibrations[i]

        return calibrated

    def fit_transform(self, scores: np.ndarray, labels: np.ndarray) -> np.ndarray:
        """拟合并转换"""
        self.fit(scores, labels)
        return self.transform(scores)

### 3.3 Beta Calibration

```python
class BetaCalibration:
    """
    Beta 校准

    使用 Beta 分布进行校准:
    P = F_beta(α * f + β; a, b)

    其中 F_beta 是 Beta 分布的 CDF
    """

    def __init__(self):
        self.params = None

    def fit(self, scores: np.ndarray, labels: np.ndarray):
        """训练"""
        from scipy.optimize import minimize
        from scipy.stats import beta as beta_dist

        def neg_log_likelihood(params):
            a, b, alpha, c = params

            # 变换分数
            x = alpha * scores + c
            x = np.clip(x, 1e-10, 1 - 1e-10)

            # Beta CDF
            try:
                calibrated = beta_dist.cdf(x, a, b)
            except:
                return float('inf')

            calibrated = np.clip(calibrated, 1e-10, 1 - 1e-10)

            # NLL
            nll = -np.mean(
                labels * np.log(calibrated) +
                (1 - labels) * np.log(1 - calibrated)
            )

            return nll

        # 初始参数
        x0 = [2.0, 2.0, 1.0, 0.0]

        # 优化
        result = minimize(
            neg_log_likelihood,
            x0,
            method='L-BFGS-B',
            bounds=[(0.1, 10), (0.1, 10), (0.1, 10), (-1, 1)]
        )

        self.params = result.x

        return self

    def transform(self, scores: np.ndarray) -> np.ndarray:
        """校准"""
        from scipy.stats import beta as beta_dist

        a, b, alpha, c = self.params

        x = alpha * scores + c
        x = np.clip(x, 1e-10, 1 - 1e-10)

        calibrated = beta_dist.cdf(x, a, b)
        calibrated = np.clip(calibrated, 1e-10, 1 - 1e-10)

        return calibrated

    def fit_transform(self, scores: np.ndarray, labels: np.ndarray) -> np.ndarray:
        """拟合并转换"""
        self.fit(scores, labels)
        return self.transform(scores)
```

## 4. 推荐系统应用

### 4.1 CTR 预测校准

```python
class CTRCalibrator:
    """
    CTR 预测校准器
    """

    def __init__(self, method: str = 'temperature'):
        """
        参数:
            method: 校准方法 ('temperature', 'platt', 'isotonic', 'histogram')
        """
        self.method = method
        self.calibrator = None

    def fit(self, predictions: np.ndarray, labels: np.ndarray):
        """
        训练校准器

        predictions: 模型预测的 CTR（可以是 logits 或概率）
        labels: 实际点击标签
        """
        if self.method == 'temperature':
            # 假设输入是 logits
            self.calibrator = TemperatureScaling()
            self.calibrator.fit(predictions, labels)
        elif self.method == 'platt':
            self.calibrator = PlattScaling()
            self.calibrator.fit(predictions, labels)
        elif self.method == 'isotonic':
            self.calibrator = IsotonicCalibration()
            self.calibrator.fit(predictions, labels)
        elif self.method == 'histogram':
            self.calibrator = HistogramBinning()
            self.calibrator.fit(predictions, labels)
        else:
            raise ValueError(f"Unknown method: {self.method}")

        return self

    def calibrate(self, predictions: np.ndarray) -> np.ndarray:
        """校准预测"""
        return self.calibrator.transform(predictions)

    def evaluate(self, predictions: np.ndarray, labels: np.ndarray
                ) -> Dict[str, float]:
        """评估校准效果"""
        calibrated = self.calibrate(predictions)

        return {
            'ece': CalibrationEvaluator.compute_ece(labels, calibrated),
            'mce': CalibrationEvaluator.compute_mce(labels, calibrated),
            'brier_score': CalibrationEvaluator.compute_brier_score(labels, calibrated)
        }


class MultiModelCalibrator:
    """
    多模型校准与融合
    """

    def __init__(self, models: List[str], calibration_method: str = 'temperature'):
        self.models = models
        self.calibrators = {
            model: CTRCalibrator(calibration_method)
            for model in models
        }

    def fit(self, predictions: Dict[str, np.ndarray], labels: np.ndarray):
        """训练所有校准器"""
        for model, preds in predictions.items():
            self.calibrators[model].fit(preds, labels)

        return self

    def calibrate(self, predictions: Dict[str, np.ndarray]
                 ) -> Dict[str, np.ndarray]:
        """校准所有模型预测"""
        return {
            model: self.calibrators[model].calibrate(preds)
            for model, preds in predictions.items()
        }

    def ensemble(self, predictions: Dict[str, np.ndarray],
                weights: Optional[Dict[str, float]] = None) -> np.ndarray:
        """
        加权集成

        校准后概率可以直接相加
        """
        calibrated = self.calibrate(predictions)

        if weights is None:
            # 等权重
            weights = {model: 1.0 / len(self.models) for model in self.models}

        ensemble_pred = np.zeros_like(list(calibrated.values())[0])

        for model, pred in calibrated.items():
            ensemble_pred += weights[model] * pred

        return ensemble_pred
```

## 5. 学习总结

### 5.1 方法对比

```
方法        适用场景            优点            缺点
────────────────────────────────────────────────────
温度缩放    神经网络输出        简单高效        单参数
Platt       SVM、逻辑回归      有理论支撑      需要足够数据
保序回归    不规则校准曲线      灵活非参数      可能过拟合
直方图      快速校准            直观            分箱敏感
Beta        非对称偏差          灵活            计算复杂
```

### 5.2 最佳实践

```
1. 在验证集上训练校准器，不在训练集
2. 评估使用 ECE 和 Brier Score
3. 神经网络推荐温度缩放
4. 树模型推荐保序回归
5. 校准后绘制可靠性图验证
```

### 5.3 推荐系统中的应用

```
1. CTR预估: 校准后概率用于竞价
2. 多任务学习: 各任务概率需要校准
3. 模型融合: 校准后的概率才能融合
4. A/B测试: 准确概率便于效果评估
```
