# 推荐系统离线训练 学习文档

## 1. 训练流程概述

### 1.1 完整训练流程

```
数据采集 → 数据清洗 → 特征工程 → 样本构造 → 模型训练 → 模型评估 → 模型部署
```

### 1.2 训练模式

| 模式 | 特点 | 适用场景 |
|------|------|----------|
| 全量训练 | 使用所有数据重新训练 | 模型迭代 |
| 增量训练 | 基于旧模型继续训练 | 实时更新 |
| 在线学习 | 实时样本实时更新 | 高时效需求 |

```python
class DataCleaner:
    """
    数据清洗器
    """

    def __init__(self):
        self.rules = []

    def add_rule(self, name: str, func):
        """添加清洗规则"""
        self.rules.append((name, func))

    def clean(self, df: pd.DataFrame) -> pd.DataFrame:
        """执行清洗"""
        for name, func in self.rules:
            before = len(df)
            df = func(df)
            after = len(df)
            print(f"规则 {name}: 过滤 {before - after} 条数据")

        return df

    @staticmethod
    def remove_duplicates(df: pd.DataFrame) -> pd.DataFrame:
        """去重"""
        return df.drop_duplicates()

    @staticmethod
    def filter_time_range(df: pd.DataFrame, timestamp_col: str,
                         start_time: int, end_time: int) -> pd.DataFrame:
        """时间范围过滤"""
        return df[(df[timestamp_col] >= start_time) &
                  (df[timestamp_col] <= end_time)]

    @staticmethod
    def filter_bots(df: pd.DataFrame, user_col: str,
                   threshold: int = 1000) -> pd.DataFrame:
        """过滤机器人用户"""
        user_counts = df[user_col].value_counts()
        bot_users = user_counts[user_counts > threshold].index
        return df[~df[user_col].isin(bot_users)]
```

```python
class FeatureEngineer:
    """
    特征工程
    """

    def __init__(self):
        self.feature_configs = {}

    def add_feature(self, name: str, feature_type: str, config: Dict):
        """添加特征配置"""
        self.feature_configs[name] = {
            'type': feature_type,
            'config': config
        }

    def extract_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """提取特征"""
        for name, config in self.feature_configs.items():
            if config['type'] == 'numerical':
                df[name] = self._extract_numerical(df, config['config'])
            elif config['type'] == 'categorical':
                df[name] = self._extract_categorical(df, config['config'])
            elif config['type'] == 'sequence':
                df[name] = self._extract_sequence(df, config['config'])
            elif config['type'] == 'statistical':
                df[name] = self._extract_statistical(df, config['config'])

        return df

class SequenceFeatureExtractor:
    """
    序列特征提取
    """

    def __init__(self, max_seq_len: int = 50):
        self.max_seq_len = max_seq_len

    def extract_user_behavior_sequence(self, df: pd.DataFrame,
                                       user_col: str,
                                       item_col: str,
                                       time_col: str) -> Dict:
        """
        提取用户行为序列

        返回:
            {user_id: [item_id1, item_id2, ...]}
        """
        df_sorted = df.sort_values(time_col)

        user_sequences = defaultdict(list)
        for _, row in df_sorted.iterrows():
            user_sequences[row[user_col]].append(row[item_col])

        # 截断
        for user_id in user_sequences:
            user_sequences[user_id] = user_sequences[user_id][-self.max_seq_len:]

        return dict(user_sequences)

    def extract_time_gap_features(self, sequences: Dict,
                                  timestamps: Dict) -> Dict:
        """提取时间间隔特征"""
        time_gap_features = {}

        for user_id, items in sequences.items():
            if user_id not in timestamps:
                continue

            ts = timestamps[user_id]
            if len(ts) < 2:
                continue

            # 计算相邻行为的时间间隔
            gaps = [ts[i+1] - ts[i] for i in range(len(ts) - 1)]

            time_gap_features[user_id] = {
                'mean_gap': np.mean(gaps),
                'max_gap': np.max(gaps),
                'min_gap': np.min(gaps),
                'std_gap': np.std(gaps)
            }

        return time_gap_features
```

## 3. 样本构造

### 3.1 正负样本

```python
class SampleConstructor:
    """
    样本构造器
    """

    def __init__(self, negative_ratio: int = 4):
        self.negative_ratio = negative_ratio

    def construct_pointwise_samples(self,
                                    interactions: pd.DataFrame,
                                    all_items: List[int]) -> pd.DataFrame:
        """
        构造 Pointwise 样本

        正样本: 用户交互的物品
        负样本: 用户未交互的物品
        """
        samples = []

        for user_id, group in interactions.groupby('user_id'):
            pos_items = set(group['item_id'])
            neg_items = list(set(all_items) - pos_items)

            # 正样本
            for _, row in group.iterrows():
                samples.append({
                    'user_id': user_id,
                    'item_id': row['item_id'],
                    'label': 1,
                    'timestamp': row.get('timestamp', 0)
                })

            # 负采样
            n_neg = min(len(neg_items), len(pos_items) * self.negative_ratio)
            neg_sampled = np.random.choice(neg_items, n_neg, replace=False)

            for item_id in neg_sampled:
                samples.append({
                    'user_id': user_id,
                    'item_id': item_id,
                    'label': 0,
                    'timestamp': 0
                })

        return pd.DataFrame(samples)

    def construct_pairwise_samples(self,
                                   interactions: pd.DataFrame,
                                   all_items: List[int]) -> List[Tuple]:
        """
        构造 Pairwise 样本

        返回: [(user_id, pos_item_id, neg_item_id), ...]
        """
        samples = []

        for user_id, group in interactions.groupby('user_id'):
            pos_items = list(group['item_id'])
            neg_items = list(set(all_items) - set(pos_items))

            for pos_item in pos_items:
                # 每个正样本配一个负样本
                neg_item = np.random.choice(neg_items)
                samples.append((user_id, pos_item, neg_item))

        return samples
```

### 3.2 样本采样

```python
class Sampler:
    """
    样本采样器
    """

    @staticmethod
    def undersample_majority(df: pd.DataFrame,
                             label_col: str,
                             ratio: float = 0.1) -> pd.DataFrame:
        """欠采样多数类"""
        pos = df[df[label_col] == 1]
        neg = df[df[label_col] == 0]

        n_keep = int(len(neg) * ratio)
        neg_sampled = neg.sample(n=n_keep, random_state=42)

        return pd.concat([pos, neg_sampled])

    @staticmethod
    def oversample_minority(df: pd.DataFrame,
                           label_col: str,
                           target_ratio: float = 0.5) -> pd.DataFrame:
        """过采样少数类"""
        pos = df[df[label_col] == 1]
        neg = df[df[label_col] == 0]

        n_pos_needed = int(len(neg) * target_ratio / (1 - target_ratio))
        n_repeat = int(np.ceil(n_pos_needed / len(pos)))

        pos_oversampled = pd.concat([pos] * n_repeat).sample(
            n=n_pos_needed, random_state=42
        )

        return pd.concat([pos_oversampled, neg])
```

```python
import torch
import torch.distributed as dist
from torch.nn.parallel import DistributedDataParallel as DDP


class DistributedTrainer:
    """
    分布式训练器
    """

    def __init__(self, model, local_rank: int):
        self.local_rank = local_rank
        self.model = model.to(local_rank)

        # DDP 包装
        self.model = DDP(
            self.model,
            device_ids=[local_rank],
            output_device=local_rank
        )

        self.optimizer = torch.optim.Adam(self.model.parameters())

    def train_epoch(self, dataloader):
        """训练一个 epoch"""
        self.model.train()

        for batch in dataloader:
            self.optimizer.zero_grad()

            # 移动数据到设备
            batch = {k: v.to(self.local_rank) for k, v in batch.items()}

            # 前向传播
            loss = self.model(**batch)

            # 反向传播
            loss.backward()
            self.optimizer.step()
```

### 5.1 离线评估

```python
class OfflineEvaluator:
    """
    离线评估器
    """

    def __init__(self):
        self.metrics = {}

    def evaluate(self, model, test_df: pd.DataFrame) -> Dict:
        """评估模型"""
        from sklearn.metrics import (
            roc_auc_score, log_loss,
            precision_score, recall_score
        )

        # 预测
        predictions = model.predict(test_df)
        labels = test_df['label'].values

        # 计算指标
        self.metrics = {
            'auc': roc_auc_score(labels, predictions),
            'logloss': log_loss(labels, predictions),
            'gauc': self._compute_gauc(test_df, predictions, labels)
        }

        return self.metrics

    def _compute_gauc(self, df: pd.DataFrame,
                     predictions: np.ndarray,
                     labels: np.ndarray) -> float:
        """计算 Group AUC"""
        user_aucs = []
        user_counts = []

        for user_id, group in df.groupby('user_id'):
            indices = group.index.tolist()
            user_pred = predictions[indices]
            user_label = labels[indices]

            if len(np.unique(user_label)) > 1:
                auc = roc_auc_score(user_label, user_pred)
                user_aucs.append(auc)
                user_counts.append(len(group))

        # 加权平均
        if user_aucs:
            return np.average(user_aucs, weights=user_counts)
        return 0
```

```python
class ModelVersionManager:
    """
    模型版本管理
    """

    def __init__(self, base_dir: str):
        self.base_dir = base_dir
        self.current_version = None

    def save_version(self, model, metrics: Dict, version: str = None):
        """保存模型版本"""
        import json
        import os
        from datetime import datetime

        if version is None:
            version = datetime.now().strftime("%Y%m%d_%H%M%S")

        version_dir = f"{self.base_dir}/{version}"
        os.makedirs(version_dir, exist_ok=True)

        # 保存模型
        torch.save(model.state_dict(), f"{version_dir}/model.pt")

        # 保存元数据
        metadata = {
            'version': version,
            'metrics': metrics,
            'timestamp': datetime.now().isoformat()
        }
        with open(f"{version_dir}/metadata.json", 'w') as f:
            json.dump(metadata, f)

        self.current_version = version
        return version

    def load_version(self, model, version: str):
        """加载指定版本"""
        model.load_state_dict(
            torch.load(f"{self.base_dir}/{version}/model.pt")
        )
        return model
```
