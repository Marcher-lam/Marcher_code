# 模型压缩与量化 学习文档

## 1. 模型压缩概述

### 1.1 为什么需要模型压缩？

```
推荐系统模型压缩的需求:

1. 推理延迟要求
   - CTR预估: < 10ms
   - 召回: < 100ms
   - 用户等待不耐烦

2. 资源限制
   - 内存有限
   - 计算资源有限
   - 成本控制

3. 部署场景
   - 边缘设备
   - 移动端
   - 嵌入式系统

压缩效果:
- 模型大小: 10x - 100x
- 推理速度: 2x - 10x
- 精度损失: < 1%
```

### 1.2 压缩方法分类

```python
"""
模型压缩方法:

1. 量化 (Quantization)
   - FP32 → FP16/INT8/INT4
   - 精度换取速度和空间

2. 剪枝 (Pruning)
   - 移除不重要的权重
   - 稀疏化模型

3. 知识蒸馏 (Knowledge Distillation)
   - 大模型 → 小模型
   - 保留教师模型知识

4. 低秩分解 (Low-Rank Decomposition)
   - 大矩阵分解为小矩阵乘积
   - 减少参数量
"""
```

## 2. 量化 (Quantization)

### 2.1 量化基础

```python
import torch
import torch.nn as nn
import numpy as np
from typing import Tuple, Optional


class QuantizationBase:
    """
    量化基础
    """

    @staticmethod
    def quantize_symmetric(x: np.ndarray, n_bits: int = 8) -> Tuple[np.ndarray, float]:
        """
        对称量化

        FP32 → INT8

        公式:
        x_q = round(x / scale)
        scale = max(|x|) / (2^(n_bits-1) - 1)
        """
        # 计算缩放因子
        max_val = np.max(np.abs(x))
        q_max = 2 ** (n_bits - 1) - 1

        scale = max_val / q_max

        # 量化
        x_q = np.round(x / scale).astype(np.int8)

        return x_q, scale

    @staticmethod
    def dequantize_symmetric(x_q: np.ndarray, scale: float) -> np.ndarray:
        """
        反量化

        INT8 → FP32
        """
        return x_q.astype(np.float32) * scale

    @staticmethod
    def quantize_asymmetric(x: np.ndarray, n_bits: int = 8
                           ) -> Tuple[np.ndarray, float, float]:
        """
        非对称量化

        使用零点偏移

        公式:
        x_q = round((x - zero_point) / scale)
        scale = (max(x) - min(x)) / (2^n_bits - 1)
        zero_point = min(x)
        """
        x_min = np.min(x)
        x_max = np.max(x)

        q_min = 0
        q_max = 2 ** n_bits - 1

        scale = (x_max - x_min) / (q_max - q_min)
        zero_point = x_min

        x_q = np.round((x - zero_point) / scale).astype(np.uint8)

        return x_q, scale, zero_point

    @staticmethod
    def dequantize_asymmetric(x_q: np.ndarray, scale: float,
                              zero_point: float) -> np.ndarray:
        """非对称反量化"""
        return x_q.astype(np.float32) * scale + zero_point
```

### 2.2 PyTorch 动态量化

```python
class DynamicQuantization:
    """
    PyTorch 动态量化

    - 权重提前量化
    - 激活值运行时量化
    - 简单易用，精度损失小
    """

    @staticmethod
    def quantize_linear(model: nn.Module) -> nn.Module:
        """
        量化线性层
        """
        quantized_model = torch.quantization.quantize_dynamic(
            model,
            {nn.Linear, nn.Embedding, nn.LSTM},
            dtype=torch.qint8
        )
        return quantized_model

    @staticmethod
    def quantize_lstm(model: nn.Module) -> nn.Module:
        """
        量化 LSTM
        """
        quantized_model = torch.quantization.quantize_dynamic(
            model,
            {nn.LSTM, nn.Linear},
            dtype=torch.qint8
        )
        return quantized_model


class QuantizedLinear(nn.Module):
    """
    手工实现的量化线性层
    """

    def __init__(self, in_features: int, out_features: int):
        super().__init__()

        self.in_features = in_features
        self.out_features = out_features

        # 量化权重 (INT8)
        self.register_buffer('weight_q', torch.zeros(out_features, in_features, dtype=torch.int8))
        self.register_buffer('weight_scale', torch.tensor(1.0))

        # 偏置 (FP32)
        self.bias = nn.Parameter(torch.zeros(out_features))

    def quantize_weight(self, weight: torch.Tensor):
        """量化权重"""
        max_val = weight.abs().max()
        scale = max_val / 127.0

        self.weight_q = (weight / scale).round().clamp(-128, 127).to(torch.int8)
        self.weight_scale = scale

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        # 反量化权重
        weight = self.weight_q.float() * self.weight_scale

        return torch.nn.functional.linear(x, weight, self.bias)
```

### 2.3 静态量化

```python
class StaticQuantization:
    """
    PyTorch 静态量化

    - 权重和激活都提前量化
    - 需要校准数据
    - 推理速度最快
    """

    def __init__(self, model: nn.Module):
        self.model = model
        self.model.eval()

    def prepare(self):
        """准备量化"""
        # 插入观察器
        self.model.qconfig = torch.quantization.get_default_qconfig('fbgemm')
        torch.quantization.prepare(self.model, inplace=True)

    def calibrate(self, dataloader, n_batches: int = 100):
        """校准"""
        with torch.no_grad():
            for i, batch in enumerate(dataloader):
                if i >= n_batches:
                    break
                self.model(batch)

    def convert(self) -> nn.Module:
        """转换"""
        torch.quantization.convert(self.model, inplace=True)
        return self.model


# 推荐系统模型量化示例
class QuantizedDeepFM(nn.Module):
    """
    量化的 DeepFM
    """

    def __init__(self, n_features: int, embed_dim: int = 16,
                 hidden_dims: list = [256, 128]):
        super().__init__()

        # Embedding 层 (通常不量化)
        self.embeddings = nn.Embedding(n_features, embed_dim)

        # FM 部分
        self.fm_linear = nn.Linear(n_features, 1)

        # Deep 部分 (量化)
        layers = []
        input_dim = embed_dim * n_features
        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(input_dim, hidden_dim))
            layers.append(nn.ReLU())
            layers.append(nn.BatchNorm1d(hidden_dim))
            input_dim = hidden_dim
        layers.append(nn.Linear(input_dim, 1))

        self.deep = nn.Sequential(*layers)

        # 量化配置
        self.quant = torch.quantization.QuantStub()
        self.dequant = torch.quantization.DeQuantStub()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # 量化输入
        x = self.quant(x)

        # Embedding
        embed = self.embeddings(x)

        # FM
        fm_out = self.fm_linear(x.float())

        # Deep
        deep_out = self.deep(embed.view(embed.size(0), -1))

        # 合并
        output = fm_out + deep_out

        # 反量化输出
        return self.dequant(output)
```

## 3. 剪枝 (Pruning)

### 3.1 结构化剪枝

```python
class StructuredPruning:
    """
    结构化剪枝

    - 移除整个神经元/通道
    - 不需要特殊硬件支持
    - 推理加速明显
    """

    @staticmethod
    def prune_linear_layer(model: nn.Module, layer_name: str,
                           amount: float = 0.3):
        """
        剪枝线性层

        amount: 剪枝比例
        """
        import torch.nn.utils.prune as prune

        layer = getattr(model, layer_name)

        # L1 非结构化剪枝
        prune.l1_unstructured(layer, name='weight', amount=amount)

        # 使剪枝永久化
        prune.remove(layer, 'weight')

        return model

    @staticmethod
    def prune_by_importance(model: nn.Module,
                            importance_scores: dict,
                            threshold: float = 0.01):
        """
        按重要性剪枝

        移除重要性低于阈值的神经元
        """
        for name, param in model.named_parameters():
            if name in importance_scores:
                scores = importance_scores[name]
                mask = scores > threshold

                # 创建新模型（这里简化处理）
                print(f"Layer {name}: {mask.sum()}/{len(mask)} neurons kept")

        return model


class MagnitudePruning:
    """
    幅度剪枝

    移除权重绝对值小的连接
    """

    def __init__(self, model: nn.Module, prune_ratio: float = 0.5):
        self.model = model
        self.prune_ratio = prune_ratio

    def compute_importance(self) -> dict:
        """计算权重重要性"""
        importance = {}
        for name, param in self.model.named_parameters():
            if 'weight' in name:
                importance[name] = param.abs()
        return importance

    def get_pruning_mask(self, importance: dict) -> dict:
        """获取剪枝掩码"""
        masks = {}
        for name, scores in importance.items():
            threshold = torch.quantile(scores.flatten(), self.prune_ratio)
            masks[name] = scores > threshold
        return masks

    def apply_masks(self, masks: dict):
        """应用剪枝掩码"""
        for name, param in self.model.named_parameters():
            if name in masks:
                param.data *= masks[name].float()

    def prune(self) -> dict:
        """执行剪枝"""
        importance = self.compute_importance()
        masks = self.get_pruning_mask(importance)
        self.apply_masks(masks)

        # 统计稀疏度
        sparsity = {}
        for name, mask in masks.items():
            sparsity[name] = 1 - mask.float().mean().item()

        return sparsity
```

## 4. 低秩分解

### 4.1 SVD 分解

```python
class LowRankDecomposition:
    """
    低秩分解

    将大矩阵分解为小矩阵乘积
    """

    @staticmethod
    def decompose_linear(layer: nn.Linear, rank: int
                        ) -> Tuple[nn.Linear, nn.Linear]:
        """
        分解线性层

        原始: (in, out)
        分解: (in, rank) x (rank, out)

        参数量: in*out → in*rank + rank*out
        当 rank << min(in, out) 时显著减少
        """
        weight = layer.weight.data  # (out, in)

        # SVD 分解
        U, S, Vh = torch.linalg.svd(weight, full_matrices=False)

        # 截断
        U_r = U[:, :rank]  # (out, rank)
        S_r = S[:rank]     # (rank,)
        Vh_r = Vh[:rank, :]  # (rank, in)

        # 创建两个小线性层
        # 第一层: in -> rank (使用 Vh_r)
        layer1 = nn.Linear(layer.in_features, rank, bias=False)
        layer1.weight.data = Vh_r

        # 第二层: rank -> out (使用 U_r @ diag(S_r))
        layer2 = nn.Linear(rank, layer.out_features, bias=True)
        layer2.weight.data = U_r @ torch.diag(S_r)
        if layer.bias is not None:
            layer2.bias.data = layer.bias.data.clone()

        return layer1, layer2

    @staticmethod
    def decompose_embedding(embedding: nn.Embedding, rank: int
                           ) -> Tuple[nn.Embedding, nn.Linear]:
        """
        分解 Embedding 层

        原始: (n_items, embed_dim)
        分解: (n_items, rank) -> (rank, embed_dim)
        """
        weight = embedding.weight.data  # (n_items, embed_dim)

        # SVD
        U, S, Vh = torch.linalg.svd(weight, full_matrices=False)

        # 截断
        U_r = U[:, :rank]
        S_r = S[:rank]
        Vh_r = Vh[:rank, :]

        # 分解
        embed_small = nn.Embedding(embedding.num_embeddings, rank)
        embed_small.weight.data = U_r * S_r.unsqueeze(0)

        projection = nn.Linear(rank, embedding.embedding_dim, bias=False)
        projection.weight.data = Vh_r

        return embed_small, projection


class DecomposedLinear(nn.Module):
    """
    分解后的线性层
    """

    def __init__(self, in_features: int, out_features: int, rank: int):
        super().__init__()

        self.layer1 = nn.Linear(in_features, rank, bias=False)
        self.layer2 = nn.Linear(rank, out_features, bias=True)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.layer1(x)
        x = self.layer2(x)
        return x


# 应用示例
def compress_deepfm(model: nn.Module, rank_ratio: float = 0.5):
    """
    压缩 DeepFM 模型
    """
    for name, module in model.named_modules():
        if isinstance(module, nn.Linear):
            in_features = module.in_features
            out_features = module.out_features
            rank = int(min(in_features, out_features) * rank_ratio)

            # 分解
            layer1, layer2 = LowRankDecomposition.decompose_linear(module, rank)

            # 替换（这里简化，实际需要修改模型结构）

    return model
```

## 5. 推荐系统压缩实践

### 5.1 压缩流水线

```python
class ModelCompressionPipeline:
    """
    模型压缩流水线
    """

    def __init__(self, model: nn.Module):
        self.model = model
        self.compressed_model = None

    def compress(self,
                 method: str = 'quantization',
                 calibrate_loader=None,
                 **kwargs):
        """
        执行压缩
        """
        if method == 'quantization':
            self.compressed_model = self._quantize(calibrate_loader, **kwargs)
        elif method == 'pruning':
            self.compressed_model = self._prune(**kwargs)
        elif method == 'decomposition':
            self.compressed_model = self._decompose(**kwargs)
        elif method == 'combined':
            self.compressed_model = self._combined(**kwargs)

        return self.compressed_model

    def _quantize(self, calibrate_loader, n_calibrate_batches: int = 100,
                  **kwargs):
        """量化"""
        quantizer = StaticQuantization(self.model)
        quantizer.prepare()
        quantizer.calibrate(calibrate_loader, n_calibrate_batches)
        return quantizer.convert()

    def _prune(self, prune_ratio: float = 0.5, **kwargs):
        """剪枝"""
        pruner = MagnitudePruning(self.model, prune_ratio)
        pruner.prune()
        return self.model

    def _decompose(self, rank_ratio: float = 0.5, **kwargs):
        """低秩分解"""
        return compress_deepfm(self.model, rank_ratio)

    def _combined(self, calibrate_loader, **kwargs):
        """组合压缩"""
        # 1. 先剪枝
        model = self._prune(**kwargs)

        # 2. 再量化
        quantizer = StaticQuantization(model)
        quantizer.prepare()
        quantizer.calibrate(calibrate_loader, kwargs.get('n_calibrate_batches', 100))

        return quantizer.convert()

    def benchmark(self, input_shape: tuple, n_runs: int = 1000):
        """性能基准测试"""
        import time

        device = next(self.model.parameters()).device
        dummy_input = torch.randn(input_shape).to(device)

        # 原始模型
        self.model.eval()
        with torch.no_grad():
            # 预热
            for _ in range(10):
                _ = self.model(dummy_input)

            start = time.time()
            for _ in range(n_runs):
                _ = self.model(dummy_input)
            original_time = (time.time() - start) / n_runs

        # 压缩后模型
        if self.compressed_model:
            self.compressed_model.eval()
            with torch.no_grad():
                # 预热
                for _ in range(10):
                    _ = self.compressed_model(dummy_input)

                start = time.time()
                for _ in range(n_runs):
                    _ = self.compressed_model(dummy_input)
                compressed_time = (time.time() - start) / n_runs
        else:
            compressed_time = original_time

        # 模型大小
        original_size = sum(p.numel() * p.element_size() for p in self.model.parameters())

        if self.compressed_model:
            compressed_size = sum(p.numel() * p.element_size()
                                 for p in self.compressed_model.parameters())
        else:
            compressed_size = original_size

        return {
            'original_time_ms': original_time * 1000,
            'compressed_time_ms': compressed_time * 1000,
            'speedup': original_time / compressed_time if compressed_time > 0 else 1.0,
            'original_size_mb': original_size / 1024 / 1024,
            'compressed_size_mb': compressed_size / 1024 / 1024,
            'compression_ratio': original_size / compressed_size if compressed_size > 0 else 1.0
        }
```

## 6. 学习总结

### 6.1 方法对比

```
方法        压缩比     速度提升   精度损失   适用场景
────────────────────────────────────────────────────
INT8量化    4x         2-4x       <1%       通用
INT4量化    8x         4-8x       1-2%      对精度要求不高
结构化剪枝  2-10x      2-5x       1-3%      有冗余的模型
非结构剪枝  10-100x    需要稀疏库  <1%       有稀疏计算支持
低秩分解    2-5x       1.5-3x     <1%       全连接层
知识蒸馏    10-100x    5-20x      1-2%      有大模型
```

### 6.2 最佳实践

```
1. 量化优先: INT8 量化是最简单有效的方法
2. 渐进压缩: 先量化，再剪枝，最后分解
3. 评估精度: 每步压缩后都要评估精度
4. 微调恢复: 压缩后进行微调恢复精度
5. 硬件适配: 选择硬件支持的压缩方法
```
