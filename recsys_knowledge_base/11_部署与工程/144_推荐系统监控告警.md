# 推荐系统监控告警 学习文档

## 1. 监控体系概述

### 1.1 监控目标

```
推荐系统需要监控的维度:

1. 业务指标
   - CTR、CVR、GMV
   - 用户留存、活跃度
   - 推荐多样性、覆盖率

2. 系统指标
   - QPS、延迟 P99/P95
   - 错误率、超时率
   - 资源使用率

3. 模型指标
   - 预测分数分布
   - AUC、LogLoss
   - 特征分布漂移

4. 数据指标
   - 特征覆盖率
   - 数据新鲜度
   - 样本分布
```

### 1.2 监控架构

```python
from typing import List, Dict, Any, Optional
from collections import defaultdict
from datetime import datetime, timedelta
import numpy as np
import json


class MonitoringSystem:
    """
    推荐系统监控框架
    """

    def __init__(self):
        self.metrics_store = defaultdict(list)
        self.alert_rules = []
        self.alert_handlers = []

    def register_metric(self, name: str, value: float,
                       tags: Dict[str, str] = None,
                       timestamp: datetime = None):
        """
        注册指标

        参数:
            name: 指标名称
            value: 指标值
            tags: 标签（如服务名、模型版本）
            timestamp: 时间戳
        """
        if timestamp is None:
            timestamp = datetime.now()

        metric = {
            'name': name,
            'value': value,
            'tags': tags or {},
            'timestamp': timestamp.isoformat()
        }

        self.metrics_store[name].append(metric)

    def add_alert_rule(self, rule):
        """添加告警规则"""
        self.alert_rules.append(rule)

    def add_alert_handler(self, handler):
        """添加告警处理器"""
        self.alert_handlers.append(handler)

    def check_alerts(self):
        """检查告警"""
        alerts = []

        for rule in self.alert_rules:
            if rule.check(self.metrics_store):
                alert = rule.create_alert()
                alerts.append(alert)

                # 触发处理器
                for handler in self.alert_handlers:
                    handler.handle(alert)

        return alerts

    def get_metric_stats(self, name: str,
                        window_minutes: int = 60) -> Dict:
        """
        获取指标统计
        """
        if name not in self.metrics_store:
            return {}

        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        values = []
        for metric in self.metrics_store[name]:
            ts = datetime.fromisoformat(metric['timestamp'])
            if ts >= cutoff:
                values.append(metric['value'])

        if not values:
            return {}

        return {
            'count': len(values),
            'mean': np.mean(values),
            'std': np.std(values),
            'min': np.min(values),
            'max': np.max(values),
            'p50': np.percentile(values, 50),
            'p95': np.percentile(values, 95),
            'p99': np.percentile(values, 99)
        }
```

```python
class BusinessMetricsCollector:
    """
    业务指标收集器
    """

    def __init__(self):
        self.impressions = []
        self.clicks = []
        self.conversions = []

    def log_impression(self, user_id: int, item_id: int,
                      position: int, context: Dict = None):
        """记录曝光"""
        self.impressions.append({
            'user_id': user_id,
            'item_id': item_id,
            'position': position,
            'timestamp': datetime.now(),
            'context': context or {}
        })

    def log_click(self, user_id: int, item_id: int,
                 position: int, context: Dict = None):
        """记录点击"""
        self.clicks.append({
            'user_id': user_id,
            'item_id': item_id,
            'position': position,
            'timestamp': datetime.now(),
            'context': context or {}
        })

    def log_conversion(self, user_id: int, item_id: int,
                      value: float, context: Dict = None):
        """记录转化"""
        self.conversions.append({
            'user_id': user_id,
            'item_id': item_id,
            'value': value,
            'timestamp': datetime.now(),
            'context': context or {}
        })

    def compute_ctr(self, window_minutes: int = 60) -> float:
        """计算 CTR"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        recent_impressions = [i for i in self.impressions
                            if i['timestamp'] >= cutoff]
        recent_clicks = [c for c in self.clicks
                        if c['timestamp'] >= cutoff]

        if not recent_impressions:
            return 0

        return len(recent_clicks) / len(recent_impressions)

    def compute_cvr(self, window_minutes: int = 60) -> float:
        """计算 CVR"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        recent_clicks = [c for c in self.clicks
                        if c['timestamp'] >= cutoff]
        recent_conversions = [c for c in self.conversions
                             if c['timestamp'] >= cutoff]

        if not recent_clicks:
            return 0

        return len(recent_conversions) / len(recent_clicks)

    def compute_position_ctr(self, window_minutes: int = 60) -> Dict[int, float]:
        """计算各位置的 CTR"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        position_impressions = defaultdict(int)
        position_clicks = defaultdict(int)

        for imp in self.impressions:
            if imp['timestamp'] >= cutoff:
                position_impressions[imp['position']] += 1

        for click in self.clicks:
            if click['timestamp'] >= cutoff:
                position_clicks[click['position']] += 1

        position_ctr = {}
        for pos in position_impressions:
            if position_impressions[pos] > 0:
                position_ctr[pos] = position_clicks[pos] / position_impressions[pos]

        return position_ctr

    def compute_coverage(self, window_minutes: int = 60,
                        total_items: int = 10000) -> float:
        """计算推荐覆盖率"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        recommended_items = set()
        for imp in self.impressions:
            if imp['timestamp'] >= cutoff:
                recommended_items.add(imp['item_id'])

        return len(recommended_items) / total_items


class DiversityMonitor:
    """
    多样性监控
    """

    def __init__(self):
        self.recommendations = []

    def log_recommendation(self, user_id: int, items: List[int],
                          categories: Dict[int, int]):
        """记录推荐列表"""
        self.recommendations.append({
            'user_id': user_id,
            'items': items,
            'categories': [categories.get(i, -1) for i in items],
            'timestamp': datetime.now()
        })

    def compute_category_diversity(self, window_minutes: int = 60) -> float:
        """计算类别多样性"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        all_categories = []
        for rec in self.recommendations:
            if rec['timestamp'] >= cutoff:
                all_categories.extend(rec['categories'])

        if not all_categories:
            return 0

        # 熵作为多样性度量
        from collections import Counter
        counts = Counter(all_categories)
        total = len(all_categories)

        entropy = -sum((c/total) * np.log(c/total + 1e-10)
                      for c in counts.values())

        # 归一化
        max_entropy = np.log(len(counts))

        return entropy / max_entropy if max_entropy > 0 else 0

    def compute_ild(self, item_features: np.ndarray,
                   window_minutes: int = 60) -> float:
        """
        计算列表内多样性 (Intra-List Diversity)
        """
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        ild_scores = []

        for rec in self.recommendations:
            if rec['timestamp'] >= cutoff and len(rec['items']) > 1:
                items = rec['items']
                distances = []

                for i in range(len(items)):
                    for j in range(i+1, len(items)):
                        feat_i = item_features[items[i]]
                        feat_j = item_features[items[j]]
                        dist = 1 - np.dot(feat_i, feat_j) / (
                            np.linalg.norm(feat_i) * np.linalg.norm(feat_j) + 1e-10
                        )
                        distances.append(dist)

                ild_scores.append(np.mean(distances))

        return np.mean(ild_scores) if ild_scores else 0
```

```python
import time
from functools import wraps


class PerformanceMonitor:
    """
    性能监控
    """

    def __init__(self):
        self.latencies = defaultdict(list)
        self.error_counts = defaultdict(int)
        self.request_counts = defaultdict(int)

    def track_latency(self, operation: str):
        """
        延迟追踪装饰器
        """
        def decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                start_time = time.time()
                try:
                    result = func(*args, **kwargs)
                    latency = (time.time() - start_time) * 1000  # ms
                    self.latencies[operation].append(latency)
                    self.request_counts[operation] += 1
                    return result
                except Exception as e:
                    self.error_counts[operation] += 1
                    raise
            return wrapper
        return decorator

    def get_latency_stats(self, operation: str) -> Dict:
        """获取延迟统计"""
        if operation not in self.latencies:
            return {}

        latencies = self.latencies[operation]
        if not latencies:
            return {}

        return {
            'count': len(latencies),
            'mean': np.mean(latencies),
            'p50': np.percentile(latencies, 50),
            'p95': np.percentile(latencies, 95),
            'p99': np.percentile(latencies, 99),
            'max': np.max(latencies)
        }

    def get_error_rate(self, operation: str) -> float:
        """获取错误率"""
        if operation not in self.request_counts:
            return 0

        total = self.request_counts[operation]
        errors = self.error_counts[operation]

        return errors / total if total > 0 else 0


class QPSMonitor:
    """
    QPS 监控
    """

    def __init__(self, window_size: int = 60):
        self.window_size = window_size
        self.requests = []

    def record_request(self, timestamp: datetime = None):
        """记录请求"""
        if timestamp is None:
            timestamp = datetime.now()
        self.requests.append(timestamp)

        # 清理旧记录
        cutoff = datetime.now() - timedelta(seconds=self.window_size)
        self.requests = [r for r in self.requests if r >= cutoff]

    def get_qps(self) -> float:
        """获取当前 QPS"""
        return len(self.requests) / self.window_size
```

```python
class ModelPerformanceMonitor:
    """
    模型性能监控
    """

    def __init__(self):
        self.predictions = []
        self.labels = []

    def log_prediction(self, prediction: float, label: int = None,
                      user_id: int = None, item_id: int = None):
        """记录预测"""
        self.predictions.append({
            'value': prediction,
            'label': label,
            'user_id': user_id,
            'item_id': item_id,
            'timestamp': datetime.now()
        })

        if label is not None:
            self.labels.append(label)

    def compute_auc(self, window_minutes: int = 60) -> float:
        """计算 AUC"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        preds = []
        labels = []

        for p in self.predictions:
            if p['timestamp'] >= cutoff and p['label'] is not None:
                preds.append(p['value'])
                labels.append(p['label'])

        if len(set(labels)) < 2:
            return 0.5

        from sklearn.metrics import roc_auc_score
        return roc_auc_score(labels, preds)

    def compute_logloss(self, window_minutes: int = 60) -> float:
        """计算 LogLoss"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        preds = []
        labels = []

        for p in self.predictions:
            if p['timestamp'] >= cutoff and p['label'] is not None:
                preds.append(np.clip(p['value'], 1e-10, 1-1e-10))
                labels.append(p['label'])

        if not preds:
            return 0

        from sklearn.metrics import log_loss
        return log_loss(labels, preds)

    def get_prediction_distribution(self, window_minutes: int = 60) -> Dict:
        """获取预测分数分布"""
        now = datetime.now()
        cutoff = now - timedelta(minutes=window_minutes)

        values = [p['value'] for p in self.predictions
                 if p['timestamp'] >= cutoff]

        if not values:
            return {}

        return {
            'mean': np.mean(values),
            'std': np.std(values),
            'min': np.min(values),
            'max': np.max(values),
            'histogram': np.histogram(values, bins=10)[0].tolist()
        }


class FeatureDriftMonitor:
    """
    特征漂移监控
    """

    def __init__(self):
        self.baseline_features = {}
        self.current_features = defaultdict(list)

    def set_baseline(self, feature_name: str, values: np.ndarray):
        """设置基线"""
        self.baseline_features[feature_name] = {
            'mean': np.mean(values),
            'std': np.std(values),
            'histogram': np.histogram(values, bins=20)[0]
        }

    def record_feature(self, feature_name: str, value: float):
        """记录特征值"""
        self.current_features[feature_name].append(value)

    def compute_psi(self, feature_name: str) -> float:
        """
        计算人口稳定性指数 (PSI)

        PSI > 0.1: 轻微漂移
        PSI > 0.25: 显著漂移
        """
        if feature_name not in self.baseline_features:
            return 0

        baseline = self.baseline_features[feature_name]
        current_values = self.current_features[feature_name]

        if not current_values:
            return 0

        # 计算当前分布
        current_hist, _ = np.histogram(current_values, bins=20)

        # 归一化
        baseline_pct = baseline['histogram'] / baseline['histogram'].sum()
        current_pct = current_hist / current_hist.sum()

        # PSI
        psi = 0
        for b, c in zip(baseline_pct, current_pct):
            if b > 0 and c > 0:
                psi += (c - b) * np.log(c / b)

        return psi

    def check_drift(self, threshold: float = 0.1) -> List[Dict]:
        """检查漂移"""
        drifted_features = []

        for feature_name in self.baseline_features:
            psi = self.compute_psi(feature_name)

            if psi > threshold:
                drifted_features.append({
                    'feature': feature_name,
                    'psi': psi,
                    'severity': 'high' if psi > 0.25 else 'medium'
                })

        return drifted_features
```

## 5. 告警系统

### 5.1 告警规则

```python
class AlertRule:
    """
    告警规则基类
    """

    def __init__(self, name: str, severity: str = 'warning'):
        self.name = name
        self.severity = severity
        self.last_triggered = None

    def check(self, metrics_store: Dict) -> bool:
        raise NotImplementedError

    def create_alert(self) -> Dict:
        return {
            'rule': self.name,
            'severity': self.severity,
            'timestamp': datetime.now().isoformat(),
            'message': ''
        }


class ThresholdAlertRule(AlertRule):
    """
    阈值告警规则
    """

    def __init__(self, name: str, metric_name: str,
                 threshold: float, comparison: str = 'gt',
                 severity: str = 'warning'):
        super().__init__(name, severity)
        self.metric_name = metric_name
        self.threshold = threshold
        self.comparison = comparison

    def check(self, metrics_store: Dict) -> bool:
        if self.metric_name not in metrics_store:
            return False

        values = [m['value'] for m in metrics_store[self.metric_name]]
        if not values:
            return False

        current_value = values[-1]

        if self.comparison == 'gt':
            triggered = current_value > self.threshold
        elif self.comparison == 'lt':
            triggered = current_value < self.threshold
        else:
            triggered = False

        return triggered

    def create_alert(self) -> Dict:
        alert = super().create_alert()
        alert['message'] = f"{self.metric_name} 超过阈值 {self.threshold}"
        return alert


class AnomalyAlertRule(AlertRule):
    """
    异常检测告警规则
    """

    def __init__(self, name: str, metric_name: str,
                 n_std: float = 3.0, window_size: int = 100,
                 severity: str = 'warning'):
        super().__init__(name, severity)
        self.metric_name = metric_name
        self.n_std = n_std
        self.window_size = window_size

    def check(self, metrics_store: Dict) -> bool:
        if self.metric_name not in metrics_store:
            return False

        values = [m['value'] for m in metrics_store[self.metric_name]]
        if len(values) < self.window_size:
            return False

        recent = values[-self.window_size:]
        mean = np.mean(recent[:-1])
        std = np.std(recent[:-1])

        if std == 0:
            return False

        current = recent[-1]
        z_score = abs(current - mean) / std

        return z_score > self.n_std


class RateOfChangeAlertRule(AlertRule):
    """
    变化率告警规则
    """

    def __init__(self, name: str, metric_name: str,
                 max_change_rate: float = 0.5,
                 severity: str = 'warning'):
        super().__init__(name, severity)
        self.metric_name = metric_name
        self.max_change_rate = max_change_rate

    def check(self, metrics_store: Dict) -> bool:
        if self.metric_name not in metrics_store:
            return False

        values = [m['value'] for m in metrics_store[self.metric_name]]
        if len(values) < 2:
            return False

        current = values[-1]
        previous = values[-2]

        if previous == 0:
            return False

        change_rate = abs(current - previous) / previous

        return change_rate > self.max_change_rate
```

```python
class AlertHandler:
    """
    告警处理器基类
    """

    def handle(self, alert: Dict):

class LogAlertHandler(AlertHandler):
    """
    日志告警处理器

class EmailAlertHandler(AlertHandler):

    邮件告警处理器
    """

class SlackAlertHandler(AlertHandler):

    Slack 告警处理器
    """

    def __init__(self, webhook_url: str):
        self.webhook_url = webhook_url

    def handle(self, alert: Dict):
        # 发送到 Slack

            'text': f"*{alert['severity'].upper()}*: {alert['message']}"
        }
        print(f"发送到 Slack: {message}")


class AlertManager:
    """
    告警管理器
    """

    def __init__(self):
        self.rules = []
        self.handlers = []
        self.alert_history = []
        self.cooldown_minutes = 15
        self.last_alert_times = {}

    def add_rule(self, rule: AlertRule):
        self.rules.append(rule)

    def add_handler(self, handler: AlertHandler):
        self.handlers.append(handler)

    def check_and_alert(self, metrics_store: Dict):
        """检查并触发告警"""
        for rule in self.rules:
            # 检查冷却期
            rule_key = rule.name
            if rule_key in self.last_alert_times:
                time_since = datetime.now() - self.last_alert_times[rule_key]
                if time_since < timedelta(minutes=self.cooldown_minutes):
                    continue

            if rule.check(metrics_store):
                alert = rule.create_alert()
                self.alert_history.append(alert)
                self.last_alert_times[rule_key] = datetime.now()

                # 触发处理器
                for handler in self.handlers:
                    try:
                        handler.handle(alert)
                    except Exception as e:
                        print(f"告警处理失败: {e}")
```

## 6. 学习总结

### 6.1 核心要点

1. **多维监控**: 业务、系统、模型、数据
2. **实时性**: 秒级/分钟级监控
3. **告警分级**: 不同严重程度不同处理
4. **可视化**: Dashboard 展示

### 6.2 监控指标分类

```
类别        关键指标
──────────────────────────────────────
业务        CTR, CVR, 覆盖率, 多样性
系统        QPS, 延迟P99, 错误率
模型        AUC, LogLoss, 预测分布
数据        特征覆盖率, PSI, 数据新鲜度
```

### 6.3 最佳实践

```
1. 监控先行: 上线前建立监控
2. 告警分级: 区分紧急和普通
3. 避免告警风暴: 合理设置冷却期
4. 定期回顾: 调整阈值和规则
```
