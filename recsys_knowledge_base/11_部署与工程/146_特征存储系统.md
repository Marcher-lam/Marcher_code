# 特征存储系统 学习文档

## 1. 特征存储概述

### 1.1 为什么需要特征存储？

```
特征存储 (Feature Store):

- 统一管理特征的存储和服务
- 解决特征一致性问题
- 支持离线训练和在线推理

问题场景:
1. 训练和推理使用不同的特征计算逻辑
2. 特征重复开发，浪费资源
3. 特征血缘不清晰，难以追溯
4. 特征更新不及时

解决方案:
- 离线特征: 存储在数据仓库 (Hive/Parquet)
- 在线特征: 存储在 KV 存储 (Redis/HBase)
- 特征注册: 统一的特征元数据管理
```

### 1.2 特征类型

```python
"""
特征类型:

1. 离线特征 (Batch Features):
   - 历史聚合特征
   - T+1 更新
   - 例: 用户过去30天点击数

2. 近线特征 (Near-line Features):
   - 流式计算
   - 分钟级延迟
   - 例: 用户最近1小时行为

3. 实时特征 (Real-time Features):
   - 请求时计算
   - 毫秒级延迟
   - 例: 当前请求上下文
"""

from typing import Dict, List, Any, Optional
from dataclasses import dataclass
from datetime import datetime
import json
import numpy as np


@dataclass
class FeatureDefinition:
    """特征定义"""
    name: str                    # 特征名
    dtype: str                   # 数据类型
    description: str             # 描述
    entity: str                  # 关联实体 (user/item/context)
    source: str                  # 数据源
    transform: str               # 转换逻辑
    update_freq: str             # 更新频率 (daily/hourly/realtime)
    ttl: int                     # 过期时间 (秒)
    default_value: Any           # 默认值


class FeatureRegistry:
    """
    特征注册中心

    管理所有特征的元数据
    """

    def __init__(self):
        self.features: Dict[str, FeatureDefinition] = {}

    def register(self, feature_def: FeatureDefinition):
        """注册特征"""
        self.features[feature_def.name] = feature_def
        print(f"Registered feature: {feature_def.name}")

    def get_feature(self, name: str) -> Optional[FeatureDefinition]:
        """获取特征定义"""
        return self.features.get(name)

    def list_features(self, entity: str = None) -> List[FeatureDefinition]:
        """列出特征"""
        features = list(self.features.values())
        if entity:
            features = [f for f in features if f.entity == entity]
        return features

    def export_schema(self) -> Dict:
        """导出特征 Schema"""
        return {
            name: {
                'dtype': f.dtype,
                'entity': f.entity,
                'description': f.description,
                'update_freq': f.update_freq,
                'default_value': str(f.default_value)
            }
            for name, f in self.features.items()
        }
```

## 2. 离线特征存储

### 2.1 批量特征计算

```python
import pandas as pd
from typing import Callable


class BatchFeatureStore:
    """
    批量特征存储

    用于离线特征的计算和存储
    """

    def __init__(self, registry: FeatureRegistry):
        self.registry = registry

    def compute_features(self,
                        df: pd.DataFrame,
                        feature_names: List[str]) -> pd.DataFrame:
        """
        计算特征

        根据特征定义计算
        """
        result = df.copy()

        for name in feature_names:
            feature_def = self.registry.get_feature(name)
            if feature_def is None:
                print(f"Warning: Feature {name} not found")
                continue

            # 执行转换
            result[name] = self._apply_transform(
                df,
                feature_def.source,
                feature_def.transform
            )

        return result

    def _apply_transform(self,
                        df: pd.DataFrame,
                        source: str,
                        transform: str) -> pd.Series:
        """应用特征转换"""
        # 简化的转换逻辑
        # 实际中可以使用表达式引擎或自定义函数

        if transform == 'count':
            return df.groupby(source).cumcount()
        elif transform == 'sum':
            return df[source].expanding().sum()
        elif transform == 'mean':
            return df[source].expanding().mean()
        else:
            return df.get(source, pd.Series(index=df.index))

    def save_to_parquet(self,
                       df: pd.DataFrame,
                       path: str,
                       partition_cols: List[str] = None):
        """保存到 Parquet"""
        if partition_cols:
            df.to_parquet(path, partition_cols=partition_cols, index=False)
        else:
            df.to_parquet(path, index=False)
        print(f"Saved {len(df)} records to {path}")

    def load_from_parquet(self, path: str) -> pd.DataFrame:
        """加载 Parquet"""
        return pd.read_parquet(path)


class UserFeatureComputer:
    """
    用户特征计算器
    """

    def __init__(self):
        self.feature_functions = {
            'user_click_count_7d': self._click_count_7d,
            'user_click_count_30d': self._click_count_30d,
            'user_avg_price': self._avg_price,
            'user_preferred_category': self._preferred_category,
            'user_active_days': self._active_days,
        }

    def compute(self,
               interactions: pd.DataFrame,
               feature_names: List[str]) -> pd.DataFrame:
        """
        计算用户特征

        interactions: 用户行为数据
        feature_names: 要计算的特征列表
        """
        result = pd.DataFrame({'user_id': interactions['user_id'].unique()})

        for name in feature_names:
            if name in self.feature_functions:
                result = result.merge(
                    self.feature_functions[name](interactions),
                    on='user_id',
                    how='left'
                )

        return result

    def _click_count_7d(self, interactions: pd.DataFrame) -> pd.DataFrame:
        """过去7天点击数"""
        # 简化实现
        return interactions.groupby('user_id').size().reset_index(
            name='user_click_count_7d'
        )

    def _click_count_30d(self, interactions: pd.DataFrame) -> pd.DataFrame:
        """过去30天点击数"""
        return interactions.groupby('user_id').size().reset_index(
            name='user_click_count_30d'
        )

    def _avg_price(self, interactions: pd.DataFrame) -> pd.DataFrame:
        """平均消费"""
        if 'price' not in interactions.columns:
            return pd.DataFrame(columns=['user_id', 'user_avg_price'])

        return interactions.groupby('user_id')['price'].mean().reset_index(
            name='user_avg_price'
        )

    def _preferred_category(self, interactions: pd.DataFrame) -> pd.DataFrame:
        """偏好类目"""
        if 'category' not in interactions.columns:
            return pd.DataFrame(columns=['user_id', 'user_preferred_category'])

        return interactions.groupby('user_id')['category'].agg(
            lambda x: x.mode()[0] if len(x.mode()) > 0 else 'unknown'
        ).reset_index(name='user_preferred_category')

    def _active_days(self, interactions: pd.DataFrame) -> pd.DataFrame:
        """活跃天数"""
        if 'date' not in interactions.columns:
            return pd.DataFrame(columns=['user_id', 'user_active_days'])

        return interactions.groupby('user_id')['date'].nunique().reset_index(
            name='user_active_days'
        )


class ItemFeatureComputer:
    """
    物品特征计算器
    """

    def compute(self,
               items: pd.DataFrame,
               interactions: pd.DataFrame,
               feature_names: List[str]) -> pd.DataFrame:
        """计算物品特征"""
        result = items.copy()

        for name in feature_names:
            if name == 'item_click_count':
                click_count = interactions.groupby('item_id').size().reset_index(
                    name='item_click_count'
                )
                result = result.merge(click_count, on='item_id', how='left')

            elif name == 'item_ctr':
                # 曝光和点击
                if 'is_click' in interactions.columns:
                    ctr = interactions.groupby('item_id').agg({
                        'is_click': ['sum', 'count']
                    }).reset_index()
                    ctr.columns = ['item_id', 'clicks', 'exposures']
                    ctr['item_ctr'] = ctr['clicks'] / ctr['exposures']
                    result = result.merge(
                        ctr[['item_id', 'item_ctr']],
                        on='item_id',
                        how='left'
                    )

            elif name == 'item_age_days':
                if 'publish_time' in items.columns:
                    result['item_age_days'] = (
                        datetime.now() - pd.to_datetime(items['publish_time'])
                    ).dt.days

        return result
```

## 3. 在线特征服务

### 3.1 Redis 特征存储

```python
import redis
import json
import pickle
from typing import Dict, List, Any


class OnlineFeatureStore:
    """
    在线特征存储

    使用 Redis 存储实时特征
    """

    def __init__(self,
                 host: str = 'localhost',
                 port: int = 6379,
                 db: int = 0,
                 prefix: str = 'feat'):
        self.client = redis.Redis(host=host, port=port, db=db)
        self.prefix = prefix

    def _make_key(self, entity_type: str, entity_id: str) -> str:
        """生成 Redis key"""
        return f"{self.prefix}:{entity_type}:{entity_id}"

    def set_features(self,
                    entity_type: str,
                    entity_id: str,
                    features: Dict[str, Any],
                    ttl: int = None):
        """
        设置特征

        entity_type: user / item / context
        entity_id: 实体ID
        features: 特征字典
        ttl: 过期时间（秒）
        """
        key = self._make_key(entity_type, entity_id)
        value = json.dumps(features)

        if ttl:
            self.client.setex(key, ttl, value)
        else:
            self.client.set(key, value)

    def get_features(self,
                    entity_type: str,
                    entity_id: str,
                    feature_names: List[str] = None) -> Dict[str, Any]:
        """
        获取特征
        """
        key = self._make_key(entity_type, entity_id)
        value = self.client.get(key)

        if value is None:
            return {}

        features = json.loads(value)

        if feature_names:
            return {k: v for k, v in features.items() if k in feature_names}

        return features

    def mget_features(self,
                     entity_type: str,
                     entity_ids: List[str],
                     feature_names: List[str] = None) -> Dict[str, Dict[str, Any]]:
        """
        批量获取特征
        """
        keys = [self._make_key(entity_type, eid) for eid in entity_ids]
        values = self.client.mget(keys)

        results = {}
        for entity_id, value in zip(entity_ids, values):
            if value is not None:
                features = json.loads(value)
                if feature_names:
                    features = {k: v for k, v in features.items() if k in feature_names}
                results[entity_id] = features

        return results

    def delete_features(self, entity_type: str, entity_id: str):
        """删除特征"""
        key = self._make_key(entity_type, entity_id)
        self.client.delete(key)

    def set_vector(self,
                  entity_type: str,
                  entity_id: str,
                  vector: np.ndarray,
                  ttl: int = None):
        """
        存储向量特征

        使用 Redis 的 bytes 存储
        """
        key = f"{self._make_key(entity_type, entity_id)}:vec"
        value = pickle.dumps(vector)

        if ttl:
            self.client.setex(key, ttl, value)
        else:
            self.client.set(key, value)

    def get_vector(self, entity_type: str, entity_id: str) -> Optional[np.ndarray]:
        """获取向量特征"""
        key = f"{self._make_key(entity_type, entity_id)}:vec"
        value = self.client.get(key)

        if value is None:
            return None

        return pickle.loads(value)
```

### 3.2 特征服务 API

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Dict, Any, Optional
import asyncio


class FeatureRequest(BaseModel):
    """特征请求"""
    entity_type: str
    entity_ids: List[str]
    feature_names: Optional[List[str]] = None


class FeatureResponse(BaseModel):
    """特征响应"""
    features: Dict[str, Dict[str, Any]]
    latency_ms: float


class FeatureService:
    """
    特征服务

    提供特征查询 API
    """

    def __init__(self, online_store: OnlineFeatureStore):
        self.online_store = online_store

    async def get_features(self, request: FeatureRequest) -> FeatureResponse:
        """获取特征"""
        import time
        start = time.time()

        features = self.online_store.mget_features(
            request.entity_type,
            request.entity_ids,
            request.feature_names
        )

        latency = (time.time() - start) * 1000

        return FeatureResponse(features=features, latency_ms=latency)


def create_feature_app() -> FastAPI:
    """创建特征服务应用"""
    app = FastAPI(title="Feature Service")

    # 初始化存储
    online_store = OnlineFeatureStore()
    feature_service = FeatureService(online_store)

    @app.post("/features", response_model=FeatureResponse)
    async def get_features(request: FeatureRequest):
        return await feature_service.get_features(request)

    @app.get("/health")
    async def health():
        return {"status": "healthy"}

    return app
```

## 4. 特征同步

### 4.1 离线到在线同步

```python
class FeatureSyncPipeline:
    """
    特征同步流水线

    将离线特征同步到在线存储
    """

    def __init__(self,
                 offline_store: BatchFeatureStore,
                 online_store: OnlineFeatureStore,
                 registry: FeatureRegistry):
        self.offline_store = offline_store
        self.online_store = online_store
        self.registry = registry

    def sync_features(self,
                     entity_type: str,
                     feature_df: pd.DataFrame,
                     id_column: str = 'user_id'):
        """
        同步特征

        将 DataFrame 中的特征写入在线存储
        """
        # 获取需要同步的特征
        feature_names = [col for col in feature_df.columns if col != id_column]

        # 获取 TTL
        ttl = self._get_max_ttl(feature_names)

        print(f"Syncing {len(feature_df)} {entity_type} records...")

        # 批量写入
        batch_size = 1000
        for i in range(0, len(feature_df), batch_size):
            batch = feature_df.iloc[i:i + batch_size]

            for _, row in batch.iterrows():
                entity_id = str(row[id_column])
                features = {name: row[name] for name in feature_names}

                # 处理 NaN
                features = {
                    k: self._serialize_value(v)
                    for k, v in features.items()
                    if pd.notna(v)
                }

                self.online_store.set_features(
                    entity_type, entity_id, features, ttl
                )

            if (i + batch_size) % 10000 == 0:
                print(f"  Synced {i + batch_size} records...")

        print(f"Sync completed: {len(feature_df)} records")

    def _get_max_ttl(self, feature_names: List[str]) -> int:
        """获取最大 TTL"""
        max_ttl = 86400  # 默认1天

        for name in feature_names:
            feature_def = self.registry.get_feature(name)
            if feature_def and feature_def.ttl > max_ttl:
                max_ttl = feature_def.ttl

        return max_ttl

    def _serialize_value(self, value: Any) -> Any:
        """序列化值"""
        if isinstance(value, (np.integer, np.floating)):
            return value.item()
        elif isinstance(value, np.ndarray):
            return value.tolist()
        return value


class IncrementalFeatureUpdater:
    """
    增量特征更新器

    处理流式特征更新
    """

    def __init__(self, online_store: OnlineFeatureStore):
        self.online_store = online_store

    def update_user_features(self,
                            user_id: str,
                            action: str,
                            item_features: Dict[str, Any]):
        """
        更新用户特征

        基于用户行为增量更新
        """
        # 获取当前特征
        current = self.online_store.get_features('user', user_id)

        # 更新计数
        if action == 'click':
            current['click_count'] = current.get('click_count', 0) + 1

            # 更新偏好类目
            if 'category' in item_features:
                # 简化: 直接更新最后点击类目
                current['last_click_category'] = item_features['category']

        elif action == 'purchase':
            current['purchase_count'] = current.get('purchase_count', 0) + 1

            if 'price' in item_features:
                current['total_purchase'] = current.get('total_purchase', 0) + item_features['price']

        # 写回
        self.online_store.set_features('user', user_id, current)
```

### 5.1 特征监控器

```python
class FeatureMonitor:
    """
    特征监控器

    监控特征质量和分布
    """

    def __init__(self, registry: FeatureRegistry):
        self.registry = registry
        self.statistics = {}

    def compute_statistics(self,
                          feature_df: pd.DataFrame,
                          feature_names: List[str]):
        """计算特征统计"""
        for name in feature_names:
            if name not in feature_df.columns:
                continue

            col = feature_df[name]

            stats = {
                'count': len(col),
                'null_rate': col.isna().mean(),
                'unique_count': col.nunique(),
            }

            if col.dtype in [np.float64, np.int64]:
                stats.update({
                    'mean': col.mean(),
                    'std': col.std(),
                    'min': col.min(),
                    'max': col.max(),
                    'median': col.median(),
                })

            self.statistics[name] = stats

    def check_feature_quality(self,
                             feature_df: pd.DataFrame,
                             feature_names: List[str]) -> Dict[str, List[str]]:
        """
        检查特征质量

        返回各特征的问题列表
        """
        issues = {}

        for name in feature_names:
            if name not in feature_df.columns:
                issues[name] = ["Feature not found"]
                continue

            col = feature_df[name]
            problems = []

            # 检查空值率
            null_rate = col.isna().mean()
            if null_rate > 0.5:
                problems.append(f"High null rate: {null_rate:.2%}")

            # 检查唯一性
            unique_rate = col.nunique() / len(col)
            if unique_rate < 0.01:
                problems.append(f"Low variance: unique rate {unique_rate:.2%}")

            # 检查异常值
            if col.dtype in [np.float64, np.int64]:
                q1, q3 = col.quantile([0.25, 0.75])
                iqr = q3 - q1
                outlier_rate = ((col < q1 - 1.5 * iqr) | (col > q3 + 1.5 * iqr)).mean()
                if outlier_rate > 0.1:
                    problems.append(f"High outlier rate: {outlier_rate:.2%}")

            if problems:
                issues[name] = problems

        return issues

    def detect_distribution_shift(self,
                                  feature_name: str,
                                  current_df: pd.DataFrame,
                                  baseline_stats: Dict) -> float:
        """
        检测分布漂移

        返回漂移分数 (0-1)
        """
        if feature_name not in current_df.columns:
            return 1.0

        current = current_df[feature_name]

        # PSI (Population Stability Index)
        # 简化实现

        if current.dtype not in [np.float64, np.int64]:
            return 0.0

        # 分箱
        n_bins = 10
        bins = np.linspace(baseline_stats.get('min', 0),
                          baseline_stats.get('max', 1),
                          n_bins + 1)

        # 计算分布
        baseline_counts = np.histogram(
            np.random.normal(
                baseline_stats.get('mean', 0.5),
                baseline_stats.get('std', 0.1),
                10000
            ),
            bins=bins
        )[0]

        current_counts = np.histogram(current.dropna(), bins=bins)[0]

        # PSI
        psi = 0.0
        for b, c in zip(baseline_counts, current_counts):
            if b > 0 and c > 0:
                p_b = b / baseline_counts.sum()
                p_c = c / current_counts.sum()
                psi += (p_c - p_b) * np.log(p_c / p_b)

        return min(1.0, psi / 0.25)  # 归一化
```
