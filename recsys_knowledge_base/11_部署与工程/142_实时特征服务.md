# 推荐系统实时特征 学习文档

## 1. 基础认知

### 1.1 什么是实时特征？

实时特征是指**在请求时刻动态计算**的特征，反映了用户当前状态和最新的行为。与离线特征相比，实时特征能捕捉用户的即时兴趣。

### 1.2 实时特征 vs 离线特征

| 维度 | 离线特征 | 实时特征 |
|------|----------|----------|
| 更新频率 | 天级/小时级 | 秒级/分钟级 |
| 计算位置 | 离线批处理 | 在线服务 |
| 数据新鲜度 | 较低 | 极高 |
| 复杂度 | 可较高 | 需简单 |
| 成本 | 低 | 较高 |

### 1.3 实时特征的类型

```
实时特征类型
├── 用户实时特征
│   ├── 最近点击序列
│   ├── 最近搜索词
│   ├── 当前会话行为
│   └── 实时偏好变化
├── 物品实时特征
│   ├── 实时热度
│   ├── 最近交互数
│   └── 实时转化率
└── 上下文特征
    ├── 当前时间
    ├── 地理位置
    ├── 设备信息
    └── 网络状态
```

## 2. 实时特征架构

### 2.1 整体架构

```
用户请求
    ↓
┌─────────────────────────────────┐
│         特征服务层              │
│  ┌─────────┐  ┌──────────────┐  │
│  │用户特征 │  │ 物品特征     │  │
│  │服务     │  │ 服务         │  │
│  └────┬────┘  └──────┬───────┘  │
│       ↓              ↓          │
│  ┌────────────────────────────┐ │
│  │      特征合并层            │ │
│  └────────────────────────────┘ │
└─────────────────────────────────┘
    ↓
推荐模型
    ↓
推荐结果
```

### 2.2 数据流

```python
from typing import Dict, List, Any
from dataclasses import dataclass
from datetime import datetime
import json


@dataclass
class FeatureRequest:
    """特征请求"""
    user_id: str
    item_ids: List[str]
    context: Dict[str, Any]
    request_id: str
    timestamp: datetime


@dataclass
class FeatureResponse:
    """特征响应"""
    user_features: Dict[str, Any]
    item_features: Dict[str, Dict[str, Any]]
    context_features: Dict[str, Any]
    latency_ms: float


class RealtimeFeatureService:
    """
    实时特征服务
    """

    def __init__(self, redis_client, db_client):
        self.redis = redis_client
        self.db = db_client

        # 特征提供者
        self.user_feature_provider = UserFeatureProvider(redis_client, db_client)
        self.item_feature_provider = ItemFeatureProvider(redis_client, db_client)
        self.context_feature_provider = ContextFeatureProvider()

    def get_features(self, request: FeatureRequest) -> FeatureResponse:
        """
        获取特征

        参数:
            request: 特征请求

        返回:
            FeatureResponse: 特征响应
        """
        start_time = datetime.now()

        # 并行获取特征
        user_features = self.user_feature_provider.get_features(request.user_id)
        item_features = {}
        for item_id in request.item_ids:
            item_features[item_id] = self.item_feature_provider.get_features(item_id)

        context_features = self.context_feature_provider.get_features(request.context)

        latency = (datetime.now() - start_time).total_seconds() * 1000

        return FeatureResponse(
            user_features=user_features,
            item_features=item_features,
            context_features=context_features,
            latency_ms=latency
        )
```

## 3. 用户实时特征

### 3.1 最近行为序列

```python
from collections import deque
from typing import List, Dict
import time


class UserRealtimeBehaviorTracker:
    """
    用户实时行为追踪器

    追踪用户的最近行为序列
    """

    def __init__(self, max_seq_len=50, window_seconds=3600):
        """
        参数:
            max_seq_len: 最大序列长度
            window_seconds: 时间窗口（秒）
        """
        self.max_seq_len = max_seq_len
        self.window_seconds = window_seconds

        # 存储结构: {user_id: deque of (timestamp, action)}
        self.behaviors = {}

    def record_behavior(self, user_id: str, item_id: str,
                       action_type: str, timestamp: float = None):
        """
        记录用户行为

        参数:
            user_id: 用户 ID
            item_id: 物品 ID
            action_type: 行为类型 (click, purchase, etc.)
            timestamp: 时间戳
        """
        if timestamp is None:
            timestamp = time.time()

        if user_id not in self.behaviors:
            self.behaviors[user_id] = deque(maxlen=self.max_seq_len)

        behavior = {
            'item_id': item_id,
            'action': action_type,
            'timestamp': timestamp
        }

        self.behaviors[user_id].append(behavior)

    def get_recent_behaviors(self, user_id: str, n: int = None,
                            action_filter: str = None) -> List[Dict]:
        """
        获取最近行为

        参数:
            user_id: 用户 ID
            n: 返回数量
            action_filter: 行为类型过滤

        返回:
            行为列表
        """
        if user_id not in self.behaviors:
            return []

        current_time = time.time()
        behaviors = []

        # 反向遍历（最新的在前）
        for behavior in reversed(self.behaviors[user_id]):
            # 时间窗口过滤
            if current_time - behavior['timestamp'] > self.window_seconds:
                break

            # 行为类型过滤
            if action_filter and behavior['action'] != action_filter:
                continue

            behaviors.append(behavior)

            if n and len(behaviors) >= n:
                break

        return behaviors

    def get_recent_items(self, user_id: str, n: int = 20,
                        action_filter: str = None) -> List[str]:
        """获取最近交互的物品"""
        behaviors = self.get_recent_behaviors(user_id, n, action_filter)
        return [b['item_id'] for b in behaviors]

    def compute_session_features(self, user_id: str) -> Dict:
        """
        计算会话级特征

        返回:
            会话特征字典
        """
        behaviors = self.get_recent_behaviors(user_id)

        if not behaviors:
            return {
                'session_length': 0,
                'session_duration': 0,
                'click_count': 0,
                'unique_items': 0
            }

        # 会话长度
        session_length = len(behaviors)

        # 会话时长
        session_duration = behaviors[-1]['timestamp'] - behaviors[0]['timestamp']

        # 点击数
        click_count = sum(1 for b in behaviors if b['action'] == 'click')

        # 唯一物品数
        unique_items = len(set(b['item_id'] for b in behaviors))

        return {
            'session_length': session_length,
            'session_duration': session_duration,
            'click_count': click_count,
            'unique_items': unique_items,
            'avg_time_between_actions': session_duration / max(session_length - 1, 1)
        }


class UserFeatureProvider:
    """
    用户特征提供者

    从多个数据源获取用户特征
    """

    def __init__(self, redis_client, db_client):
        self.redis = redis_client
        self.db = db_client
        self.behavior_tracker = UserRealtimeBehaviorTracker()

    def get_features(self, user_id: str) -> Dict:
        """
        获取用户特征

        参数:
            user_id: 用户 ID

        返回:
            用户特征字典
        """
        features = {}

        # 离线特征（从 Redis 缓存）
        offline_features = self._get_offline_features(user_id)
        features.update(offline_features)

        # 实时特征
        realtime_features = self._get_realtime_features(user_id)
        features.update(realtime_features)

        return features

    def _get_offline_features(self, user_id: str) -> Dict:
        """获取离线特征"""
        # 从 Redis 获取
        cache_key = f"user:offline:{user_id}"
        cached = self.redis.get(cache_key)

        if cached:
            return json.loads(cached)

        # 从数据库获取
        # 这里简化，实际会查询数据库
        return {
            'user_age': 25,
            'user_gender': 'male',
            'user_level': 3,
            'historical_ctr': 0.05
        }

    def _get_realtime_features(self, user_id: str) -> Dict:
        """获取实时特征"""
        # 最近行为
        recent_items = self.behavior_tracker.get_recent_items(user_id, n=20)

        # 会话特征
        session_features = self.behavior_tracker.compute_session_features(user_id)

        return {
            'recent_items': recent_items,
            'recent_item_count': len(recent_items),
            **session_features
        }

    def record_behavior(self, user_id: str, item_id: str, action: str):
        """记录行为"""
        self.behavior_tracker.record_behavior(user_id, item_id, action)
```

## 4. 物品实时特征

### 4.1 实时热度计算

```python
import time
from collections import defaultdict


class ItemRealtimeStats:
    """
    物品实时统计

    跟踪物品的实时热度指标
    """

    def __init__(self, window_seconds=3600):
        """
        参数:
            window_seconds: 统计时间窗口
        """
        self.window_seconds = window_seconds

        # 存储结构
        # {item_id: [(timestamp, event_type), ...]}
        self.events = defaultdict(list)

    def record_event(self, item_id: str, event_type: str, timestamp: float = None):
        """
        记录事件

        参数:
            item_id: 物品 ID
            event_type: 事件类型 (exposure, click, etc.)
            timestamp: 时间戳
        """
        if timestamp is None:
            timestamp = time.time()

        self.events[item_id].append((timestamp, event_type))

    def get_realtime_ctr(self, item_id: str) -> float:
        """
        获取实时 CTR

        返回:
            实时 CTR
        """
        current_time = time.time()

        clicks = 0
        exposures = 0

        for ts, event in self.events.get(item_id, []):
            if current_time - ts > self.window_seconds:
                continue

            if event == 'click':
                clicks += 1
            elif event == 'exposure':
                exposures += 1

        if exposures == 0:
            return 0.0

        return clicks / exposures

    def get_realtime_popularity(self, item_id: str) -> int:
        """
        获取实时热度

        返回:
            时间窗口内的点击数
        """
        current_time = time.time()
        count = 0

        for ts, event in self.events.get(item_id, []):
            if current_time - ts > self.window_seconds:
                continue
            if event == 'click':
                count += 1

        return count

    def cleanup_old_events(self):
        """清理过期事件"""
        current_time = time.time()

        for item_id in list(self.events.keys()):
            self.events[item_id] = [
                (ts, event) for ts, event in self.events[item_id]
                if current_time - ts <= self.window_seconds
            ]


class ItemFeatureProvider:
    """
    物品特征提供者
    """

    def __init__(self, redis_client, db_client):
        self.redis = redis_client
        self.db = db_client
        self.realtime_stats = ItemRealtimeStats()

    def get_features(self, item_id: str) -> Dict:
        """
        获取物品特征

        参数:
            item_id: 物品 ID

        返回:
            物品特征字典
        """
        features = {}

        # 离线特征
        offline_features = self._get_offline_features(item_id)
        features.update(offline_features)

        # 实时特征
        realtime_features = self._get_realtime_features(item_id)
        features.update(realtime_features)

        return features

    def _get_offline_features(self, item_id: str) -> Dict:
        """获取离线特征"""
        cache_key = f"item:offline:{item_id}"
        cached = self.redis.get(cache_key)

        if cached:
            return json.loads(cached)

        return {
            'category': 'electronics',
            'price': 99.99,
            'age_days': 30,
            'historical_ctr': 0.03
        }

    def _get_realtime_features(self, item_id: str) -> Dict:
        """获取实时特征"""
        return {
            'realtime_ctr': self.realtime_stats.get_realtime_ctr(item_id),
            'realtime_popularity': self.realtime_stats.get_realtime_popularity(item_id)
        }

    def record_event(self, item_id: str, event_type: str):
        """记录事件"""
        self.realtime_stats.record_event(item_id, event_type)
```

## 5. 上下文特征

### 5.1 上下文特征提取

```python
from datetime import datetime
from typing import Dict, Any


class ContextFeatureProvider:
    """
    上下文特征提供者
    """

    def get_features(self, context: Dict[str, Any]) -> Dict[str, Any]:
        """
        获取上下文特征

        参数:
            context: 原始上下文信息

        返回:
            特征字典
        """
        features = {}

        # 时间特征
        features.update(self._extract_time_features())

        # 地理特征
        if 'location' in context:
            features.update(self._extract_geo_features(context['location']))

        # 设备特征
        if 'device' in context:
            features.update(self._extract_device_features(context['device']))

        # 网络特征
        if 'network' in context:
            features.update(self._extract_network_features(context['network']))

        return features

    def _extract_time_features(self) -> Dict:
        """提取时间特征"""
        now = datetime.now()

        return {
            'hour': now.hour,
            'day_of_week': now.weekday(),
            'is_weekend': now.weekday() >= 5,
            'is_morning': 6 <= now.hour < 12,
            'is_afternoon': 12 <= now.hour < 18,
            'is_evening': 18 <= now.hour < 24,
            'is_night': 0 <= now.hour < 6,
            'minute_of_hour': now.minute,
            'is_meal_time': now.hour in [7, 8, 12, 13, 18, 19]
        }

    def _extract_geo_features(self, location: Dict) -> Dict:
        """提取地理特征"""
        return {
            'country': location.get('country', 'unknown'),
            'city': location.get('city', 'unknown'),
            'latitude': location.get('latitude', 0),
            'longitude': location.get('longitude', 0),
            'is_home': location.get('is_home', False),
            'is_work': location.get('is_work', False)
        }

    def _extract_device_features(self, device: Dict) -> Dict:
        """提取设备特征"""
        return {
            'device_type': device.get('type', 'unknown'),
            'os': device.get('os', 'unknown'),
            'app_version': device.get('app_version', '0'),
            'is_ios': device.get('os', '').lower() == 'ios',
            'is_android': device.get('os', '').lower() == 'android',
            'screen_width': device.get('screen_width', 0),
            'screen_height': device.get('screen_height', 0)
        }

    def _extract_network_features(self, network: Dict) -> Dict:
        """提取网络特征"""
        return {
            'network_type': network.get('type', 'unknown'),
            'is_wifi': network.get('type', '') == 'wifi',
            'is_mobile': network.get('type', '') == 'mobile'
        }
```

## 6. 特征存储与缓存

### 6.1 Redis 特征存储

```python
import redis
import json
from typing import Dict, Any, Optional


class FeatureCache:
    """
    特征缓存

    使用 Redis 缓存特征
    """

    def __init__(self, redis_host='localhost', redis_port=6379,
                 default_ttl=3600):
        """
        参数:
            redis_host: Redis 主机
            redis_port: Redis 端口
            default_ttl: 默认过期时间（秒）
        """
        self.redis = redis.Redis(host=redis_host, port=redis_port, decode_responses=True)
        self.default_ttl = default_ttl

    def get(self, key: str) -> Optional[Dict]:
        """获取特征"""
        value = self.redis.get(key)
        if value:
            return json.loads(value)
        return None

    def set(self, key: str, value: Dict, ttl: int = None):
        """设置特征"""
        if ttl is None:
            ttl = self.default_ttl

        self.redis.setex(key, ttl, json.dumps(value))

    def mget(self, keys: list) -> Dict[str, Dict]:
        """批量获取"""
        values = self.redis.mget(keys)
        result = {}
        for key, value in zip(keys, values):
            if value:
                result[key] = json.loads(value)
        return result

    def mset(self, mapping: Dict[str, Dict], ttl: int = None):
        """批量设置"""
        ttl = ttl or self.default_ttl

        pipe = self.redis.pipeline()
        for key, value in mapping.items():
            pipe.setex(key, ttl, json.dumps(value))
        pipe.execute()

    def delete(self, key: str):
        """删除特征"""
        self.redis.delete(key)

    def get_user_features(self, user_id: str) -> Optional[Dict]:
        """获取用户特征"""
        return self.get(f"user:features:{user_id}")

    def set_user_features(self, user_id: str, features: Dict, ttl: int = None):
        """设置用户特征"""
        self.set(f"user:features:{user_id}", features, ttl)

    def get_item_features(self, item_id: str) -> Optional[Dict]:
        """获取物品特征"""
        return self.get(f"item:features:{item_id}")

    def set_item_features(self, item_id: str, features: Dict, ttl: int = None):
        """设置物品特征"""
        self.set(f"item:features:{item_id}", features, ttl)


class FeaturePipeline:
    """
    特征流水线

    组合多个特征源
    """

    def __init__(self, cache: FeatureCache):
        self.cache = cache
        self.providers = []

    def add_provider(self, provider):
        """添加特征提供者"""
        self.providers.append(provider)

    def get_features(self, entity_id: str, entity_type: str) -> Dict:
        """
        获取特征

        参数:
            entity_id: 实体 ID
            entity_type: 实体类型 (user, item)

        返回:
            特征字典
        """
        # 先从缓存获取
        cache_key = f"{entity_type}:features:{entity_id}"
        cached = self.cache.get(cache_key)
        if cached:
            return cached

        # 从各提供者获取
        features = {}
        for provider in self.providers:
            provider_features = provider.get_features(entity_id)
            features.update(provider_features)

        # 写入缓存
        self.cache.set(cache_key, features)

        return features
```

## 7. 学习总结

### 7.1 核心要点

1. **实时性很重要**：捕捉用户的即时兴趣
2. **缓存策略**：平衡新鲜度和性能
3. **分层设计**：离线特征 + 实时特征

### 7.2 架构原则

- 实时特征计算要简单快速
- 使用缓存减少计算
- 合理设置过期时间
