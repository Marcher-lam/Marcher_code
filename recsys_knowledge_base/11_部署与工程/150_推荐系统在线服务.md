# 推荐系统在线服务 学习文档

## 1. 基础认知

### 1.1 什么是在线服务？

在线服务是推荐系统面向用户的最后一环，负责在用户请求时实时生成推荐结果。它需要满足：

- **低延迟**：通常要求 < 100ms
- **高可用**：99.99% 以上的可用性
- **高并发**：支持每秒数万甚至数十万请求
- **准确性**：提供高质量的推荐结果

### 1.2 在线服务架构

```
用户请求
    ↓
┌─────────────────────────────────────────────┐
│              API 网关 / 负载均衡              │
└─────────────────────────────────────────────┘
    ↓
┌─────────────────────────────────────────────┐
│              推荐服务层                       │
│  ┌─────────┐ ┌─────────┐ ┌─────────┐       │
│  │ 召回服务 │ │ 排序服务 │ │ 重排服务 │       │
│  └─────────┘ └─────────┘ └─────────┘       │
└─────────────────────────────────────────────┘
    ↓
┌─────────────────────────────────────────────┐
│              数据存储层                       │
│  ┌─────────┐ ┌─────────┐ ┌─────────┐       │
│  │  Redis  │ │   KV    │ │ Feature │       │
│  │  Cache  │ │  Store  │ │  Store  │       │
│  └─────────┘ └─────────┘ └─────────┘       │
└─────────────────────────────────────────────┘
```

```python
def recommend_request_handler(user_id, context):
    """
    推荐请求处理
    """

    return final_items
```

## 2. 核心组件

### 2.1 召回服务

召回服务负责从海量物品中快速筛选出候选集。

```python
import numpy as np
from typing import List, Dict
import time

class RecallService:
    """召回服务"""

    def __init__(self, recallers: List, top_k: int = 1000):
        """
        参数:
            recallers: 召回器列表
            top_k: 每个召回器返回的物品数
        """
        self.recallers = recallers
        self.top_k = top_k

    def recall(self, user_id: str, context: Dict) -> List[str]:
        """
        多路召回
        """
        all_items = {}

        for recaller in self.recallers:
            start_time = time.time()
            items = recaller.recall(user_id, context, self.top_k)
            latency = (time.time() - start_time) * 1000

            # 记录延迟
            print(f"{recaller.name} recall latency: {latency:.2f}ms")

            # 合并结果
            for item_id, score in items:
                if item_id in all_items:
                    all_items[item_id] = max(all_items[item_id], score)
                else:
                    all_items[item_id] = score

        # 排序返回
        sorted_items = sorted(all_items.items(), key=lambda x: x[1], reverse=True)
        return [item_id for item_id, _ in sorted_items[:self.top_k]]


class ItemCFRecaller:
    """基于物品协同过滤的召回器"""

    def __init__(self, name: str = "ItemCF"):
        self.name = name
        self.item_similarity = {}  # 物品相似度矩阵
        self.user_history = {}     # 用户历史行为

    def recall(self, user_id: str, context: Dict, top_k: int) -> List:
        """召回"""
        if user_id not in self.user_history:
            return []

        history_items = self.user_history[user_id]
        scores = {}

        for item_id in history_items:
            if item_id in self.item_similarity:
                for sim_item, sim_score in self.item_similarity[item_id]:
                    if sim_item not in scores:
                        scores[sim_item] = 0
                    scores[sim_item] += sim_score

        sorted_items = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return sorted_items[:top_k]


class VectorRecaller:
    """向量召回器（Embedding召回）"""

    def __init__(self, name: str = "Vector"):
        self.name = name
        self.user_embeddings = {}  # 用户 Embedding
        self.item_embeddings = {}  # 物品 Embedding
        self.index = None          # 向量索引

    def recall(self, user_id: str, context: Dict, top_k: int) -> List:
        """向量召回"""
        if user_id not in self.user_embeddings:
            return []

        user_emb = self.user_embeddings[user_id]

        # 使用向量索引检索
        scores, indices = self.index.search(
            user_emb.reshape(1, -1).astype('float32'),
            top_k
        )

        return [(idx, score) for idx, score in zip(indices[0], scores[0])]


class HotRecaller:
    """热门召回器"""

    def __init__(self, name: str = "Hot"):
        self.name = name
        self.hot_items = []  # 热门物品列表

    def recall(self, user_id: str, context: Dict, top_k: int) -> List:
        """热门召回"""
        return self.hot_items[:top_k]
```

### 2.2 排序服务

排序服务对召回物品进行精细排序。

```python
import torch
import torch.nn as nn
import numpy as np
from typing import List, Dict

class RankingService:
    """排序服务"""

    def __init__(self, model, feature_extractor, top_k: int = 100):
        """
        参数:
            model: 排序模型
            feature_extractor: 特征提取器
            top_k: 返回的物品数
        """
        self.model = model
        self.feature_extractor = feature_extractor
        self.top_k = top_k

    def rank(self, user_id: str, items: List[str],
             user_features: Dict, context: Dict) -> List:
        """
        排序
        """
        if not items:
            return []

        # 批量特征提取
        features_list = []
        for item_id in items:
            features = self.feature_extractor.extract(
                user_id, item_id, user_features, context
            )
            features_list.append(features)

        # 批量预测
        with torch.no_grad():
            features_tensor = torch.FloatTensor(features_list)
            scores = self.model(features_tensor).numpy().flatten()

        # 排序
        item_scores = list(zip(items, scores))
        item_scores.sort(key=lambda x: x[1], reverse=True)

        return item_scores[:self.top_k]


class FeatureExtractor:
    """特征提取器"""

    def __init__(self):
        self.user_feature_store = None
        self.item_feature_store = None
        self.context_feature_store = None

    def extract(self, user_id: str, item_id: str,
                user_features: Dict, context: Dict) -> np.ndarray:
        """
        提取特征
        """
        features = []

        # 用户特征
        features.extend(self._get_user_features(user_id, user_features))

        # 物品特征
        features.extend(self._get_item_features(item_id))

        # 上下文特征
        features.extend(self._get_context_features(context))

        # 交叉特征
        features.extend(self._get_cross_features(user_id, item_id, context))

        return np.array(features, dtype=np.float32)

    def _get_user_features(self, user_id: str, user_features: Dict) -> List:
        """获取用户特征"""
        # 从缓存或存储获取
        return list(user_features.values())

    def _get_item_features(self, item_id: str) -> List:
        """获取物品特征"""
        # 从物品特征存储获取
        return []

    def _get_context_features(self, context: Dict) -> List:
        """获取上下文特征"""
        return []

    def _get_cross_features(self, user_id: str, item_id: str,
                            context: Dict) -> List:
        """获取交叉特征"""
        return []
```

### 2.3 重排服务

重排服务对排序结果进行业务规则处理和多样性控制。

```python
from typing import List, Tuple
import random

class RerankService:
    """重排服务"""

    def __init__(self):
        self.rules = []
        self.diversity_config = {
            'enabled': True,
            'category_window': 3,  # 相邻 N 个物品不能同类目
            'similarity_threshold': 0.8
        }

    def add_rule(self, rule):
        """添加业务规则"""
        self.rules.append(rule)

    def rerank(self, items: List[Tuple[str, float]],
               context: Dict) -> List[Tuple[str, float]]:
        """
        重排
        """
        result = items.copy()

        # 应用业务规则
        for rule in self.rules:
            result = rule.apply(result, context)

        # 多样性控制
        if self.diversity_config['enabled']:
            result = self._apply_diversity(result)

        return result

    def _apply_diversity(self, items: List[Tuple[str, float]]) -> List:
        """多样性控制"""
        result = []
        window = []  # 滑动窗口，记录最近的物品类目

        for item_id, score in items:
            item_category = self._get_category(item_id)

            # 检查是否与窗口内的物品同类目
            if item_category not in window:
                result.append((item_id, score))
                window.append(item_category)

                # 保持窗口大小
                if len(window) > self.diversity_config['category_window']:
                    window.pop(0)
            else:
                # 同类目物品降权，放到后面
                result.append((item_id, score * 0.5))

        return result

    def _get_category(self, item_id: str) -> str:
        """获取物品类目"""
        # 从缓存获取
        return "category_" + item_id.split("_")[0]


class BusinessRule:
    """业务规则基类"""

    def apply(self, items: List, context: Dict) -> List:
        raise NotImplementedError


class FilterRule(BusinessRule):
    """过滤规则"""

    def __init__(self, filter_func):
        self.filter_func = filter_func

    def apply(self, items: List, context: Dict) -> List:
        return [(iid, score) for iid, score in items if self.filter_func(iid, context)]


class PositionRule(BusinessRule):
    """位置规则（如：某类物品只能在前3位）"""

    def __init__(self, item_type: str, max_position: int):
        self.item_type = item_type
        self.max_position = max_position

    def apply(self, items: List, context: Dict) -> List:
        # 实现位置约束逻辑
        return items
```

### 3.1 缓存策略

```python
import redis
import json
from typing import Any, Optional
import hashlib

class RecommendCache:
    """推荐结果缓存"""

    def __init__(self, redis_client, ttl: int = 300):
        """
        参数:
            redis_client: Redis 客户端
            ttl: 缓存过期时间（秒）
        """
        self.redis = redis_client
        self.ttl = ttl

    def get(self, user_id: str, scene: str, context: Dict) -> Optional[List]:
        """获取缓存"""
        cache_key = self._build_key(user_id, scene, context)
        cached = self.redis.get(cache_key)

        if cached:
            return json.loads(cached)
        return None

    def set(self, user_id: str, scene: str, context: Dict, items: List):
        """设置缓存"""
        cache_key = self._build_key(user_id, scene, context)
        self.redis.setex(cache_key, self.ttl, json.dumps(items))

    def _build_key(self, user_id: str, scene: str, context: Dict) -> str:
        """构建缓存 key"""
        # 只用稳定的上下文特征作为 key
        stable_context = {
            'device': context.get('device'),
            'time_period': context.get('time_period')
        }
        context_hash = hashlib.md5(
            json.dumps(stable_context, sort_keys=True).encode()
        ).hexdigest()[:8]

        return f"rec:{scene}:{user_id}:{context_hash}"


class FeatureCache:
    """特征缓存"""

    def __init__(self, redis_client, ttl: int = 3600):
        self.redis = redis_client
        self.ttl = ttl

    def get_user_features(self, user_id: str) -> Optional[Dict]:
        """获取用户特征"""
        key = f"user_features:{user_id}"
        cached = self.redis.get(key)
        return json.loads(cached) if cached else None

    def set_user_features(self, user_id: str, features: Dict):
        """设置用户特征"""
        key = f"user_features:{user_id}"
        self.redis.setex(key, self.ttl, json.dumps(features))

    def get_item_features(self, item_id: str) -> Optional[Dict]:
        """获取物品特征"""
        key = f"item_features:{item_id}"
        cached = self.redis.get(key)
```

### 3.2 并行处理

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor
from typing import List, Dict
import time

class ParallelRecommendService:
    """并行推荐服务"""

    def __init__(self, recall_service, ranking_service, max_workers: int = 10):
        self.recall_service = recall_service
        self.ranking_service = ranking_service
        self.executor = ThreadPoolExecutor(max_workers=max_workers)

    async def recommend(self, user_id: str, context: Dict) -> List:
        """异步推荐"""
        # 并行召回
        recall_tasks = [
            self._async_recall(recaller, user_id, context)
            for recaller in self.recall_service.recallers
        ]

        recall_results = await asyncio.gather(*recall_tasks)

        # 合并召回结果
        all_items = self._merge_recall_results(recall_results)

        # 排序
        ranked_items = await self._async_rank(user_id, all_items, context)

        return ranked_items

    async def _async_recall(self, recaller, user_id: str, context: Dict):
        """异步召回"""
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self.executor,
            recaller.recall,
            user_id,
            context,
            100
        )

    async def _async_rank(self, user_id: str, items: List, context: Dict):
        """异步排序"""
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self.executor,
            self.ranking_service.rank,
            user_id,
            items,
            {},
            context
        )

    def _merge_recall_results(self, results: List) -> List:
        """合并召回结果"""
        all_items = set()
        for items in results:
            all_items.update(item for item, _ in items)
        return list(all_items)
```

```python
class BatchPredictor:
    """批量预测器"""

    def __init__(self, model, batch_size: int = 256):
        self.model = model
        self.batch_size = batch_size

    def predict(self, features_list: List) -> np.ndarray:
        """批量预测"""
        all_predictions = []

        for i in range(0, len(features_list), self.batch_size):
            batch_features = features_list[i:i + self.batch_size]

            with torch.no_grad():
                batch_tensor = torch.FloatTensor(batch_features)
                predictions = self.model(batch_tensor).numpy()

            all_predictions.append(predictions)

        return np.concatenate(all_predictions, axis=0)
```

### 4.1 指标监控

```python
from dataclasses import dataclass
from typing import Dict, List
import time
import threading

@dataclass
class MetricPoint:
    """指标数据点"""
    timestamp: float
    value: float
    tags: Dict[str, str]


class RecommendMetrics:
    """推荐服务指标"""

    def __init__(self):
        self.metrics = {
            'request_count': [],
            'latency': [],
            'recall_latency': [],
            'ranking_latency': [],
            'error_count': [],
            'cache_hit_rate': []
        }
        self.lock = threading.Lock()

    def record_latency(self, metric_name: str, value: float, tags: Dict = None):
        """记录延迟指标"""
        with self.lock:
            point = MetricPoint(
                timestamp=time.time(),
                value=value,
                tags=tags or {}
            )
            self.metrics[metric_name].append(point)

    def record_request(self, success: bool = True, tags: Dict = None):
        """记录请求"""
        with self.lock:
            point = MetricPoint(
                timestamp=time.time(),
                value=1,
                tags=tags or {}
            )
            self.metrics['request_count'].append(point)

            if not success:
                self.metrics['error_count'].append(point)

    def get_stats(self, metric_name: str, window_seconds: int = 60) -> Dict:
        """获取统计信息"""
        now = time.time()
        window_start = now - window_seconds

        with self.lock:
            points = [
                p for p in self.metrics[metric_name]
                if p.timestamp >= window_start
            ]

        if not points:
            return {'count': 0, 'mean': 0, 'max': 0, 'min': 0}

        values = [p.value for p in points]

        return {
            'count': len(values),
            'mean': sum(values) / len(values),
            'max': max(values),
            'min': min(values),
            'p50': self._percentile(values, 50),
            'p99': self._percentile(values, 99)
        }

    def _percentile(self, values: List[float], p: int) -> float:
        """计算百分位"""
        sorted_values = sorted(values)
        idx = int(len(sorted_values) * p / 100)
        return sorted_values[min(idx, len(sorted_values) - 1)]


class AlertManager:
    """告警管理"""

    def __init__(self, metrics: RecommendMetrics):
        self.metrics = metrics
        self.alert_rules = []

    def add_rule(self, rule):
        """添加告警规则"""
        self.alert_rules.append(rule)

    def check_alerts(self) -> List[Dict]:
        """检查告警"""
        alerts = []

        for rule in self.alert_rules:
            if rule.check(self.metrics):
                alerts.append(rule.get_alert())

        return alerts


class LatencyAlertRule:
    """延迟告警规则"""

    def __init__(self, metric_name: str, threshold: float, window_seconds: int = 60):
        self.metric_name = metric_name
        self.threshold = threshold
        self.window_seconds = window_seconds

    def check(self, metrics: RecommendMetrics) -> bool:
        stats = metrics.get_stats(self.metric_name, self.window_seconds)
        return stats['p99'] > self.threshold

    def get_alert(self) -> Dict:
        return {
            'type': 'latency',
            'metric': self.metric_name,
            'message': f"P99 latency exceeds {self.threshold}ms"
        }
```

## 5. 容错与降级

### 5.1 熔断器

```python
import time
from enum import Enum
from threading import Lock

class CircuitState(Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开


class CircuitBreaker:
    """熔断器"""

    def __init__(self, failure_threshold: int = 5,
                 recovery_timeout: int = 30):
        """
        参数:
            failure_threshold: 失败次数阈值
            recovery_timeout: 恢复超时时间（秒）
        """
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout

        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = 0
        self.lock = Lock()

    def call(self, func, *args, **kwargs):
        """调用函数"""
        with self.lock:
            if self.state == CircuitState.OPEN:
                # 检查是否可以进入半开状态
                if time.time() - self.last_failure_time > self.recovery_timeout:
                    self.state = CircuitState.HALF_OPEN
                else:
                    raise Exception("Circuit breaker is OPEN")

        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise e

    def _on_success(self):
        """成功时处理"""
        with self.lock:
            if self.state == CircuitState.HALF_OPEN:
                self.state = CircuitState.CLOSED
            self.failure_count = 0

    def _on_failure(self):
        """失败时处理"""
        with self.lock:
            self.failure_count += 1
            self.last_failure_time = time.time()

            if self.failure_count >= self.failure_threshold:
                self.state = CircuitState.OPEN
```

### 5.2 降级策略

```python
class DegradationStrategy:
    """降级策略"""

    def __init__(self):
        self.strategies = {}

    def register(self, service_name: str, fallback_func):
        """注册降级策略"""
        self.strategies[service_name] = fallback_func

    def get_fallback(self, service_name: str):
        """获取降级函数"""
        return self.strategies.get(service_name)


class RecommendDegradation:
    """推荐降级"""

    def __init__(self):
        self.strategy = DegradationStrategy()

        # 注册降级策略
        self.strategy.register('recall', self._fallback_recall)
        self.strategy.register('ranking', self._fallback_ranking)

    def _fallback_recall(self, user_id: str, context: Dict) -> List:
        """召回降级：返回热门物品"""
        return self._get_hot_items(100)

    def _fallback_ranking(self, user_id: str, items: List, context: Dict) -> List:
        """排序降级：随机排序"""
        import random
        random.shuffle(items)
        return items[:50]

    def _get_hot_items(self, count: int) -> List:
        """获取热门物品"""
        # 从缓存或配置获取
        return [f"hot_item_{i}" for i in range(count)]
```

## 6. 完整服务示例

```python
from flask import Flask, request, jsonify
import uuid
import time

app = Flask(__name__)

# 初始化组件
recall_service = RecallService([...])
ranking_service = RankingService(...)
rerank_service = RerankService()
cache = RecommendCache(redis_client)
metrics = RecommendMetrics()
logger = RecommendLogger()
circuit_breaker = CircuitBreaker()
degradation = RecommendDegradation()


@app.route('/recommend', methods=['POST'])
def recommend():
    """推荐接口"""
    request_id = str(uuid.uuid4())
    start_time = time.time()

    try:
        # 解析请求
        data = request.json
        user_id = data['user_id']
        scene = data.get('scene', 'home')
        context = data.get('context', {})

        # 检查缓存
        cached = cache.get(user_id, scene, context)
        if cached:
            metrics.record_latency('latency', (time.time() - start_time) * 1000)
            return jsonify({
                'request_id': request_id,
                'items': cached,
                'cached': True
            })

        # 获取用户特征
        user_features = get_user_features(user_id)

        # 召回（带熔断）
        try:
            recall_items = circuit_breaker.call(
                recall_service.recall, user_id, context
            )
        except Exception:
            recall_items = degradation.strategy.get_fallback('recall')(user_id, context)

        # 排序
        ranked_items = ranking_service.rank(user_id, recall_items, user_features, context)

        # 重排
        final_items = rerank_service.rerank(ranked_items, context)

        # 设置缓存
        cache.set(user_id, scene, context, final_items)

        # 记录指标
        latency = (time.time() - start_time) * 1000
        metrics.record_latency('latency', latency)
        metrics.record_request(success=True)

        logger.log_request(user_id, scene, request_id, latency, final_items)

        return jsonify({
            'request_id': request_id,
            'items': final_items,
            'latency_ms': latency
        })

    except Exception as e:
        metrics.record_request(success=False)
        logger.log_error(request_id, type(e).__name__, str(e))
        return jsonify({'error': str(e)}), 500


@app.route('/health', methods=['GET'])
def health():
    """健康检查"""
    return jsonify({'status': 'ok'})


@app.route('/metrics', methods=['GET'])
def get_metrics():
    """获取指标"""
    return jsonify({
        'latency': metrics.get_stats('latency'),
        'request_count': metrics.get_stats('request_count'),
        'error_count': metrics.get_stats('error_count')
    })


if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8080)
```

## 7. 学习总结

### 7.1 核心要点

1. **低延迟**：缓存、并行、批处理
