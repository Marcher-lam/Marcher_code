# 模型版本管理 学习文档

## 1. 模型版本管理概述

### 1.1 为什么需要模型版本管理？

```
模型版本管理需求:

1. 可追溯: 知道哪个版本部署了什么
2. 可回滚: 出问题能快速回退
3. 可复现: 能复现训练结果
4. 可比较: 不同版本效果对比
5. 可协作: 团队协作开发

典型场景:
- 线上模型出问题，快速回滚
- A/B测试对比新旧模型
- 重新训练模型，保持一致性
- 审计需要查看历史模型
```

### 1.2 版本管理内容

```python
"""
需要版本管理的内容:

1. 模型文件
   - 模型权重
   - 模型结构
   - 预处理/后处理代码

2. 配置文件
   - 模型超参数
   - 特征配置
   - 训练配置

3. 元数据
   - 训练数据版本
   - 训练时间
   - 评估指标
   - 提交信息

4. 关联信息
   - Git commit
   - 数据版本
   - 依赖版本
"""

from typing import Dict, List, Optional, Any
from dataclasses import dataclass, asdict
from datetime import datetime
import json
import hashlib
import os
import shutil


@dataclass
class ModelVersion:
    """模型版本信息"""
    version: str                    # 版本号
    model_name: str                 # 模型名称
    created_at: str                 # 创建时间
    created_by: str                 # 创建者
    description: str                # 描述
    metrics: Dict[str, float]       # 评估指标
    config: Dict[str, Any]          # 配置
    data_version: str               # 数据版本
    git_commit: str                 # Git commit
    dependencies: Dict[str, str]    # 依赖版本
    tags: List[str]                 # 标签
    status: str                     # 状态 (staging/production/archived)


class ModelRegistry:
    """
    模型注册中心

    管理所有模型版本
    """

    def __init__(self, registry_path: str = "model_registry"):
        self.registry_path = registry_path
        self.versions: Dict[str, ModelVersion] = {}

        os.makedirs(registry_path, exist_ok=True)
        self._load_registry()

    def _load_registry(self):
        """加载注册表"""
        registry_file = os.path.join(self.registry_path, "registry.json")
        if os.path.exists(registry_file):
            with open(registry_file, 'r') as f:
                data = json.load(f)
                for version_id, version_data in data.items():
                    self.versions[version_id] = ModelVersion(**version_data)

    def _save_registry(self):
        """保存注册表"""
        registry_file = os.path.join(self.registry_path, "registry.json")
        with open(registry_file, 'w') as f:
            json.dump(
                {vid: asdict(v) for vid, v in self.versions.items()},
                f,
                indent=2
            )

    def register_model(self,
                      model_path: str,
                      model_name: str,
                      description: str = "",
                      metrics: Dict[str, float] = None,
                      config: Dict[str, Any] = None,
                      data_version: str = "",
                      git_commit: str = "",
                      tags: List[str] = None,
                      created_by: str = "unknown") -> ModelVersion:
        """
        注册新模型版本
        """
        # 生成版本号
        version = self._generate_version(model_name)

        # 创建版本记录
        model_version = ModelVersion(
            version=version,
            model_name=model_name,
            created_at=datetime.now().isoformat(),
            created_by=created_by,
            description=description,
            metrics=metrics or {},
            config=config or {},
            data_version=data_version,
            git_commit=git_commit,
            dependencies=self._get_dependencies(),
            tags=tags or [],
            status="staging"
        )

        # 复制模型文件
        version_dir = os.path.join(self.registry_path, model_name, version)
        os.makedirs(version_dir, exist_ok=True)
        self._copy_model_files(model_path, version_dir)

        # 保存元数据
        with open(os.path.join(version_dir, "metadata.json"), 'w') as f:
            json.dump(asdict(model_version), f, indent=2)

        # 更新注册表
        version_id = f"{model_name}/{version}"
        self.versions[version_id] = model_version
        self._save_registry()

        print(f"Registered model: {version_id}")
        return model_version

    def _generate_version(self, model_name: str) -> str:
        """生成版本号"""
        # 格式: vYYYYMMDD.N
        today = datetime.now().strftime("%Y%m%d")

        # 计算今天的版本数
        today_versions = [
            v for vid, v in self.versions.items()
            if vid.startswith(model_name) and v.version.startswith(f"v{today}")
        ]

        n = len(today_versions) + 1
        return f"v{today}.{n:03d}"

    def _copy_model_files(self, src_path: str, dst_path: str):
        """复制模型文件"""
        if os.path.isdir(src_path):
            shutil.copytree(src_path, dst_path, dirs_exist_ok=True)
        else:
            shutil.copy2(src_path, dst_path)

    def _get_dependencies(self) -> Dict[str, str]:
        """获取依赖版本"""
        import torch
        return {
            "torch": torch.__version__,
            "python": os.sys.version.split()[0]
        }

    def get_version(self, model_name: str, version: str) -> Optional[ModelVersion]:
        """获取指定版本"""
        version_id = f"{model_name}/{version}"
        return self.versions.get(version_id)

    def list_versions(self, model_name: str = None,
                     status: str = None,
                     tags: List[str] = None) -> List[ModelVersion]:
        """列出模型版本"""
        results = []

        for version_id, version in self.versions.items():
            # 过滤
            if model_name and version.model_name != model_name:
                continue
            if status and version.status != status:
                continue
            if tags and not all(t in version.tags for t in tags):
                continue

            results.append(version)

        # 按时间排序
        results.sort(key=lambda x: x.created_at, reverse=True)
        return results

    def get_latest_version(self, model_name: str,
                          status: str = "production") -> Optional[ModelVersion]:
        """获取最新版本"""
        versions = self.list_versions(model_name, status)
        return versions[0] if versions else None

    def promote_to_production(self, model_name: str, version: str):
        """将模型升级为生产版本"""
        version_id = f"{model_name}/{version}"

        if version_id not in self.versions:
            raise ValueError(f"Version not found: {version_id}")

        # 将其他生产版本降级
        for vid, v in self.versions.items():
            if v.model_name == model_name and v.status == "production":
                v.status = "archived"

        # 升级指定版本
        self.versions[version_id].status = "production"
        self._save_registry()

        print(f"Promoted {version_id} to production")

    def rollback(self, model_name: str, target_version: str):
        """回滚到指定版本"""
        self.promote_to_production(model_name, target_version)
        print(f"Rolled back {model_name} to {target_version}")
```

```python
import torch
import pickle
import yaml


class ModelSerializer:
    """
    模型序列化器
    """

    @staticmethod
    def save_pytorch(model: torch.nn.Module,
                    path: str,
                    include_optimizer: bool = False,
                    optimizer: torch.optim.Optimizer = None):
        """保存 PyTorch 模型"""
        checkpoint = {
            'model_state_dict': model.state_dict(),
            'model_class': model.__class__.__name__,
        }

        if include_optimizer and optimizer is not None:
            checkpoint['optimizer_state_dict'] = optimizer.state_dict()

        torch.save(checkpoint, path)

    @staticmethod
    def load_pytorch(model_class: type,
                    path: str,
                    model_args: dict = None) -> torch.nn.Module:
        """加载 PyTorch 模型"""
        checkpoint = torch.load(path)

        # 创建模型实例
        if model_args:
            model = model_class(**model_args)
        else:
            model = model_class()

        model.load_state_dict(checkpoint['model_state_dict'])

        return model

    @staticmethod
    def export_onnx(model: torch.nn.Module,
                   path: str,
                   input_shape: tuple,
                   input_names: List[str] = None,
                   output_names: List[str] = None):
        """导出为 ONNX 格式"""
        model.eval()
        dummy_input = torch.randn(input_shape)

        torch.onnx.export(
            model,
            dummy_input,
            path,
            input_names=input_names or ['input'],
            output_names=output_names or ['output'],
            dynamic_axes={
                'input': {0: 'batch_size'},
                'output': {0: 'batch_size'}
            }
        )

        print(f"Exported ONNX model to {path}")


class ConfigManager:
    """
    配置管理器
    """

    def __init__(self, config_path: str = "configs"):
        self.config_path = config_path
        os.makedirs(config_path, exist_ok=True)

    def save_config(self, config: Dict, name: str, version: str = None):
        """保存配置"""
        if version:
            filename = f"{name}_{version}.yaml"
        else:
            filename = f"{name}.yaml"

        filepath = os.path.join(self.config_path, filename)

        with open(filepath, 'w') as f:
            yaml.dump(config, f, default_flow_style=False)

        return filepath

    def load_config(self, name: str, version: str = None) -> Dict:
        """加载配置"""
        if version:
            filename = f"{name}_{version}.yaml"
        else:
            filename = f"{name}.yaml"

        filepath = os.path.join(self.config_path, filename)

        with open(filepath, 'r') as f:
            return yaml.safe_load(f)
```

## 3. 版本对比

### 3.1 模型对比工具

```python
class ModelComparator:
    """
    模型对比工具
    """

    def __init__(self, registry: ModelRegistry):
        self.registry = registry

    def compare_versions(self,
                        model_name: str,
                        version1: str,
                        version2: str) -> Dict[str, Any]:
        """对比两个版本"""
        v1 = self.registry.get_version(model_name, version1)
        v2 = self.registry.get_version(model_name, version2)

        if not v1 or not v2:
            raise ValueError("Version not found")

        comparison = {
            "versions": {
                "version1": version1,
                "version2": version2
            },
            "metrics_diff": {},
            "config_diff": {},
            "time_diff": None
        }

        # 指标对比
        all_metrics = set(v1.metrics.keys()) | set(v2.metrics.keys())
        for metric in all_metrics:
            val1 = v1.metrics.get(metric, 0)
            val2 = v2.metrics.get(metric, 0)
            comparison["metrics_diff"][metric] = {
                "v1": val1,
                "v2": val2,
                "diff": val2 - val1,
                "diff_pct": (val2 - val1) / val1 * 100 if val1 != 0 else 0
            }

        # 配置对比
        all_configs = set(v1.config.keys()) | set(v2.config.keys())
        for key in all_configs:
            if v1.config.get(key) != v2.config.get(key):
                comparison["config_diff"][key] = {
                    "v1": v1.config.get(key),
                    "v2": v2.config.get(key)
                }

        # 时间对比
        t1 = datetime.fromisoformat(v1.created_at)
        t2 = datetime.fromisoformat(v2.created_at)
        comparison["time_diff"] = str(t2 - t1)

        return comparison

    def print_comparison(self, comparison: Dict):
        """打印对比结果"""
        print("=" * 60)
        print(f"Model Comparison: {comparison['versions']['version1']} vs {comparison['versions']['version2']}")
        print("=" * 60)

        print("\n📊 Metrics Comparison:")
        print("-" * 40)
        for metric, vals in comparison["metrics_diff"].items():
            diff_symbol = "↑" if vals["diff"] > 0 else "↓" if vals["diff"] < 0 else "="
            print(f"  {metric}:")
            print(f"    v1: {vals['v1']:.4f}")
            print(f"    v2: {vals['v2']:.4f}")
            print(f"    diff: {diff_symbol} {abs(vals['diff_pct']):.2f}%")

        if comparison["config_diff"]:
            print("\n⚙️ Config Differences:")
            print("-" * 40)
            for key, vals in comparison["config_diff"].items():
                print(f"  {key}: {vals['v1']} → {vals['v2']}")

        print(f"\n⏱️ Time difference: {comparison['time_diff']}")
```

```python
class ModelDeploymentPipeline:
    """
    模型部署流水线
    """

    def __init__(self, registry: ModelRegistry):
        self.registry = registry
        self.stages = ["dev", "staging", "production"]

    def deploy(self,
              model_name: str,
              version: str,
              target_stage: str,
              auto_rollback: bool = True):
        """
        部署模型

        target_stage: dev / staging / production
        """
        model_version = self.registry.get_version(model_name, version)

        if not model_version:
            raise ValueError(f"Version not found: {model_name}/{version}")

        print(f"Deploying {model_name}/{version} to {target_stage}...")

        # 部署前检查
        self._pre_deploy_check(model_version, target_stage)

        # 执行部署
        try:
            self._deploy_to_stage(model_version, target_stage)

            # 部署后验证
            self._post_deploy_validation(model_version, target_stage)

            # 更新状态
            if target_stage == "production":
                self.registry.promote_to_production(model_name, version)

            print(f"✅ Deployment successful!")

        except Exception as e:
            print(f"❌ Deployment failed: {e}")

            if auto_rollback and target_stage == "production":
                print("Initiating automatic rollback...")
                self._rollback(model_name)

            raise
```

## 5. 实验跟踪

### 5.1 实验记录

```python
@dataclass
class Experiment:
    """实验记录"""
    experiment_id: str
    model_name: str
    description: str
    config: Dict[str, Any]
    metrics: Dict[str, float]
    start_time: str
    end_time: str
    status: str
    artifacts: List[str]
    tags: List[str]


class ExperimentTracker:
    """
    实验跟踪器

    记录训练实验
    """

    def __init__(self, tracking_path: str = "experiments"):
        self.tracking_path = tracking_path
        self.experiments: Dict[str, Experiment] = {}
        os.makedirs(tracking_path, exist_ok=True)

    def start_experiment(self,
                        model_name: str,
                        description: str,
                        config: Dict[str, Any],
                        tags: List[str] = None) -> str:
        """开始实验"""
        experiment_id = self._generate_id()

        experiment = Experiment(
            experiment_id=experiment_id,
            model_name=model_name,
            description=description,
            config=config,
            metrics={},
            start_time=datetime.now().isoformat(),
            end_time="",
            status="running",
            artifacts=[],
            tags=tags or []
        )

        self.experiments[experiment_id] = experiment
        self._save_experiment(experiment)

        return experiment_id

    def log_metrics(self, experiment_id: str, metrics: Dict[str, float]):
        """记录指标"""
        if experiment_id not in self.experiments:
            raise ValueError(f"Experiment not found: {experiment_id}")

        self.experiments[experiment_id].metrics.update(metrics)
        self._save_experiment(self.experiments[experiment_id])

    def log_artifact(self, experiment_id: str, artifact_path: str):
        """记录产物"""
        if experiment_id not in self.experiments:
            raise ValueError(f"Experiment not found: {experiment_id}")

        self.experiments[experiment_id].artifacts.append(artifact_path)
        self._save_experiment(self.experiments[experiment_id])

    def end_experiment(self, experiment_id: str, status: str = "completed"):
        """结束实验"""
        if experiment_id not in self.experiments:
            raise ValueError(f"Experiment not found: {experiment_id}")

        self.experiments[experiment_id].end_time = datetime.now().isoformat()
        self.experiments[experiment_id].status = status
        self._save_experiment(self.experiments[experiment_id])

    def _generate_id(self) -> str:
        """生成实验ID"""
        import uuid
        return uuid.uuid4().hex[:8]

    def _save_experiment(self, experiment: Experiment):
        """保存实验记录"""
        filepath = os.path.join(
            self.tracking_path,
            f"{experiment.experiment_id}.json"
        )
        with open(filepath, 'w') as f:
            json.dump(asdict(experiment), f, indent=2)
```

## 6. 学习总结

### 6.1 核心要点

```
1. 版本命名: 使用语义化版本或时间戳
2. 元数据: 记录完整的训练信息
3. 回滚机制: 支持快速回退
4. 环境隔离: dev/staging/production
5. 自动化: CI/CD 集成
```

### 6.2 最佳实践

```
1. 每次训练都注册新版本
2. 记录完整的配置和指标
3. 生产部署前进行验证
4. 保留历史版本以便回滚
5. 使用标签组织模型
6. 定期清理过期版本
```
