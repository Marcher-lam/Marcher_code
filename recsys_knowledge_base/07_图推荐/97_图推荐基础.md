# 图推荐基础 学习文档

## 1. 基础认知

### 1.1 什么是图推荐？

图推荐是将推荐系统建模为**图结构**，利用图算法来发现用户和物品之间的关系。用户-物品交互可以自然地表示为二部图。

### 1.2 为什么用图？

**图的优势：**
1. **自然表示**：用户-物品交互本身就是图
2. **高阶关系**：可以发现多跳关联
3. **结构信息**：利用图的拓扑结构
4. **丰富算法**：图论有丰富的算法库

### 1.3 图推荐的类型

```
图推荐方法
├── 基于随机游走
│   ├── DeepWalk
│   ├── Node2Vec
│   └── PersonalRank
├── 基于矩阵分解
│   ├── Graph Regularized MF
│   └── TrustSVD
└── 图神经网络
    ├── GCN
    ├── GraphSAGE
    ├── GAT
    ├── NGCF
    └── LightGCN
```

## 2. 图的构建

### 2.1 用户-物品二部图

```python
import numpy as np
from collections import defaultdict
import networkx as nx


class UserItemGraph:
    """
    用户-物品二部图构建器
    """

    def __init__(self):
        self.graph = nx.Graph()
        self.user_nodes = set()
        self.item_nodes = set()
        self.user_to_idx = {}
        self.item_to_idx = {}
        self.idx_to_user = {}
        self.idx_to_item = {}

    def build_from_interactions(self, interactions, user_col='user_id',
                                item_col='item_id', weight_col=None):
        """
        从交互数据构建图

        参数:
            interactions: DataFrame 或列表，交互数据
            user_col: 用户列名
            item_col: 物品列名
            weight_col: 权重列名（可选）
        """
        # 处理输入
        if hasattr(interactions, 'iterrows'):
            # DataFrame
            data = [(row[user_col], row[item_col],
                    row.get(weight_col, 1) if weight_col else 1)
                   for _, row in interactions.iterrows()]
        else:
            # 列表
            data = interactions

        # 收集节点
        users = set()
        items = set()
        for item in data:
            users.add(item[0])
            items.add(item[1])

        # 构建索引
        self.user_to_idx = {u: i for i, u in enumerate(users)}
        self.item_to_idx = {i: len(users) + j for j, i in enumerate(items)}
        self.idx_to_user = {i: u for u, i in self.user_to_idx.items()}
        self.idx_to_item = {i: it for it, i in self.item_to_idx.items()}

        # 添加节点
        for user in users:
            user_node = f"u_{user}"
            self.graph.add_node(user_node, type='user', idx=self.user_to_idx[user])
            self.user_nodes.add(user_node)

        for item in items:
            item_node = f"i_{item}"
            self.graph.add_node(item_node, type='item', idx=self.item_to_idx[item])
            self.item_nodes.add(item_node)

        # 添加边
        for user, item, weight in data:
            user_node = f"u_{user}"
            item_node = f"i_{item}"
            self.graph.add_edge(user_node, item_node, weight=weight)

        self.n_users = len(users)
        self.n_items = len(items)
        self.n_nodes = self.n_users + self.n_items

        return self

    def get_adjacency_matrix(self):
        """获取邻接矩阵"""
        nodes = list(self.graph.nodes())
        adj = nx.adjacency_matrix(self.graph, nodes)
        return adj

    def get_normalized_adjacency(self, norm='symmetric'):
        """
        获取归一化邻接矩阵

        参数:
            norm: 归一化方式 ('symmetric', 'left', 'right')
        """
        adj = self.get_adjacency_matrix().astype(np.float32)

        if norm == 'symmetric':
            # D^{-1/2} A D^{-1/2}
            degree = np.array(adj.sum(axis=1)).flatten()
            d_inv_sqrt = np.power(degree, -0.5)
            d_inv_sqrt[np.isinf(d_inv_sqrt)] = 0.
            d_mat_inv_sqrt = np.diag(d_inv_sqrt)
            normalized = d_mat_inv_sqrt @ adj @ d_mat_inv_sqrt
        elif norm == 'left':
            # D^{-1} A
            degree = np.array(adj.sum(axis=1)).flatten()
            d_inv = np.power(degree, -1)
            d_inv[np.isinf(d_inv)] = 0.
            d_mat_inv = np.diag(d_inv)
            normalized = d_mat_inv @ adj
        else:
            normalized = adj

        return normalized

    def get_neighbors(self, node):
        """获取节点的邻居"""
        return list(self.graph.neighbors(node))

    def get_degree(self, node):
        """获取节点的度"""
        return self.graph.degree(node)
```

```python
class GraphAnalyzer:
    """
    图性质分析器
    """

    def __init__(self, graph_builder):
        self.graph = graph_builder.graph
        self.builder = graph_builder

    def analyze(self):
        """分析图的性质"""
        results = {}

        # 基本信息
        results['n_nodes'] = self.graph.number_of_nodes()
        results['n_edges'] = self.graph.number_of_edges()

        # 度分布
        degrees = [d for n, d in self.graph.degree()]
        results['avg_degree'] = np.mean(degrees)
        results['max_degree'] = np.max(degrees)
        results['min_degree'] = np.min(degrees)

        # 连通性
        results['is_connected'] = nx.is_connected(self.graph)
        results['n_components'] = nx.number_connected_components(self.graph)

        # 聚类系数
        results['avg_clustering'] = nx.average_clustering(self.graph)

        # 二部图特性
        results['is_bipartite'] = nx.is_bipartite(self.graph)

        return results

    def get_user_degree_distribution(self):
        """获取用户度分布"""
        degrees = [self.graph.degree(node) for node in self.builder.user_nodes]
        return degrees

    def get_item_degree_distribution(self):
        """获取物品度分布"""
        degrees = [self.graph.degree(node) for node in self.builder.item_nodes]
        return degrees

    def find_high_degree_items(self, top_k=10):
        """找高度物品（热门物品）"""
        item_degrees = [(node, self.graph.degree(node))
                       for node in self.builder.item_nodes]
        return sorted(item_degrees, key=lambda x: x[1], reverse=True)[:top_k]
```

### 3.1 PersonalRank

```python
import numpy as np
from collections import defaultdict


class PersonalRank:
    """
    PersonalRank 随机游走推荐算法

    基于随机游走的图推荐方法
    """

    def __init__(self, alpha=0.85, max_iter=100, tol=1e-6):
        """
        参数:
            alpha: 继续游走的概率
            max_iter: 最大迭代次数
            tol: 收敛阈值
        """
        self.alpha = alpha
        self.max_iter = max_iter
        self.tol = tol
        self.graph = None
        self.user_to_idx = None
        self.item_to_idx = None

    def fit(self, graph_builder):
        """
        训练模型

        参数:
            graph_builder: UserItemGraph 实例
        """
        self.graph_builder = graph_builder
        self.graph = graph_builder.graph
        self.user_to_idx = graph_builder.user_to_idx
        self.item_to_idx = graph_builder.item_to_idx
        self.idx_to_user = graph_builder.idx_to_user
        self.idx_to_item = graph_builder.idx_to_item
        self.n_users = graph_builder.n_users
        self.n_items = graph_builder.n_items

        # 构建转移矩阵
        self._build_transition_matrix()

    def _build_transition_matrix(self):
        """构建转移矩阵"""
        n = self.n_users + self.n_items
        self.transition = np.zeros((n, n))

        # 构建用户->物品的转移
        for user_node in self.graph_builder.user_nodes:
            user_idx = self.user_to_idx[user_node[2:]]  # 去掉 'u_' 前缀
            neighbors = list(self.graph.neighbors(user_node))
            if neighbors:
                prob = 1.0 / len(neighbors)
                for item_node in neighbors:
                    item_idx = self.item_to_idx[item_node[2:]]  # 去掉 'i_' 前缀
                    self.transition[user_idx, item_idx] = prob

        # 构建物品->用户的转移
        for item_node in self.graph_builder.item_nodes:
            item_idx = self.item_to_idx[item_node[2:]]
            neighbors = list(self.graph.neighbors(item_node))
            if neighbors:
                prob = 1.0 / len(neighbors)
                for user_node in neighbors:
                    user_idx = self.user_to_idx[user_node[2:]]
                    self.transition[item_idx, user_idx] = prob

    def recommend(self, user_id, top_k=10, exclude_interacted=True):
        """
        为用户推荐物品

        参数:
            user_id: 用户 ID
            top_k: 推荐数量
            exclude_interacted: 是否排除已交互物品

        返回:
            推荐物品列表
        """
        if user_id not in self.user_to_idx:
            return []

        user_idx = self.user_to_idx[user_id]

        # 初始化概率分布
        n = self.n_users + self.n_items
        r = np.zeros(n)
        r[user_idx] = 1.0  # 从目标用户开始

        # 迭代计算
        for _ in range(self.max_iter):
            r_new = (1 - self.alpha) * np.eye(n)[user_idx] + self.alpha * self.transition.T @ r

            # 检查收敛
            if np.linalg.norm(r_new - r) < self.tol:
                break
            r = r_new

        # 提取物品分数
        item_scores = r[self.n_users:]  # 物品在后面的索引

        # 排除已交互物品
        if exclude_interacted:
            user_node = f"u_{user_id}"
            interacted = {n[2:] for n in self.graph.neighbors(user_node)}
            for item in interacted:
                if item in self.item_to_idx:
                    item_scores[self.item_to_idx[item]] = -np.inf

        # 获取 top-k
        top_indices = np.argsort(item_scores)[::-1][:top_k]

        return [(self.idx_to_item[self.n_users + idx], item_scores[idx])
                for idx in top_indices]

    def recommend_batch(self, user_ids, top_k=10):
        """批量推荐"""
        results = {}
        for user_id in user_ids:
            results[user_id] = self.recommend(user_id, top_k)
        return results


# 使用示例
def test_personal_rank():
    # 构建图
    interactions = [
        ('u1', 'i1', 1), ('u1', 'i2', 1), ('u1', 'i3', 1),
        ('u2', 'i1', 1), ('u2', 'i4', 1),
        ('u3', 'i2', 1), ('u3', 'i3', 1), ('u3', 'i5', 1),
        ('u4', 'i4', 1), ('u4', 'i5', 1),
    ]

    graph_builder = UserItemGraph()
    graph_builder.build_from_interactions(interactions)

    # PersonalRank
    pr = PersonalRank(alpha=0.85, max_iter=50)
    pr.fit(graph_builder)

    # 推荐
    recommendations = pr.recommend('u4', top_k=3)
    print("为 u4 的推荐:")
    for item, score in recommendations:
        print(f"  {item}: {score:.4f}")


if __name__ == "__main__":
    test_personal_rank()
```

### 3.2 DeepWalk

```python
import numpy as np
import random
from collections import defaultdict
from gensim.models import Word2Vec


class DeepWalk:
    """
    DeepWalk: 在图上进行随机游走并学习节点嵌入

    论文: DeepWalk: Online Learning of Social Representations (KDD 2014)
    """

    def __init__(self, walk_length=80, num_walks=10, window_size=5,
                 embed_dim=64, min_count=1, workers=4):
        """
        参数:
            walk_length: 每次游走的长度
            num_walks: 每个节点的游走次数
            window_size: Skip-gram 窗口大小
            embed_dim: 嵌入维度
            min_count: 最小出现次数
            workers: 并行数
        """
        self.walk_length = walk_length
        self.num_walks = num_walks
        self.window_size = window_size
        self.embed_dim = embed_dim
        self.min_count = min_count
        self.workers = workers

        self.graph = None
        self.model = None

    def fit(self, graph_builder):
        """训练"""
        self.graph_builder = graph_builder
        self.graph = graph_builder.graph

        # 生成随机游走序列
        walks = self._generate_walks()

        # 使用 Word2Vec 学习嵌入
        walks_str = [[str(node) for node in walk] for walk in walks]
        self.model = Word2Vec(
            walks_str,
            vector_size=self.embed_dim,
            window=self.window_size,
            min_count=self.min_count,
            sg=1,  # Skip-gram
            workers=self.workers
        )

        return self

    def _generate_walks(self):
        """生成随机游走序列"""
        walks = []
        nodes = list(self.graph.nodes())

        for _ in range(self.num_walks):
            random.shuffle(nodes)
            for node in nodes:
                walk = self._random_walk(node)
                walks.append(walk)

        return walks

    def _random_walk(self, start_node):
        """从指定节点开始随机游走"""
        walk = [start_node]
        current = start_node

        for _ in range(self.walk_length - 1):
            neighbors = list(self.graph.neighbors(current))
            if not neighbors:
                break
            current = random.choice(neighbors)
            walk.append(current)

        return walk

    def get_embedding(self, node):
        """获取节点嵌入"""
        try:
            return self.model.wv[str(node)]
        except KeyError:
            return None

    def get_all_embeddings(self):
        """获取所有节点嵌入"""
        embeddings = {}
        for node in self.graph.nodes():
            emb = self.get_embedding(node)
            if emb is not None:
                embeddings[node] = emb
        return embeddings

    def recommend(self, user_id, top_k=10, exclude_interacted=True):
        """
        为用户推荐物品

        参数:
            user_id: 用户 ID
            top_k: 推荐数量
            exclude_interacted: 是否排除已交互物品

        返回:
            推荐物品列表
        """
        user_node = f"u_{user_id}"
        user_emb = self.get_embedding(user_node)

        if user_emb is None:
            return []

        # 获取已交互物品
        interacted = set()
        if exclude_interacted:
            interacted = {n for n in self.graph.neighbors(user_node)}

        # 计算与所有物品的相似度
        scores = []
        for item_node in self.graph_builder.item_nodes:
            if item_node in interacted:
                continue
            item_emb = self.get_embedding(item_node)
            if item_emb is not None:
                similarity = np.dot(user_emb, item_emb) / (
                    np.linalg.norm(user_emb) * np.linalg.norm(item_emb)
                )
                scores.append((item_node[2:], similarity))  # 去掉 'i_' 前缀

        # 排序
        scores.sort(key=lambda x: x[1], reverse=True)

        return scores[:top_k]


class Node2Vec(DeepWalk):
    """
    Node2Vec: 带偏置的随机游走

    论文: node2vec: Scalable Feature Learning for Networks (KDD 2016)

    通过 p 和 q 参数控制游走策略:
    - p: 返回参数，控制回到上一个节点的概率
    - q: 出入参数，控制探索新节点还是回到已访问节点的概率
    """

    def __init__(self, p=1.0, q=1.0, walk_length=80, num_walks=10,
                 window_size=5, embed_dim=64, min_count=1, workers=4):
        super().__init__(walk_length, num_walks, window_size,
                        embed_dim, min_count, workers)
        self.p = p
        self.q = q

    def fit(self, graph_builder):
        """训练"""
        self.graph_builder = graph_builder
        self.graph = graph_builder.graph

        # 预计算转移概率
        self._precompute_transition_probs()

        # 生成游走序列
        walks = self._generate_walks()

        # Word2Vec
        walks_str = [[str(node) for node in walk] for walk in walks]
        self.model = Word2Vec(
            walks_str,
            vector_size=self.embed_dim,
            window=self.window_size,
            min_count=self.min_count,
            sg=1,
            workers=self.workers
        )

        return self

    def _precompute_transition_probs(self):
        """预计算转移概率"""
        self.alias_nodes = {}
        self.alias_edges = {}

        # 计算节点到邻居的转移概率
        for node in self.graph.nodes():
            neighbors = list(self.graph.neighbors(node))
            if neighbors:
                probs = [1.0 / len(neighbors)] * len(neighbors)
                self.alias_nodes[node] = self._create_alias_table(probs, neighbors)

    def _create_alias_table(self, probs, items):
        """创建 Alias 采样表"""
        n = len(probs)
        prob = np.array(probs) * n
        smaller = []
        larger = []

        for i, p in enumerate(prob):
            if p < 1.0:
                smaller.append(i)
            else:
                larger.append(i)

        alias = [0] * n
        while smaller and larger:
            small = smaller.pop()
            large = larger.pop()

            prob[small] = 1.0
            alias[small] = large

            prob[large] = prob[large] + prob[small] - 1.0
            if prob[large] < 1.0:
                smaller.append(large)
            else:
                larger.append(large)

        return {'prob': prob, 'alias': alias, 'items': items}

    def _alias_sample(self, alias_table):
        """Alias 采样"""
        n = len(alias_table['items'])
        i = random.randint(0, n - 1)

        if random.random() < alias_table['prob'][i]:
            return alias_table['items'][i]
        else:
            return alias_table['items'][alias_table['alias'][i]]

    def _random_walk(self, start_node):
        """带偏置的随机游走"""
        walk = [start_node]

        while len(walk) < self.walk_length:
            current = walk[-1]
            neighbors = list(self.graph.neighbors(current))

            if not neighbors:
                break

            if len(walk) == 1:
                # 第一步，均匀采样
                next_node = random.choice(neighbors)
            else:
                # 后续步骤，使用 Node2Vec 采样
                prev = walk[-2]
                next_node = self._node2vec_sample(prev, current, neighbors)

            walk.append(next_node)

        return walk

    def _node2vec_sample(self, prev, current, neighbors):
        """Node2Vec 采样"""
        # 计算转移概率
        probs = []
        for neighbor in neighbors:
            if neighbor == prev:
                # 返回上一个节点
                probs.append(1.0 / self.p)
            elif self.graph.has_edge(prev, neighbor):
                # 距离为 1（共同邻居）
                probs.append(1.0)
            else:
                # 距离为 2（新节点）
                probs.append(1.0 / self.q)

        # 归一化
        total = sum(probs)
        probs = [p / total for p in probs]

        # 采样
        return random.choices(neighbors, weights=probs, k=1)[0]
```

## 4. 图神经网络基础

### 4.1 GCN 原理

图卷积网络（GCN）的传播规则：

$$H^{(l+1)} = \sigma(\tilde{D}^{-1/2}\tilde{A}\tilde{D}^{-1/2}H^{(l)}W^{(l)})$$

其中：
- $\tilde{A} = A + I$：添加自环的邻接矩阵
- $\tilde{D}$：度矩阵
- $H^{(l)}$：第 l 层的节点表示
- $W^{(l)}$：可学习权重

### 4.2 GCN 实现

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import scipy.sparse as sp


class GCNLayer(nn.Module):
    """
    GCN 层
    """

    def __init__(self, in_features, out_features, bias=True):
        super().__init__()
        self.weight = nn.Parameter(torch.FloatTensor(in_features, out_features))
        if bias:
            self.bias = nn.Parameter(torch.FloatTensor(out_features))
        else:
            self.register_parameter('bias', None)

        self.reset_parameters()

    def reset_parameters(self):
        nn.init.xavier_uniform_(self.weight)
        if self.bias is not None:
            nn.init.zeros_(self.bias)

    def forward(self, x, adj):
        """
        前向传播

        参数:
            x: (n_nodes, in_features) 节点特征
            adj: (n_nodes, n_nodes) 归一化邻接矩阵
        """
        support = torch.mm(x, self.weight)
        output = torch.spmm(adj, support)

        if self.bias is not None:
            output = output + self.bias

        return output


class GCN(nn.Module):
    """
    图卷积网络
    """

    def __init__(self, n_features, hidden_dim, n_classes, n_layers=2, dropout=0.5):
        super().__init__()

        self.layers = nn.ModuleList()

        # 第一层
        self.layers.append(GCNLayer(n_features, hidden_dim))

        # 中间层
        for _ in range(n_layers - 2):
            self.layers.append(GCNLayer(hidden_dim, hidden_dim))

        # 输出层
        self.layers.append(GCNLayer(hidden_dim, n_classes))

        self.dropout = dropout

    def forward(self, x, adj):
        for i, layer in enumerate(self.layers[:-1]):
            x = layer(x, adj)
            x = F.relu(x)
            x = F.dropout(x, self.dropout, training=self.training)

        x = self.layers[-1](x)

        return x


def normalize_adjacency(adj):
    """对称归一化邻接矩阵"""
    adj = adj + sp.eye(adj.shape[0])
    degree = np.array(adj.sum(axis=1)).flatten()
    d_inv_sqrt = np.power(degree, -0.5)
    d_inv_sqrt[np.isinf(d_inv_sqrt)] = 0.
    d_mat_inv_sqrt = sp.diags(d_inv_sqrt)
    normalized = adj.dot(d_mat_inv_sqrt).transpose().dot(d_mat_inv_sqrt)
    return normalized.tocoo()
```

## 5. 学习总结

### 5.1 核心要点

1. **图表示**：用户-物品交互自然形成二部图
2. **随机游走**：通过游走发现高阶关系
3. **图神经网络**：学习节点表示进行推荐

### 5.2 方法选择

| 方法 | 优点 | 适用场景 |
|------|------|----------|
| PersonalRank | 简单、可解释 | 小规模、需要解释 |
| DeepWalk | 无监督、可扩展 | 中大规模 |
| Node2Vec | 灵活、效果好 | 需要调参 |
| GCN | 端到端、效果好 | 有特征 |
