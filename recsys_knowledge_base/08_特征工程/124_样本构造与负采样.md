# 样本构造与负采样 学习文档

## 1. 基础认知

### 1.1 为什么需要负采样？

在推荐系统中，我们通常只有用户的**正反馈**（点击、购买等），而负反馈（不感兴趣）是隐式的。为了训练模型，需要构造负样本。

### 1.2 样本构造的挑战

1. **数据稀疏**：正样本远少于可能的用户-物品对
2. **偏差问题**：观察到的数据是有偏的
3. **负样本定义**：什么算作真正的负样本？

### 1.3 常见策略

```
样本构造策略
├── 全量负采样
│   ├── 随机负采样
│   └── 批内负采样
├── 基于流行度负采样
│   ├── 按流行度采样
│   └── 按逆流行度采样
├── Hard Negative Sampling
│   ├── 基于相似度
│   └── 基于模型
└── 混合策略
```

## 2. 正样本构造

### 2.1 交互数据到样本

```python
import numpy as np
import pandas as pd
from datetime import datetime, timedelta
from collections import defaultdict


class PositiveSampleConstructor:
    """
    正样本构造器
    """

    def __init__(self, time_decay=True, decay_rate=0.1):
        """
        参数:
            time_decay: 是否使用时间衰减
            decay_rate: 衰减率
        """
        self.time_decay = time_decay
        self.decay_rate = decay_rate

    def construct_from_interactions(self, interactions, min_interactions=5):
        """
        从交互数据构造正样本

        参数:
            interactions: DataFrame, 包含 user_id, item_id, timestamp, action
            min_interactions: 用户最小交互数

        返回:
            positive_samples: list of (user_id, item_id, weight)
        """
        # 过滤低活跃用户
        user_counts = interactions['user_id'].value_counts()
        valid_users = user_counts[user_counts >= min_interactions].index
        interactions = interactions[interactions['user_id'].isin(valid_users)]

        # 时间衰减权重
        if self.time_decay:
            max_time = interactions['timestamp'].max()
            interactions['days_ago'] = (max_time - interactions['timestamp']).dt.days
            interactions['weight'] = np.exp(-self.decay_rate * interactions['days_ago'])
        else:
            interactions['weight'] = 1.0

        # 构造样本
        samples = []
        for _, row in interactions.iterrows():
            samples.append((row['user_id'], row['item_id'], row['weight']))

        return samples

    def construct_sequence_samples(self, interactions, sequence_length=10):
        """
        构造序列样本

        参数:
            interactions: DataFrame
            sequence_length: 序列长度

        返回:
            sequence_samples: list of {'user_id', 'sequence', 'target'}
        """
        # 按用户分组并排序
        user_sequences = interactions.sort_values('timestamp').groupby('user_id')

        samples = []
        for user_id, group in user_sequences:
            items = group['item_id'].tolist()

            # 滑动窗口构造样本
            for i in range(sequence_length, len(items)):
                sequence = items[i - sequence_length:i]
                target = items[i]

                samples.append({
                    'user_id': user_id,
                    'sequence': sequence,
                    'target': target
                })

        return samples


class ImplicitFeedbackConstructor:
    """
    隐式反馈样本构造器
    """

    def __init__(self, positive_action='click', negative_action=None):
        self.positive_action = positive_action
        self.negative_action = negative_action

    def construct(self, interactions, action_col='action'):
        """
        构造隐式反馈样本

        参数:
            interactions: DataFrame
            action_col: 行为类型列名

        返回:
            samples: DataFrame
        """
        # 标记正样本
        interactions['label'] = (interactions[action_col] == self.positive_action).astype(int)

        # 如果有明确的负行为（如 dislike）
        if self.negative_action:
            neg_mask = interactions[action_col] == self.negative_action
            interactions.loc[neg_mask, 'label'] = 0

        return interactions
```

## 3. 负采样策略

### 3.1 随机负采样

```python
import numpy as np
from collections import defaultdict


class RandomNegativeSampler:
    """
    随机负采样器
    """

    def __init__(self, n_items, n_negatives=4):
        """
        参数:
            n_items: 物品总数
            n_negatives: 每个正样本对应的负样本数
        """
        self.n_items = n_items
        self.n_negatives = n_negatives

    def sample(self, positive_items, exclude_items=None):
        """
        采样负样本

        参数:
            positive_items: 正样本物品集合
            exclude_items: 要排除的物品集合

        返回:
            负样本列表
        """
        exclude = positive_items | (exclude_items or set())

        negatives = []
        while len(negatives) < self.n_negatives:
            item = np.random.randint(0, self.n_items)
            if item not in exclude:
                negatives.append(item)

        return negatives

    def sample_batch(self, positive_items_batch, exclude_items_batch=None):
        """批量采样"""
        all_negatives = []

        for i, pos_items in enumerate(positive_items_batch):
            exclude = pos_items
            if exclude_items_batch:
                exclude |= exclude_items_batch[i]

            negatives = []
            while len(negatives) < self.n_negatives:
                item = np.random.randint(0, self.n_items)
                if item not in exclude:
                    negatives.append(item)

            all_negatives.append(negatives)

        return all_negatives


class PopularityBasedSampler:
    """
    基于流行度的负采样器
    """

    def __init__(self, item_frequencies, n_negatives=4, mode='pop'):
        """
        参数:
            item_frequencies: dict, {item_id: frequency}
            n_negatives: 负样本数
            mode: 'pop' (按流行度) 或 'unpop' (按逆流行度)
        """
        self.n_negatives = n_negatives
        self.mode = mode

        items = list(item_frequencies.keys())
        freqs = np.array([item_frequencies[i] for i in items], dtype=np.float64)

        if mode == 'pop':
            # 按流行度采样
            probs = freqs / freqs.sum()
        elif mode == 'unpop':
            # 按逆流行度采样（更倾向于冷门物品）
            probs = 1.0 / (freqs + 1)
            probs = probs / probs.sum()
        else:
            probs = np.ones(len(items)) / len(items)

        self.items = items
        self.probs = probs

    def sample(self, exclude_items=None, n_samples=None):
        """
        采样

        参数:
            exclude_items: 要排除的物品
            n_samples: 采样数量（默认用 n_negatives）

        返回:
            负样本列表
        """
        n_samples = n_samples or self.n_negatives
        exclude = exclude_items or set()

        # 创建排除掩码
        exclude_indices = [self.items.index(i) for i in exclude if i in self.items]

        # 调整概率
        adjusted_probs = self.probs.copy()
        for idx in exclude_indices:
            adjusted_probs[idx] = 0
        adjusted_probs = adjusted_probs / adjusted_probs.sum()

        # 采样
        indices = np.random.choice(len(self.items), size=n_samples, p=adjusted_probs, replace=False)
        return [self.items[i] for i in indices]
```

### 3.2 批内负采样

```python
import torch


class InBatchNegativeSampler:
    """
    批内负采样器

    使用同一个 batch 内其他样本的正样本作为负样本
    """

    def __init__(self, temperature=1.0):
        """
        参数:
            temperature: softmax 温度
        """
        self.temperature = temperature

    def compute_loss(self, user_embeddings, item_embeddings, labels=None):
        """
        计算批内负采样损失

        参数:
            user_embeddings: (batch, dim)
            item_embeddings: (batch, dim)
            labels: (batch,) 可选标签

        返回:
            loss: 标量
        """
        batch_size = user_embeddings.size(0)

        # 计算相似度矩阵
        # (batch, dim) @ (dim, batch) = (batch, batch)
        similarity = torch.matmul(user_embeddings, item_embeddings.t()) / self.temperature

        # 对角线是正样本
        if labels is None:
            labels = torch.arange(batch_size, device=user_embeddings.device)

        # Cross Entropy 损失
        loss = torch.nn.functional.cross_entropy(similarity, labels)

        return loss


class CrossBatchNegativeSampler:
    """
    跨批负采样器

    使用历史 batch 的样本作为负样本
    """

    def __init__(self, embed_dim, queue_size=65536, temperature=0.07):
        """
        参数:
            embed_dim: 嵌入维度
            queue_size: 队列大小
            temperature: 温度
        """
        self.embed_dim = embed_dim
        self.queue_size = queue_size
        self.temperature = temperature

        # 初始化队列
        self.register_buffer("queue", torch.randn(embed_dim, queue_size))
        self.queue = torch.nn.functional.normalize(self.queue, dim=0)
        self.register_buffer("queue_ptr", torch.zeros(1, dtype=torch.long))

    def register_buffer(self, name, tensor):
        """注册缓冲区"""
        setattr(self, name, tensor)

    @torch.no_grad()
    def dequeue_and_enqueue(self, keys):
        """更新队列"""
        batch_size = keys.shape[0]
        ptr = int(self.queue_ptr)

        # 替换队列中的键
        if ptr + batch_size > self.queue_size:
            # 分两次替换
            first_part = self.queue_size - ptr
            self.queue[:, ptr:] = keys[:first_part].t()
            self.queue[:, :batch_size - first_part] = keys[first_part:].t()
            ptr = batch_size - first_part
        else:
            self.queue[:, ptr:ptr + batch_size] = keys.t()
            ptr = (ptr + batch_size) % self.queue_size

        self.queue_ptr[0] = ptr

    def compute_loss(self, q, k):
        """
        计算损失

        参数:
            q: queries (batch, dim)
            k: keys (batch, dim)
        """
        batch_size = q.shape[0]

        # 正样本 logits
        l_pos = torch.einsum('nc,nc->n', [q, k]).unsqueeze(-1)  # (batch, 1)

        # 负样本 logits（从队列）
        l_neg = torch.einsum('nc,ck->nk', [q, self.queue.clone().detach()])  # (batch, K)

        # logits
        logits = torch.cat([l_pos, l_neg], dim=1) / self.temperature  # (batch, 1+K)

        # labels
        labels = torch.zeros(batch_size, dtype=torch.long, device=q.device)

        loss = torch.nn.functional.cross_entropy(logits, labels)

        # 更新队列
        self.dequeue_and_enqueue(k)

        return loss
```

### 3.3 Hard Negative Sampling

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity


class HardNegativeSampler:
    """
    Hard Negative 采样器

    选择与正样本相似但非正样本的物品作为负样本
    """

    def __init__(self, item_embeddings, n_negatives=4,
                 hard_ratio=0.5, similarity_threshold=0.8):
        """
        参数:
            item_embeddings: (n_items, dim) 物品嵌入
            n_negatives: 负样本数
            hard_ratio: hard negative 比例
            similarity_threshold: 相似度阈值
        """
        self.item_embeddings = item_embeddings
        self.n_negatives = n_negatives
        self.hard_ratio = hard_ratio
        self.similarity_threshold = similarity_threshold
        self.n_items = item_embeddings.shape[0]

        # 预计算相似度矩阵（对于小规模数据）
        if self.n_items <= 10000:
            self.similarity_matrix = cosine_similarity(item_embeddings)
        else:
            self.similarity_matrix = None

    def sample(self, positive_item, user_interacted_items=None, n_samples=None):
        """
        采样

        参数:
            positive_item: 正样本物品
            user_interacted_items: 用户已交互物品
            n_samples: 采样数量

        返回:
            负样本列表
        """
        n_samples = n_samples or self.n_negatives
        n_hard = int(n_samples * self.hard_ratio)
        n_random = n_samples - n_hard

        exclude = user_interacted_items or set()
        exclude.add(positive_item)

        negatives = []

        # Hard negatives
        if self.similarity_matrix is not None:
            # 使用预计算矩阵
            similarities = self.similarity_matrix[positive_item]
            # 找相似的物品
            candidate_items = np.where(similarities > self.similarity_threshold)[0]
            # 排除已交互
            candidate_items = [i for i in candidate_items if i not in exclude]
            # 随机选择
            if candidate_items:
                hard_negs = np.random.choice(
                    candidate_items,
                    min(n_hard, len(candidate_items)),
                    replace=False
                ).tolist()
                negatives.extend(hard_negs)
        else:
            # 在线计算
            pos_emb = self.item_embeddings[positive_item:positive_item + 1]
            similarities = cosine_similarity(pos_emb, self.item_embeddings)[0]
            candidate_items = np.where(similarities > self.similarity_threshold)[0]
            candidate_items = [i for i in candidate_items if i not in exclude]
            if candidate_items:
                hard_negs = np.random.choice(
                    candidate_items,
                    min(n_hard, len(candidate_items)),
                    replace=False
                ).tolist()
                negatives.extend(hard_negs)

        # Random negatives
        while len(negatives) < n_samples:
            item = np.random.randint(0, self.n_items)
            if item not in exclude and item not in negatives:
                negatives.append(item)

        return negatives[:n_samples]


class ModelBasedHardNegativeSampler:
    """
    基于模型的 Hard Negative 采样器

    使用模型预测的分数来选择难分样本
    """

    def __init__(self, model, n_items, n_negatives=4, top_k=100):
        """
        参数:
            model: 推荐模型
            n_items: 物品数
            n_negatives: 负样本数
            top_k: 候选集大小
        """
        self.model = model
        self.n_items = n_items
        self.n_negatives = n_negatives
        self.top_k = top_k

    def sample(self, user_id, positive_items, device='cpu'):
        """
        采样

        参数:
            user_id: 用户 ID
            positive_items: 正样本物品集合
            device: 设备

        返回:
            负样本列表
        """
        # 获取模型对所有物品的预测
        with torch.no_grad():
            user_tensor = torch.LongTensor([user_id]).to(device)
            all_items = torch.LongTensor(list(range(self.n_items))).to(device)

            scores = self.model.predict(user_tensor.expand(self.n_items), all_items)
            scores = scores.cpu().numpy()

        # 排除正样本
        for item in positive_items:
            scores[item] = -np.inf

        # 选择分数最高的 top-k（模型认为可能是正样本但实际不是）
        top_indices = np.argsort(scores)[::-1][:self.top_k]

        # 从 top-k 中随机选择
        selected = np.random.choice(top_indices, self.n_negatives, replace=False)

        return selected.tolist()
```

## 4. 样本平衡

### 4.1 样本不平衡处理

```python
class SampleBalancer:
    """
    样本平衡器
    """

    def __init__(self, strategy='oversample', ratio=1.0):
        """
        参数:
            strategy: 'oversample' (过采样) 或 'undersample' (欠采样)
            ratio: 正负样本比例
        """
        self.strategy = strategy
        self.ratio = ratio

    def balance(self, positive_samples, negative_samples):
        """
        平衡样本

        参数:
            positive_samples: 正样本列表
            negative_samples: 负样本列表

        返回:
            balanced_samples: 平衡后的样本
        """
        n_pos = len(positive_samples)
        n_neg = len(negative_samples)

        target_neg = int(n_pos / self.ratio)

        if self.strategy == 'oversample':
            # 过采样少数类
            if n_neg < target_neg:
                # 过采样负样本
                extra_neg = np.random.choice(
                    negative_samples,
                    target_neg - n_neg,
                    replace=True
                ).tolist()
                negative_samples = negative_samples + extra_neg
            elif n_pos < n_neg * self.ratio:
                # 过采样正样本
                extra_pos = np.random.choice(
                    positive_samples,
                    int(n_neg * self.ratio) - n_pos,
                    replace=True
                ).tolist()
                positive_samples = positive_samples + extra_pos

        elif self.strategy == 'undersample':
            # 欠采样多数类
            if n_neg > target_neg:
                negative_samples = np.random.choice(
                    negative_samples,
                    target_neg,
                    replace=False
                ).tolist()

        return positive_samples, negative_samples
```

## 5. 完整的样本构造流水线

```python
class SamplePipeline:
    """
    完整的样本构造流水线
    """

    def __init__(self, n_items, n_negatives=4, neg_sample_strategy='mixed',
                 time_decay=True, hard_ratio=0.3):
        """
        参数:
            n_items: 物品数
            n_negatives: 负样本数
            neg_sample_strategy: 'random', 'pop', 'hard', 'mixed'
            time_decay: 是否时间衰减
            hard_ratio: hard negative 比例
        """
        self.n_items = n_items
        self.n_negatives = n_negatives
        self.neg_sample_strategy = neg_sample_strategy

        # 初始化各组件
        self.pos_constructor = PositiveSampleConstructor(time_decay=time_decay)

        if neg_sample_strategy == 'random':
            self.neg_sampler = RandomNegativeSampler(n_items, n_negatives)
        elif neg_sample_strategy == 'pop':
            # 需要先设置 item_frequencies
            self.neg_sampler = None
        elif neg_sample_strategy == 'hard':
            # 需要先设置 item_embeddings
            self.neg_sampler = None
        elif neg_sample_strategy == 'mixed':
            self.random_sampler = RandomNegativeSampler(n_items, n_negatives - int(n_negatives * hard_ratio))
            self.hard_sampler = None
            self.hard_ratio = hard_ratio

    def set_item_frequencies(self, item_frequencies):
        """设置物品频率"""
        if self.neg_sample_strategy == 'pop':
            self.neg_sampler = PopularityBasedSampler(item_frequencies, self.n_negatives)

    def set_item_embeddings(self, item_embeddings):
        """设置物品嵌入"""
        if self.neg_sample_strategy == 'hard':
            self.neg_sampler = HardNegativeSampler(
                item_embeddings, self.n_negatives, hard_ratio=0.5
            )
        elif self.neg_sample_strategy == 'mixed':
            self.hard_sampler = HardNegativeSampler(
                item_embeddings, int(self.n_negatives * self.hard_ratio), hard_ratio=1.0
            )

    def construct_training_samples(self, interactions):
        """
        构造训练样本

        参数:
            interactions: DataFrame, 交互数据

        返回:
            training_samples: list of {'user_id', 'item_id', 'label', 'weight'}
        """
        # 构造正样本
        positive_samples = self.pos_constructor.construct_from_interactions(interactions)

        # 用户交互字典
        user_items = defaultdict(set)
        for user, item, _ in positive_samples:
            user_items[user].add(item)

        training_samples = []

        # 构造负样本
        for user, item, weight in positive_samples:
            # 正样本
            training_samples.append({
                'user_id': user,
                'item_id': item,
                'label': 1,
                'weight': weight
            })

            # 负样本
            if self.neg_sample_strategy == 'mixed':
                # 一部分随机，一部分 hard
                random_negs = self.random_sampler.sample(user_items[user])
                hard_negs = self.hard_sampler.sample(item, user_items[user])
                neg_items = random_negs + hard_negs
            else:
                neg_items = self.neg_sampler.sample(user_items[user])

            for neg_item in neg_items:
                training_samples.append({
                    'user_id': user,
                    'item_id': neg_item,
                    'label': 0,
                    'weight': 1.0
                })

        return training_samples
```

## 6. 学习总结

### 6.1 核心要点

1. **正样本构造**：考虑时间衰减、序列性
2. **负采样策略**：随机、流行度、hard negative
3. **样本平衡**：处理正负样本不平衡

### 6.2 策略选择

| 策略 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| 随机 | 简单、无偏 | 可能太简单 | 快速实验 |
| 流行度 | 符合真实分布 | 可能偏向热门 | 通用 |
| Hard | 难分样本、效果好 | 计算量大 | 精细化 |
