# 特征筛选方法 学习文档

## 1. 特征筛选概述

### 1.1 为什么要做特征筛选？

```
特征筛选的目的:
1. 提升模型效果: 去除噪声特征
2. 加速训练: 减少特征数量
3. 降低过拟合: 简化模型
4. 可解释性: 聚焦重要特征
5. 工程效率: 减少存储和计算

推荐系统特征数量:
- 小规模: 几十到几百个
- 大规模: 几千到上万个

不做筛选的问题:
- 维度灾难
- 计算成本高
- 噪声干扰
```

## 2. 过滤法 (Filter)

### 2.1 方差筛选

```python
class VarianceSelector:
    """
    方差筛选

    移除方差过小的特征（变化不大的特征价值低）
    """

    def __init__(self, threshold: float = 0.0):
        self.threshold = threshold
        self.selected_features = None

    def fit(self, X: pd.DataFrame):
        """拟合并选择特征"""
        variances = X.var()

        # 选择方差大于阈值的特征
        self.selected_features = variances[
            variances > self.threshold
        ].index.tolist()

        print(f"原始特征数: {X.shape[1]}")
        print(f"筛选后特征数: {len(self.selected_features)}")
        print(f"移除特征: {X.shape[1] - len(self.selected_features)}")

        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]

    def fit_transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """拟合并转换"""
        self.fit(X)
        return self.transform(X)


class CoefficientVariationSelector:
    """
    变异系数筛选

    CV = std / mean
    移除变异系数过小的特征
    """

    def __init__(self, threshold: float = 0.1):
        self.threshold = threshold
        self.selected_features = None

    def fit(self, X: pd.DataFrame):
        """拟合"""
        cv = X.std() / (X.mean() + 1e-10)
        self.selected_features = cv[cv > self.threshold].index.tolist()
        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]
```

### 2.2 相关性筛选

```python
class CorrelationSelector:
    """
    相关性筛选

    1. 与目标变量的相关性
    2. 特征之间的共线性
    """

    def __init__(self, corr_threshold: float = 0.95,
                 target_corr_threshold: float = 0.01):
        self.corr_threshold = corr_threshold
        self.target_corr_threshold = target_corr_threshold
        self.selected_features = None

    def fit(self, X: pd.DataFrame, y: pd.Series = None):
        """拟合"""
        # 1. 与目标的相关性
        if y is not None:
            target_corr = X.corrwith(y).abs()
            high_target_corr = target_corr[
                target_corr > self.target_corr_threshold
            ].index.tolist()
            X = X[high_target_corr]

        # 2. 特征间共线性
        corr_matrix = X.corr().abs()

        # 找出高度相关的特征对
        to_drop = set()
        features = X.columns.tolist()

        for i in range(len(features)):
            for j in range(i + 1, len(features)):
                if corr_matrix.iloc[i, j] > self.corr_threshold:
                    # 保留与目标相关性更高的特征
                    if y is not None:
                        if target_corr[features[i]] < target_corr[features[j]]:
                            to_drop.add(features[i])
                        else:
                            to_drop.add(features[j])
                    else:
                        to_drop.add(features[j])

        self.selected_features = [f for f in features if f not in to_drop]

        print(f"因共线性移除: {len(to_drop)} 个特征")

        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]
```

### 2.3 统计检验筛选

```python
class StatisticalSelector:
    """
    统计检验筛选

    使用统计检验评估特征与目标的关系
    """

    @staticmethod
    def chi2_selection(X: np.ndarray, y: np.ndarray,
                       k: int = 100) -> np.ndarray:
        """
        卡方检验

        适用于分类特征
        """
        selector = SelectKBest(chi2, k=k)
        X_selected = selector.fit_transform(X, y)

        # 获取分数
        scores = selector.scores_
        pvalues = selector.pvalues_

        return X_selected, scores, pvalues

    @staticmethod
    def f_test_selection(X: np.ndarray, y: np.ndarray,
                         k: int = 100) -> np.ndarray:
        """
        F检验 (ANOVA)

        适用于连续特征 vs 分类目标
        """
        selector = SelectKBest(f_classif, k=k)
        X_selected = selector.fit_transform(X, y)

        scores = selector.scores_

        return X_selected, scores

    @staticmethod
    def mutual_info_selection(X: np.ndarray, y: np.ndarray,
                               k: int = 100) -> np.ndarray:
        """
        互信息

        能捕捉非线性关系
        """
        selector = SelectKBest(mutual_info_classif, k=k)
        X_selected = selector.fit_transform(X, y)

        scores = selector.scores_

        return X_selected, scores
```

### 3.1 递归特征消除

```python
class RecursiveFeatureElimination:
    """
    递归特征消除 (RFE)

    反复训练模型，每次移除最不重要的特征
    """

    def __init__(self, n_features_to_select: int = 50,
                 step: int = 5):
        self.n_features_to_select = n_features_to_select
        self.step = step
        self.selected_features = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """拟合"""
        # 使用 LightGBM 作为基模型
        model = lgb.LGBMClassifier(
            n_estimators=100,
            learning_rate=0.1,
            random_state=42,
            verbose=-1
        )

        # RFE
        rfe = RFE(
            estimator=model,
            n_features_to_select=self.n_features_to_select,
            step=self.step
        )

        rfe.fit(X, y)

        self.selected_features = X.columns[rfe.support_].tolist()
        self.ranking_ = rfe.ranking_

        print(f"选择特征数: {len(self.selected_features)}")

        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]

    def get_ranking(self) -> pd.DataFrame:
        """获取特征排名"""
        return pd.DataFrame({
            'feature': self.selected_features,
            'ranking': self.ranking_[self.ranking_ == 1]
        })
```

class SequentialFeatureSelector:
    """
    序列特征选择

    前向选择: 从空集开始，逐步添加
    后向选择: 从全集开始，逐步移除
    """

    def __init__(self, n_features: int = 50,
                 direction: str = 'forward',
                 cv: int = 3):
        self.n_features = n_features
        self.direction = direction
        self.cv = cv
        self.selected_features = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """拟合"""
        from sklearn.feature_selection import SequentialFeatureSelector as SFS

        model = lgb.LGBMClassifier(
            n_estimators=50,
            learning_rate=0.1,
            random_state=42,
            verbose=-1
        )

        sfs = SFS(
            model,
            n_features_to_select=self.n_features,
            direction=self.direction,
            cv=self.cv,
            scoring='roc_auc'
        )

        sfs.fit(X, y)

        self.selected_features = X.columns[sfs.get_support()].tolist()

        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]
```

class TreeBasedSelector:
    """
    基于树模型的特征选择

    利用 Gini importance 或 Split importance
    """

    def __init__(self, threshold: str = 'median',
                 n_estimators: int = 100):
        self.threshold = threshold
        self.n_estimators = n_estimators
        self.model = None
        self.selected_features = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """拟合"""
        # 训练模型
        self.model = lgb.LGBMClassifier(
            n_estimators=self.n_estimators,
            learning_rate=0.1,
            random_state=42,
            verbose=-1
        )
        self.model.fit(X, y)

        # 获取特征重要性
        importance = pd.DataFrame({
            'feature': X.columns,
            'importance': self.model.feature_importances_
        }).sort_values('importance', ascending=False)

        self.feature_importance_ = importance

        # 选择特征
        selector = SelectFromModel(
            self.model,
            threshold=self.threshold,
            prefit=True
        )

        self.selected_features = X.columns[selector.get_support()].tolist()

        print(f"选择特征数: {len(self.selected_features)}")
        print("\nTop 10 重要特征:")
        print(importance.head(10))

        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]

    def plot_importance(self, top_n: int = 20):
        """可视化特征重要性"""
        import matplotlib.pyplot as plt

        top_features = self.feature_importance_.head(top_n)

        plt.figure(figsize=(10, 8))
        plt.barh(range(len(top_features)), top_features['importance'].values)
        plt.yticks(range(len(top_features)), top_features['feature'].values)
        plt.xlabel('Importance')
        plt.title(f'Top {top_n} Feature Importance')
        plt.tight_layout()
        plt.show()
```

### 4.2 L1 正则化筛选

```python
class L1RegularizationSelector:
    """
    L1 正则化筛选

    L1 正则化会产生稀疏解，自动做特征选择
    """

    def __init__(self, C: float = 0.1, threshold: float = 1e-5):
        self.C = C
        self.threshold = threshold
        self.model = None
        self.selected_features = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """拟合"""
        # L1 正则化逻辑回归
        self.model = LogisticRegression(
            penalty='l1',
            C=self.C,
            solver='saga',
            max_iter=1000,
            random_state=42
        )
        self.model.fit(X, y)

        # 获取非零系数特征
        coef = np.abs(self.model.coef_[0])
        self.selected_features = X.columns[coef > self.threshold].tolist()

        # 特征系数
        self.feature_coef_ = pd.DataFrame({
            'feature': X.columns,
            'coef': self.model.coef_[0]
        }).sort_values('coef', key=abs, ascending=False)

        print(f"非零系数特征数: {len(self.selected_features)}")

        return self

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        """转换"""
        return X[self.selected_features]
```

## 5. 推荐系统特征筛选

### 5.1 特征筛选流水线

```python
class RecSysFeatureSelector:
    """
    推荐系统特征筛选流水线
    """

    def __init__(self, n_features: int = 100):
        self.n_features = n_features
        self.selected_features = None

    def select_features(self, X: pd.DataFrame, y: pd.Series,
                        method: str = 'combined') -> pd.DataFrame:
        """
        特征筛选

        method:
        - 'variance': 方差筛选
        - 'correlation': 相关性筛选
        - 'tree': 树模型重要性
        - 'l1': L1正则化
        - 'combined': 组合方法
        """
        if method == 'variance':
            selector = VarianceSelector(threshold=0.01)
            selector.fit(X)
            return selector.transform(X)

        elif method == 'correlation':
            selector = CorrelationSelector(
                corr_threshold=0.95,
                target_corr_threshold=0.01
            )
            selector.fit(X, y)
            return selector.transform(X)

        elif method == 'tree':
            selector = TreeBasedSelector(threshold='median')
            selector.fit(X, y)
            return selector.transform(X)

        elif method == 'l1':
            selector = L1RegularizationSelector(C=0.1)
            selector.fit(X, y)
            return selector.transform(X)

        elif method == 'combined':
            # 组合多种方法
            return self._combined_selection(X, y)

        else:
            raise ValueError(f"Unknown method: {method}")

    def _combined_selection(self, X: pd.DataFrame,
                            y: pd.Series) -> pd.DataFrame:
        """
        组合筛选方法

        1. 先用方差筛选去除常量
        2. 再用相关性去除高度相关特征
        3. 最后用树模型选择 Top-K
        """
        print("Step 1: 方差筛选...")
        X1 = VarianceSelector(threshold=0.001).fit_transform(X)

        print("\nStep 2: 相关性筛选...")
        selector2 = CorrelationSelector(corr_threshold=0.95)
        selector2.fit(X1, y)
        X2 = selector2.transform(X1)

        print("\nStep 3: 树模型筛选...")
        selector3 = TreeBasedSelector(
            threshold=f'mean'  # 选择重要性高于均值的
        )
        selector3.fit(X2, y)
        X3 = selector3.transform(X2)

        self.selected_features = selector3.selected_features
        self.feature_importance = selector3.feature_importance_

        return X3
```

### 5.2 特征分组筛选

```python
class GroupFeatureSelector:
    """
    分组特征筛选

    推荐系统特征通常分为:
    - 用户特征
    - 物品特征
    - 上下文特征
    - 交叉特征
    """

    def __init__(self, feature_groups: Dict[str, List[str]],
                 max_features_per_group: int = 30):
        self.feature_groups = feature_groups
        self.max_features = max_features_per_group
        self.selected_features = {}

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """分组拟合"""
        for group_name, features in self.feature_groups.items():
            # 获取该组特征
            available_features = [f for f in features if f in X.columns]
            if not available_features:
                continue

            X_group = X[available_features]

            # 使用树模型筛选
            model = lgb.LGBMClassifier(
                n_estimators=50,
                verbose=-1
            )
            model.fit(X_group, y)

            # 选择 Top-K
            importance = pd.DataFrame({
                'feature': available_features,
                'importance': model.feature_importances_
            }).sort_values('importance', ascending=False)

            top_features = importance.head(self.max_features)['feature'].tolist()
            self.selected_features[group_name] = top_features

            print(f"{group_name}: {len(top_features)} features selected")

        return self

    def get_all_selected_features(self) -> List[str]:
        """获取所有选中的特征"""
        all_features = []
        for features in self.selected_features.values():
            all_features.extend(features)
        return all_features
```

## 6. 特征筛选评估

### 6.1 评估方法

```python
class FeatureSelectionEvaluator:
    """
    特征筛选效果评估
    """

    @staticmethod
    def evaluate_selection(X_train: pd.DataFrame, y_train: pd.Series,
                           X_test: pd.DataFrame, y_test: pd.Series,
                           selected_features: List[str]) -> Dict:
        """
        评估特征筛选效果
        """
        # 使用全部特征训练
        model_full = lgb.LGBMClassifier(
            n_estimators=100,
            learning_rate=0.1,
            verbose=-1
        )
        model_full.fit(X_train, y_train)
        pred_full = model_full.predict_proba(X_test)[:, 1]

        # 使用筛选后特征训练
        model_selected = lgb.LGBMClassifier(
            n_estimators=100,
            learning_rate=0.1,
            verbose=-1
        )
        model_selected.fit(X_train[selected_features], y_train)
        pred_selected = model_selected.predict_proba(
            X_test[selected_features]
        )[:, 1]

        # 计算指标
        from sklearn.metrics import roc_auc_score, log_loss

        results = {
            'full_features': {
                'n_features': X_train.shape[1],
                'auc': roc_auc_score(y_test, pred_full),
                'logloss': log_loss(y_test, pred_full)
            },
            'selected_features': {
                'n_features': len(selected_features),
                'auc': roc_auc_score(y_test, pred_selected),
                'logloss': log_loss(y_test, pred_selected)
            },
            'reduction_ratio': 1 - len(selected_features) / X_train.shape[1]
        }

        return results
```

## 7. 学习总结

### 7.1 方法对比

```
方法        优点                缺点                适用场景
────────────────────────────────────────────────────────────
方差筛选    快速简单            忽略与目标关系      初步筛选
相关性      直观                只能捕捉线性        去除共线性
统计检验    有理论保证          计算量大            特征数不多
RFE         效果好              计算非常慢          小规模
树模型      非线性、速度快      可能过拟合          大规模
L1正则      自动稀疏            需要调参            高维稀疏
```

### 7.2 最佳实践

```
1. 多种方法组合使用
2. 保留领域知识重要特征
3. 评估筛选对模型效果的影响
4. 考虑特征的稳定性
5. 定期重新评估特征重要性
```
