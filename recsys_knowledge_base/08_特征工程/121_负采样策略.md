# 负采样策略 学习文档

## 1. 基础认知

### 1.1 什么是负采样？

在推荐系统中，正样本（用户交互的物品）相对稀少，而负样本（用户未交互的物品）数量巨大。负采样是从大量负样本中选择代表性样本的策略。

### 1.2 为什么需要负采样？

```
问题:
- 全量负样本计算代价高
- 随机负样本信息量低
- 简单负样本容易分类

解决:
- 智能选择有信息量的负样本
- 提高训练效率
- 提升模型区分能力
```

## 2. 负采样方法

### 2.1 随机负采样

```python
import numpy as np
from typing import List, Tuple, Dict, Optional
from collections import defaultdict
import random


class RandomNegativeSampler:
    """
    随机负采样
    """

    def __init__(self, n_items: int, sample_ratio: float = 1.0):
        """
        参数:
            n_items: 物品总数
            sample_ratio: 负正样本比例
        """
        self.n_items = n_items
        self.sample_ratio = sample_ratio

    def sample(self, user_id: int, pos_items: List[int],
               n_neg: int = None) -> List[int]:
        """
        采样负样本

        参数:
            user_id: 用户ID
            pos_items: 正样本物品
            n_neg: 负样本数量

        返回:
            负样本列表
        """
        if n_neg is None:
            n_neg = int(len(pos_items) * self.sample_ratio)

        pos_set = set(pos_items)
        neg_items = []

        while len(neg_items) < n_neg:
            item = random.randint(0, self.n_items - 1)
            if item not in pos_set:
                neg_items.append(item)

        return neg_items
```

### 2.2 流行度负采样

```python
class PopularityNegativeSampler:
    """
    基于流行度的负采样
    """

    def __init__(self, item_popularity: Dict[int, int],
                 alpha: float = 0.75):
        """
        参数:
            item_popularity: {item_id: popularity}
            alpha: 流行度平滑系数
        """
        self.items = list(item_popularity.keys())
        self.n_items = len(self.items)

        # 计算采样概率
        pop_values = np.array([item_popularity.get(i, 1) for i in self.items])
        pop_values = np.power(pop_values, alpha)
        self.probs = pop_values / pop_values.sum()

    def sample(self, pos_items: List[int], n_neg: int) -> List[int]:
        """采样"""
        pos_set = set(pos_items)
        neg_items = []

        while len(neg_items) < n_neg:
            idx = np.random.choice(self.n_items, p=self.probs)
            item = self.items[idx]
            if item not in pos_set:
                neg_items.append(item)

        return neg_items
```

### 2.3 批次内负采样

```python
import torch


class InBatchNegativeSampler:
    """
    批次内负采样

    使用同一批次中其他用户的正样本作为负样本
    """

    def __init__(self, temperature: float = 1.0):
        self.temperature = temperature

    def compute_loss(self, user_embeds: torch.Tensor,
                    item_embeds: torch.Tensor) -> torch.Tensor:
        """
        计算批次内负采样损失

        参数:
            user_embeds: (batch, dim) 用户嵌入
            item_embeds: (batch, dim) 物品嵌入

        返回:
            损失值
        """
        batch_size = user_embeds.size(0)

        # 归一化
        user_embeds = torch.nn.functional.normalize(user_embeds, dim=-1)
        item_embeds = torch.nn.functional.normalize(item_embeds, dim=-1)

        # 计算相似度矩阵
        sim_matrix = torch.matmul(user_embeds, item_embeds.T) / self.temperature

        # 对角线是正样本，其他是负样本
        labels = torch.arange(batch_size, device=user_embeds.device)

        # 交叉熵损失
        loss = torch.nn.functional.cross_entropy(sim_matrix, labels)

        return loss


class MixedNegativeSampler:
    """
    混合负采样：批次内 + 随机
    """

    def __init__(self, n_items: int, batch_neg_ratio: float = 0.5,
                 random_neg_ratio: float = 0.5):
        self.n_items = n_items
        self.batch_neg_ratio = batch_neg_ratio
        self.random_neg_ratio = random_neg_ratio

    def sample(self, batch_item_ids: torch.Tensor,
               n_batch_neg: int, n_random_neg: int) -> torch.Tensor:
        """
        混合采样
        """
        batch_size = batch_item_ids.size(0)

        # 批次内负样本
        batch_negs = batch_item_ids[torch.randperm(batch_size)[:n_batch_neg]]

        # 随机负样本
        random_negs = torch.randint(0, self.n_items, (n_random_neg,))

        return torch.cat([batch_negs, random_negs])
```

### 2.4 困难负采样

```python
class HardNegativeSampler:
    """
    困难负采样
    """

    def __init__(self, model, n_items: int,
                 hard_ratio: float = 0.5,
                 cache_size: int = 1000):
        """
        参数:
            model: 推荐模型
            n_items: 物品数量
            hard_ratio: 困难负样本比例
            cache_size: 候选缓存大小
        """
        self.model = model
        self.n_items = n_items
        self.hard_ratio = hard_ratio
        self.cache_size = cache_size

        # 候选池缓存
        self.candidate_pool = list(range(n_items))

    @torch.no_grad()
    def sample(self, user_embed: torch.Tensor,
               pos_items: List[int],
               n_neg: int) -> List[int]:
        """
        采样困难负样本

        参数:
            user_embed: 用户嵌入
            pos_items: 正样本
            n_neg: 负样本数量

        返回:
            负样本列表
        """
        # 计算困难负样本数量
        n_hard = int(n_neg * self.hard_ratio)
        n_random = n_neg - n_hard

        pos_set = set(pos_items)

        # 从候选池中采样
        candidates = random.sample(
            self.candidate_pool,
            min(self.cache_size, len(self.candidate_pool))
        )

        # 过滤正样本
        candidates = [c for c in candidates if c not in pos_set]

        if len(candidates) < n_hard:
            # 候选不足，补充随机样本
            while len(candidates) < n_neg:
                item = random.randint(0, self.n_items - 1)
                if item not in pos_set and item not in candidates:
                    candidates.append(item)
            return candidates[:n_neg]

        # 计算候选物品的分数
        candidate_tensor = torch.tensor(candidates)
        scores = self.model.predict_single(user_embed, candidate_tensor)

        # 选择分数最高的（最困难的）
        _, hard_indices = torch.topk(scores, n_hard)
        hard_negs = [candidates[i] for i in hard_indices.tolist()]

        # 补充随机负样本
        random_negs = []
        while len(random_negs) < n_random:
            item = random.randint(0, self.n_items - 1)
            if item not in pos_set and item not in hard_negs:
                random_negs.append(item)

        return hard_negs + random_negs


class DynamicHardNegativeSampler:
    """
    动态困难负采样

    训练过程中动态调整困难程度
    """

    def __init__(self, model, n_items: int,
                 initial_alpha: float = 0.0,
                 final_alpha: float = 1.0,
                 total_epochs: int = 100):
        self.model = model
        self.n_items = n_items
        self.initial_alpha = initial_alpha
        self.final_alpha = final_alpha
        self.total_epochs = total_epochs
        self.current_epoch = 0

    def update_epoch(self, epoch: int):
        """更新当前 epoch"""
        self.current_epoch = epoch

    def get_alpha(self) -> float:
        """获取当前的困难程度系数"""
        progress = self.current_epoch / self.total_epochs
        return self.initial_alpha + (self.final_alpha - self.initial_alpha) * progress

    def sample_with_scores(self, user_embed: torch.Tensor,
                          candidates: List[int],
                          n_neg: int) -> Tuple[List[int], torch.Tensor]:
        """带分数的采样"""
        # 计算分数
        candidate_tensor = torch.tensor(candidates)
        scores = self.model.predict_single(user_embed, candidate_tensor)

        # 动态调整采样分布
        alpha = self.get_alpha()
        # 分数越高，被采样概率越大
        adjusted_scores = torch.softmax(scores * alpha, dim=0)

        # 采样
        indices = torch.multinomial(adjusted_scores, n_neg, replacement=False)

        return [candidates[i] for i in indices.tolist()], scores[indices]
```

## 3. 图负采样

### 3.1 基于图的负采样

```python
class GraphBasedNegativeSampler:
    """
    基于图结构的负采样
    """

    def __init__(self, adjacency_dict: Dict[int, List[int]], n_items: int):
        """
        参数:
            adjacency_dict: 邻接表 {item: [neighbors]}
            n_items: 物品数量
        """
        self.adjacency = adjacency_dict
        self.n_items = n_items

        # 计算物品相似度（基于共同邻居）
        self.item_sim = self._compute_item_similarity()

    def _compute_item_similarity(self) -> Dict[int, List[Tuple[int, float]]]:
        """计算物品相似度"""
        item_sim = defaultdict(list)

        all_items = list(self.adjacency.keys())

        for i in range(len(all_items)):
            for j in range(i + 1, len(all_items)):
                item_i = all_items[i]
                item_j = all_items[j]

                # Jaccard 相似度
                neighbors_i = set(self.adjacency.get(item_i, []))
                neighbors_j = set(self.adjacency.get(item_j, []))

                if neighbors_i and neighbors_j:
                    intersection = len(neighbors_i & neighbors_j)
                    union = len(neighbors_i | neighbors_j)
                    sim = intersection / union if union > 0 else 0

                    if sim > 0:
                        item_sim[item_i].append((item_j, sim))
                        item_sim[item_j].append((item_i, sim))

        # 排序
        for item in item_sim:
            item_sim[item].sort(key=lambda x: x[1], reverse=True)

        return dict(item_sim)

    def sample_hard_negatives(self, item_id: int, n_neg: int) -> List[int]:
        """采样与给定物品相似但未交互的物品"""
        if item_id not in self.item_sim:
            return []

        similar_items = self.item_sim[item_id][:n_neg]
        return [item for item, _ in similar_items]


class RandomWalkNegativeSampler:
    """
    基于随机游走的负采样
    """

    def __init__(self, user_item_dict: Dict[int, List[int]],
                 item_user_dict: Dict[int, List[int]],
                 walk_length: int = 3):
        self.user_item = user_item_dict
        self.item_user = item_user_dict
        self.walk_length = walk_length

    def random_walk(self, start_user: int) -> List[int]:
        """随机游走"""
        visited_items = []
        current = start_user
        is_user = True

        for _ in range(self.walk_length):
            if is_user:
                # 用户 -> 随机物品
                items = self.user_item.get(current, [])
                if items:
                    current = random.choice(items)
                    visited_items.append(current)
            else:
                # 物品 -> 随机用户
                users = self.item_user.get(current, [])
                if users:
                    current = random.choice(users)

            is_user = not is_user

        return visited_items

    def sample(self, user_id: int, pos_items: List[int],
               n_neg: int, n_walks: int = 10) -> List[int]:
        """通过随机游走采样负样本"""
        candidates = set()

        for _ in range(n_walks):
            items = self.random_walk(user_id)
            candidates.update(items)

        # 移除正样本
        pos_set = set(pos_items)
        candidates = list(candidates - pos_set)

        if len(candidates) >= n_neg:
            return random.sample(candidates, n_neg)
        else:
            # 补充随机样本
            while len(candidates) < n_neg:
                item = random.randint(0, max(self.item_user.keys()))
                if item not in pos_set and item not in candidates:
                    candidates.append(item)
            return candidates
```

## 4. 负采样策略对比

### 4.1 实验对比

| 策略 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| 随机 | 简单、快速 | 信息量低 | 基线 |
| 流行度 | 偏向热门物品 | 可能过拟合 | 热门物品惩罚 |
| 批次内 | 高效、无需额外计算 | 偏差 | 大规模训练 |
| 困难 | 信息量大 | 计算代价高 | 精排模型 |

### 4.2 混合策略

```python
class HybridNegativeSampler:
    """
    混合负采样策略
    """

    def __init__(self, samplers: List[Tuple[object, float]]):
        """
        参数:
            samplers: [(sampler, weight), ...]
        """
        self.samplers = samplers
        total_weight = sum(w for _, w in samplers)
        self.weights = [w / total_weight for _, w in samplers]

    def sample(self, **kwargs) -> List[int]:
        """混合采样"""
        all_negs = []

        for (sampler, _), weight in zip(self.samplers, self.weights):
            n = int(kwargs.get('n_neg', 10) * weight)
            negs = sampler.sample(n=n, **kwargs)
            all_negs.extend(negs)

        return all_negs


def create_default_sampler(n_items: int, item_popularity: Dict,
                          model=None) -> HybridNegativeSampler:
    """创建默认混合采样器"""
    samplers = [
        (RandomNegativeSampler(n_items), 0.3),
        (PopularityNegativeSampler(item_popularity), 0.3),
    ]

    if model is not None:
        samplers.append((HardNegativeSampler(model, n_items), 0.4))

    return HybridNegativeSampler(samplers)
```

## 5. 学习总结

### 5.1 核心要点

1. **随机采样**: 简单但信息量低
2. **流行度采样**: 偏向热门物品
3. **批次内采样**: 高效、无额外开销
4. **困难采样**: 信息量大、计算成本高

### 5.2 选择建议

```
召回模型: 批次内 + 随机
精排模型: 困难 + 流行度
冷启动: 基于内容的相似负样本
```
