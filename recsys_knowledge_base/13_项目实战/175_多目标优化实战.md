## 2. 多任务模型

### 2.1 MMOE 实现

```python
class MMoE(nn.Module):
    """
    Multi-gate Mixture-of-Experts

    多门控混合专家模型
    """

    def __init__(self,
                 feature_dim: int,
                 n_experts: int = 8,
                 expert_dim: int = 64,
                 n_tasks: int = 3,
                 tower_dims: List[int] = None):
        super().__init__()

        self.n_tasks = n_tasks
        self.n_experts = n_experts

        # 专家网络 (共享)
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, expert_dim),
                nn.ReLU(),
                nn.Linear(expert_dim, expert_dim),
                nn.ReLU()
            ) for _ in range(n_experts)
        ])

        # 门控网络 (每个任务一个)
        self.gates = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, n_experts),
                nn.Softmax(dim=-1)
            ) for _ in range(n_tasks)
        ])

        # 任务塔 (每个任务一个)
        tower_dims = tower_dims or [64, 32]
        self.towers = nn.ModuleList([
            self._build_tower(expert_dim, tower_dims)
            for _ in range(n_tasks)
        ])

    def _build_tower(self, input_dim: int, hidden_dims: List[int]) -> nn.Module:
        """构建任务塔"""
        layers = []
        dim = input_dim
        for hidden_dim in hidden_dims:
            layers.extend([
                nn.Linear(dim, hidden_dim),
                nn.ReLU(),
                nn.Dropout(0.2)
            ])
            dim = hidden_dim
        layers.append(nn.Linear(dim, 1))
        return nn.Sequential(*layers)

    def forward(self, features: torch.Tensor) -> List[torch.Tensor]:
        """
        前向传播

        返回: 每个任务的预测值列表
        """
        batch_size = features.size(0)

        # 专家输出
        expert_outputs = []
        for expert in self.experts:
            expert_outputs.append(expert(features))
        expert_outputs = torch.stack(expert_outputs, dim=1)  # (batch, n_experts, expert_dim)

        # 各任务预测
        task_outputs = []
        for task_id in range(self.n_tasks):
            # 门控权重
            gate_weights = self.gates[task_id](features)  # (batch, n_experts)

            # 加权融合专家输出
            gate_weights = gate_weights.unsqueeze(-1)  # (batch, n_experts, 1)
            weighted_expert = torch.sum(gate_weights * expert_outputs, dim=1)  # (batch, expert_dim)

            # 任务塔
            task_output = self.towers[task_id](weighted_expert)
            task_outputs.append(task_output.squeeze(-1))

        return task_outputs


class PLE(nn.Module):
    """
    Progressive Layered Extraction

    带任务特定专家的 MMOE 变体
    """

    def __init__(self,
                 feature_dim: int,
                 n_shared_experts: int = 4,
                 n_task_experts: int = 2,
                 expert_dim: int = 64,
                 n_tasks: int = 3):
        super().__init__()

        self.n_tasks = n_tasks

        # 共享专家
        self.shared_experts = nn.ModuleList([
            self._build_expert(feature_dim, expert_dim)
            for _ in range(n_shared_experts)
        ])

        # 任务特定专家
        self.task_experts = nn.ModuleList([
            nn.ModuleList([
                self._build_expert(feature_dim, expert_dim)
                for _ in range(n_task_experts)
            ]) for _ in range(n_tasks)
        ])

        # 门控
        total_experts = n_shared_experts + n_task_experts
        self.gates = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, total_experts),
                nn.Softmax(dim=-1)
            ) for _ in range(n_tasks)
        ])

        # 任务塔
        self.towers = nn.ModuleList([
            nn.Sequential(
                nn.Linear(expert_dim, 32),
                nn.ReLU(),
                nn.Linear(32, 1)
            ) for _ in range(n_tasks)
        ])

    def _build_expert(self, input_dim: int, output_dim: int) -> nn.Module:
        return nn.Sequential(
            nn.Linear(input_dim, output_dim),
            nn.ReLU(),
            nn.Linear(output_dim, output_dim)
        )

    def forward(self, features: torch.Tensor) -> List[torch.Tensor]:
        """前向传播"""
        batch_size = features.size(0)

        task_outputs = []

        for task_id in range(self.n_tasks):
            # 共享专家输出
            shared_outputs = [expert(features) for expert in self.shared_experts]

            # 任务特定专家输出
            task_specific_outputs = [
                expert(features) for expert in self.task_experts[task_id]
            ]

            # 合并所有专家
            all_expert_outputs = shared_outputs + task_specific_outputs
            all_expert_outputs = torch.stack(all_expert_outputs, dim=1)

            # 门控
            gate_weights = self.gates[task_id](features).unsqueeze(-1)

            # 加权融合
            weighted_output = torch.sum(gate_weights * all_expert_outputs, dim=1)

            # 任务塔
            task_output = self.towers[task_id](weighted_output)
            task_outputs.append(task_output.squeeze(-1))

        return task_outputs
```

### 2.2 ESMM 实现

```python
class ESMM(nn.Module):
    """
    Entire Space Multi-Task Model

    专门用于 CTR + CVR 多任务
    """

    def __init__(self,
                 feature_dim: int,
                 embed_dim: int = 64,
                 hidden_dims: List[int] = None):
        super().__init__()

        hidden_dims = hidden_dims or [256, 128]

        # 共享嵌入层
        self.embedding = nn.Linear(feature_dim, embed_dim)

        # CTR 网络
        self.ctr_net = self._build_network(embed_dim, hidden_dims)

        # CVR 网络
        self.cvr_net = self._build_network(embed_dim, hidden_dims)

    def _build_network(self, input_dim: int, hidden_dims: List[int]) -> nn.Module:
        """构建网络"""
        layers = []
        dim = input_dim
        for hidden_dim in hidden_dims:
            layers.extend([
                nn.Linear(dim, hidden_dim),
                nn.ReLU(),
                nn.Dropout(0.2)
            ])
            dim = hidden_dim
        layers.extend([
            nn.Linear(dim, 1),
            nn.Sigmoid()
        ])
        return nn.Sequential(*layers)

    def forward(self, features: torch.Tensor) -> Dict[str, torch.Tensor]:
        """
        前向传播

        返回:
            ctr: P(click | impression)
            cvr: P(conversion | click)
            ctcvr: P(conversion | impression) = ctr * cvr
        """
        embed = self.embedding(features)

        # CTR
        ctr = self.ctr_net(embed).squeeze(-1)

        # CVR
        cvr = self.cvr_net(embed).squeeze(-1)

        # CTCVR = CTR × CVR
        ctcvr = ctr * cvr

        return {
            'ctr': ctr,
            'cvr': cvr,
            'ctcvr': ctcvr
        }

    def compute_loss(self,
                    outputs: Dict[str, torch.Tensor],
                    click_labels: torch.Tensor,
                    conversion_labels: torch.Tensor) -> torch.Tensor:
        """
        ESMM 损失

        L = L_CTR + L_CTCVR

        注意: CVR 不直接有损失，通过 CTCVR 间接学习
        """
        # CTR 损失 (所有曝光样本)
        ctr_loss = F.binary_cross_entropy(
            outputs['ctr'],
            click_labels.float()
        )

        # CTCVR 损失 (所有曝光样本)
        ctcvr_loss = F.binary_cross_entropy(
            outputs['ctcvr'],
            conversion_labels.float()
        )

        return ctr_loss + ctcvr_loss
```

## 3. 目标融合

### 3.1 加权融合

```python
@dataclass
class ObjectiveConfig:
    """目标配置"""
    name: str
    weight: float
    higher_is_better: bool = True


class WeightedLoss:
    """
    加权损失

    简单的线性加权
    """

    def __init__(self, objectives: List[ObjectiveConfig]):
        self.objectives = objectives

    def compute(self,
               predictions: Dict[str, torch.Tensor],
               labels: Dict[str, torch.Tensor]) -> Tuple[torch.Tensor, Dict[str, float]]:
        """计算加权损失"""
        total_loss = 0.0
        individual_losses = {}

        for obj in self.objectives:
            pred = predictions[obj.name]
            label = labels[obj.name]

            # BCE 损失
            loss = F.binary_cross_entropy_with_logits(pred, label.float())

            # 加权
            weighted_loss = loss * obj.weight
            total_loss += weighted_loss

            individual_losses[obj.name] = loss.item()

        return total_loss, individual_losses


class LearnableWeightedLoss(nn.Module):
    """
    可学习权重损失

    自动学习任务权重 (Kendall et al.)
    """

    def __init__(self, n_tasks: int):
        super().__init__()

        # 可学习的 log 方差
        self.log_vars = nn.Parameter(torch.zeros(n_tasks))

    def forward(self,
               losses: List[torch.Tensor]) -> torch.Tensor:
        """
        计算加权损失

        weight = 1 / (2 * var) = exp(-log_var)
        loss = Σ weight * loss + log_var
        """
        total_loss = 0.0

        for i, loss in enumerate(losses):
            precision = torch.exp(-self.log_vars[i])
            total_loss += precision * loss + self.log_vars[i]

        return total_loss


class GradientNormalization:
    """
    梯度归一化 (GradNorm)

    动态调整权重使各任务梯度范数相近
    """

    def __init__(self, n_tasks: int, alpha: float = 1.5):
        self.n_tasks = n_tasks
        self.alpha = alpha
        self.initial_losses = None
        self.weights = nn.Parameter(torch.ones(n_tasks))

    def compute_weights(self, losses: List[float]) -> torch.Tensor:
        """计算动态权重"""
        if self.initial_losses is None:
            self.initial_losses = losses

        # 损失比率
        loss_ratios = [
            loss / (init_loss + 1e-10)
            for loss, init_loss in zip(losses, self.initial_losses)
        ]

        # 平均比率
        avg_ratio = sum(loss_ratios) / len(loss_ratios)

        # 计算权重
        weights = []
        for ratio in loss_ratios:
            w = (ratio / avg_ratio) ** self.alpha
            weights.append(w)

        # 归一化
        weights = torch.tensor(weights)
        weights = weights / weights.sum() * self.n_tasks

        return weights
```

### 3.2 推理融合

```python
class InferenceFusion:
    """
    推理时的分数融合
    """

    def __init__(self, weights: Dict[str, float]):
        self.weights = weights

    def fuse(self, scores: Dict[str, float]) -> float:
        """
        线性融合

        final_score = Σ w_i * score_i
        """
        total = 0.0
        for task, score in scores.items():
            weight = self.weights.get(task, 0.0)
            total += weight * score
        return total


class RankingScoreFusion:
    """
    排序分数融合

    用于重排阶段
    """

    def __init__(self,
                 ctr_weight: float = 0.5,
                 cvr_weight: float = 0.3,
                 diversity_weight: float = 0.2):
        self.ctr_weight = ctr_weight
        self.cvr_weight = cvr_weight
        self.diversity_weight = diversity_weight

    def compute_final_score(self,
                           ctr: float,
                           cvr: float,
                           diversity_score: float = 0.0,
                           price: float = 1.0) -> float:
        """
        计算最终排序分数

        考虑 GMV = CTR × CVR × Price
        """
        # 点击期望
        click_score = ctr * self.ctr_weight

        # 转化期望 (GMV 归一化)
        gmv_score = ctr * cvr * np.log1p(price) * self.cvr_weight

        # 多样性
        div_score = diversity_score * self.diversity_weight

        return click_score + gmv_score + div_score

    def rank(self,
            items: List[Dict],
            top_k: int = 50) -> List[Dict]:
        """
        排序
        """
        for item in items:
            item['final_score'] = self.compute_final_score(
                ctr=item.get('ctr', 0),
                cvr=item.get('cvr', 0),
                diversity_score=item.get('diversity_score', 0),
                price=item.get('price', 1)
            )

        # 排序
        items.sort(key=lambda x: x['final_score'], reverse=True)

        return items[:top_k]


class LearnableFusion(nn.Module):
    """
    可学习的融合网络
    """

    def __init__(self, n_tasks: int, hidden_dim: int = 32):
        super().__init__()

        self.fusion_net = nn.Sequential(
            nn.Linear(n_tasks, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1)
        )

    def forward(self, task_scores: torch.Tensor) -> torch.Tensor:
        """
        融合多任务分数

        task_scores: (batch, n_tasks)
        """
        return self.fusion_net(task_scores).squeeze(-1)
```

## 4. 训练与评估

### 4.1 训练器

```python
class MultiTaskTrainer:
    """
    多任务训练器
    """

    def __init__(self,
                 model: nn.Module,
                 tasks: List[str],
                 task_weights: Dict[str, float] = None,
                 lr: float = 0.001):
        self.model = model
        self.tasks = tasks

        if task_weights is None:
            self.task_weights = {task: 1.0 for task in tasks}
        else:
            self.task_weights = task_weights

        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    def train_epoch(self, dataloader) -> Dict[str, float]:
        """训练一个 epoch"""
        self.model.train()

        total_losses = {task: 0.0 for task in self.tasks}
        total_losses['total'] = 0.0
        n_batches = 0

        for batch in dataloader:
            self.optimizer.zero_grad()

            features = batch['features']

            # 前向传播
            if isinstance(self.model, ESMM):
                outputs = self.model(features)
                loss = self.model.compute_loss(
                    outputs,
                    batch['click'],
                    batch['conversion']
                )
                total_loss = loss
            else:
                predictions = self.model(features)

                # 计算各任务损失
                losses = {}
                total_loss = 0.0

                for task in self.tasks:
                    task_loss = F.binary_cross_entropy_with_logits(
                        predictions[self.tasks.index(task)],
                        batch[task].float()
                    )
                    losses[task] = task_loss
                    total_loss += task_loss * self.task_weights[task]
                    total_losses[task] += task_loss.item()

                loss = total_loss

            # 反向传播
            loss.backward()
            self.optimizer.step()

            total_losses['total'] += loss.item()
            n_batches += 1

        # 平均
        for key in total_losses:
            total_losses[key] /= n_batches

        return total_losses

    def evaluate(self, dataloader) -> Dict[str, Dict[str, float]]:
        """评估"""
        self.model.eval()

        task_preds = {task: [] for task in self.tasks}
        task_labels = {task: [] for task in self.tasks}

        with torch.no_grad():
            for batch in dataloader:
                features = batch['features']

                if isinstance(self.model, ESMM):
                    outputs = self.model(features)
                    predictions = [outputs['ctr'], outputs['ctcvr']]
                else:
                    predictions = self.model(features)

                for i, task in enumerate(self.tasks):
                    pred = torch.sigmoid(predictions[i])
                    task_preds[task].extend(pred.cpu().numpy())
                    task_labels[task].extend(batch[task].cpu().numpy())

        # 计算各任务指标
        from sklearn.metrics import roc_auc_score, log_loss

        metrics = {}
        for task in self.tasks:
            preds = np.array(task_preds[task])
            labels = np.array(task_labels[task])

            if len(np.unique(labels)) > 1:
                metrics[task] = {
                    'auc': roc_auc_score(labels, preds),
                    'logloss': log_loss(labels, preds)
                }
            else:
                metrics[task] = {'auc': 0.5, 'logloss': 0.0}

        return metrics
```

### 4.2 多目标评估

```python
class MultiObjectiveEvaluator:
    """
    多目标评估器
    """

    def __init__(self, objectives: List[str]):
        self.objectives = objectives

    def evaluate_ranking(self,
                        predictions: List[Dict],
                        ground_truth: Dict[str, List[int]],
                        k_list: List[int] = [10, 50, 100]) -> Dict[str, Dict[str, float]]:
        """
        评估排序效果

        predictions: 排序后的预测结果列表
        ground_truth: {task: [item_ids], ...} 各目标的真实相关物品
        """
        metrics = {}

        for task in self.objectives:
            relevant = set(ground_truth.get(task, []))
            task_metrics = {}

            for k in k_list:
                # 获取 Top-K
                top_k_items = [p['item_id'] for p in predictions[:k]]

                # Recall@K
                hit = len(set(top_k_items) & relevant)
                recall = hit / len(relevant) if relevant else 0
                task_metrics[f'recall@{k}'] = recall

                # Precision@K
                precision = hit / k
                task_metrics[f'precision@{k}'] = precision

            metrics[task] = task_metrics

        # 计算综合分数
        metrics['overall'] = self._compute_overall_score(metrics)

        return metrics

    def _compute_overall_score(self, metrics: Dict) -> Dict[str, float]:
        """计算综合分数"""
        # 简单平均
        overall = {}

        for k in [10, 50, 100]:
            recall_key = f'recall@{k}'
            avg_recall = np.mean([
                metrics[task].get(recall_key, 0)
                for task in self.objectives
            ])
            overall[f'avg_recall@{k}'] = avg_recall

        return overall
```

## 5. 学习总结

### 5.1 核心要点

```
1. 模型选择: MMOE通用、ESMM适合CTR+CVR
2. 权重设置: 等权重开始，逐步调整
3. 损失融合: 简单加权或可学习权重
4. 推理融合: 考虑业务目标 (如GMV)
5. 评估: 各任务独立评估 + 综合评估
```

### 5.2 最佳实践

```
1. 任务相关性: 相关任务才适合多任务
2. 负迁移: 监控是否有任务效果下降
3. 权重调优: 通过验证集调整
4. 在线评估: 最终以A/B测试为准
5. 业务理解: 融合权重应符合业务目标
```
