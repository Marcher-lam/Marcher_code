### 2.1 数据加载与预处理

```python
import pandas as pd
import numpy as np
from typing import List, Dict, Tuple
from sklearn.preprocessing import LabelEncoder, MinMaxScaler
from sklearn.model_selection import train_test_split
import pickle
import os


class CTRDataProcessor:
    """
    CTR 数据处理器
    """

    def __init__(self, data_path: str, cache_dir: str = './cache'):
        self.data_path = data_path
        self.cache_dir = cache_dir
        os.makedirs(cache_dir, exist_ok=True)

        # 特征配置
        self.sparse_features = []
        self.dense_features = []
        self.target_col = 'click'

        # 编码器
        self.label_encoders = {}
        self.scalers = {}

        # 特征统计
        self.feature_stats = {}

    def load_data(self) -> pd.DataFrame:
        """加载数据"""
        print("加载数据...")
        df = pd.read_csv(self.data_path)
        print(f"数据形状: {df.shape}")
        return df

    def identify_features(self, df: pd.DataFrame,
                         sparse_threshold: int = 100):
        """自动识别稀疏/稠密特征"""
        for col in df.columns:
            if col == self.target_col:
                continue

            if df[col].dtype == 'object' or df[col].nunique() <= sparse_threshold:
                self.sparse_features.append(col)
            else:
                self.dense_features.append(col)

        print(f"稀疏特征 ({len(self.sparse_features)}): {self.sparse_features[:5]}...")
        print(f"稠密特征 ({len(self.dense_features)}): {self.dense_features[:5]}...")

    def preprocess(self, df: pd.DataFrame) -> pd.DataFrame:
        """数据预处理"""
        print("预处理数据...")

        # 处理缺失值
        for col in self.sparse_features:
            df[col] = df[col].fillna('-1')

        for col in self.dense_features:
            df[col] = df[col].fillna(0)

        # 稀疏特征编码
        for col in self.sparse_features:
            if col not in self.label_encoders:
                self.label_encoders[col] = LabelEncoder()
                df[col] = self.label_encoders[col].fit_transform(df[col].astype(str))
                # 保存编码器
                with open(f'{self.cache_dir}/{col}_encoder.pkl', 'wb') as f:
                    pickle.dump(self.label_encoders[col], f)
            else:
                df[col] = df[col].map(
                    lambda x: self.label_encoders[col].transform([str(x)])[0]
                    if str(x) in self.label_encoders[col].classes_ else 0
                )

        # 稠密特征归一化
        for col in self.dense_features:
            if col not in self.scalers:
                self.scalers[col] = MinMaxScaler()
                df[col] = self.scalers[col].fit_transform(df[[col]])
                with open(f'{self.cache_dir}/{col}_scaler.pkl', 'wb') as f:
                    pickle.dump(self.scalers[col], f)
            else:
                df[col] = self.scalers[col].transform(df[[col]])

        # 保存特征统计
        self.feature_stats = {
            'sparse_features': self.sparse_features,
            'dense_features': self.dense_features,
            'sparse_dims': {col: df[col].nunique() for col in self.sparse_features}
        }

        with open(f'{self.cache_dir}/feature_stats.pkl', 'wb') as f:
            pickle.dump(self.feature_stats, f)

        return df

    def split_data(self, df: pd.DataFrame,
                   test_size: float = 0.2,
                   val_size: float = 0.1) -> Tuple:
        """划分数据集"""
        print("划分数据集...")

        # 先划分出测试集
        train_df, test_df = train_test_split(
            df, test_size=test_size,
            random_state=42, stratify=df[self.target_col]
        )

        # 再从训练集划分验证集
        val_ratio = val_size / (1 - test_size)
        train_df, val_df = train_test_split(
            train_df, test_size=val_ratio,
            random_state=42, stratify=train_df[self.target_col]
        )

        print(f"训练集: {len(train_df)}, 验证集: {len(val_df)}, 测试集: {len(test_df)}")

        return train_df, val_df, test_df


class FeatureGenerator:
    """
    特征生成器
    """

    def __init__(self):
        self.cross_features = []

    def add_cross_feature(self, features: List[str]):
        """添加交叉特征"""
        self.cross_features.append(features)

    def generate_cross_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """生成交叉特征"""
        for features in self.cross_features:
            name = '_'.join(features)
            df[name] = df[features[0]].astype(str) + '_' + df[features[1]].astype(str)
            df[name] = pd.factorize(df[name])[0]

        return df

    @staticmethod
    def generate_time_features(df: pd.DataFrame,
                               timestamp_col: str) -> pd.DataFrame:
        """生成时间特征"""
        df['hour'] = pd.to_datetime(df[timestamp_col]).dt.hour
        df['day_of_week'] = pd.to_datetime(df[timestamp_col]).dt.dayofweek
        df['is_weekend'] = (df['day_of_week'] >= 5).astype(int)

        # 时间段
        df['time_period'] = pd.cut(
            df['hour'],
            bins=[0, 6, 12, 18, 24],
            labels=['night', 'morning', 'afternoon', 'evening']
        )

        return df


class CTRDataset(torch.utils.data.Dataset):
    """
    CTR 数据集
    """

    def __init__(self, df: pd.DataFrame,
                 sparse_features: List[str],
                 dense_features: List[str],
                 target_col: str = 'click'):
        self.sparse_features = df[sparse_features].values.astype(np.int64)
        self.dense_features = df[dense_features].values.astype(np.float32)
        self.targets = df[target_col].values.astype(np.float32)

    def __len__(self):
        return len(self.targets)

    def __getitem__(self, idx):
        return {
            'sparse': torch.tensor(self.sparse_features[idx]),
            'dense': torch.tensor(self.dense_features[idx]),
            'target': torch.tensor(self.targets[idx])
        }


def create_dataloaders(train_df, val_df, test_df,
                      sparse_features, dense_features,
                      batch_size=256):
    """创建数据加载器"""
    train_dataset = CTRDataset(train_df, sparse_features, dense_features)
    val_dataset = CTRDataset(val_df, sparse_features, dense_features)
    test_dataset = CTRDataset(test_df, sparse_features, dense_features)

    train_loader = torch.utils.data.DataLoader(
        train_dataset, batch_size=batch_size, shuffle=True, num_workers=4
    )
    val_loader = torch.utils.data.DataLoader(
        val_dataset, batch_size=batch_size, shuffle=False, num_workers=4
    )
    test_loader = torch.utils.data.DataLoader(
        test_dataset, batch_size=batch_size, shuffle=False, num_workers=4
    )

    return train_loader, val_loader, test_loader
```

### 2.2 模型定义

```python
import torch
import torch.nn as nn
import torch.nn.functional as F


class DeepFM(nn.Module):
    """
    DeepFM 模型
    """

    def __init__(self, sparse_dims: Dict[str, int],
                 dense_dim: int,
                 embed_dim: int = 10,
                 mlp_dims: List[int] = [256, 128, 64],
                 dropout: float = 0.2):
        super().__init__()

        self.sparse_dims = sparse_dims
        self.dense_dim = dense_dim
        self.embed_dim = embed_dim

        # 稀疏特征嵌入
        self.embeddings = nn.ModuleDict({
            name: nn.Embedding(dim, embed_dim)
            for name, dim in sparse_dims.items()
        })

        # 线性部分
        self.linear_sparse = nn.ModuleDict({
            name: nn.Embedding(dim, 1)
            for name, dim in sparse_dims.items()
        })
        self.linear_dense = nn.Linear(dense_dim, 1)
        self.bias = nn.Parameter(torch.zeros(1))

        # FM 部分
        # 隐式通过嵌入层实现

        # Deep 部分
        total_embed_dim = len(sparse_dims) * embed_dim + dense_dim
        self.mlp = nn.Sequential(
            nn.Linear(total_embed_dim, mlp_dims[0]),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(mlp_dims[0], mlp_dims[1]),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(mlp_dims[1], 1)
        )

    def forward(self, sparse_features: Dict[str, torch.Tensor],
               dense_features: torch.Tensor) -> torch.Tensor:
        batch_size = dense_features.size(0)

        # 线性部分
        linear_part = self.bias
        for name, emb in self.linear_sparse.items():
            linear_part = linear_part + emb(sparse_features[name])
        linear_part = linear_part + self.linear_dense(dense_features)

        # 嵌入
        sparse_embeds = []
        for name, emb in self.embeddings.items():
            sparse_embeds.append(emb(sparse_features[name]))

        sparse_embeds = torch.stack(sparse_embeds, dim=1)  # (batch, n_sparse, embed_dim)

        # FM 二阶交互
        square_of_sum = torch.sum(sparse_embeds, dim=1) ** 2  # (batch, embed_dim)
        sum_of_square = torch.sum(sparse_embeds ** 2, dim=1)  # (batch, embed_dim)
        fm_part = 0.5 * torch.sum(square_of_sum - sum_of_square, dim=1, keepdim=True)

        # Deep 部分
        sparse_embeds_flat = sparse_embeds.view(batch_size, -1)
        deep_input = torch.cat([sparse_embeds_flat, dense_features], dim=1)
        deep_part = self.mlp(deep_input)

        # 组合
        output = linear_part + fm_part + deep_part

        return output.squeeze(-1)

    def predict(self, sparse_features, dense_features):
        """预测概率"""
        logits = self.forward(sparse_features, dense_features)
        return torch.sigmoid(logits)
```

### 2.3 训练流程

```python
from sklearn.metrics import roc_auc_score, log_loss
import tqdm


class CTRTrainer:
    """
    CTR 模型训练器
    """

    def __init__(self, model, device='cuda', lr=0.001, weight_decay=1e-5):
        self.model = model.to(device)
        self.device = device

        self.optimizer = torch.optim.Adam(
            model.parameters(),
            lr=lr,
            weight_decay=weight_decay
        )
        self.criterion = nn.BCEWithLogitsLoss()

        self.best_auc = 0
        self.best_model_state = None

    def train_epoch(self, dataloader):
        """训练一个 epoch"""
        self.model.train()
        total_loss = 0

        for batch in tqdm.tqdm(dataloader, desc="Training"):
            sparse = {k: v.to(self.device) for k, v in batch['sparse'].items()}
            dense = batch['dense'].to(self.device)
            targets = batch['target'].to(self.device)

            self.optimizer.zero_grad()

            logits = self.model(sparse, dense)
            loss = self.criterion(logits, targets)

            loss.backward()
            self.optimizer.step()

            total_loss += loss.item()

        return total_loss / len(dataloader)

    def evaluate(self, dataloader):
        """评估"""
        self.model.eval()
        all_preds = []
        all_targets = []

        with torch.no_grad():
            for batch in dataloader:
                sparse = {k: v.to(self.device) for k, v in batch['sparse'].items()}
                dense = batch['dense'].to(self.device)
                targets = batch['target']

                preds = self.model.predict(sparse, dense)

                all_preds.extend(preds.cpu().numpy())
                all_targets.extend(targets.numpy())

        all_preds = np.array(all_preds)
        all_targets = np.array(all_targets)

        auc = roc_auc_score(all_targets, all_preds)
        logloss = log_loss(all_targets, all_preds)

        return {'auc': auc, 'logloss': logloss}

    def train(self, train_loader, val_loader, epochs=10, early_stop=3):
        """完整训练流程"""
        no_improve = 0

        for epoch in range(epochs):
            print(f"\nEpoch {epoch+1}/{epochs}")

            # 训练
            train_loss = self.train_epoch(train_loader)
            print(f"Train Loss: {train_loss:.4f}")

            # 验证
            val_metrics = self.evaluate(val_loader)
            print(f"Val AUC: {val_metrics['auc']:.4f}, LogLoss: {val_metrics['logloss']:.4f}")

            # 保存最佳模型
            if val_metrics['auc'] > self.best_auc:
                self.best_auc = val_metrics['auc']
                self.best_model_state = self.model.state_dict().copy()
                no_improve = 0
            else:
                no_improve += 1

            # 早停
            if no_improve >= early_stop:
                print(f"Early stopping at epoch {epoch+1}")
                break

        # 恢复最佳模型
        if self.best_model_state:
            self.model.load_state_dict(self.best_model_state)

    def save_model(self, path):
        """保存模型"""
        torch.save({
            'model_state': self.model.state_dict(),
            'optimizer_state': self.optimizer.state_dict(),
            'best_auc': self.best_auc
        }, path)
        print(f"模型已保存到 {path}")

    def load_model(self, path):
        """加载模型"""
        checkpoint = torch.load(path)
        self.model.load_state_dict(checkpoint['model_state'])
        self.optimizer.load_state_dict(checkpoint['optimizer_state'])
        self.best_auc = checkpoint['best_auc']
        print(f"模型已从 {path} 加载")


def run_pipeline(data_path: str, output_dir: str = './output'):
    """
    运行完整训练流程
    """
    os.makedirs(output_dir, exist_ok=True)

    # 1. 数据处理
    processor = CTRDataProcessor(data_path)
    df = processor.load_data()
    processor.identify_features(df)
    df = processor.preprocess(df)
    train_df, val_df, test_df = processor.split_data(df)

    # 2. 创建数据加载器
    train_loader, val_loader, test_loader = create_dataloaders(
        train_df, val_df, test_df,
        processor.sparse_features,
        processor.dense_features
    )

    # 3. 创建模型
    model = DeepFM(
        sparse_dims=processor.feature_stats['sparse_dims'],
        dense_dim=len(processor.dense_features)
    )

    # 4. 训练
    trainer = CTRTrainer(model)
    trainer.train(train_loader, val_loader, epochs=10)

    # 5. 测试
    test_metrics = trainer.evaluate(test_loader)
    print(f"\nTest AUC: {test_metrics['auc']:.4f}")

    # 6. 保存
    trainer.save_model(f'{output_dir}/model.pt')

    return trainer, test_metrics


if __name__ == "__main__":
```

### 3.2 特征服务

```python
import redis
import json
from typing import Dict, Optional


class FeatureService:
    """
    特征服务

    从 Redis 获取实时特征
    """

    def __init__(self, redis_host: str = 'localhost', redis_port: int = 6379):
        self.redis = redis.Redis(host=redis_host, port=redis_port, db=0)

    def get_user_features(self, user_id: str) -> Dict:
        """获取用户特征"""
        key = f"user:{user_id}"
        features = self.redis.hgetall(key)
        return {k.decode(): v.decode() for k, v in features.items()}

    def get_item_features(self, item_id: str) -> Dict:
        """获取物品特征"""
        key = f"item:{item_id}"
        features = self.redis.hgetall(key)
        return {k.decode(): v.decode() for k, v in features.items()}

    def update_user_realtime_features(self, user_id: str, features: Dict):
        """更新用户实时特征"""
        key = f"user_realtime:{user_id}"
        self.redis.hmset(key, features)
        self.redis.expire(key, 3600)  # 1小时过期

    def get_context_features(self) -> Dict:
        """获取上下文特征（时间等）"""
        from datetime import datetime
        now = datetime.now()
        return {
            'hour': now.hour,
            'day_of_week': now.weekday(),
            'is_weekend': int(now.weekday() >= 5)
        }
```

## 4. 监控与评估

```python
class CTRMonitor:
    """
    CTR 监控器
    """

    def __init__(self):
        self.predictions = []
        self.actuals = []

    def log_prediction(self, pred: float, actual: int = None):
        """记录预测"""
        self.predictions.append(pred)
        if actual is not None:
            self.actuals.append(actual)

    def compute_metrics(self) -> Dict:
        """计算指标"""
        if not self.actuals:
            return {}

        preds = np.array(self.predictions[-1000:])  # 最近1000条
        actuals = np.array(self.actuals[-1000:])

        return {
            'auc': roc_auc_score(actuals, preds),
            'avg_ctr': np.mean(preds),
            'actual_ctr': np.mean(actuals)
        }
```
