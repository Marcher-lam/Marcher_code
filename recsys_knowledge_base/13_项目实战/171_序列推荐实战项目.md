## 2. 数据准备

### 2.1 数据加载与预处理

```python
import pandas as pd
import numpy as np
from typing import List, Dict, Tuple
from collections import defaultdict
import torch
from torch.utils.data import Dataset, DataLoader
import os


class SequenceDataProcessor:
    """
    序列推荐数据处理器
    """

    def __init__(self, data_path: str, max_seq_len: int = 50):
        self.data_path = data_path
        self.max_seq_len = max_seq_len

        # ID 映射
        self.user2id = {}
        self.item2id = {}
        self.id2item = {}

        # 统计信息
        self.n_users = 0
        self.n_items = 0

    def load_data(self) -> pd.DataFrame:
        """加载数据"""
        print("加载数据...")
        df = pd.read_csv(self.data_path)
        print(f"原始数据: {len(df)} 条交互")
        return df

    def preprocess(self, df: pd.DataFrame) -> pd.DataFrame:
        """预处理"""
        print("预处理数据...")

        # 去重
        df = df.drop_duplicates(subset=['user_id', 'item_id', 'timestamp'])

        # 排序
        df = df.sort_values(['user_id', 'timestamp'])

        # 过滤低频物品
        item_counts = df['item_id'].value_counts()
        valid_items = item_counts[item_counts >= 5].index
        df = df[df['item_id'].isin(valid_items)]

        # 构建 ID 映射
        unique_users = df['user_id'].unique()
        unique_items = df['item_id'].unique()

        self.user2id = {u: i for i, u in enumerate(unique_users)}
        self.item2id = {item: i for i, item in enumerate(unique_items)}
        self.id2item = {i: item for item, i in self.item2id.items()}

        self.n_users = len(unique_users)
        self.n_items = len(unique_items)

        print(f"用户数: {self.n_users}, 物品数: {self.n_items}")

        # 映射 ID
        df['user_id_enc'] = df['user_id'].map(self.user2id)
        df['item_id_enc'] = df['item_id'].map(self.item2id)

        return df

    def build_sequences(self, df: pd.DataFrame) -> Dict[int, List[int]]:
        """
        构建用户行为序列
        """
        print("构建序列...")

        user_sequences = defaultdict(list)

        for _, row in df.iterrows():
            user_id = row['user_id_enc']
            item_id = row['item_id_enc']
            user_sequences[user_id].append(item_id)

        # 截断/填充
        processed_sequences = {}
        for user_id, seq in user_sequences.items():
            if len(seq) > self.max_seq_len:
                seq = seq[-self.max_seq_len:]
            processed_sequences[user_id] = seq

        print(f"构建了 {len(processed_sequences)} 个用户序列")

        return processed_sequences

    def split_data(self, sequences: Dict[int, List[int]],
                   test_ratio: float = 0.2) -> Tuple:
        """
        划分训练/测试数据

        对于序列: 使用最后一个物品作为测试目标
        """
        print("划分数据...")

        train_data = []
        test_data = []

        for user_id, seq in sequences.items():
            if len(seq) < 2:
                continue

            # 训练数据: 前n-1个预测第n个
            for i in range(1, len(seq) - 1):
                train_data.append({
                    'user_id': user_id,
                    'sequence': seq[:i],
                    'target': seq[i]
                })

            # 测试数据: 前n-1个预测最后一个
            test_data.append({
                'user_id': user_id,
                'sequence': seq[:-1],
                'target': seq[-1]
            })

        print(f"训练样本: {len(train_data)}, 测试样本: {len(test_data)}")

        return train_data, test_data


class SequenceDataset(Dataset):
    """
    序列数据集
    """

    def __init__(self, data: List[Dict], n_items: int,
                 max_seq_len: int = 50, mask_token: int = 0):
        self.data = data
        self.n_items = n_items
        self.max_seq_len = max_seq_len
        self.mask_token = mask_token

    def __len__(self):
        return len(self.data)

    def __getitem__(self, idx):
        item = self.data[idx]

        seq = item['sequence']
        target = item['target']

        # 填充
        seq_len = len(seq)
        padded_seq = [self.mask_token] * (self.max_seq_len - seq_len) + seq

        # 注意力掩码
        attention_mask = [0] * (self.max_seq_len - seq_len) + [1] * seq_len

        return {
            'user_id': torch.tensor(item['user_id'], dtype=torch.long),
            'sequence': torch.tensor(padded_seq, dtype=torch.long),
            'attention_mask': torch.tensor(attention_mask, dtype=torch.long),
            'target': torch.tensor(target, dtype=torch.long),
            'seq_len': torch.tensor(seq_len, dtype=torch.long)
        }


def create_dataloaders(train_data, test_data, n_items,
                      max_seq_len, batch_size=256):
    """创建数据加载器"""
    train_dataset = SequenceDataset(train_data, n_items, max_seq_len)
    test_dataset = SequenceDataset(test_data, n_items, max_seq_len)

    train_loader = DataLoader(
        train_dataset, batch_size=batch_size,
        shuffle=True, num_workers=4
    )
    test_loader = DataLoader(
        test_dataset, batch_size=batch_size,
        shuffle=False, num_workers=4
    )

    return train_loader, test_loader
```

## 3. 模型定义

### 3.1 GRU4Rec 模型

```python
import torch
import torch.nn as nn
import torch.nn.functional as F


class GRU4Rec(nn.Module):
    """
    GRU4Rec 模型

    基于 GRU 的序列推荐
    """

    def __init__(self, n_items: int, embed_dim: int = 64,
                 hidden_dim: int = 128, n_layers: int = 2,
                 dropout: float = 0.2):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim
        self.hidden_dim = hidden_dim

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items + 1, embed_dim, padding_idx=0)

        # GRU
        self.gru = nn.GRU(
            embed_dim, hidden_dim,
            num_layers=n_layers,
            dropout=dropout if n_layers > 1 else 0,
            batch_first=True
        )

        # 输出层
        self.output_layer = nn.Sequential(
            nn.Linear(hidden_dim, embed_dim),
            nn.ReLU()
        )

        self.dropout = nn.Dropout(dropout)

    def forward(self, sequence: torch.Tensor,
               attention_mask: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播

        参数:
            sequence: (batch, seq_len)
            attention_mask: (batch, seq_len)

        返回:
            logits: (batch, n_items)
        """
        # 嵌入
        embeds = self.item_embedding(sequence)  # (batch, seq_len, embed_dim)
        embeds = self.dropout(embeds)

        # GRU
        if attention_mask is not None:
            lengths = attention_mask.sum(dim=1).cpu()
            packed = nn.utils.rnn.pack_padded_sequence(
                embeds, lengths, batch_first=True, enforce_sorted=False
            )
            packed_output, hidden = self.gru(packed)
            output, _ = nn.utils.rnn.pad_packed_sequence(packed_output, batch_first=True)
        else:
            output, hidden = self.gru(embeds)

        # 取最后一个有效位置
        if attention_mask is not None:
            batch_size = sequence.size(0)
            last_indices = attention_mask.sum(dim=1) - 1
            last_hidden = output[torch.arange(batch_size), last_indices]
        else:
            last_hidden = output[:, -1, :]

        # 输出
        output_embed = self.output_layer(last_hidden)

        # 与所有物品计算分数
        item_embeds = self.item_embedding.weight[1:]  # 排除 padding
        logits = torch.matmul(output_embed, item_embeds.T)

        return logits

    def predict_next(self, sequence: torch.Tensor,
                    attention_mask: torch.Tensor = None,
                    top_k: int = 10) -> torch.Tensor:
        """预测下一个物品"""
        logits = self.forward(sequence, attention_mask)
        _, top_indices = torch.topk(logits, top_k, dim=-1)
        return top_indices
```

### 3.2 SASRec 模型

```python
class SASRec(nn.Module):
    """
    SASRec: Self-Attentive Sequential Recommendation

    基于 Transformer 的序列推荐
    """

    def __init__(self, n_items: int, embed_dim: int = 64,
                 n_heads: int = 2, n_layers: int = 2,
                 max_seq_len: int = 50, dropout: float = 0.2):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim
        self.max_seq_len = max_seq_len

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items + 1, embed_dim, padding_idx=0)

        # 位置嵌入
        self.position_embedding = nn.Embedding(max_seq_len, embed_dim)

        # Transformer 编码器
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim,
            nhead=n_heads,
            dim_feedforward=embed_dim * 4,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)

        # Layer Norm
        self.layer_norm = nn.LayerNorm(embed_dim)

        self.dropout = nn.Dropout(dropout)

        # 初始化
        self._init_weights()

    def _init_weights(self):
        nn.init.normal_(self.item_embedding.weight, std=0.02)
        nn.init.normal_(self.position_embedding.weight, std=0.02)

    def forward(self, sequence: torch.Tensor,
               attention_mask: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播
        """
        batch_size, seq_len = sequence.size()

        # 物品嵌入
        item_embeds = self.item_embedding(sequence)

        # 位置嵌入
        positions = torch.arange(seq_len, device=sequence.device).unsqueeze(0)
        pos_embeds = self.position_embedding(positions)

        # 组合
        embeds = item_embeds + pos_embeds
        embeds = self.layer_norm(embeds)
        embeds = self.dropout(embeds)

        # 因果掩码（防止看到未来）
        causal_mask = torch.triu(
            torch.ones(seq_len, seq_len, device=sequence.device) * float('-inf'),
            diagonal=1
        )

        # 填充掩码
        if attention_mask is not None:
            key_padding_mask = (attention_mask == 0)
        else:
            key_padding_mask = None

        # Transformer
        output = self.transformer(
            embeds,
            mask=causal_mask,
            src_key_padding_mask=key_padding_mask
        )

        # 取最后一个有效位置
        if attention_mask is not None:
            last_indices = attention_mask.sum(dim=1) - 1
            last_output = output[torch.arange(batch_size), last_indices]
        else:
            last_output = output[:, -1, :]

        # 与所有物品计算分数
        item_embeds = self.item_embedding.weight[1:]
        logits = torch.matmul(last_output, item_embeds.T)

        return logits

    def predict_next(self, sequence: torch.Tensor,
                    attention_mask: torch.Tensor = None,
                    top_k: int = 10) -> torch.Tensor:
        """预测下一个物品"""
        logits = self.forward(sequence, attention_mask)
        _, top_indices = torch.topk(logits, top_k, dim=-1)
        return top_indices


class BERT4Rec(nn.Module):
    """
    BERT4Rec: Bidirectional Encoder Representations from Transformer for Recommendation

    双向 Transformer + Masked Item Prediction
    """

    def __init__(self, n_items: int, embed_dim: int = 64,
                 n_heads: int = 2, n_layers: int = 2,
                 max_seq_len: int = 50, dropout: float = 0.2,
                 mask_ratio: float = 0.2):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim
        self.mask_ratio = mask_ratio

        # 特殊 token
        self.mask_token = n_items  # mask token ID
        self.pad_token = 0

        # 嵌入
        self.item_embedding = nn.Embedding(n_items + 2, embed_dim, padding_idx=0)
        self.position_embedding = nn.Embedding(max_seq_len, embed_dim)

        # Transformer
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim,
            nhead=n_heads,
            dim_feedforward=embed_dim * 4,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)

        self.layer_norm = nn.LayerNorm(embed_dim)
        self.dropout = nn.Dropout(dropout)

        # 输出层
        self.output_layer = nn.Linear(embed_dim, n_items + 1)

    def forward(self, sequence: torch.Tensor,
               attention_mask: torch.Tensor = None,
               masked_positions: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播
        """
        batch_size, seq_len = sequence.size()

        # 嵌入
        item_embeds = self.item_embedding(sequence)
        positions = torch.arange(seq_len, device=sequence.device).unsqueeze(0)
        pos_embeds = self.position_embedding(positions)

        embeds = item_embeds + pos_embeds
        embeds = self.layer_norm(embeds)
        embeds = self.dropout(embeds)

        # 注意力掩码
        if attention_mask is not None:
            key_padding_mask = (attention_mask == 0)
        else:
            key_padding_mask = None

        # Transformer（双向）
        output = self.transformer(embeds, src_key_padding_mask=key_padding_mask)

        # 输出
        logits = self.output_layer(output)

        return logits

    def mask_sequence(self, sequence: torch.Tensor,
                     attention_mask: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        随机遮掩序列
        """
        batch_size, seq_len = sequence.size()

        # 创建遮掩
        masked_sequence = sequence.clone()
        masked_positions = torch.zeros_like(sequence)

        for i in range(batch_size):
            valid_len = attention_mask[i].sum().item()
            n_mask = int(valid_len * self.mask_ratio)

            if n_mask > 0:
                mask_indices = torch.randperm(int(valid_len))[:n_mask]
                masked_sequence[i, mask_indices] = self.mask_token
                masked_positions[i, mask_indices] = 1

        return masked_sequence, masked_positions
```

## 4. 训练流程

### 4.1 训练器

```python
from sklearn.metrics import ndcg_score
import tqdm


class SequenceRecTrainer:
    """
    序列推荐训练器
    """

    def __init__(self, model, device='cuda', lr=0.001,
                 weight_decay=1e-5, warmup_steps=1000):
        self.model = model.to(device)
        self.device = device

        # 优化器
        self.optimizer = torch.optim.AdamW(
            model.parameters(),
            lr=lr,
            weight_decay=weight_decay
        )

        # 学习率调度器
        self.scheduler = torch.optim.lr_scheduler.OneCycleLR(
            self.optimizer,
            max_lr=lr,
            total_steps=warmup_steps * 10,
            pct_start=0.1
        )

        # 损失函数
        self.criterion = nn.CrossEntropyLoss(ignore_index=0)

        # 最佳模型
        self.best_metrics = {}
        self.best_model_state = None

    def train_epoch(self, dataloader) -> float:
        """训练一个 epoch"""
        self.model.train()
        total_loss = 0
        n_batches = 0

        for batch in tqdm.tqdm(dataloader, desc="Training"):
            sequence = batch['sequence'].to(self.device)
            attention_mask = batch['attention_mask'].to(self.device)
            target = batch['target'].to(self.device)

            self.optimizer.zero_grad()

            logits = self.model(sequence, attention_mask)
            loss = self.criterion(logits, target)

            loss.backward()
            torch.nn.utils.clip_grad_norm_(self.model.parameters(), 5.0)
            self.optimizer.step()
            self.scheduler.step()

            total_loss += loss.item()
            n_batches += 1

        return total_loss / n_batches

    def evaluate(self, dataloader, k_list=[5, 10, 20]) -> Dict:
        """评估"""
        self.model.eval()

        all_preds = []
        all_targets = []

        with torch.no_grad():
            for batch in tqdm.tqdm(dataloader, desc="Evaluating"):
                sequence = batch['sequence'].to(self.device)
                attention_mask = batch['attention_mask'].to(self.device)
                target = batch['target'].to(self.device)

                logits = self.model(sequence, attention_mask)

                all_preds.append(logits.cpu())
                all_targets.append(target.cpu())

        all_preds = torch.cat(all_preds, dim=0)
        all_targets = torch.cat(all_targets, dim=0)

        # 计算指标
        metrics = {}

        for k in k_list:
            hr = self._hit_rate(all_preds, all_targets, k)
            ndcg = self._ndcg(all_preds, all_targets, k)
            mrr = self._mrr(all_preds, all_targets, k)

            metrics[f'HR@{k}'] = hr
            metrics[f'NDCG@{k}'] = ndcg
            metrics[f'MRR@{k}'] = mrr

        return metrics

    def _hit_rate(self, preds: torch.Tensor, targets: torch.Tensor, k: int) -> float:
        """Hit Rate@K"""
        _, top_k = torch.topk(preds, k, dim=-1)
        hits = (top_k == targets.unsqueeze(1)).any(dim=1).float()
        return hits.mean().item()

    def _ndcg(self, preds: torch.Tensor, targets: torch.Tensor, k: int) -> float:
        """NDCG@K"""
        _, top_k = torch.topk(preds, k, dim=-1)

        ndcg_scores = []
        for i in range(len(targets)):
            target = targets[i].item()
            rank = (top_k[i] == target).nonzero(as_tuple=True)[0]

            if len(rank) > 0:
                ndcg_scores.append(1.0 / np.log2(rank[0].item() + 2))
            else:
                ndcg_scores.append(0.0)

        return np.mean(ndcg_scores)

    def _mrr(self, preds: torch.Tensor, targets: torch.Tensor, k: int) -> float:
        """MRR@K"""
        _, top_k = torch.topk(preds, k, dim=-1)

        rr_scores = []
        for i in range(len(targets)):
            target = targets[i].item()
            rank = (top_k[i] == target).nonzero(as_tuple=True)[0]

            if len(rank) > 0:
                rr_scores.append(1.0 / (rank[0].item() + 1))
            else:
                rr_scores.append(0.0)

        return np.mean(rr_scores)

    def train(self, train_loader, val_loader, epochs=10, early_stop=3):
        """完整训练"""
        no_improve = 0
        best_hr = 0

        for epoch in range(epochs):
            print(f"\nEpoch {epoch+1}/{epochs}")

            # 训练
            train_loss = self.train_epoch(train_loader)
            print(f"Train Loss: {train_loss:.4f}")

            # 验证
            val_metrics = self.evaluate(val_loader)
            print(f"Val HR@10: {val_metrics['HR@10']:.4f}, "
                  f"NDCG@10: {val_metrics['NDCG@10']:.4f}")

            # 保存最佳模型
            if val_metrics['HR@10'] > best_hr:
                best_hr = val_metrics['HR@10']
                self.best_metrics = val_metrics
                self.best_model_state = self.model.state_dict().copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= early_stop:
                print(f"Early stopping at epoch {epoch+1}")
                break

        # 恢复最佳模型
        if self.best_model_state:
            self.model.load_state_dict(self.best_model_state)

    def save_model(self, path: str):
        """保存模型"""
        torch.save({
            'model_state': self.model.state_dict(),
            'best_metrics': self.best_metrics
        }, path)
        print(f"模型已保存到 {path}")

    def load_model(self, path: str):
        """加载模型"""
        checkpoint = torch.load(path)
        self.model.load_state_dict(checkpoint['model_state'])
        print(f"模型已从 {path} 加载")
```

def run_sequence_rec_pipeline(data_path: str, output_dir: str = './output'):
    """
    运行完整序列推荐流程
    """
    os.makedirs(output_dir, exist_ok=True)

    # 1. 数据处理
    processor = SequenceDataProcessor(data_path, max_seq_len=50)
    df = processor.load_data()
    df = processor.preprocess(df)
    sequences = processor.build_sequences(df)
    train_data, test_data = processor.split_data(sequences)

    # 2. 创建数据加载器
    train_loader, test_loader = create_dataloaders(
        train_data, test_data,
        processor.n_items, 50, batch_size=256
    )

    # 3. 创建模型
    model = SASRec(
        n_items=processor.n_items,
        embed_dim=64,
        n_heads=2,
        n_layers=2,
        max_seq_len=50
    )

    # 4. 训练
    trainer = SequenceRecTrainer(model)
    trainer.train(train_loader, test_loader, epochs=10)

    # 5. 评估
    test_metrics = trainer.evaluate(test_loader)
    print(f"\nTest Metrics:")
    for k, v in test_metrics.items():
        print(f"  {k}: {v:.4f}")

    # 6. 保存
    trainer.save_model(f'{output_dir}/sasrec_model.pt')

    return trainer, test_metrics

## 6. 学习总结

### 6.1 核心要点

1. **序列建模**: 捕捉用户行为时序
2. **Transformer**: 注意力机制建模依赖
3. **评估指标**: HR, NDCG, MRR
4. **训练技巧**: 梯度裁剪、学习率调度

### 6.2 模型对比

```
模型        特点                优势
────────────────────────────────────────
GRU4Rec     RNN 序列建模        简单高效
SASRec      单向注意力          并行计算
BERT4Rec    双向注意力+Mask     更强表示
```

### 6.3 最佳实践

```
1. 序列长度: 20-100 之间权衡
2. 嵌入维度: 64-256
3. 负采样: 全量负样本 vs 采样
4. 数据增强: 序列裁剪、遮掩
```
