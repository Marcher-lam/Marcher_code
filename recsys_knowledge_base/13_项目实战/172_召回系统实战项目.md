## 1. 项目概述

### 1.1 项目目标

构建一个完整的多路召回系统：
- ItemCF 召回
- 向量召回（双塔模型）
- 热门召回
- 召回结果融合与去重

import pandas as pd
import numpy as np
from typing import List, Dict, Tuple
from collections import defaultdict
from sklearn.model_selection import train_test_split


class RecallDataProcessor:
    """
    召回系统数据处理器
    """

    def __init__(self, data_path: str):
        self.data_path = data_path

        # 映射
        self.user2id = {}
        self.item2id = {}
        self.id2user = {}
        self.id2item = {}

    def load_and_process(self) -> pd.DataFrame:
        """加载并处理数据"""
        print("加载数据...")
        df = pd.read_csv(self.data_path)

        # 基本预处理
        df = df.drop_duplicates(subset=['user_id', 'item_id', 'timestamp'])
        df = df.sort_values(['user_id', 'timestamp'])

        # 过滤低频
        item_counts = df['item_id'].value_counts()
        user_counts = df['user_id'].value_counts()

        df = df[df['item_id'].isin(item_counts[item_counts >= 5].index)]
        df = df[df['user_id'].isin(user_counts[user_counts >= 5].index)]

        # 构建映射
        users = df['user_id'].unique()
        items = df['item_id'].unique()

        self.user2id = {u: i for i, u in enumerate(users)}
        self.item2id = {item: i for i, item in enumerate(items)}
        self.id2user = {i: u for u, i in self.user2id.items()}
        self.id2item = {i: item for item, i in self.item2id.items()}

        df['user_id_enc'] = df['user_id'].map(self.user2id)
        df['item_id_enc'] = df['item_id'].map(self.item2id)

        print(f"用户数: {len(self.user2id)}, 物品数: {len(self.item2id)}")

        return df

    def build_interaction_dict(self, df: pd.DataFrame) -> Tuple[Dict, Dict]:
        """
        构建交互字典

        返回:
            user_items: {user_id: [item_id, ...]}
            item_users: {item_id: [user_id, ...]}
        """
        user_items = defaultdict(list)
        item_users = defaultdict(list)

        for _, row in df.iterrows():
            user_id = row['user_id_enc']
            item_id = row['item_id_enc']

            user_items[user_id].append(item_id)
            item_users[item_id].append(user_id)

        return dict(user_items), dict(item_users)

    def split_by_time(self, df: pd.DataFrame,
                     test_ratio: float = 0.2) -> Tuple[pd.DataFrame, pd.DataFrame]:
        """按时间划分"""
        # 按时间排序后划分
        n_test = int(len(df) * test_ratio)

        train_df = df.iloc[:-n_test]
        test_df = df.iloc[-n_test:]

        return train_df, test_df

    def get_test_users_items(self, test_df: pd.DataFrame) -> Dict[int, List[int]]:
        """获取测试用户的目标物品"""
        test_ground_truth = defaultdict(set)

        for _, row in test_df.iterrows():
            test_ground_truth[row['user_id_enc']].add(row['item_id_enc'])

        return {k: list(v) for k, v in test_ground_truth.items()}

class ItemCFRecaller:
    """
    Item-based Collaborative Filtering 召回
    """

    def __init__(self, top_k: int = 50, n_rec: int = 200):
        self.top_k = top_k
        self.n_rec = n_rec
        self.item_sim_matrix = {}
        self.user_items = {}

    def fit(self, user_items: Dict[int, List[int]]):
        """
        训练：计算物品相似度
        """
        self.user_items = user_items

        # 统计共现
        co_occurrence = defaultdict(lambda: defaultdict(int))
        item_counts = defaultdict(int)

        for user_id, items in user_items.items():
            for i in range(len(items)):
                item_counts[items[i]] += 1
                for j in range(i + 1, len(items)):
                    co_occurrence[items[i]][items[j]] += 1
                    co_occurrence[items[j]][items[i]] += 1

        # 计算相似度
        self.item_sim_matrix = {}

        for item_i in co_occurrence:
            sim_scores = []
            for item_j, co_count in co_occurrence[item_i].items():
                # Jaccard 相似度
                sim = co_count / (item_counts[item_i] + item_counts[item_j] - co_count)
                sim_scores.append((item_j, sim))

            # 保留 top_k
            sim_scores.sort(key=lambda x: x[1], reverse=True)
            self.item_sim_matrix[item_i] = sim_scores[:self.top_k]

        print(f"ItemCF 训练完成，共 {len(self.item_sim_matrix)} 个物品")

    def recall(self, user_id: int, n: int = None) -> List[Tuple[int, float]]:
        """
        召回
        """
        if n is None:
            n = self.n_rec

        if user_id not in self.user_items:
            return []

        user_items = set(self.user_items[user_id])
        scores = defaultdict(float)

        for item_id in user_items:
            if item_id not in self.item_sim_matrix:
                continue

            for sim_item, sim_score in self.item_sim_matrix[item_id]:
                if sim_item not in user_items:
                    scores[sim_item] += sim_score

        # 排序
        sorted_items = sorted(scores.items(), key=lambda x: x[1], reverse=True)

        return sorted_items[:n]


class SwingRecaller:
    """
    Swing 召回

    基于用户共现，减少热门物品影响
    """

    def __init__(self, alpha: float = 0.5, top_k: int = 50, n_rec: int = 200):
        self.alpha = alpha
        self.top_k = top_k
        self.n_rec = n_rec
        self.swing_scores = {}
        self.user_items = {}

    def fit(self, user_items: Dict[int, List[int]]):
        """计算 Swing 分数"""
        self.user_items = user_items

        # 物品-用户倒排
        item_users = defaultdict(set)
        for user_id, items in user_items.items():
            for item_id in items:
                item_users[item_id].add(user_id)

        # 计算物品对的 Swing 分数
        swing_scores = defaultdict(lambda: defaultdict(float))

        items = list(item_users.keys())
        n_items = len(items)

        for i in range(n_items):
            for j in range(i + 1, n_items):
                item_i, item_j = items[i], items[j]

                # 共同用户
                common_users = item_users[item_i] & item_users[item_j]

                if len(common_users) >= 2:
                    # Swing = 1 / (|U| + α)
                    score = 1.0 / (len(common_users) + self.alpha)

                    swing_scores[item_i][item_j] = score
                    swing_scores[item_j][item_i] = score

        # 保留 top_k
        self.swing_scores = {}
        for item_id in swing_scores:
            sorted_swing = sorted(
                swing_scores[item_id].items(),
                key=lambda x: x[1],
                reverse=True
            )[:self.top_k]
            self.swing_scores[item_id] = sorted_swing

        print(f"Swing 训练完成")

    def recall(self, user_id: int, n: int = None) -> List[Tuple[int, float]]:
        """召回"""
        if n is None:
            n = self.n_rec

        if user_id not in self.user_items:
            return []

        user_items = set(self.user_items[user_id])
        scores = defaultdict(float)

        for item_id in user_items:
            if item_id not in self.swing_scores:
                continue

            for sim_item, swing_score in self.swing_scores[item_id]:
                if sim_item not in user_items:
                    scores[sim_item] += swing_score

        sorted_items = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return sorted_items[:n]

### 3.2 向量召回

```python
import torch
import torch.nn as nn


class TwoTowerRecaller:
    """
    双塔模型向量召回
    """

    def __init__(self, n_users: int, n_items: int,
                 embed_dim: int = 64, hidden_dim: int = 128):
        self.n_users = n_users
        self.n_items = n_items
        self.embed_dim = embed_dim

        # 模型
        self.user_tower = nn.Sequential(
            nn.Embedding(n_users, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim)
        )

        self.item_tower = nn.Sequential(
            nn.Embedding(n_items, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim)
        )

        # 索引
        self.item_embeddings = None
        self.index = None

    def fit(self, user_items: Dict[int, List[int]],
            n_epochs: int = 10, batch_size: int = 1024):
        """训练"""
        optimizer = torch.optim.Adam(
            list(self.user_tower.parameters()) + list(self.item_tower.parameters()),
            lr=0.001
        )

        # 生成训练样本
        samples = []
        all_items = list(range(self.n_items))

        for user_id, items in user_items.items():
            pos_items = items
            for pos_item in pos_items:
                samples.append((user_id, pos_item, 1))

            # 负采样
            neg_items = list(set(all_items) - set(pos_items))
            n_neg = min(len(neg_items), len(pos_items) * 4)
            for neg_item in np.random.choice(neg_items, n_neg, replace=False):
                samples.append((user_id, neg_item, 0))

        # 训练
        for epoch in range(n_epochs):
            np.random.shuffle(samples)
            total_loss = 0

            for i in range(0, len(samples), batch_size):
                batch = samples[i:i+batch_size]

                user_ids = torch.tensor([s[0] for s in batch])
                item_ids = torch.tensor([s[1] for s in batch])
                labels = torch.tensor([s[2] for s in batch], dtype=torch.float)

                # 编码
                user_embed = self.user_tower[0](user_ids)
                user_embed = self.user_tower[1:](user_embed)

                item_embed = self.item_tower[0](item_ids)
                item_embed = self.item_tower[1:](item_embed)

                # 归一化
                user_embed = torch.nn.functional.normalize(user_embed, dim=-1)
                item_embed = torch.nn.functional.normalize(item_embed, dim=-1)

                # 预测
                pred = torch.sum(user_embed * item_embed, dim=-1)

                # 损失
                loss = nn.BCEWithLogitsLoss()(pred, labels)

                optimizer.zero_grad()
                loss.backward()
                optimizer.step()

                total_loss += loss.item()

            print(f"Epoch {epoch+1}: Loss = {total_loss / (len(samples) // batch_size):.4f}")

        # 计算物品嵌入
        self._compute_item_embeddings()

    def _compute_item_embeddings(self):
        """计算所有物品嵌入"""
        item_ids = torch.arange(self.n_items)
        with torch.no_grad():
            item_embed = self.item_tower[0](item_ids)
            item_embed = self.item_tower[1:](item_embed)
            item_embed = torch.nn.functional.normalize(item_embed, dim=-1)
            self.item_embeddings = item_embed.numpy()

    def build_index(self):
        """构建 Faiss 索引"""
        try:
            import faiss

            dim = self.embed_dim
            self.index = faiss.IndexFlatIP(dim)
            self.index.add(self.item_embeddings.astype('float32'))

            print("Faiss 索引构建完成")
        except ImportError:
            print("Faiss 未安装，使用暴力搜索")
            self.index = None

    def recall(self, user_id: int, n: int = 200,
              user_items: Dict[int, List[int]] = None) -> List[Tuple[int, float]]:
        """召回"""
        # 获取用户嵌入
        user_tensor = torch.tensor([user_id])
        with torch.no_grad():
            user_embed = self.user_tower[0](user_tensor)
            user_embed = self.user_tower[1:](user_embed)
            user_embed = torch.nn.functional.normalize(user_embed, dim=-1)
            user_embed = user_embed.numpy()

        # 搜索
        if self.index is not None:
            scores, indices = self.index.search(user_embed.astype('float32'), n)
            results = [(int(idx), float(score)) for idx, score in zip(indices[0], scores[0])]
        else:
            # 暴力搜索
            scores = np.dot(self.item_embeddings, user_embed[0])
            top_indices = np.argsort(scores)[::-1][:n]
            results = [(int(idx), float(scores[idx])) for idx in top_indices]

        # 过滤已交互物品
        if user_items and user_id in user_items:
            interacted = set(user_items[user_id])
            results = [(item_id, score) for item_id, score in results
                      if item_id not in interacted]

        return results
```

class HotRecaller:
    """
    热门召回
    """

    def __init__(self, n_rec: int = 100):
        self.n_rec = n_rec
        self.hot_items = []

    def fit(self, user_items: Dict[int, List[int]]):
        """计算热门物品"""
        item_counts = defaultdict(int)

        for items in user_items.values():
            for item_id in items:
                item_counts[item_id] += 1

        sorted_items = sorted(item_counts.items(), key=lambda x: x[1], reverse=True)
        self.hot_items = sorted_items[:self.n_rec * 2]

        print(f"热门召回: {len(self.hot_items)} 个物品")

    def recall(self, n: int = None) -> List[Tuple[int, float]]:
        """召回"""
        if n is None:
            n = self.n_rec

        return self.hot_items[:n]


class CategoryRecaller:
    """
    品类召回

    用户偏好品类下的热门物品
    """

    def __init__(self, n_rec: int = 100, top_categories: int = 3):
        self.n_rec = n_rec
        self.top_categories = top_categories

        self.user_category_pref = {}
        self.category_items = {}

    def fit(self, user_items: Dict[int, List[int]],
           item_categories: Dict[int, int]):
        """训练"""
        # 计算用户品类偏好
        for user_id, items in user_items.items():
            category_counts = defaultdict(int)
            for item_id in items:
                if item_id in item_categories:
                    category_counts[item_categories[item_id]] += 1

            sorted_cats = sorted(category_counts.items(), key=lambda x: x[1], reverse=True)
            self.user_category_pref[user_id] = sorted_cats[:self.top_categories]

        # 计算品类热门物品
        category_item_counts = defaultdict(lambda: defaultdict(int))
        for item_id, category in item_categories.items():
            # 计算物品流行度（需要额外信息）
            category_item_counts[category][item_id] = 1  # 简化

        for category in category_item_counts:
            sorted_items = sorted(
                category_item_counts[category].items(),
                key=lambda x: x[1],
                reverse=True
            )
            self.category_items[category] = sorted_items

        print(f"品类召回训练完成")

    def recall(self, user_id: int, n: int = None) -> List[Tuple[int, float]]:
        """召回"""
        if n is None:
            n = self.n_rec

        if user_id not in self.user_category_pref:
            return []

        results = []
        for category, pref_score in self.user_category_pref[user_id]:
            if category in self.category_items:
                for item_id, pop_score in self.category_items[category]:
                    results.append((item_id, pref_score * pop_score))

        results.sort(key=lambda x: x[1], reverse=True)
        return results[:n]

class MultiChannelRecaller:
    """
    多路召回协调器
    """

    def __init__(self):
        self.recallers = {}
        self.weights = {}

    def register_recaller(self, name: str, recaller, weight: float = 1.0):
        """注册召回器"""
        self.recallers[name] = recaller
        self.weights[name] = weight

    def recall(self, user_id: int, n_per_channel: int = 200,
              final_n: int = 500, user_items: Dict = None) -> List[Tuple[int, float]]:
        """
        多路召回并融合

        参数:
            user_id: 用户ID
            n_per_channel: 每路召回数量
            final_n: 最终返回数量
        """
        all_items = defaultdict(float)
        channel_results = {}

        # 各路召回
        for name, recaller in self.recallers.items():
            if name == 'hot':
                # 热门召回与用户无关
                results = recaller.recall(n_per_channel)
            else:
                results = recaller.recall(user_id, n_per_channel)

            channel_results[name] = results

            # 加权融合
            weight = self.weights[name]
            for item_id, score in results:
                all_items[item_id] += score * weight

        # 排序
        sorted_items = sorted(all_items.items(), key=lambda x: x[1], reverse=True)

        return sorted_items[:final_n], channel_results


class RecallDeduplicator:
    """
    召回去重
    """

    def __init__(self):
        pass

    def deduplicate(self, items: List[Tuple[int, float]],
                   user_history: List[int] = None,
                   max_per_category: int = None,
                   item_categories: Dict[int, int] = None) -> List[Tuple[int, float]]:
        """
        去重

        参数:
            items: 召回物品列表
            user_history: 用户历史（排除已交互）
            max_per_category: 每类最大数量
            item_categories: 物品类目映射
        """
        result = []
        seen = set()
        category_counts = defaultdict(int)

        if user_history:
            seen = set(user_history)

        for item_id, score in items:
            # 去重
            if item_id in seen:
                continue

            # 类目限制
            if max_per_category and item_categories:
                category = item_categories.get(item_id)
                if category and category_counts[category] >= max_per_category:
                    continue
                category_counts[category] += 1

            result.append((item_id, score))
            seen.add(item_id)

        return result

## 5. 召回评估

### 5.1 评估指标

```python
class RecallEvaluator:
    """
    召回评估器
    """

    def __init__(self):
        pass

    @staticmethod
    def recall_at_k(recalled: List[int], ground_truth: List[int], k: int) -> float:
        """
        Recall@K

        召回的物品中有多少在真实集合中
        """
        if not ground_truth:
            return 0

        recalled_k = set(recalled[:k])
        truth_set = set(ground_truth)

        return len(recalled_k & truth_set) / len(truth_set)

    @staticmethod
    def hit_rate_at_k(recalled: List[int], ground_truth: List[int], k: int) -> float:
        """
        Hit Rate@K

        是否至少召回了一个正确物品
        """
        recalled_k = set(recalled[:k])
        truth_set = set(ground_truth)

        return 1.0 if recalled_k & truth_set else 0.0

    @staticmethod
    def precision_at_k(recalled: List[int], ground_truth: List[int], k: int) -> float:
        """Precision@K"""
        if k == 0:
            return 0

        recalled_k = set(recalled[:k])
        truth_set = set(ground_truth)

        return len(recalled_k & truth_set) / k

    @staticmethod
    def coverage(all_recalled: List[List[int]], total_items: int) -> float:
        """
        覆盖率

        所有召回结果覆盖了多少物品
        """
        unique_items = set()
        for items in all_recalled:
            unique_items.update(items)

        return len(unique_items) / total_items

    @staticmethod
    def novelty(recalled: List[int], item_popularity: Dict[int, float]) -> float:
        """
        新颖性

        召回物品的平均流行度的倒数
        """
        if not recalled:
            return 0

        novelty_scores = []
        for item_id in recalled:
            pop = item_popularity.get(item_id, 0.5)
            novelty_scores.append(-np.log(pop + 1e-10))

        return np.mean(novelty_scores)


class RecallSystemEvaluator:
    """
    召回系统评估
    """

    def __init__(self, recaller, ground_truth: Dict[int, List[int]],
                 total_items: int, item_popularity: Dict[int, float]):
        self.recaller = recaller
        self.ground_truth = ground_truth
        self.total_items = total_items
        self.item_popularity = item_popularity

    def evaluate(self, k_list: List[int] = [50, 100, 200, 500]) -> Dict:
        """评估"""
        all_recalled = []
        metrics = {f'recall@{k}': [] for k in k_list}
        metrics.update({f'hit_rate@{k}': [] for k in k_list})
        metrics.update({f'precision@{k}': [] for k in k_list})

        for user_id, truth in self.ground_truth.items():
            recalled, _ = self.recaller.recall(user_id, final_n=max(k_list))
            recalled_items = [item_id for item_id, _ in recalled]
            all_recalled.append(recalled_items)

            for k in k_list:
                metrics[f'recall@{k}'].append(
                    RecallEvaluator.recall_at_k(recalled_items, truth, k)
                )
                metrics[f'hit_rate@{k}'].append(
                    RecallEvaluator.hit_rate_at_k(recalled_items, truth, k)
                )
                metrics[f'precision@{k}'].append(
                    RecallEvaluator.precision_at_k(recalled_items, truth, k)
                )

        # 平均
        result = {k: np.mean(v) for k, v in metrics.items()}

        # 覆盖率
        result['coverage'] = RecallEvaluator.coverage(all_recalled, self.total_items)

        # 新颖性
        all_items = []
        for items in all_recalled:
            all_items.extend(items[:100])
        result['novelty'] = RecallEvaluator.novelty(all_items, self.item_popularity)

        return result
```

### 6.1 运行脚本

def run_recall_pipeline(data_path: str):
    """运行完整召回流程"""
    # 1. 数据处理
    processor = RecallDataProcessor(data_path)
    df = processor.load_and_process()
    train_df, test_df = processor.split_by_time(df)

    user_items, item_users = processor.build_interaction_dict(train_df)
    test_ground_truth = processor.get_test_users_items(test_df)

    # 2. 训练各路召回
    # ItemCF
    itemcf = ItemCFRecaller(top_k=50, n_rec=200)
    itemcf.fit(user_items)

    # Swing
    swing = SwingRecaller(alpha=0.5, top_k=50, n_rec=200)
    swing.fit(user_items)

    # 向量召回
    vector_recaller = TwoTowerRecaller(
        n_users=len(processor.user2id),
        n_items=len(processor.item2id),
        embed_dim=64
    )
    vector_recaller.fit(user_items, n_epochs=5)
    vector_recaller.build_index()

    # 热门召回
    hot = HotRecaller(n_rec=100)
    hot.fit(user_items)

    # 3. 构建多路召回
    multi_recaller = MultiChannelRecaller()
    multi_recaller.register_recaller('itemcf', itemcf, weight=1.0)
    multi_recaller.register_recaller('swing', swing, weight=0.8)
    multi_recaller.register_recaller('vector', vector_recaller, weight=1.2)
    multi_recaller.register_recaller('hot', hot, weight=0.3)

    # 4. 评估
    item_popularity = defaultdict(int)
    for items in user_items.values():
        for item_id in items:
            item_popularity[item_id] += 1
    max_pop = max(item_popularity.values())
    item_popularity = {k: v/max_pop for k, v in item_popularity.items()}

    evaluator = RecallSystemEvaluator(
        multi_recaller, test_ground_truth,
        len(processor.item2id), item_popularity
    )

    metrics = evaluator.evaluate()
    print("\n召回评估结果:")
    for k, v in metrics.items():
        print(f"  {k}: {v:.4f}")

    return multi_recaller, metrics

## 7. 学习总结

### 7.1 核心要点

1. **多路召回**: 不同召回器互补
2. **加权融合**: 根据效果调整权重
3. **去重截断**: 控制召回数量
4. **评估优化**: 关注召回率和覆盖率

### 7.2 召回器对比

```
召回器       优势              劣势
──────────────────────────────────────
ItemCF       可解释性强        稀疏性
Swing        减少热门偏差      计算量大
向量召回     泛化能力强        冷启动
热门召回     简单稳定          缺乏个性化
```

### 7.3 最佳实践

```
1. 召回数量: 500-2000
2. 各路独立: 避免重复计算
3. 定期更新: 模型和索引
4. 监控指标: 召回率、覆盖率
```
