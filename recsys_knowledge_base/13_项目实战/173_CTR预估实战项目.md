```python
@dataclass
class DataConfig:
    """数据配置"""
    train_path: str = "data/train.csv"
    valid_path: str = "data/valid.csv"
    test_path: str = "data/test.csv"
    sparse_features: List[str] = None
    dense_features: List[str] = None
    target: str = "click"
    batch_size: int = 4096
    max_seq_len: int = 50


class CTRDataProcessor:
    """
    CTR 数据处理器
    """

    def __init__(self, config: DataConfig):
        self.config = config
        self.feature_stats = {}

        # 特征编码器
        self.label_encoders = {}
        self.minmax_scalers = {}

    def load_data(self, path: str) -> pd.DataFrame:
        """加载数据"""
        print(f"Loading data from {path}")

        # 支持多种格式
        if path.endswith('.csv'):
            df = pd.read_csv(path)
        elif path.endswith('.parquet'):
            df = pd.read_parquet(path)
        else:
            raise ValueError(f"Unsupported format: {path}")

        print(f"Loaded {len(df)} samples, {len(df.columns)} features")
        return df

    def analyze_features(self, df: pd.DataFrame):
        """分析特征"""
        # 自动识别稀疏和稠密特征
        if self.config.sparse_features is None:
            self.config.sparse_features = df.select_dtypes(
                include=['object', 'category']
            ).columns.tolist()

        if self.config.dense_features is None:
            self.config.dense_features = df.select_dtypes(
                include=[np.number]
            ).columns.tolist()
            # 移除目标
            if self.config.target in self.config.dense_features:
                self.config.dense_features.remove(self.config.target)

        print(f"Sparse features: {len(self.config.sparse_features)}")
        print(f"Dense features: {len(self.config.dense_features)}")

        # 统计特征基数
        for col in self.config.sparse_features:
            self.feature_stats[col] = {
                'nunique': df[col].nunique(),
                'top_values': df[col].value_counts().head(10).to_dict()
            }

    def preprocess(self, df: pd.DataFrame, fit: bool = False) -> pd.DataFrame:
        """预处理"""
        df = df.copy()

        # 1. 处理稀疏特征
        for col in self.config.sparse_features:
            if col not in df.columns:
                continue

            # 填充缺失值
            df[col] = df[col].fillna('missing')

            if fit:
                from sklearn.preprocessing import LabelEncoder
                le = LabelEncoder()
                df[col] = le.fit_transform(df[col].astype(str))
                self.label_encoders[col] = le
            else:
                le = self.label_encoders[col]
                # 处理未见过的类别
                df[col] = df[col].astype(str).apply(
                    lambda x: le.transform([x])[0] if x in le.classes_ else 0
                )

        # 2. 处理稠密特征
        for col in self.config.dense_features:
            if col not in df.columns:
                continue

            # 填充缺失值
            df[col] = df[col].fillna(df[col].median())

            if fit:
                from sklearn.preprocessing import MinMaxScaler
                scaler = MinMaxScaler()
                df[col] = scaler.fit_transform(df[[col]])
                self.minmax_scalers[col] = scaler
            else:
                scaler = self.minmax_scalers[col]
                df[col] = scaler.transform(df[[col]])

        return df

    def create_feature_columns(self) -> Dict[str, int]:
        """创建特征列配置"""
        feature_columns = {}

        for col in self.config.sparse_features:
            if col in self.feature_stats:
                feature_columns[col] = self.feature_stats[col]['nunique'] + 1

        return feature_columns

    def get_dataloader(self, df: pd.DataFrame,
                       shuffle: bool = True) -> torch.utils.data.DataLoader:
        """创建数据加载器"""
        from torch.utils.data import Dataset, DataLoader

        class CTRDataset(Dataset):
            def __init__(self, df, sparse_features, dense_features, target):
                self.sparse_data = df[sparse_features].values.astype(np.int64)
                self.dense_data = df[dense_features].values.astype(np.float32)
                self.labels = df[target].values.astype(np.float32)

            def __len__(self):
                return len(self.labels)

            def __getitem__(self, idx):
                return {
                    'sparse_features': torch.tensor(self.sparse_data[idx]),
                    'dense_features': torch.tensor(self.dense_data[idx]),
                    'label': torch.tensor(self.labels[idx])
                }

        dataset = CTRDataset(
            df,
            self.config.sparse_features,
            self.config.dense_features,
            self.config.target
        )

        return DataLoader(
            dataset,
            batch_size=self.config.batch_size,
            shuffle=shuffle,
            num_workers=4,
            pin_memory=True
        )
```

## 3. 模型实现

### 3.1 DeepFM 模型

```python
class DeepFM(nn.Module):
    """
    DeepFM: 结合 FM 和 DNN

    FM 部分学习二阶交叉
    Deep 部分学习高阶特征
    """

    def __init__(self,
                 sparse_feature_info: Dict[str, int],
                 dense_feature_dim: int,
                 embed_dim: int = 16,
                 hidden_units: List[int] = [256, 128, 64],
                 dropout: float = 0.2):
        super().__init__()

        self.sparse_feature_info = sparse_feature_info
        self.dense_feature_dim = dense_feature_dim
        self.embed_dim = embed_dim

        # Sparse 特征嵌入
        self.sparse_embeddings = nn.ModuleDict({
            name: nn.Embedding(nunique, embed_dim)
            for name, nunique in sparse_feature_info.items()
        })

        # 初始化
        for embed in self.sparse_embeddings.values():
            nn.init.xavier_uniform_(embed.weight)

        # FM 一阶部分
        self.fm_first_order_sparse = nn.ModuleDict({
            name: nn.Embedding(nunique, 1)
            for name, nunique in sparse_feature_info.items()
        })

        self.fm_first_order_dense = nn.Linear(dense_feature_dim, 1)

        # Deep 部分
        n_sparse_features = len(sparse_feature_info)
        deep_input_dim = n_sparse_features * embed_dim + dense_feature_dim

        deep_layers = []
        input_dim = deep_input_dim
        for hidden_dim in hidden_units:
            deep_layers.extend([
                nn.Linear(input_dim, hidden_dim),
                nn.BatchNorm1d(hidden_dim),
                nn.ReLU(),
                nn.Dropout(dropout)
            ])
            input_dim = hidden_dim

        self.deep = nn.Sequential(*deep_layers)

        # 输出层
        self.output = nn.Linear(hidden_units[-1] + 1, 1)

    def forward(self, sparse_features: Dict[str, torch.Tensor],
                dense_features: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        参数:
            sparse_features: Dict[str, (batch,)] 各稀疏特征
            dense_features: (batch, dense_dim) 稠密特征
        """
        batch_size = dense_features.size(0)

        # 1. FM 一阶部分
        fm_first_sparse = sum(
            embed(sparse_features[name]).squeeze(-1)
            for name, embed in self.fm_first_order_sparse.items()
        )
        fm_first_dense = self.fm_first_order_dense(dense_features)
        fm_first = fm_first_sparse + fm_first_dense

        # 2. FM 二阶部分
        sparse_embeds = [
            embed(sparse_features[name])
            for name, embed in self.sparse_embeddings.items()
        ]
        sparse_embeds = torch.stack(sparse_embeds, dim=1)  # (batch, n_sparse, embed_dim)

        # sum of squares
        sum_of_squares = torch.sum(sparse_embeds, dim=1) ** 2
        # square of sum
        square_of_sum = torch.sum(sparse_embeds ** 2, dim=1)
        # 二阶交叉
        fm_second = 0.5 * torch.sum(sum_of_squares - square_of_sum, dim=1, keepdim=True)

        # 3. Deep 部分
        sparse_embeds_flat = sparse_embeds.view(batch_size, -1)
        deep_input = torch.cat([sparse_embeds_flat, dense_features], dim=1)
        deep_output = self.deep(deep_input)

        # 4. 合并
        combined = torch.cat([fm_second, deep_output], dim=1)
        output = self.output(combined)

        return output.squeeze(-1)

    def predict_proba(self, sparse_features: Dict[str, torch.Tensor],
                      dense_features: torch.Tensor) -> torch.Tensor:
        """预测概率"""
        logits = self.forward(sparse_features, dense_features)
        return torch.sigmoid(logits)
```

### 3.2 模型包装器

```python
class CTRModel:
    """
    CTR 模型包装器
    """

    def __init__(self, model: nn.Module, device: str = 'cuda'):
        self.model = model.to(device)
        self.device = device

    def train_epoch(self, dataloader, optimizer, criterion) -> float:
        """训练一个 epoch"""
        self.model.train()
        total_loss = 0

        for batch in dataloader:
            # 准备数据
            sparse_features = {
                name: batch['sparse_features'][:, i].to(self.device)
                for i, name in enumerate(self.model.sparse_feature_info.keys())
            }
            dense_features = batch['dense_features'].to(self.device)
            labels = batch['label'].to(self.device)

            # 前向传播
            optimizer.zero_grad()
            logits = self.model(sparse_features, dense_features)
            loss = criterion(logits, labels)

            # 反向传播
            loss.backward()
            nn.utils.clip_grad_norm_(self.model.parameters(), 5.0)
            optimizer.step()

            total_loss += loss.item()

        return total_loss / len(dataloader)

    def evaluate(self, dataloader) -> Dict[str, float]:
        """评估"""
        from sklearn.metrics import roc_auc_score, log_loss

        self.model.eval()
        all_preds = []
        all_labels = []

        with torch.no_grad():
            for batch in dataloader:
                sparse_features = {
                    name: batch['sparse_features'][:, i].to(self.device)
                    for i, name in enumerate(self.model.sparse_feature_info.keys())
                }
                dense_features = batch['dense_features'].to(self.device)
                labels = batch['label']

                preds = self.model.predict_proba(sparse_features, dense_features)

                all_preds.extend(preds.cpu().numpy())
                all_labels.extend(labels.numpy())

        all_preds = np.array(all_preds)
        all_labels = np.array(all_labels)

        return {
            'auc': roc_auc_score(all_labels, all_preds),
            'logloss': log_loss(all_labels, all_preds),
            'mean_pred': all_preds.mean(),
            'std_pred': all_preds.std()
        }

    def predict_batch(self, sparse_features: Dict[str, np.ndarray],
                      dense_features: np.ndarray) -> np.ndarray:
        """批量预测"""
        self.model.eval()

        with torch.no_grad():
            sparse_tensors = {
                name: torch.tensor(data, dtype=torch.long, device=self.device)
                for name, data in sparse_features.items()
            }
            dense_tensor = torch.tensor(dense_features, dtype=torch.float32, device=self.device)

            probs = self.model.predict_proba(sparse_tensors, dense_tensor)

        return probs.cpu().numpy()
```

## 4. 训练流程

### 4.1 训练器

```python
class CTRTrainer:
    """
    CTR 训练器
    """

    def __init__(self, model: CTRModel, config: Dict):
        self.model = model
        self.config = config

        # 优化器
        self.optimizer = torch.optim.Adam(
            model.model.parameters(),
            lr=config.get('lr', 0.001),
            weight_decay=config.get('weight_decay', 1e-5)
        )

        # 学习率调度器
        self.scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
            self.optimizer,
            mode='max',
            factor=0.5,
            patience=2,
            verbose=True
        )

        # 损失函数
        self.criterion = nn.BCEWithLogitsLoss()

        # 早停
        self.best_auc = 0
        self.patience_counter = 0
        self.patience = config.get('patience', 5)

        # 日志
        self.history = {
            'train_loss': [],
            'valid_auc': [],
            'valid_logloss': []
        }

    def train(self, train_loader, valid_loader, n_epochs: int):
        """完整训练"""
        print("Starting training...")

        for epoch in range(n_epochs):
            # 训练
            train_loss = self.model.train_epoch(
                train_loader, self.optimizer, self.criterion
            )
            self.history['train_loss'].append(train_loss)

            # 验证
            valid_metrics = self.model.evaluate(valid_loader)
            self.history['valid_auc'].append(valid_metrics['auc'])
            self.history['valid_logloss'].append(valid_metrics['logloss'])

            # 日志
            print(f"\nEpoch {epoch + 1}/{n_epochs}")
            print(f"  Train Loss: {train_loss:.4f}")
            print(f"  Valid AUC: {valid_metrics['auc']:.4f}")
            print(f"  Valid LogLoss: {valid_metrics['logloss']:.4f}")

            # 学习率调整
            self.scheduler.step(valid_metrics['auc'])

            # 早停检查
            if valid_metrics['auc'] > self.best_auc:
                self.best_auc = valid_metrics['auc']
                self.patience_counter = 0

                # 保存最佳模型
                self._save_model('best_model.pt')
                print(f"  New best model saved! AUC: {self.best_auc:.4f}")
            else:
                self.patience_counter += 1
                if self.patience_counter >= self.patience:
                    print(f"\nEarly stopping after {epoch + 1} epochs")
                    break

        print(f"\nTraining completed. Best AUC: {self.best_auc:.4f}")

    def _save_model(self, path: str):
        """保存模型"""
        torch.save({
            'model_state_dict': self.model.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'best_auc': self.best_auc,
            'config': self.config
        }, path)

    def load_model(self, path: str):
        """加载模型"""
        checkpoint = torch.load(path)
        self.model.model.load_state_dict(checkpoint['model_state_dict'])
        self.optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
        self.best_auc = checkpoint['best_auc']
```

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import uvicorn
import time


class PredictRequest(BaseModel):
    """预测请求"""
    user_id: int
    item_ids: List[int]
    features: Dict[str, List]


class PredictResponse(BaseModel):
    """预测响应"""
    predictions: List[float]
    latency_ms: float


class CTRInferenceService:
    """
    CTR 推理服务
    """

    def __init__(self, model_path: str, device: str = 'cuda'):
        # 加载模型
        checkpoint = torch.load(model_path)
        self.config = checkpoint['config']

        # 创建模型
        self.model = self._create_model()
        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.model.eval()
        self.model.to(device)

        self.device = device

        # 特征处理器
        self.processor = self._load_processor()

    def _create_model(self) -> nn.Module:
        """创建模型"""
        # 根据配置创建模型
        return DeepFM(**self.config['model_config'])

    def predict(self, request: PredictRequest) -> PredictResponse:
        """预测"""
        start_time = time.time()

        # 处理特征
        sparse_features, dense_features = self._process_features(request)

        # 推理
        with torch.no_grad():
            probs = self.model.predict_proba(sparse_features, dense_features)

        latency = (time.time() - start_time) * 1000

        return PredictResponse(
            predictions=probs.tolist(),
            latency_ms=latency
        )

    def _process_features(self, request: PredictRequest) -> Tuple[Dict, np.ndarray]:
        """处理特征"""
        sparse_features = {}
        dense_features = []

        # 处理稀疏特征
        for name, values in request.features.items():
            if name in self.processor.label_encoders:
                encoded = [self.processor.label_encoders[name].transform([v])[0]
                          for v in values]
                sparse_features[name] = np.array(encoded)
            else:
                dense_features.append(values)

        dense_features = np.column_stack(dense_features) if dense_features else np.zeros((len(request.item_ids), 0))

        return sparse_features, dense_features


def create_app(model_path: str) -> FastAPI:
    """创建 FastAPI 应用"""
    app = FastAPI(title="CTR Prediction Service")

    # 初始化服务
    service = CTRInferenceService(model_path)

    @app.post("/predict", response_model=PredictResponse)
    def predict(request: PredictRequest):
        return service.predict(request)

    @app.get("/health")
    def health():
        return {"status": "healthy"}

    return app


if __name__ == "__main__":
    app = create_app("best_model.pt")
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
