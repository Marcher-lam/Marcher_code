## 1. 概率基础

### 1.1 基本概念

```python
import numpy as np
from typing import List, Dict, Tuple
from collections import Counter
import matplotlib.pyplot as plt


class ProbabilityBasics:
    """
    概率论基础概念
    """

    @staticmethod
    def probability(event_count: int, total_count: int) -> float:
        """
        古典概型概率

        P(A) = 事件A发生次数 / 总可能次数
        """
        if total_count == 0:
            return 0
        return event_count / total_count

    @staticmethod
    def conditional_probability(joint_count: int,
                                condition_count: int) -> float:
        """
        条件概率

        P(A|B) = P(A∩B) / P(B)
        """
        if condition_count == 0:
            return 0
        return joint_count / condition_count

    @staticmethod
    def bayes_theorem(p_b_given_a: float,
                      p_a: float,
                      p_b: float) -> float:
        """
        贝叶斯定理

        P(A|B) = P(B|A) * P(A) / P(B)
        """
        if p_b == 0:
            return 0
        return p_b_given_a * p_a / p_b


class BayesClassifier:
    """
    朴素贝叶斯分类器示例
    """

    def __init__(self):
        self.class_priors = {}  # P(Class)
        self.feature_probs = {}  # P(Feature|Class)

    def fit(self, X: List[List], y: List):
        """训练"""
        n_samples = len(y)
        classes = set(y)

        # 计算类先验概率
        class_counts = Counter(y)
        for cls, count in class_counts.items():
            self.class_priors[cls] = count / n_samples

        # 计算特征条件概率
        n_features = len(X[0])
        for cls in classes:
            cls_samples = [X[i] for i in range(n_samples) if y[i] == cls]
            self.feature_probs[cls] = []

            for f_idx in range(n_features):
                feature_values = [sample[f_idx] for sample in cls_samples]
                value_counts = Counter(feature_values)
                total = len(feature_values)

                # 拉普拉斯平滑
                probs = {v: (c + 1) / (total + len(value_counts))
                        for v, c in value_counts.items()}
                self.feature_probs[cls].append(probs)

    def predict(self, x: List) -> str:
        """预测"""
        best_class = None
        best_prob = -1

        for cls in self.class_priors:
            prob = self.class_priors[cls]

            for f_idx, f_val in enumerate(x):
                if f_val in self.feature_probs[cls][f_idx]:
                    prob *= self.feature_probs[cls][f_idx][f_val]
                else:
                    # 未知值，使用平滑概率
                    prob *= 1 / (len(self.feature_probs[cls][f_idx]) + 1)

            if prob > best_prob:
                best_prob = prob
                best_class = cls

        return best_class
```

### 2.1 离散分布

```python
class DiscreteDistributions:
    """
    离散概率分布
    """

    @staticmethod
    def bernoulli(p: float, n: int = 10000) -> np.ndarray:
        """
        伯努利分布

        P(X=1) = p
        P(X=0) = 1-p
        """
        return np.random.binomial(1, p, n)

    @staticmethod
    def binomial(n: int, p: float, size: int = 10000) -> np.ndarray:
        """
        二项分布

        P(X=k) = C(n,k) * p^k * (1-p)^(n-k)

        n次试验中成功k次的概率
        """
        return np.random.binomial(n, p, size)

    @staticmethod
    def poisson(lam: float, size: int = 10000) -> np.ndarray:
        """
        泊松分布

        P(X=k) = (λ^k * e^(-λ)) / k!

        单位时间内事件发生次数
        """
        return np.random.poisson(lam, size)

    @staticmethod
    def geometric(p: float, size: int = 10000) -> np.ndarray:
        """
        几何分布

        P(X=k) = (1-p)^(k-1) * p

        首次成功所需的试验次数
        """
        return np.random.geometric(p, size)


class DistributionStats:
    """
    分布统计量
    """

    @staticmethod
    def compute_stats(samples: np.ndarray) -> Dict:
        """计算统计量"""
        return {
            'mean': np.mean(samples),
            'variance': np.var(samples),
            'std': np.std(samples),
            'median': np.median(samples),
            'skewness': (np.mean((samples - np.mean(samples))**3) /
                        np.std(samples)**3),
            'kurtosis': (np.mean((samples - np.mean(samples))**4) /
                        np.std(samples)**4 - 3)
        }
```

### 2.2 连续分布

```python
class ContinuousDistributions:
    """
    连续概率分布
    """

    @staticmethod
    def uniform(low: float, high: float, size: int = 10000) -> np.ndarray:
        """
        均匀分布

        f(x) = 1/(b-a), a ≤ x ≤ b
        """
        return np.random.uniform(low, high, size)

    @staticmethod
    def normal(mu: float, sigma: float, size: int = 10000) -> np.ndarray:
        """
        正态分布（高斯分布）

        f(x) = (1/√(2πσ²)) * e^(-(x-μ)²/(2σ²))
        """
        return np.random.normal(mu, sigma, size)

    @staticmethod
    def exponential(scale: float, size: int = 10000) -> np.ndarray:
        """
        指数分布

        f(x) = λ * e^(-λx), x ≥ 0

        常用于建模用户行为间隔时间
        """
        return np.random.exponential(scale, size)

    @staticmethod
    def beta(a: float, b: float, size: int = 10000) -> np.ndarray:
        """
        Beta 分布

        f(x) = x^(a-1) * (1-x)^(b-1) / B(a,b)

        常用于建模概率分布（如点击率）
        """
        return np.random.beta(a, b, size)

    @staticmethod
    def lognormal(mean: float, sigma: float, size: int = 10000) -> np.ndarray:
        """
        对数正态分布

        常用于建模用户活跃度、物品流行度
        """
        return np.random.lognormal(mean, sigma, size)


# 推荐系统中的分布应用示例
class RecSysDistributionExample:
    """
    推荐系统中概率分布的应用
    """

    def __init__(self):
        pass

    def model_ctr_distribution(self, observed_clicks: int,
                               observed_impressions: int,
                               prior_alpha: float = 1.0,
                               prior_beta: float = 1.0) -> Dict:
        """
        使用 Beta 分布建模 CTR

        先验: Beta(α, β)
        观测: clicks 次成功, impressions-clicks 次失败
        后验: Beta(α + clicks, β + impressions - clicks)
        """
        posterior_alpha = prior_alpha + observed_clicks
        posterior_beta = prior_beta + observed_impressions - observed_clicks

        # 后验分布采样
        samples = np.random.beta(posterior_alpha, posterior_beta, 10000)

        return {
            'posterior_alpha': posterior_alpha,
            'posterior_beta': posterior_beta,
            'mean': posterior_alpha / (posterior_alpha + posterior_beta),
            'variance': (posterior_alpha * posterior_beta) /
                       ((posterior_alpha + posterior_beta)**2 *
                        (posterior_alpha + posterior_beta + 1)),
            'samples': samples
        }

    def model_session_length(self) -> Dict:
        """
        使用指数分布建模用户会话长度
        """
        # 假设平均会话时长为 10 分钟
        mean_session = 10
        samples = np.random.exponential(mean_session, 10000)

        return {
            'mean': np.mean(samples),
            'median': np.median(samples),
            'percentile_90': np.percentile(samples, 90)
        }

    def model_item_popularity(self, n_items: int = 1000) -> np.ndarray:
        """
        使用对数正态分布建模物品流行度

        符合长尾分布特征
        """
        # 参数设置使分布呈现长尾特征
        popularity = np.random.lognormal(mean=0, sigma=1.5, size=n_items)
        return popularity / popularity.sum()  # 归一化为概率
```

### 3.1 参数估计

```python
class ParameterEstimation:
    """
    参数估计方法
    """

    @staticmethod
    def mle_normal(samples: np.ndarray) -> Tuple[float, float]:
        """
        正态分布的最大似然估计

        μ_hat = (1/n) * Σ x_i
        σ²_hat = (1/n) * Σ (x_i - μ_hat)²
        """
        mu = np.mean(samples)
        sigma_sq = np.mean((samples - mu)**2)
        return mu, np.sqrt(sigma_sq)

    @staticmethod
    def mle_bernoulli(samples: np.ndarray) -> float:
        """
        伯努利分布的 MLE

        p_hat = 成功次数 / 总次数
        """
        return np.mean(samples)

    @staticmethod
    def mle_exponential(samples: np.ndarray) -> float:
        """
        指数分布的 MLE

        λ_hat = 1 / mean(samples)
        """
        return 1 / np.mean(samples)


class BayesianEstimation:
    """
    贝叶斯参数估计
    """

    def __init__(self, prior_alpha: float = 1.0, prior_beta: float = 1.0):
        """
        初始化 Beta 先验（用于估计概率）
        """
        self.alpha = prior_alpha
        self.beta = prior_beta

    def update(self, successes: int, failures: int):
        """
        贝叶斯更新

        后验 = 先验 × 似然
        Beta(α', β') = Beta(α, β) × Bernoulli
        """
        self.alpha += successes
        self.beta += failures

    def get_posterior_mean(self) -> float:
        """后验均值（点估计）"""
        return self.alpha / (self.alpha + self.beta)

    def get_credible_interval(self, confidence: float = 0.95) -> Tuple[float, float]:
        """可信区间"""
        from scipy import stats
        lower = (1 - confidence) / 2
        upper = 1 - lower
        return (stats.beta.ppf(lower, self.alpha, self.beta),
                stats.beta.ppf(upper, self.alpha, self.beta))
```

### 3.2 假设检验

```python
class HypothesisTesting:
    """
    假设检验
    """

    @staticmethod
    def t_test(sample1: np.ndarray, sample2: np.ndarray) -> Dict:
        """
        t 检验（比较两组均值）

        用于 A/B 测试中比较两组用户的点击率
        """
        from scipy import stats
        t_stat, p_value = stats.ttest_ind(sample1, sample2)

        return {
            't_statistic': t_stat,
            'p_value': p_value,
            'significant_0.05': p_value < 0.05
        }

    @staticmethod
    def chi_square_test(observed: np.ndarray, expected: np.ndarray) -> Dict:
        """
        卡方检验（检验分布是否一致）

        用于检验推荐结果分布是否符合预期
        """
        from scipy import stats
        chi2, p_value = stats.chisquare(observed, expected)

        return {
            'chi2_statistic': chi2,
            'p_value': p_value,
            'significant_0.05': p_value < 0.05
        }

    @staticmethod
    def proportion_test(successes1: int, n1: int,
                        successes2: int, n2: int) -> Dict:
        """
        比例检验（A/B 测试常用）

        检验两组比例是否有显著差异
        """
        from scipy import stats

        p1 = successes1 / n1
        p2 = successes2 / n2
        p_pooled = (successes1 + successes2) / (n1 + n2)

        # Z 统计量
        se = np.sqrt(p_pooled * (1 - p_pooled) * (1/n1 + 1/n2))
        z_stat = (p1 - p2) / se

        # 双侧 p 值
        p_value = 2 * (1 - stats.norm.cdf(abs(z_stat)))

        return {
            'proportion1': p1,
            'proportion2': p2,
            'z_statistic': z_stat,
            'p_value': p_value,
            'significant_0.05': p_value < 0.05
        }


class ABTestingFramework:
    """
    A/B 测试框架
    """

    def __init__(self, alpha: float = 0.05, power: float = 0.8):
        self.alpha = alpha
        self.power = power

    def sample_size_for_proportion(self, p1: float, p2: float) -> int:
        """
        计算比例检验所需样本量

        参数:
            p1: 对照组比例
            p2: 实验组比例（预期提升后）
        """
        from scipy import stats

        z_alpha = stats.norm.ppf(1 - self.alpha/2)
        z_beta = stats.norm.ppf(self.power)

        p_avg = (p1 + p2) / 2

        n = ((z_alpha * np.sqrt(2 * p_avg * (1 - p_avg)) +
              z_beta * np.sqrt(p1 * (1 - p1) + p2 * (1 - p2)))**2 /
             (p2 - p1)**2)

        return int(np.ceil(n))

    def minimum_detectable_effect(self, n: int, p1: float) -> float:
        """
        给定样本量，计算最小可检测效应
        """
        from scipy import stats

        z_alpha = stats.norm.ppf(1 - self.alpha/2)
        z_beta = stats.norm.ppf(self.power)

        # 简化计算
        mde = (z_alpha + z_beta) * np.sqrt(2 * p1 * (1 - p1) / n)

        return mde
```

class CorrelationAnalysis:
    """
    相关性分析
    """

    @staticmethod
    def pearson_correlation(x: np.ndarray, y: np.ndarray) -> float:
        """
        Pearson 相关系数

        r = Σ((x_i - x̄)(y_i - ȳ)) / √(Σ(x_i - x̄)² × Σ(y_i - ȳ)²)

        度量线性相关性
        """
        return np.corrcoef(x, y)[0, 1]

    @staticmethod
    def spearman_correlation(x: np.ndarray, y: np.ndarray) -> float:
        """
        Spearman 秩相关系数

        基于秩次的相关性，对异常值更鲁棒
        """
        from scipy import stats
        return stats.spearmanr(x, y)[0]

    @staticmethod
    def kendall_correlation(x: np.ndarray, y: np.ndarray) -> float:
        """
        Kendall 秩相关系数

        适用于排序评估
        """
        from scipy import stats
        return stats.kendalltau(x, y)[0]


class CovarianceMatrix:
    """
    协方差矩阵
    """

    @staticmethod
    def compute(data: np.ndarray) -> np.ndarray:
        """
        计算协方差矩阵

        Cov(X, Y) = E[(X - μ_X)(Y - μ_Y)]
        """
        return np.cov(data, rowvar=False)

    @staticmethod
    def correlation_matrix(data: np.ndarray) -> np.ndarray:
        """
        计算相关系数矩阵
        """
        return np.corrcoef(data, rowvar=False)
```

class UserBehaviorStats:
    """
    用户行为统计分析
    """

    def __init__(self, user_behaviors: List[Dict]):
        """
        参数:
            user_behaviors: 用户行为列表
        """
        self.behaviors = user_behaviors

    def compute_session_stats(self) -> Dict:
        """会话统计"""
        session_lengths = []

        for behavior in self.behaviors:
            if 'session_length' in behavior:
                session_lengths.append(behavior['session_length'])

        if not session_lengths:
            return {}

        return {
            'mean': np.mean(session_lengths),
            'median': np.median(session_lengths),
            'std': np.std(session_lengths),
            'percentiles': {
                '25': np.percentile(session_lengths, 25),
                '50': np.percentile(session_lengths, 50),
                '75': np.percentile(session_lengths, 75),
                '90': np.percentile(session_lengths, 90)
            }
        }

    def compute_inter_arrival_time(self) -> Dict:
        """
        计算用户行为到达间隔时间

        常用于建模用户活跃度
        """
        timestamps = sorted([b['timestamp'] for b in self.behaviors
                           if 'timestamp' in b])

        if len(timestamps) < 2:
            return {}

        intervals = [timestamps[i+1] - timestamps[i]
                    for i in range(len(timestamps)-1)]

        return {
            'mean_interval': np.mean(intervals),
            'median_interval': np.median(intervals),
            'std_interval': np.std(intervals)
        }


class ItemStatistics:
    """
    物品统计分析
    """

    def __init__(self, interactions: List[Tuple]):
        """
        参数:
            interactions: [(user_id, item_id, rating/timestamp), ...]
        """
        self.interactions = interactions

    def popularity_distribution(self) -> Dict:
        """物品流行度分布"""
        from collections import Counter

        item_counts = Counter([item for _, item, _ in self.interactions])
        counts = list(item_counts.values())

        return {
            'mean_popularity': np.mean(counts),
            'median_popularity': np.median(counts),
            'gini_coefficient': self._gini_coefficient(counts),
            'entropy': self._entropy(counts),
            'long_tail_ratio': self._long_tail_ratio(counts)
        }

    @staticmethod
    def _gini_coefficient(values: List) -> float:
        """计算基尼系数"""
        values = np.sort(values)
        n = len(values)
        cumsum = np.cumsum(values)
        return (2 * np.sum((np.arange(1, n+1)) * values) -
                (n+1) * np.sum(values)) / (n * np.sum(values))

    @staticmethod
    def _entropy(values: List) -> float:
        """计算熵"""
        probs = np.array(values) / sum(values)
        probs = probs[probs > 0]
        return -np.sum(probs * np.log2(probs))

    @staticmethod
    def _long_tail_ratio(values: List, threshold: float = 0.8) -> float:
        """长尾比例"""
        values = np.sort(values)[::-1]
        cumsum = np.cumsum(values)
        n_head = np.searchsorted(cumsum / cumsum[-1], threshold)
        return 1 - n_head / len(values)
```

## 6. 学习总结

### 6.1 核心要点

1. **概率基础**: 条件概率、贝叶斯定理是推荐系统核心
2. **分布模型**: Beta分布用于CTR建模，指数分布用于会话建模
3. **参数估计**: MLE 和贝叶斯估计各有应用场景
4. **假设检验**: A/B测试的基础

### 6.2 推荐系统应用

```
概率概念 → 推荐应用
────────────────────────────
条件概率 → 协同过滤
贝叶斯更新 → 在线学习、冷启动
Beta分布 → CTR平滑、汤普森采样
假设检验 → A/B测试
相关性 → 相似度计算
```
