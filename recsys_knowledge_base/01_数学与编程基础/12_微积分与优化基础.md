# 微积分与优化基础 学习文档

## 1. 为什么推荐系统需要优化？

推荐系统的核心是**最优化问题**：
- 最小化预测误差
- 最大化推荐效果
- 学习最优参数

## 2. 导数基础

### 2.1 导数定义

导数表示函数在某点的**变化率**：

$$f'(x) = \lim_{h \to 0} \frac{f(x+h) - f(x)}{h}$$

```python
import numpy as np
import matplotlib.pyplot as plt

def numerical_derivative(f, x, h=1e-5):
    """数值导数"""
    return (f(x + h) - f(x - h)) / (2 * h)

# 示例函数
def f(x):
    return x ** 2

# 在 x=3 处的导数
x = 3
deriv = numerical_derivative(f, x)
print(f"f(x) = x² 在 x={x} 处的导数: {deriv}")  # 应该接近 2*3=6
```

### 2.3 链式法则

```python
def sigmoid(x):
    """Sigmoid 函数"""
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(x):
    """Sigmoid 导数（使用链式法则）"""
```

## 3. 偏导数与梯度

### 3.1 偏导数

对于多变量函数，偏导数是对单个变量的导数：

$$\frac{\partial f}{\partial x_i} = \lim_{h \to 0} \frac{f(x_1, ..., x_i+h, ..., x_n) - f(x_1, ..., x_i, ..., x_n)}{h}$$

```python
def numerical_gradient(f, x, h=1e-5):
    """
    数值梯度

    参数:
        f: 多变量函数
        x: 向量（numpy 数组）
        h: 步长

    返回:
        梯度向量
    """
    grad = np.zeros_like(x)

    for i in range(len(x)):
        # 对第 i 个变量求偏导
        x_plus = x.copy()
        x_plus[i] += h
        x_minus = x.copy()
        x_minus[i] -= h

        grad[i] = (f(x_plus) - f(x_minus)) / (2 * h)

    return grad

# 示例: f(x, y) = x² + y²
def f(xy):
    return xy[0]**2 + xy[1]**2

x = np.array([3.0, 4.0])
grad = numerical_gradient(f, x)
print(f"在 (3, 4) 处的梯度: {grad}")  # 应该是 [6, 8]
```

### 3.2 梯度的意义

梯度指向函数**增长最快**的方向：


## 4. 梯度下降

### 4.1 基本原理

梯度下降通过**沿负梯度方向**迭代更新参数：

$$\theta_{t+1} = \theta_t - \eta \nabla f(\theta_t)$$

其中 η 是学习率。

```python
def gradient_descent(f, grad_f, x0, learning_rate=0.1,
                    max_iters=100, tol=1e-6):
    """
    梯度下降

    参数:
        f: 目标函数
        grad_f: 梯度函数
        x0: 初始点
        learning_rate: 学习率
        max_iters: 最大迭代次数
        tol: 收敛阈值

    返回:
        最优解, 历史记录
    """
    x = x0.copy()
    history = [x.copy()]

    for i in range(max_iters):
        # 计算梯度
        grad = grad_f(x)

        # 更新参数
        x = x - learning_rate * grad
        history.append(x.copy())

        # 检查收敛
        if np.linalg.norm(grad) < tol:
            print(f"在 {i+1} 次迭代后收敛")
            break

    return x, history

# 示例: 最小化 f(x) = x² + y²
def f(xy):
    return xy[0]**2 + xy[1]**2

def grad_f(xy):
    return np.array([2*xy[0], 2*xy[1]])

x0 = np.array([3.0, 4.0])
x_opt, history = gradient_descent(f, grad_f, x0, learning_rate=0.1)
print(f"最优解: {x_opt}")
print(f"最优值: {f(x_opt)}")
```

## 5. 优化算法

### 5.1 SGD（随机梯度下降）

```python
class SGD:
    """随机梯度下降"""

    def __init__(self, learning_rate=0.01):
        self.lr = learning_rate

    def update(self, params, grads):
        """更新参数"""
        for key in params.keys():
            params[key] -= self.lr * grads[key]
        return params


class Momentum:
    """动量法"""

    def __init__(self, learning_rate=0.01, momentum=0.9):
        self.lr = learning_rate
        self.momentum = momentum
        self.v = None

    def update(self, params, grads):
        if self.v is None:
            self.v = {key: np.zeros_like(val) for key, val in params.items()}

        for key in params.keys():
            self.v[key] = self.momentum * self.v[key] - self.lr * grads[key]
            params[key] += self.v[key]

        return params


class Adam:
    """Adam 优化器"""

    def __init__(self, learning_rate=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = learning_rate
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None  # 一阶矩
        self.v = None  # 二阶矩
        self.t = 0     # 时间步

    def update(self, params, grads):
        if self.m is None:
            self.m = {key: np.zeros_like(val) for key, val in params.items()}
            self.v = {key: np.zeros_like(val) for key, val in params.items()}

        self.t += 1

        for key in params.keys():
            # 更新一阶矩
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * grads[key]
            # 更新二阶矩
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * (grads[key] ** 2)

            # 偏差校正
            m_hat = self.m[key] / (1 - self.beta1 ** self.t)
            v_hat = self.v[key] / (1 - self.beta2 ** self.t)

            # 更新参数
            params[key] -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

        return params
```

## 6. 正则化

### 6.1 L1 正则化（Lasso）

$$L = L_{loss} + \lambda \sum_{i} |w_i|$$

```python
def l1_regularization(weights, lambda_reg):
    """L1 正则化"""
    return lambda_reg * np.sum(np.abs(weights))

def l1_gradient(weights, lambda_reg):
    """L1 正则化梯度"""
    return lambda_reg * np.sign(weights)
```

### 6.2 L2 正则化（Ridge）

$$L = L_{loss} + \lambda \sum_{i} w_i^2$$

```python
def l2_regularization(weights, lambda_reg):
    """L2 正则化"""
    return lambda_reg * np.sum(weights ** 2)

def l2_gradient(weights, lambda_reg):
    """L2 正则化梯度"""
    return 2 * lambda_reg * weights
```

### 6.3 Elastic Net

$$L = L_{loss} + \lambda_1 \sum_{i} |w_i| + \lambda_2 \sum_{i} w_i^2$$

```python
def elastic_net_regularization(weights, lambda1, lambda2):
    """Elastic Net 正则化"""
    return lambda1 * np.sum(np.abs(weights)) + lambda2 * np.sum(weights ** 2)
```

## 7. 损失函数

### 7.1 均方误差（MSE）

```python
def mse_loss(y_pred, y_true):
    """均方误差"""
    return np.mean((y_pred - y_true) ** 2)

def mse_gradient(y_pred, y_true):
    """MSE 梯度"""
    return 2 * (y_pred - y_true) / len(y_true)
```

### 7.2 交叉熵损失

```python
def binary_cross_entropy(y_pred, y_true, eps=1e-15):
    """二元交叉熵"""
    y_pred = np.clip(y_pred, eps, 1 - eps)
    return -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))

def bce_gradient(y_pred, y_true, eps=1e-15):
    """BCE 梯度"""
    y_pred = np.clip(y_pred, eps, 1 - eps)
    return (y_pred - y_true) / (y_pred * (1 - y_pred))
```

### 7.3 Softmax 交叉熵

```python
def softmax(x):
    """Softmax 函数"""
    exp_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
    return exp_x / np.sum(exp_x, axis=-1, keepdims=True)

def cross_entropy_loss(logits, labels):
    """交叉熵损失（带 softmax）"""
    probs = softmax(logits)
    return -np.mean(np.log(probs[np.arange(len(labels)), labels] + 1e-10))
```

## 8. 完整训练示例

```python
def train_linear_model(X, y, learning_rate=0.01, epochs=100, reg_lambda=0.01):
    """
    训练线性回归模型

    参数:
        X: 特征矩阵 (n_samples, n_features)
        y: 标签 (n_samples,)
        learning_rate: 学习率
        epochs: 训练轮数
        reg_lambda: L2 正则化系数
    """
    n_samples, n_features = X.shape

    # 初始化权重
    W = np.random.randn(n_features) * 0.01
    b = 0

    losses = []

    for epoch in range(epochs):
        # 前向传播
        y_pred = X @ W + b

        # 计算损失
        loss = mse_loss(y_pred, y) + l2_regularization(W, reg_lambda)
        losses.append(loss)

        # 计算梯度
        dW = mse_gradient(y_pred, y) @ X + l2_gradient(W, reg_lambda)
        db = np.mean(2 * (y_pred - y))

        # 更新参数
        W -= learning_rate * dW
        b -= learning_rate * db

        if (epoch + 1) % 20 == 0:
            print(f"Epoch {epoch+1}/{epochs}, Loss: {loss:.4f}")

    return W, b, losses


# 示例
np.random.seed(42)
X = np.random.randn(100, 3)
true_W = np.array([1.5, -2.0, 1.0])
y = X @ true_W + np.random.randn(100) * 0.1

W, b, losses = train_linear_model(X, y, learning_rate=0.1, epochs=100)
print(f"\n学习到的权重: {W}")
print(f"真实权重: {true_W}")
```

## 9. 学习总结

### 9.1 核心要点

1. **导数**：函数变化率
2. **梯度**：多变量函数的方向导数
3. **梯度下降**：沿负梯度方向优化
4. **正则化**：防止过拟合

### 9.2 推荐系统应用

| 优化概念 | 推荐系统应用 |
|---------|-------------|
| 梯度下降 | 模型训练 |
| 学习率 | 超参数调优 |
| 正则化 | 防止过拟合 |
| 损失函数 | 优化目标 |
