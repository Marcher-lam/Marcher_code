# 知识图谱与推荐系统 学习文档

## 1. 知识图谱概述

### 1.1 什么是知识图谱？

```
知识图谱 (Knowledge Graph):

- 结构化的语义知识库
- 实体 (Entity) + 关系 (Relation) + 属性 (Attribute)
- 图结构存储知识

在推荐系统中的作用:
1. 解决数据稀疏
2. 提供语义关联
3. 增强可解释性
4. 发现潜在兴趣

例子:
实体: 电影《盗梦空间》
关系: 导演 → 克里斯托弗·诺兰
关系: 主演 → 莱昂纳多·迪卡普里奥
关系: 类型 → 科幻
关系: 相似 → 《星际穿越》
```

### 1.2 知识图谱表示

```python
from typing import List, Dict, Tuple, Optional, Set
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F


class Triple:
    """
    知识图谱三元组

    (head, relation, tail)
    """

    def __init__(self, head: str, relation: str, tail: str):
        self.head = head
        self.relation = relation
        self.tail = tail

    def __repr__(self):
        return f"({self.head}, {self.relation}, {self.tail})"


class KnowledgeGraph:
    """
    知识图谱
    """

    def __init__(self):
        self.entities: Set[str] = set()
        self.relations: Set[str] = set()
        self.triples: List[Triple] = []

        # 索引
        self.entity2id: Dict[str, int] = {}
        self.relation2id: Dict[str, int] = {}

        # 邻接表
        self.outgoing: Dict[int, List[Tuple[int, int]]] = {}  # entity -> [(relation, tail)]
        self.incoming: Dict[int, List[Tuple[int, int]]] = {}  # entity -> [(relation, head)]

    def add_triple(self, head: str, relation: str, tail: str):
        """添加三元组"""
        triple = Triple(head, relation, tail)
        self.triples.append(triple)

        # 添加实体和关系
        self.entities.add(head)
        self.entities.add(tail)
        self.relations.add(relation)

        # 更新索引
        if head not in self.entity2id:
            self.entity2id[head] = len(self.entity2id)
        if tail not in self.entity2id:
            self.entity2id[tail] = len(self.entity2id)
        if relation not in self.relation2id:
            self.relation2id[relation] = len(self.relation2id)

        # 更新邻接表
        h_id = self.entity2id[head]
        r_id = self.relation2id[relation]
        t_id = self.entity2id[tail]

        if h_id not in self.outgoing:
            self.outgoing[h_id] = []
        self.outgoing[h_id].append((r_id, t_id))

        if t_id not in self.incoming:
            self.incoming[t_id] = []
        self.incoming[t_id].append((r_id, h_id))

    def get_neighbors(self, entity_id: int) -> List[Tuple[int, int]]:
        """获取邻居"""
        return self.outgoing.get(entity_id, [])

    @property
    def n_entities(self) -> int:
        return len(self.entities)

    @property
    def n_relations(self) -> int:
        return len(self.relations)
```

## 2. 知识图谱嵌入

### 2.1 TransE

```python
class TransE(nn.Module):
    """
    TransE: 简单有效的知识图谱嵌入

    核心思想: h + r ≈ t

    适用于一对一关系
    """

    def __init__(self, n_entities: int, n_relations: int,
                 embed_dim: int = 128, margin: float = 1.0):
        super().__init__()

        self.n_entities = n_entities
        self.n_relations = n_relations
        self.embed_dim = embed_dim
        self.margin = margin

        # 嵌入
        self.entity_embedding = nn.Embedding(n_entities, embed_dim)
        self.relation_embedding = nn.Embedding(n_relations, embed_dim)

        # 初始化
        nn.init.xavier_uniform_(self.entity_embedding.weight)
        nn.init.xavier_uniform_(self.relation_embedding.weight)

        # 归一化
        with torch.no_grad():
            self.entity_embedding.weight.div_(
                self.entity_embedding.weight.norm(p=2, dim=1, keepdim=True)
            )

    def forward(self, heads: torch.Tensor, relations: torch.Tensor,
                tails: torch.Tensor, negative_tails: torch.Tensor) -> torch.Tensor:
        """
        计算 TransE 损失
        """
        # 嵌入
        h = self.entity_embedding(heads)
        r = self.relation_embedding(relations)
        t = self.entity_embedding(tails)
        t_neg = self.entity_embedding(negative_tails)

        # 正样本距离
        pos_score = torch.norm(h + r - t, p=2, dim=-1)

        # 负样本距离
        neg_score = torch.norm(h + r - t_neg, p=2, dim=-1)

        # Margin Loss
        loss = F.relu(self.margin + pos_score - neg_score).mean()

        return loss

    def get_entity_embed(self, entity_id: int) -> torch.Tensor:
        """获取实体嵌入"""
        return self.entity_embedding.weight[entity_id]

    def predict_tail(self, head: torch.Tensor, relation: torch.Tensor,
                     k: int = 10) -> torch.Tensor:
        """预测尾实体"""
        h = self.entity_embedding(head)
        r = self.relation_embedding(relation)

        # 计算所有候选
        all_entities = self.entity_embedding.weight
        scores = torch.norm(h + r - all_entities.unsqueeze(0), p=2, dim=-1)

        # Top-K
        _, indices = torch.topk(scores, k, largest=False)

        return indices
```

### 2.2 TransR

```python
class TransR(nn.Module):
    """
    TransR: 关系特定的投影空间

    不同关系有不同的语义空间
    """

    def __init__(self, n_entities: int, n_relations: int,
                 entity_dim: int = 128, relation_dim: int = 128,
                 margin: float = 1.0):
        super().__init__()

        self.n_entities = n_entities
        self.n_relations = n_relations
        self.entity_dim = entity_dim
        self.relation_dim = relation_dim
        self.margin = margin

        # 嵌入
        self.entity_embedding = nn.Embedding(n_entities, entity_dim)
        self.relation_embedding = nn.Embedding(n_relations, relation_dim)

        # 投影矩阵 (每个关系一个)
        self.projection = nn.Parameter(
            torch.randn(n_relations, entity_dim, relation_dim)
        )

        nn.init.xavier_uniform_(self.entity_embedding.weight)
        nn.init.xavier_uniform_(self.relation_embedding.weight)
        nn.init.xavier_uniform_(self.projection)

    def project(self, entity_embed: torch.Tensor,
                relation_id: torch.Tensor) -> torch.Tensor:
        """
        投影实体到关系空间
        """
        # (batch, entity_dim) @ (batch, entity_dim, relation_dim)
        proj_matrix = self.projection[relation_id]

        return torch.bmm(entity_embed.unsqueeze(1), proj_matrix).squeeze(1)

    def forward(self, heads: torch.Tensor, relations: torch.Tensor,
                tails: torch.Tensor, negative_tails: torch.Tensor) -> torch.Tensor:
        """计算损失"""
        # 嵌入
        h = self.entity_embedding(heads)
        r = self.relation_embedding(relations)
        t = self.entity_embedding(tails)
        t_neg = self.entity_embedding(negative_tails)

        # 投影
        h_proj = self.project(h, relations)
        t_proj = self.project(t, relations)
        t_neg_proj = self.project(t_neg, relations)

        # 距离
        pos_score = torch.norm(h_proj + r - t_proj, p=2, dim=-1)
        neg_score = torch.norm(h_proj + r - t_neg_proj, p=2, dim=-1)

        loss = F.relu(self.margin + pos_score - neg_score).mean()

        return loss
```

## 3. 知识图谱增强推荐

### 3.1 CKE (Collaborative Knowledge Base Embedding)

```python
class CKE(nn.Module):
    """
    CKE: 协同知识库嵌入

    融合协同过滤和知识图谱
    """

    def __init__(self,
                 n_users: int,
                 n_items: int,
                 n_entities: int,
                 n_relations: int,
                 embed_dim: int = 64):
        super().__init__()

        # 用户嵌入 (CF)
        self.user_embedding = nn.Embedding(n_users, embed_dim)

        # 物品/实体嵌入 (共享)
        self.item_entity_embedding = nn.Embedding(n_items, embed_dim)

        # TransE 部分
        self.relation_embedding = nn.Embedding(n_relations, embed_dim)

        # 偏置
        self.user_bias = nn.Embedding(n_users, 1)
        self.item_bias = nn.Embedding(n_items, 1)

    def cf_forward(self, user_ids: torch.Tensor,
                   item_ids: torch.Tensor) -> torch.Tensor:
        """协同过滤部分"""
        user_embed = self.user_embedding(user_ids)
        item_embed = self.item_entity_embedding(item_ids)

        # 内积
        cf_score = (user_embed * item_embed).sum(dim=-1, keepdim=True)

        # 偏置
        user_b = self.user_bias(user_ids)
        item_b = self.item_bias(item_ids)

        return cf_score + user_b + item_b

    def kg_forward(self, heads: torch.Tensor, relations: torch.Tensor,
                   tails: torch.Tensor) -> torch.Tensor:
        """知识图谱部分"""
        h = self.item_entity_embedding(heads)
        r = self.relation_embedding(relations)
        t = self.item_entity_embedding(tails)

        # TransE 分数
        score = -torch.norm(h + r - t, p=2, dim=-1)

        return score

    def forward(self, user_ids: torch.Tensor, item_ids: torch.Tensor,
                kg_heads: torch.Tensor = None, kg_relations: torch.Tensor = None,
                kg_tails: torch.Tensor = None) -> Dict[str, torch.Tensor]:
        """前向传播"""
        cf_score = self.cf_forward(user_ids, item_ids)

        result = {'cf_score': cf_score}

        if kg_heads is not None:
            kg_score = self.kg_forward(kg_heads, kg_relations, kg_tails)
            result['kg_score'] = kg_score

        return result
```

```python
    def forward(self, user_history_items: torch.Tensor,
                item_ids: torch.Tensor,
                ripple_sets: List[torch.Tensor]) -> torch.Tensor:
        """
        前向传播

        user_history_items: (batch, n_history)
        item_ids: (batch,)
        ripple_sets: 每一跳的 ripple 集合
        """
        batch_size = user_history_items.size(0)

        # 物品嵌入
        item_embed = self.item_embedding(item_ids)  # (batch, embed_dim)

        # 用户表示 (通过 ripple 传播)
        user_embed = torch.zeros(batch_size, self.embed_dim, device=item_ids.device)

        for hop_ripple in ripple_sets:
            # hop_ripple: (batch, n_memory, 3) - (head, relation, tail)
            heads = hop_ripple[:, :, 0]
            relations = hop_ripple[:, :, 1]
            tails = hop_ripple[:, :, 2]

            # 计算注意力权重
            head_embed = self.item_embedding(heads)
            relation_embed = self.relation_embedding(relations)
            tail_embed = self.item_embedding(tails)

            # 与物品的相似度作为权重
            item_embed_expand = item_embed.unsqueeze(1).expand(-1, heads.size(1), -1)
            attn_weights = torch.matmul(
                head_embed * relation_embed[:, :, :self.embed_dim],
                item_embed_expand.unsqueeze(-1)
            ).squeeze(-1)
            attn_weights = F.softmax(attn_weights, dim=-1)

            # 加权求和
            user_embed = user_embed + torch.matmul(
                attn_weights.unsqueeze(1),
                tail_embed
            ).squeeze(1)

        # 预测分数
        score = (user_embed * item_embed).sum(dim=-1)

        return score
```

### 3.3 KGAT

```python
class KGAT(nn.Module):
    """
    KGAT: Knowledge Graph Attention Network

    使用 GAT 在知识图谱上进行信息传播
    """

    def __init__(self,
                 n_users: int,
                 n_items: int,
                 n_entities: int,
                 n_relations: int,
                 embed_dim: int = 64,
                 n_layers: int = 2,
                 n_heads: int = 4):
        super().__init__()

        self.n_users = n_users
        self.n_items = n_items
        self.n_entities = n_entities
        self.embed_dim = embed_dim
        self.n_layers = n_layers

        # 嵌入
        # 用户和实体在同一空间
        self.entity_embedding = nn.Embedding(n_users + n_entities, embed_dim)
        self.relation_embedding = nn.Embedding(n_relations, embed_dim)

        # GAT 层
        self.gat_layers = nn.ModuleList([
            GATLayer(embed_dim, n_heads) for _ in range(n_layers)
        ])

    def forward(self, user_ids: torch.Tensor,
                item_ids: torch.Tensor,
                adj_matrix: torch.Tensor,
                edge_types: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        adj_matrix: 邻接矩阵 (包含用户-物品和知识图谱边)
        edge_types: 边类型 (关系)
        """
        # 初始嵌入
        all_embeds = self.entity_embedding.weight

        # GAT 传播
        for gat_layer in self.gat_layers:
            all_embeds = gat_layer(all_embeds, adj_matrix, edge_types)

        # 获取用户和物品嵌入
        user_embed = all_embeds[user_ids]
        item_embed = all_embeds[self.n_users + item_ids]

        # 预测
        score = (user_embed * item_embed).sum(dim=-1)

        return score


class GATLayer(nn.Module):
    """
    图注意力层
    """

    def __init__(self, embed_dim: int, n_heads: int = 4):
        super().__init__()

        self.n_heads = n_heads
        self.head_dim = embed_dim // n_heads

        self.q_proj = nn.Linear(embed_dim, embed_dim)
        self.k_proj = nn.Linear(embed_dim, embed_dim)
        self.v_proj = nn.Linear(embed_dim, embed_dim)
        self.out_proj = nn.Linear(embed_dim, embed_dim)

    def forward(self, x: torch.Tensor, adj: torch.Tensor,
                edge_types: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播

        x: (n_nodes, embed_dim)
        adj: (n_nodes, n_nodes) 邻接矩阵
        """
        n_nodes = x.size(0)

        # Q, K, V
        q = self.q_proj(x).view(n_nodes, self.n_heads, self.head_dim)
        k = self.k_proj(x).view(n_nodes, self.n_heads, self.head_dim)
        v = self.v_proj(x).view(n_nodes, self.n_heads, self.head_dim)

        # 注意力分数
        scores = torch.einsum('nhd,mhd->nmh', q, k) / (self.head_dim ** 0.5)

        # 只保留邻居
        mask = (adj == 0).unsqueeze(-1)
        scores = scores.masked_fill(mask, float('-inf'))

        # Softmax
        attn = F.softmax(scores, dim=1)
        attn = torch.where(torch.isnan(attn), torch.zeros_like(attn), attn)

        # 聚合
        out = torch.einsum('nmh,mhd->nhd', attn, v)
        out = out.contiguous().view(n_nodes, -1)
        out = self.out_proj(out)

        # 残差
        return x + out
```

### 4.1 可解释推荐

```python
class ExplainableKGRec:
    """
    基于知识图谱的可解释推荐
    """

    def __init__(self, kg: KnowledgeGraph, model: nn.Module):
        self.kg = kg
        self.model = model

    def explain_recommendation(self, user_id: int, item_id: int,
                               max_paths: int = 3) -> List[str]:
        """
        生成推荐解释

        返回连接用户和物品的知识路径
        """
        explanations = []

        # 获取用户历史物品
        user_history = self._get_user_history(user_id)

        # 找到历史物品到推荐物品的路径
        for history_item in user_history[:10]:  # 限制搜索
            paths = self._find_paths(history_item, item_id, max_length=3)

            for path in paths[:max_paths]:
                explanation = self._path_to_text(path)
                explanations.append(explanation)

        return explanations[:max_paths]

    def _find_paths(self, start: int, end: int,
                    max_length: int = 3) -> List[List[Tuple]]:
        """
        找到两个实体之间的路径

        BFS 搜索
        """
        from collections import deque

        paths = []
        queue = deque([(start, [])])
        visited = {start}

        while queue and len(paths) < 5:
            current, path = queue.popleft()

            if len(path) > max_length * 2:  # 每跳包含 (relation, tail)
                continue

            if current == end and path:
                paths.append(path)
                continue

            for relation, neighbor in self.kg.get_neighbors(current):
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, path + [(current, relation, neighbor)]))

        return paths

    def _path_to_text(self, path: List[Tuple]) -> str:
        """将路径转换为自然语言"""
        if not path:
            return ""

        parts = []
        for head, relation, tail in path:
            head_name = self.kg.id2entity.get(head, str(head))
            relation_name = self.kg.id2relation.get(relation, str(relation))
            tail_name = self.kg.id2entity.get(tail, str(tail))

            parts.append(f"{head_name} {relation_name} {tail_name}")

        return " → ".join(parts)

    def _get_user_history(self, user_id: int) -> List[int]:
        """获取用户历史物品"""
        # 实际实现需要访问用户行为数据
        return []
```

### 5.1 方法对比

```
方法        特点                    适用场景
────────────────────────────────────────────────
TransE      简单高效                一对一关系
TransR      关系特定空间            复杂关系
CKE         CF + KG 联合           一般场景
RippleNet   兴趣传播                长尾推荐
KGAT        图注意力                知识密集型
```
