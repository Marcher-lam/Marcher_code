# 推荐系统冷启动问题 学习文档

## 1. 问题定义

### 1.1 什么是冷启动？

冷启动是指推荐系统在没有足够历史数据时无法做出准确推荐的问题。

### 1.2 冷启动类型

| 类型 | 描述 | 难度 |
|------|------|------|
| 用户冷启动 | 新用户无行为记录 | 高 |
| 物品冷启动 | 新物品无交互记录 | 中 |
| 系统冷启动 | 新系统无任何数据 | 极高 |

### 1.3 问题本质

```
冷启动 = 数据稀疏 + 信息不足

解决思路:
1. 利用已有信息（元数据、属性）
2. 利用相似对象的信息
3. 主动探索获取信息
```

## 2. 用户冷启动解决方案

### 2.1 基于人口统计学的推荐

```python
import numpy as np
from typing import Dict, List, Tuple
from collections import defaultdict


class DemographicRecommender:
    """
    基于人口统计学的推荐
    """

    def __init__(self):
        # 用户画像存储
        self.user_profiles = {}
        # 人口统计群体 -> 物品偏好
        self.group_preferences = defaultdict(lambda: defaultdict(float))

    def add_user(self, user_id: str, demographics: Dict):
        """
        添加用户人口统计信息

        参数:
            demographics: {
                'age_group': '18-25',
                'gender': 'M',
                'location': 'Beijing',
                'occupation': 'student'
            }
        """
        self.user_profiles[user_id] = demographics

    def learn_group_preferences(self, interactions: List[Tuple]):
        """
        从历史交互中学习群体偏好

        参数:
            interactions: [(user_id, item_id, rating), ...]
        """
        group_item_counts = defaultdict(lambda: defaultdict(int))
        group_totals = defaultdict(int)

        for user_id, item_id, rating in interactions:
            if user_id not in self.user_profiles:
                continue

            # 生成群体键
            group_key = self._get_group_key(self.user_profiles[user_id])
            group_item_counts[group_key][item_id] += rating
            group_totals[group_key] += 1

        # 计算群体平均偏好
        for group_key in group_item_counts:
            for item_id in group_item_counts[group_key]:
                self.group_preferences[group_key][item_id] = (
                    group_item_counts[group_key][item_id] /
                    group_totals[group_key]
                )

    def _get_group_key(self, demographics: Dict) -> str:
        """生成群体键"""
        return f"{demographics.get('age_group', 'unknown')}_" \
               f"{demographics.get('gender', 'unknown')}_" \
               f"{demographics.get('occupation', 'unknown')}"

    def recommend_for_new_user(self, demographics: Dict, n: int = 10) -> List[Tuple]:
        """
        为新用户推荐

        参数:
            demographics: 人口统计信息
            n: 推荐数量

        返回:
            [(item_id, score), ...]
        """
        group_key = self._get_group_key(demographics)

        if group_key not in self.group_preferences:
            # 回退到更粗粒度的群体
            group_key = f"{demographics.get('age_group', 'unknown')}_" \
                       f"{demographics.get('gender', 'unknown')}"

        if group_key in self.group_preferences:
            items = self.group_preferences[group_key]
            return sorted(items.items(), key=lambda x: x[1], reverse=True)[:n]

        return []


class OnboardingQuestionnaire:
    """
    新用户引导问卷
    """

    def __init__(self):
        self.questions = []
        self.item_tags = {}

    def add_question(self, question: str, options: List[str], tag_weights: Dict):
        """
        添加问题

        参数:
            question: 问题文本
            options: 选项列表
            tag_weights: {option: {tag: weight}}
        """
        self.questions.append({
            'question': question,
            'options': options,
            'tag_weights': tag_weights
        })

    def get_user_preferences(self, answers: List[str]) -> Dict[str, float]:
        """
        根据用户回答计算偏好

        参数:
            answers: 对每个问题的回答

        返回:
            {tag: preference_score}
        """
        preferences = defaultdict(float)

        for i, answer in enumerate(answers):
            if i >= len(self.questions):
                break

            question = self.questions[i]
            if answer in question['tag_weights']:
                for tag, weight in question['tag_weights'][answer].items():
                    preferences[tag] += weight

        return dict(preferences)

    def recommend_by_preferences(self, preferences: Dict[str, float],
                                 items: List[Dict], n: int = 10) -> List[Tuple]:
        """根据偏好推荐物品"""
        scores = []

        for item in items:
            item_id = item['id']
            item_tags = item.get('tags', [])

            score = sum(preferences.get(tag, 0) for tag in item_tags)
            scores.append((item_id, score))

        return sorted(scores, key=lambda x: x[1], reverse=True)[:n]
```

```python
class RegistrationBasedRecommender:
    """
    基于注册信息的推荐
    """

    def __init__(self):
        # 关键词 -> 相关物品
        self.keyword_items = defaultdict(list)
        # 类别 -> 热门物品
        self.category_top_items = {}

    def build_index(self, items: List[Dict]):
        """构建索引"""
        for item in items:
            item_id = item['id']
            category = item.get('category')
            keywords = item.get('keywords', [])

            # 关键词索引
            for keyword in keywords:
                self.keyword_items[keyword].append(item_id)

            # 类别索引
            if category:
                if category not in self.category_top_items:
                    self.category_top_items[category] = []
                self.category_top_items[category].append(item_id)

    def recommend(self, registration_info: Dict, n: int = 10) -> List[str]:
        """
        基于注册信息推荐

        参数:
            registration_info: {
                'interests': ['科技', '游戏'],
                'occupation': '程序员',
                'age': 25
            }
        """
        candidates = defaultdict(float)

        # 基于兴趣
        interests = registration_info.get('interests', [])
        for interest in interests:
            for item_id in self.keyword_items.get(interest, []):
                candidates[item_id] += 1.0

        # 基于职业
        occupation = registration_info.get('occupation')
        if occupation:
            for item_id in self.keyword_items.get(occupation, []):
                candidates[item_id] += 0.8

        # 排序返回
        sorted_items = sorted(candidates.items(), key=lambda x: x[1], reverse=True)
        return [item_id for item_id, _ in sorted_items[:n]]
```

### 2.3 热门推荐兜底

```python
class PopularFallbackRecommender:
    """
    热门推荐兜底策略
    """

    def __init__(self, time_window: int = 7):
        self.time_window = time_window  # 天
        self.item_scores = defaultdict(float)

    def update_scores(self, item_id: str, timestamp: float, current_time: float):
        """更新物品分数（带时间衰减）"""
        days_ago = (current_time - timestamp) / (24 * 3600)
        if days_ago <= self.time_window:
            decay = np.exp(-0.1 * days_ago)
            self.item_scores[item_id] += decay

    def get_top_popular(self, n: int = 10) -> List[Tuple[str, float]]:
        """获取热门物品"""
        return sorted(
            self.item_scores.items(),
            key=lambda x: x[1],
            reverse=True
        )[:n]
```

## 3. 物品冷启动解决方案

### 3.1 基于内容的推荐

```python
class ContentBasedColdStart:
    """
    基于内容的物品冷启动
    """

    def __init__(self, embed_dim: int = 128):
        self.embed_dim = embed_dim
        self.item_features = {}
        self.item_embeddings = {}

    def extract_features(self, item_info: Dict) -> np.ndarray:
        """
        提取物品特征

        参数:
            item_info: {
                'title': str,
                'description': str,
                'category': str,
                'tags': List[str],
                'price': float
            }
        """
        features = []

        # 类别特征（one-hot）
        category = item_info.get('category', '')
        # 这里简化处理，实际应该有类别映射
        features.append(hash(category) % 100 / 100)

        # 标签特征
        tags = item_info.get('tags', [])
        tag_feature = len(tags) / 10  # 标签数量
        features.append(tag_feature)

        # 价格特征
        price = item_info.get('price', 0)
        features.append(np.log1p(price) / 10)

        # 文本特征（简化）
        text = f"{item_info.get('title', '')} {item_info.get('description', '')}"
        text_feature = len(text) / 1000
        features.append(text_feature)

        return np.array(features)

    def find_similar_items(self, new_item_features: np.ndarray,
                          existing_items: Dict[str, np.ndarray],
                          n: int = 10) -> List[Tuple[str, float]]:
        """
        找到相似的已有物品

        参数:
            new_item_features: 新物品特征
            existing_items: {item_id: features}
        """
        similarities = []

        for item_id, features in existing_items.items():
            # 余弦相似度
            sim = np.dot(new_item_features, features) / (
                np.linalg.norm(new_item_features) * np.linalg.norm(features) + 1e-10
            )
            similarities.append((item_id, sim))

        return sorted(similarities, key=lambda x: x[1], reverse=True)[:n]

    def recommend_for_new_item(self, new_item_id: str, new_item_info: Dict,
                               user_item_history: Dict[str, List[str]],
                               n_users: int = 100) -> List[str]:
        """
        为新物品找到潜在感兴趣的用户

        参数:
            new_item_info: 新物品信息
            user_item_history: {user_id: [item_id, ...]}
            n_users: 推荐用户数量
        """
        # 提取新物品特征
        new_features = self.extract_features(new_item_info)

        # 找到相似物品
        similar_items = self.find_similar_items(new_features, self.item_features)

        # 找到喜欢相似物品的用户
        user_scores = defaultdict(float)
        for similar_item_id, similarity in similar_items:
            for user_id, history in user_item_history.items():
                if similar_item_id in history:
                    user_scores[user_id] += similarity

        # 返回得分最高的用户
        return sorted(user_scores.keys(), key=lambda x: user_scores[x], reverse=True)[:n_users]
```

### 3.2 利用物品属性传播

```python
class AttributePropagation:
    """
    基于物品属性的推荐传播
    """

    def __init__(self):
        # 属性 -> 用户偏好
        self.attribute_user_affinity = defaultdict(lambda: defaultdict(float))

    def learn_from_interactions(self, interactions: List[Tuple],
                                item_attributes: Dict[str, List[str]]):
        """
        从交互中学习属性-用户关联

        参数:
            interactions: [(user_id, item_id, rating), ...]
            item_attributes: {item_id: [attr1, attr2, ...]}
        """
        for user_id, item_id, rating in interactions:
            if item_id not in item_attributes:
                continue

            for attr in item_attributes[item_id]:
                self.attribute_user_affinity[attr][user_id] += rating

    def find_potential_users(self, new_item_attrs: List[str],
                            min_affinity: float = 1.0) -> List[Tuple[str, float]]:
        """
        为具有特定属性的新物品找到潜在用户
        """
        user_scores = defaultdict(float)

        for attr in new_item_attrs:
            for user_id, affinity in self.attribute_user_affinity[attr].items():
                user_scores[user_id] += affinity

        # 过滤并排序
        return sorted(
            [(u, s) for u, s in user_scores.items() if s >= min_affinity],
            key=lambda x: x[1],
            reverse=True
        )
```

### 4.1 多臂老虎机冷启动

```python
class BanditColdStart:
    """
    使用多臂老虎机解决冷启动
    """

    def __init__(self, n_items: int, exploration_bonus: float = 1.0):
        self.n_items = n_items
        self.exploration_bonus = exploration_bonus

        # UCB 参数
        self.item_clicks = np.zeros(n_items)
        self.item_impressions = np.zeros(n_items)
        self.total_impressions = 0

    def select_items(self, n: int = 10) -> List[int]:
        """选择要展示的物品（UCB策略）"""
        scores = np.zeros(self.n_items)

        for i in range(self.n_items):
            if self.item_impressions[i] == 0:
                # 未展示过的物品给予最高优先级
                scores[i] = float('inf')
            else:
                # UCB 分数
                avg_reward = self.item_clicks[i] / self.item_impressions[i]
                exploration = np.sqrt(
                    2 * np.log(self.total_impressions + 1) / self.item_impressions[i]
                )
                scores[i] = avg_reward + self.exploration_bonus * exploration

        # 返回得分最高的物品
        top_indices = np.argsort(scores)[::-1][:n]
        return top_indices.tolist()

    def update(self, item_id: int, clicked: bool):
        """更新统计信息"""
        self.item_impressions[item_id] += 1
        self.total_impressions += 1
        if clicked:
            self.item_clicks[item_id] += 1
```

```python
class AdaptiveExploration:
    """
    适应性探索策略

    根据用户反馈动态调整探索程度
    """

    def __init__(self, initial_explore_rate: float = 0.3):
        self.explore_rate = initial_explore_rate
        self.min_explore_rate = 0.05
        self.decay_rate = 0.95

    def should_explore(self) -> bool:
        """决定是否进行探索"""
        return np.random.random() < self.explore_rate

    def update(self, engagement: float):
        """
        根据用户参与度更新探索率

        参数:
            engagement: 参与度分数 [0, 1]
        """
        if engagement > 0.5:
            # 高参与度，减少探索
            self.explore_rate = max(
                self.min_explore_rate,
                self.explore_rate * self.decay_rate
            )
        else:
            # 低参与度，增加探索
            self.explore_rate = min(0.5, self.explore_rate / self.decay_rate)
```

```python
class CrossDomainColdStart:
    """
    跨域推荐解决冷启动

    利用源域的用户行为帮助目标域推荐
    """

    def __init__(self):
        # 源域用户行为
        self.source_domain_behavior = {}
        # 域间映射
        self.domain_mapping = {}

    def learn_mapping(self, source_interactions: Dict,
                     target_interactions: Dict,
                     source_item_features: Dict,
                     target_item_features: Dict):
        """
        学习源域到目标域的映射
        """
        # 找到在两个域都有行为的用户
        common_users = set(source_interactions.keys()) & set(target_interactions.keys())

        # 基于共同用户学习物品相似度
        for user_id in common_users:
            source_items = source_interactions[user_id]
            target_items = target_interactions[user_id]

            # 建立关联
            for s_item in source_items:
                if s_item not in self.domain_mapping:
                    self.domain_mapping[s_item] = defaultdict(float)
                for t_item in target_items:
                    self.domain_mapping[s_item][t_item] += 1

    def recommend_from_source(self, source_history: List[str],
                             n: int = 10) -> List[Tuple[str, float]]:
        """
        基于源域历史推荐目标域物品
        """
        target_scores = defaultdict(float)

        for source_item in source_history:
            if source_item in self.domain_mapping:
                for target_item, count in self.domain_mapping[source_item].items():
                    target_scores[target_item] += count

        return sorted(target_scores.items(), key=lambda x: x[1], reverse=True)[:n]
```

## 6. 冷启动评估

### 6.1 评估指标

```python
def evaluate_cold_start(recommender, test_cases: List[Dict]) -> Dict:
    """
    评估冷启动推荐效果

    参数:
        test_cases: [{
            'user_id': str,
            'type': 'new_user' or 'new_item',
            'ground_truth': List[str]
        }]
    """
    metrics = {
        'hit_rate': [],
        'coverage': [],
        'diversity': []
    }

    for case in test_cases:
        if case['type'] == 'new_user':
            recs = recommender.recommend_for_new_user(case['user_info'], n=10)
        else:
            recs = recommender.recommend_for_new_item(case['item_info'], n=10)

        rec_items = set(r[0] if isinstance(r, tuple) else r for r in recs)
        truth_items = set(case['ground_truth'])

        # Hit Rate
        hit = len(rec_items & truth_items) > 0
        metrics['hit_rate'].append(hit)

    return {
        'avg_hit_rate': np.mean(metrics['hit_rate']),
        'std_hit_rate': np.std(metrics['hit_rate'])
    }
```
