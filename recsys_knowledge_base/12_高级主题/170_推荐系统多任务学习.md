# 推荐系统多任务学习 学习文档

## 1. 多任务学习概述

### 1.1 什么是多任务学习？

```
多任务学习 (Multi-Task Learning, MTL):
- 同时学习多个相关任务
- 共享表示，互相促进
- 提高泛化能力

推荐系统中的多任务:
- 点击率 (CTR) + 转化率 (CVR)
- 点击 + 点赞 + 收藏 + 分享
- 短期目标 + 长期目标
```

### 1.2 为什么需要多任务学习？

```
优势:
1. 数据效率: 多任务共享数据
2. 正则化: 减少过拟合
3. 迁移学习: 知识共享
4. 计算效率: 一次推理多任务

挑战:
1. 负迁移: 任务冲突
2. 任务权重: 如何平衡
3. 梯度冲突: 优化困难
```

## 2. 多任务模型架构

### 2.1 硬参数共享

```python
import torch
import torch.nn as nn
from typing import List, Dict, Tuple


class HardSharedMTL(nn.Module):
    """
    硬参数共享

    共享底层，各任务独立输出头
    """

    def __init__(self, feature_dim: int, hidden_dims: List[int],
                 n_tasks: int = 2):
        super().__init__()

        self.n_tasks = n_tasks

        # 共享层
        layers = []
        input_dim = feature_dim
        for hidden_dim in hidden_dims:
            layers.extend([
                nn.Linear(input_dim, hidden_dim),
                nn.BatchNorm1d(hidden_dim),
                nn.ReLU(),
                nn.Dropout(0.2)
            ])
            input_dim = hidden_dim

        self.shared_layers = nn.Sequential(*layers)

        # 任务特定层
        self.task_heads = nn.ModuleList([
            nn.Sequential(
                nn.Linear(hidden_dims[-1], hidden_dims[-1] // 2),
                nn.ReLU(),
                nn.Linear(hidden_dims[-1] // 2, 1)
            ) for _ in range(n_tasks)
        ])

    def forward(self, features: torch.Tensor) -> List[torch.Tensor]:
        """
        前向传播

        返回: [task1_output, task2_output, ...]
        """
        shared_repr = self.shared_layers(features)

        outputs = []
        for head in self.task_heads:
            outputs.append(head(shared_repr))

        return outputs


class SoftSharedMTL(nn.Module):
    """
    软参数共享

    每个任务有自己的参数，但通过正则化约束相似
    """

    def __init__(self, feature_dim: int, hidden_dim: int,
                 n_tasks: int = 2, lambda_reg: float = 0.1):
        super().__init__()

        self.n_tasks = n_tasks
        self.lambda_reg = lambda_reg

        # 每个任务独立的网络
        self.task_networks = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, hidden_dim),
                nn.ReLU(),
                nn.Linear(hidden_dim, hidden_dim // 2),
                nn.ReLU(),
                nn.Linear(hidden_dim // 2, 1)
            ) for _ in range(n_tasks)
        ])

    def forward(self, features: torch.Tensor) -> List[torch.Tensor]:
        """前向传播"""
        outputs = []
        for net in self.task_networks:
            outputs.append(net(features))
        return outputs

    def regularization_loss(self) -> torch.Tensor:
        """
        正则化损失

        约束不同任务网络的参数相似
        """
        reg_loss = 0

        # 提取各任务网络参数
        params = []
        for net in self.task_networks:
            task_params = []
            for param in net.parameters():
                task_params.append(param.view(-1))
            params.append(torch.cat(task_params))

        # 计算参数差异
        for i in range(self.n_tasks):
            for j in range(i + 1, self.n_tasks):
                reg_loss += torch.norm(params[i] - params[j])

        return self.lambda_reg * reg_loss
```

### 2.2 MMOE 架构

```python
class MMoE(nn.Module):
    """
    Multi-gate Mixture-of-Experts (MMoE)

    Google 提出的多任务学习架构
    """

    def __init__(self, feature_dim: int, n_experts: int = 8,
                 expert_dim: int = 64, n_tasks: int = 2):
        super().__init__()

        self.n_experts = n_experts
        self.n_tasks = n_tasks

        # 专家网络
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, expert_dim),
                nn.ReLU(),
                nn.Linear(expert_dim, expert_dim),
                nn.ReLU()
            ) for _ in range(n_experts)
        ])

        # 门控网络（每个任务一个）
        self.gates = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, n_experts),
                nn.Softmax(dim=-1)
            ) for _ in range(n_tasks)
        ])

        # 任务塔
        self.towers = nn.ModuleList([
            nn.Sequential(
                nn.Linear(expert_dim, expert_dim // 2),
                nn.ReLU(),
                nn.Linear(expert_dim // 2, 1)
            ) for _ in range(n_tasks)
        ])

    def forward(self, features: torch.Tensor) -> List[torch.Tensor]:
        """
        前向传播
        """
        batch_size = features.size(0)

        # 专家输出
        expert_outputs = []
        for expert in self.experts:
            expert_outputs.append(expert(features))
        expert_outputs = torch.stack(expert_outputs, dim=1)  # (batch, n_experts, expert_dim)

        outputs = []
        for task_id in range(self.n_tasks):
            # 门控权重
            gate_weights = self.gates[task_id](features)  # (batch, n_experts)

            # 加权融合专家输出
            gate_weights = gate_weights.unsqueeze(-1)  # (batch, n_experts, 1)
            weighted_expert = torch.sum(gate_weights * expert_outputs, dim=1)  # (batch, expert_dim)

            # 任务塔
            task_output = self.towers[task_id](weighted_expert)
            outputs.append(task_output)

        return outputs


class PLE(nn.Module):
    """
    Progressive Layered Extraction (PLE)

    腾讯提出的改进版 MMOE，增加任务特定专家
    """

    def __init__(self, feature_dim: int, n_shared_experts: int = 4,
                 n_task_experts: int = 4, expert_dim: int = 64,
                 n_tasks: int = 2):
        super().__init__()

        self.n_tasks = n_tasks
        self.n_shared_experts = n_shared_experts
        self.n_task_experts = n_task_experts

        # 共享专家
        self.shared_experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, expert_dim),
                nn.ReLU(),
                nn.Linear(expert_dim, expert_dim)
            ) for _ in range(n_shared_experts)
        ])

        # 任务特定专家
        self.task_experts = nn.ModuleList([
            nn.ModuleList([
                nn.Sequential(
                    nn.Linear(feature_dim, expert_dim),
                    nn.ReLU(),
                    nn.Linear(expert_dim, expert_dim)
                ) for _ in range(n_task_experts)
            ]) for _ in range(n_tasks)
        ])

        # 门控网络
        total_experts = n_shared_experts + n_task_experts
        self.gates = nn.ModuleList([
            nn.Sequential(
                nn.Linear(feature_dim, total_experts),
                nn.Softmax(dim=-1)
            ) for _ in range(n_tasks)
        ])

        # 任务塔
        self.towers = nn.ModuleList([
            nn.Sequential(
                nn.Linear(expert_dim, expert_dim // 2),
                nn.ReLU(),
                nn.Linear(expert_dim // 2, 1)
            ) for _ in range(n_tasks)
        ])

    def forward(self, features: torch.Tensor) -> List[torch.Tensor]:
        """前向传播"""
        outputs = []

        for task_id in range(self.n_tasks):
            # 共享专家输出
            shared_outputs = [expert(features) for expert in self.shared_experts]

            # 任务特定专家输出
            task_outputs = [expert(features) for expert in self.task_experts[task_id]]

            # 合并所有专家
            all_expert_outputs = shared_outputs + task_outputs
            all_expert_outputs = torch.stack(all_expert_outputs, dim=1)

            # 门控
            gate_weights = self.gates[task_id](features).unsqueeze(-1)
            weighted_output = torch.sum(gate_weights * all_expert_outputs, dim=1)

            # 任务塔
            task_output = self.towers[task_id](weighted_output)
            outputs.append(task_output)

        return outputs
```

### 2.3 ESMM 架构

```python
class ESMM(nn.Module):
    """
    Entire Space Multi-Task Model (ESMM)

    阿里提出，专门用于 CTR+CVR 多任务
    解决样本选择偏差 (SSB) 和数据稀疏 (DS) 问题
    """

    def __init__(self, feature_dim: int, hidden_dim: int = 128):
        super().__init__()

        # 嵌入层（共享）
        self.embedding = nn.Linear(feature_dim, hidden_dim)

        # CTR 网络
        self.ctr_net = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1),
            nn.Sigmoid()
        )

        # CVR 网络
        self.cvr_net = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1),
            nn.Sigmoid()
        )

    def forward(self, features: torch.Tensor) -> Dict[str, torch.Tensor]:
        """
        前向传播

        返回:
            ctr: P(click | impression)
            cvr: P(conversion | click)
            ctcvr: P(conversion | impression) = ctr * cvr
        """
        embed = self.embedding(features)

        ctr = self.ctr_net(embed)
        cvr = self.cvr_net(embed)

        # CTCVR = CTR * CVR
        ctcvr = ctr * cvr

        return {
            'ctr': ctr,
            'cvr': cvr,
            'ctcvr': ctcvr
        }

    def loss(self, outputs: Dict, click_labels: torch.Tensor,
            conversion_labels: torch.Tensor) -> torch.Tensor:
        """
        ESMM 损失

        L = L_CTR + L_CTCVR

        注意: CVR 没有直接监督，通过 CTCVR 间接学习
        """
        ctr = outputs['ctr']
        ctcvr = outputs['ctcvr']

        # CTR 损失（所有曝光样本）
        ctr_loss = nn.BCELoss()(ctr.squeeze(), click_labels.float())

        # CTCVR 损失（所有曝光样本）
        ctcvr_loss = nn.BCELoss()(ctcvr.squeeze(), conversion_labels.float())

        return ctr_loss + ctcvr_loss
```

## 3. 梯度冲突处理

### 3.1 梯度归一化

```python
class GradientNormalization:
    """
    梯度归一化 (GradNorm)

    动态调整任务权重，使各任务梯度范数相近
    """

    def __init__(self, n_tasks: int, alpha: float = 1.5):
        self.n_tasks = n_tasks
        self.alpha = alpha

        # 初始损失
        self.initial_losses = None
        # 任务权重
        self.task_weights = nn.Parameter(torch.ones(n_tasks))

    def compute_weights(self, losses: List[torch.Tensor]) -> torch.Tensor:
        """
        计算任务权重
        """
        if self.initial_losses is None:
            self.initial_losses = [loss.item() for loss in losses]

        # 计算损失比率
        loss_ratios = []
        for i, loss in enumerate(losses):
            ratio = loss.item() / (self.initial_losses[i] + 1e-10)
            loss_ratios.append(ratio)

        loss_ratios = torch.tensor(loss_ratios)
        avg_ratio = loss_ratios.mean()

        # 计算权重
        weights = (loss_ratios / avg_ratio) ** self.alpha
        weights = weights / weights.sum() * self.n_tasks

        return weights


class PCGrad:
    """
    Project Conflicting Gradients (PCGrad)

    处理梯度冲突，将冲突梯度投影
    """

    def __init__(self):
        pass

    def project_conflicting_gradients(self, grads: List[torch.Tensor]) -> List[torch.Tensor]:
        """
        投影冲突梯度
        """
        n_tasks = len(grads)
        projected_grads = [g.clone() for g in grads]

        for i in range(n_tasks):
            for j in range(n_tasks):
                if i != j:
                    # 检查梯度是否冲突
                    cos_sim = torch.dot(
                        projected_grads[i].flatten(),
                        grads[j].flatten()
                    ) / (
                        torch.norm(projected_grads[i]) *
                        torch.norm(grads[j]) + 1e-10
                    )

                    if cos_sim < 0:
                        # 投影
                        projected_grads[i] = projected_grads[i] - (
                            torch.dot(projected_grads[i].flatten(), grads[j].flatten()) /
                            (torch.norm(grads[j]) ** 2 + 1e-10)
                        ) * grads[j]

        return projected_grads
```

## 4. 多任务训练

### 4.1 训练器

```python
class MTLTrainer:
    """
    多任务训练器
    """

    def __init__(self, model, task_names: List[str],
                 task_weights: List[float] = None,
                 lr: float = 0.001):
        self.model = model
        self.task_names = task_names
        self.n_tasks = len(task_names)

        if task_weights is None:
            self.task_weights = [1.0] * self.n_tasks
        else:
            self.task_weights = task_weights

        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    def train_batch(self, features: torch.Tensor,
                   labels: Dict[str, torch.Tensor]) -> Dict:
        """
        训练一个批次
        """
        self.model.train()
        self.optimizer.zero_grad()

        # 前向传播
        outputs = self.model(features)

        # 计算各任务损失
        task_losses = {}
        total_loss = 0

        for i, task_name in enumerate(self.task_names):
            pred = outputs[i] if isinstance(outputs, list) else outputs[task_name]
            label = labels[task_name]

            loss = nn.BCEWithLogitsLoss()(pred.squeeze(), label.float())
            task_losses[task_name] = loss.item()

            total_loss += self.task_weights[i] * loss

        # 反向传播
        total_loss.backward()
        self.optimizer.step()

        return {
            'total_loss': total_loss.item(),
            'task_losses': task_losses
        }

    def evaluate(self, dataloader) -> Dict:
        """评估"""
        self.model.eval()

        task_preds = {name: [] for name in self.task_names}
        task_labels = {name: [] for name in self.task_names}

        with torch.no_grad():
            for batch in dataloader:
                features = batch['features']
                outputs = self.model(features)

                for i, task_name in enumerate(self.task_names):
                    pred = outputs[i] if isinstance(outputs, list) else outputs[task_name]
                    pred_prob = torch.sigmoid(pred.squeeze())

                    task_preds[task_name].extend(pred_prob.cpu().numpy())
                    task_labels[task_name].extend(batch[task_name].cpu().numpy())

        # 计算各任务 AUC
        from sklearn.metrics import roc_auc_score

        metrics = {}
        for task_name in self.task_names:
            preds = np.array(task_preds[task_name])
            labels = np.array(task_labels[task_name])

            if len(np.unique(labels)) > 1:
                auc = roc_auc_score(labels, preds)
                metrics[f'{task_name}_auc'] = auc
            else:
                metrics[f'{task_name}_auc'] = 0.5

        return metrics


class DynamicWeightTrainer(MTLTrainer):
    """
    动态权重调整训练器
    """

    def __init__(self, model, task_names: List[str],
                 lr: float = 0.001,
                 weight_update_freq: int = 100):
        super().__init__(model, task_names, None, lr)

        self.weight_update_freq = weight_update_freq
        self.step_count = 0

        # 记录各任务损失历史
        self.loss_history = {name: [] for name in task_names}

    def update_weights(self):
        """动态更新任务权重"""
        # 基于损失的倒数
        weights = []
        for task_name in self.task_names:
            if self.loss_history[task_name]:
                avg_loss = np.mean(self.loss_history[task_name][-100:])
                weights.append(1.0 / (avg_loss + 1e-10))
            else:
                weights.append(1.0)

        # 归一化
        total = sum(weights)
        self.task_weights = [w / total * self.n_tasks for w in weights]

    def train_batch(self, features: torch.Tensor,
                   labels: Dict[str, torch.Tensor]) -> Dict:
        """训练"""
        self.step_count += 1

        if self.step_count % self.weight_update_freq == 0:
            self.update_weights()

        result = super().train_batch(features, labels)

        # 记录损失
        for task_name in self.task_names:
            self.loss_history[task_name].append(result['task_losses'][task_name])

        return result
```

## 5. 学习总结

### 5.1 核心要点

1. **任务相关**: 多任务之间应有相关性
2. **架构选择**: MMOE、PLE、ESMM 各有适用场景
3. **权重平衡**: 动态调整任务权重
4. **梯度冲突**: 处理负迁移问题

### 5.2 架构对比

```
架构        特点                适用场景
────────────────────────────────────────────
Hard Share  简单共享            任务高度相关
MMOE        专家混合            一般场景
PLE         任务特定专家        任务差异大
ESMM        CTR+CVR专用         转化预测
```

### 5.3 最佳实践

```
1. 任务选择: 相关但不过于相似
2. 权重调优: 从等权重开始
3. 监控各任务: 确保无负迁移
4. 消融实验: 验证多任务收益
```
