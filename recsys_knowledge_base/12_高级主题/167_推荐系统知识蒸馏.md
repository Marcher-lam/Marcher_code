# 推荐系统知识蒸馏 学习文档

## 1. 知识蒸馏基础

### 1.1 什么是知识蒸馏？

```
知识蒸馏 (Knowledge Distillation):
- Teacher-Student 框架
- 将大模型(教师)的知识迁移到小模型(学生)
- 学生模型学习教师的软标签(Soft Labels)

核心思想:
- 软标签比硬标签包含更多信息
- 类别间的相似度关系是有价值的知识
```

### 1.2 推荐系统中的应用

```
应用场景:
1. 模型压缩: 大模型 → 小模型
2. 加速推理: 复杂模型 → 简单模型
3. 跨模态迁移: 多模态模型 → 单模态模型
4. 集成蒸馏: 多个模型 → 单个模型
```

## 2. 蒸馏方法

### 2.1 经典蒸馏

```python
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from typing import List, Tuple, Dict, Optional


class SoftTargetDistillation(nn.Module):
    """
    软标签蒸馏

    学生学习教师的软标签分布
    """

    def __init__(self, temperature: float = 4.0, alpha: float = 0.7):
        """
        参数:
            temperature: 蒸馏温度（越高越软）
            alpha: 蒸馏损失权重
        """
        super().__init__()
        self.temperature = temperature
        self.alpha = alpha

    def forward(self, student_logits: torch.Tensor,
               teacher_logits: torch.Tensor,
               labels: torch.Tensor) -> Tuple[torch.Tensor, Dict]:
        """
        蒸馏损失

        L = α * L_KL + (1-α) * L_CE

        参数:
            student_logits: 学生模型输出
            teacher_logits: 教师模型输出
            labels: 真实标签
        """
        # 软标签蒸馏损失
        soft_loss = F.kl_div(
            F.log_softmax(student_logits / self.temperature, dim=-1),
            F.softmax(teacher_logits / self.temperature, dim=-1),
            reduction='batchmean'
        ) * (self.temperature ** 2)

        # 硬标签损失
        hard_loss = F.cross_entropy(student_logits, labels)

        # 组合损失
        total_loss = self.alpha * soft_loss + (1 - self.alpha) * hard_loss

        return total_loss, {
            'soft_loss': soft_loss.item(),
            'hard_loss': hard_loss.item()
        }


class ResponseBasedDistillation(nn.Module):
    """
    基于响应的蒸馏

    学生学习教师的最终预测
    """

    def __init__(self, temperature: float = 2.0):
        super().__init__()
        self.temperature = temperature

    def forward(self, student_scores: torch.Tensor,
               teacher_scores: torch.Tensor,
               labels: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        MSE 损失或 KL 散度

        参数:
            student_scores: 学生预测分数 (batch,)
            teacher_scores: 教师预测分数 (batch,)
        """
        # 归一化
        student_probs = torch.sigmoid(student_scores / self.temperature)
        teacher_probs = torch.sigmoid(teacher_scores / self.temperature)

        # MSE 损失
        loss = F.mse_loss(student_probs, teacher_probs)

        return loss


class FeatureBasedDistillation(nn.Module):
    """
    基于特征的蒸馏

    学生学习教师的中间层表示
    """

    def __init__(self, student_dim: int, teacher_dim: int):
        super().__init__()

        # 投影层（对齐维度）
        self.projector = nn.Linear(student_dim, teacher_dim)

    def forward(self, student_features: torch.Tensor,
               teacher_features: torch.Tensor) -> torch.Tensor:
        """
        特征对齐损失

        参数:
            student_features: 学生中间特征 (batch, student_dim)
            teacher_features: 教师中间特征 (batch, teacher_dim)
        """
        # 投影到教师空间
        projected = self.projector(student_features)

        # L2 损失
        loss = F.mse_loss(projected, teacher_features)

        return loss


class RelationBasedDistillation(nn.Module):
    """
    基于关系的蒸馏

    学生学习教师表示之间的结构关系
    """

    def __init__(self):
        super().__init__()

    def forward(self, student_features: torch.Tensor,
               teacher_features: torch.Tensor) -> torch.Tensor:
        """
        关系蒸馏

        保持样本间的关系结构
        """
        # 计算关系矩阵（相似度矩阵）
        student_relations = self._compute_relation(student_features)
        teacher_relations = self._compute_relation(teacher_features)

        # 关系对齐损失
        loss = F.mse_loss(student_relations, teacher_relations)

        return loss

    def _compute_relation(self, features: torch.Tensor) -> torch.Tensor:
        """计算关系矩阵"""
        # 归一化
        features = F.normalize(features, dim=-1)

        # 相似度矩阵
        relation = torch.matmul(features, features.T)

        return relation
```

```python
class CTRModelDistillation:
    """
    CTR 模型蒸馏

    复杂教师模型 → 简单学生模型
    """

    def __init__(self, teacher_model: nn.Module,
                 student_model: nn.Module,
                 temperature: float = 2.0,
                 alpha: float = 0.5):
        self.teacher = teacher_model
        self.student = student_model
        self.temperature = temperature
        self.alpha = alpha

        # 冻结教师模型
        for param in self.teacher.parameters():
            param.requires_grad = False
        self.teacher.eval()

        # 损失函数
        self.response_loss = ResponseBasedDistillation(temperature)
        self.feature_loss = None  # 可选

    def distill(self, dataloader, optimizer, n_epochs: int = 10):
        """
        蒸馏训练
        """
        for epoch in range(n_epochs):
            self.student.train()
            total_loss = 0
            n_batches = 0

            for batch in dataloader:
                # 教师预测
                with torch.no_grad():
                    teacher_scores = self.teacher(batch)

                # 学生预测
                student_scores = self.student(batch)

                # 蒸馏损失
                distill_loss = self.response_loss(
                    student_scores, teacher_scores
                )

                # 硬标签损失
                labels = batch['label']
                hard_loss = F.binary_cross_entropy_with_logits(
                    student_scores, labels
                )

                # 组合
                loss = self.alpha * distill_loss + (1 - self.alpha) * hard_loss

                # 优化
                optimizer.zero_grad()
                loss.backward()
                optimizer.step()

                total_loss += loss.item()
                n_batches += 1

            avg_loss = total_loss / n_batches
            print(f"Epoch {epoch+1}: Loss = {avg_loss:.4f}")


class RankingDistillation(nn.Module):
    """
    排序蒸馏

    学生学习教师的排序关系
    """

    def __init__(self, temperature: float = 1.0):
        super().__init__()
        self.temperature = temperature

    def forward(self, student_scores: torch.Tensor,
               teacher_scores: torch.Tensor) -> torch.Tensor:
        """
        Listwise 排序蒸馏

        参数:
            student_scores: 学生分数 (batch, n_items)
            teacher_scores: 教师分数 (batch, n_items)
        """
        # 软化分布
        student_probs = F.log_softmax(student_scores / self.temperature, dim=-1)
        teacher_probs = F.softmax(teacher_scores / self.temperature, dim=-1)

        # KL 散度
        loss = F.kl_div(student_probs, teacher_probs, reduction='batchmean')

        return loss * (self.temperature ** 2)

    def pairwise_forward(self, student_scores: torch.Tensor,
                        teacher_scores: torch.Tensor) -> torch.Tensor:
        """
        Pairwise 排序蒸馏

        保持两两之间的排序关系
        """
        batch_size, n_items = student_scores.size()

        # 计算两两差异
        student_diff = student_scores.unsqueeze(2) - student_scores.unsqueeze(1)
        teacher_diff = teacher_scores.unsqueeze(2) - teacher_scores.unsqueeze(1)

        # 符号一致性
        student_sign = torch.tanh(student_diff)
        teacher_sign = torch.sign(teacher_diff)

        # 损失
        loss = F.mse_loss(student_sign, teacher_sign)

        return loss
```

```python
class SequenceDistillation(nn.Module):
    """
    序列推荐蒸馏
    """

    def __init__(self, temperature: float = 2.0, alpha: float = 0.5):
        super().__init__()
        self.temperature = temperature
        self.alpha = alpha

    def forward(self, student_logits: torch.Tensor,
               teacher_logits: torch.Tensor,
               labels: torch.Tensor) -> torch.Tensor:
        """
        序列蒸馏

        参数:
            student_logits: (batch, seq_len, n_items)
            teacher_logits: (batch, seq_len, n_items)
            labels: (batch, seq_len)
        """
        batch_size, seq_len, n_items = student_logits.size()

        # 展平
        student_logits = student_logits.view(-1, n_items)
        teacher_logits = teacher_logits.view(-1, n_items)
        labels = labels.view(-1)

        # 软标签损失
        soft_loss = F.kl_div(
            F.log_softmax(student_logits / self.temperature, dim=-1),
            F.softmax(teacher_logits / self.temperature, dim=-1),
            reduction='batchmean'
        ) * (self.temperature ** 2)

        # 硬标签损失
        hard_loss = F.cross_entropy(student_logits, labels)

        return self.alpha * soft_loss + (1 - self.alpha) * hard_loss


class AttentionDistillation(nn.Module):
    """
    注意力蒸馏

    学生学习教师的注意力分布
    """

    def __init__(self):
        super().__init__()

    def forward(self, student_attention: torch.Tensor,
               teacher_attention: torch.Tensor) -> torch.Tensor:
        """
        注意力对齐

        参数:
            student_attention: (batch, n_heads, seq_len, seq_len)
            teacher_attention: (batch, n_heads, seq_len, seq_len)
        """
        # MSE 损失
        loss = F.mse_loss(student_attention, teacher_attention)

        return loss


class HiddenStateDistillation(nn.Module):
    """
    隐藏状态蒸馏
    """

    def __init__(self, student_dim: int, teacher_dim: int):
        super().__init__()

        self.projector = nn.Sequential(
            nn.Linear(student_dim, teacher_dim),
            nn.ReLU()
        )

    def forward(self, student_hidden: torch.Tensor,
               teacher_hidden: torch.Tensor) -> torch.Tensor:
        """
        隐藏状态对齐
        """
        projected = self.projector(student_hidden)
        return F.mse_loss(projected, teacher_hidden)
```

## 4. 多教师蒸馏

### 4.1 集成蒸馏

```python
class MultiTeacherDistillation(nn.Module):
    """
    多教师蒸馏
    """

    def __init__(self, n_teachers: int, temperature: float = 2.0,
                 ensemble_method: str = 'avg'):
        super().__init__()
        self.n_teachers = n_teachers
        self.temperature = temperature
        self.ensemble_method = ensemble_method

        # 可学习的教师权重
        self.teacher_weights = nn.Parameter(
            torch.ones(n_teachers) / n_teachers
        )

    def forward(self, student_logits: torch.Tensor,
               teacher_logits_list: List[torch.Tensor]) -> torch.Tensor:
        """
        多教师蒸馏

        参数:
            student_logits: 学生输出
            teacher_logits_list: 各教师输出列表
        """
        # 整合教师预测
        if self.ensemble_method == 'avg':
            # 平均
            teacher_logits = torch.stack(teacher_logits_list, dim=0).mean(dim=0)
        elif self.ensemble_method == 'weighted':
            # 加权平均
            weights = F.softmax(self.teacher_weights, dim=0)
            teacher_logits = sum(
                w * t for w, t in zip(weights, teacher_logits_list)
            )
        elif self.ensemble_method == 'max':
            # 最大
            teacher_logits = torch.stack(teacher_logits_list, dim=0).max(dim=0)[0]

        # 蒸馏损失
        soft_loss = F.kl_div(
            F.log_softmax(student_logits / self.temperature, dim=-1),
            F.softmax(teacher_logits / self.temperature, dim=-1),
            reduction='batchmean'
        ) * (self.temperature ** 2)

        return soft_loss


class AdaptiveMultiTeacherDistillation(nn.Module):
    """
    自适应多教师蒸馏

    根据样本动态选择教师
    """

    def __init__(self, n_teachers: int, feature_dim: int,
                 temperature: float = 2.0):
        super().__init__()
        self.n_teachers = n_teachers
        self.temperature = temperature

        # 门控网络
        self.gate_network = nn.Sequential(
            nn.Linear(feature_dim, 128),
            nn.ReLU(),
            nn.Linear(128, n_teachers),
            nn.Softmax(dim=-1)
        )

    def forward(self, student_logits: torch.Tensor,
               teacher_logits_list: List[torch.Tensor],
               features: torch.Tensor) -> torch.Tensor:
        """
        自适应蒸馏
        """
        # 计算门控权重
        gate_weights = self.gate_network(features)  # (batch, n_teachers)

        # 加权教师预测
        teacher_logits = torch.stack(teacher_logits_list, dim=1)  # (batch, n_teachers, n_classes)
        weighted_teacher = torch.sum(
            gate_weights.unsqueeze(-1) * teacher_logits, dim=1
        )

        # 蒸馏损失
        soft_loss = F.kl_div(
            F.log_softmax(student_logits / self.temperature, dim=-1),
            F.softmax(weighted_teacher / self.temperature, dim=-1),
            reduction='batchmean'
        ) * (self.temperature ** 2)

        return soft_loss
```

## 5. 自蒸馏

### 5.1 自蒸馏框架

```python
class SelfDistillation(nn.Module):
    """
    自蒸馏

    模型从自身的早期版本或不同层学习
    """

    def __init__(self, model: nn.Module,
                 temperature: float = 2.0,
                 momentum: float = 0.99):
        super().__init__()
        self.student = model
        self.temperature = temperature
        self.momentum = momentum

        # 教师是学生的 EMA 版本
        self.teacher = self._copy_model(model)
        for param in self.teacher.parameters():
            param.requires_grad = False

    def _copy_model(self, model: nn.Module) -> nn.Module:
        """复制模型"""
        import copy
        return copy.deepcopy(model)

    @torch.no_grad()
    def update_teacher(self):
        """更新教师（EMA）"""
        for param_s, param_t in zip(
            self.student.parameters(),
            self.teacher.parameters()
        ):
            param_t.data = self.momentum * param_t.data + \
                          (1 - self.momentum) * param_s.data

    def forward(self, inputs: torch.Tensor,
               labels: torch.Tensor) -> Tuple[torch.Tensor, Dict]:
        """
        自蒸馏
        """
        # 学生预测
        student_logits = self.student(inputs)

        # 教师预测
        with torch.no_grad():
            teacher_logits = self.teacher(inputs)

        # 软标签损失
        soft_loss = F.kl_div(
            F.log_softmax(student_logits / self.temperature, dim=-1),
            F.softmax(teacher_logits / self.temperature, dim=-1),
            reduction='batchmean'
        ) * (self.temperature ** 2)

        # 硬标签损失
        hard_loss = F.cross_entropy(student_logits, labels)

        # 组合
        total_loss = 0.5 * soft_loss + 0.5 * hard_loss

        return total_loss, {
            'soft_loss': soft_loss.item(),
            'hard_loss': hard_loss.item()
        }


class ProgressiveSelfDistillation(nn.Module):
    """
    渐进式自蒸馏

    从浅层到深层逐步蒸馏
    """

    def __init__(self, layers: List[nn.Module],
                 temperature: float = 2.0):
        super().__init__()
        self.layers = nn.ModuleList(layers)
        self.temperature = temperature
        self.n_layers = len(layers)

    def forward(self, x: torch.Tensor,
               labels: torch.Tensor) -> Tuple[torch.Tensor, Dict]:
        """
        渐进蒸馏
        """
        losses = []
        features = x

        for i, layer in enumerate(self.layers):
            features = layer(features)

            if i > 0:
                # 当前层向最后一层学习
                with torch.no_grad():
                    final_features = self.layers[-1](features.clone())

                distill_loss = F.mse_loss(features, final_features)
                losses.append(distill_loss)

        # 分类损失
        logits = self.classifier(features)
        cls_loss = F.cross_entropy(logits, labels)
        losses.append(cls_loss)

        total_loss = sum(losses)

        return total_loss, {'layer_losses': [l.item() for l in losses]}
```

## 6. 蒸馏效果评估

### 6.1 评估指标

```python
class DistillationEvaluator:
    """
    蒸馏效果评估
    """

    def __init__(self, teacher_model: nn.Module,
                 student_model: nn.Module):
        self.teacher = teacher_model
        self.student = student_model

    def evaluate(self, dataloader) -> Dict:
        """评估"""
        self.teacher.eval()
        self.student.eval()

        teacher_preds = []
        student_preds = []
        labels = []

        with torch.no_grad():
            for batch in dataloader:
                t_pred = self.teacher(batch)
                s_pred = self.student(batch)

                teacher_preds.append(t_pred.cpu())
                student_preds.append(s_pred.cpu())
                labels.append(batch['label'].cpu())

        teacher_preds = torch.cat(teacher_preds)
        student_preds = torch.cat(student_preds)
        labels = torch.cat(labels)

        # 计算指标
        metrics = {
            'teacher_auc': self._compute_auc(teacher_preds, labels),
            'student_auc': self._compute_auc(student_preds, labels),
            'prediction_correlation': self._compute_correlation(
                teacher_preds, student_preds
            ),
            'ranking_similarity': self._compute_ranking_similarity(
                teacher_preds, student_preds
            ),
            'knowledge_transfer_ratio': self._compute_ktr(
                teacher_preds, student_preds, labels
            )
        }

        return metrics

    @staticmethod
    def _compute_auc(preds: torch.Tensor, labels: torch.Tensor) -> float:
        """计算 AUC"""
        from sklearn.metrics import roc_auc_score
        return roc_auc_score(labels.numpy(), torch.sigmoid(preds).numpy())

    @staticmethod
    def _compute_correlation(preds1: torch.Tensor,
                            preds2: torch.Tensor) -> float:
        """计算预测相关性"""
        return torch.corrcoef(torch.stack([preds1.flatten(), preds2.flatten()]))[0, 1].item()

    @staticmethod
    def _compute_ranking_similarity(preds1: torch.Tensor,
                                    preds2: torch.Tensor) -> float:
        """计算排序相似度"""
        rank1 = torch.argsort(torch.argsort(preds1, descending=True))
        rank2 = torch.argsort(torch.argsort(preds2, descending=True))

        # Spearman 相关系数
        from scipy import stats
        return stats.spearmanr(rank1.numpy(), rank2.numpy())[0]

    @staticmethod
    def _compute_ktr(teacher_preds: torch.Tensor,
                    student_preds: torch.Tensor,
                    labels: torch.Tensor) -> float:
        """
        知识迁移率 (Knowledge Transfer Ratio)

        KTR = AUC(student - random) / AUC(teacher - random)
        """
        teacher_auc = DistillationEvaluator._compute_auc(teacher_preds, labels)
        student_auc = DistillationEvaluator._compute_auc(student_preds, labels)

        return student_auc / teacher_auc if teacher_auc > 0 else 0
```

## 7. 学习总结

### 7.1 核心要点

1. **软标签**: 比硬标签包含更多信息
2. **温度参数**: 控制分布软硬程度
3. **多种蒸馏**: 响应、特征、关系蒸馏
4. **自蒸馏**: 模型自我提升

### 7.2 蒸馏类型对比

```
蒸馏类型        蒸馏内容            优势
──────────────────────────────────────────
响应蒸馏        最终预测            简单高效
特征蒸馏        中间表示            细粒度
关系蒸馏        样本间关系          结构保持
注意力蒸馏      注意力分布          可解释
```

### 7.3 最佳实践

```
1. 温度选择: 2-5 之间调优
2. 损失权重: 软标签 0.5-0.9
3. 教师选择: 性能好的大模型
4. 迭代蒸馏: 逐步提升
```
