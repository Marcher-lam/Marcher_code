# 推荐系统 PPO 详解 学习文档

## 1. PPO 算法原理

### 1.1 什么是 PPO？

```
PPO (Proximal Policy Optimization):

- OpenAI 提出的策略梯度算法
- 通过限制策略更新幅度来稳定训练
- 广泛应用于推荐系统的强化学习

核心思想:
- 每次更新不要偏离太远
- 使用 Clip 机制约束更新
- 简单高效，效果好

相比其他算法:
- vs DQN: 适合连续动作空间
- vs A2C: 更稳定的更新
- vs TRPO: 实现更简单
```

### 1.2 PPO 核心公式

```python
"""
PPO 目标函数:

L^CLIP(θ) = E_t[min(r_t(θ) * A_t, clip(r_t(θ), 1-ε, 1+ε) * A_t)]

其中:
- r_t(θ) = π_θ(a|s) / π_θ_old(a|s)  概率比
- A_t = 优势函数
- ε = clip 参数 (通常 0.1-0.2)

优势函数计算:
A_t = δ_t + (γλ)δ_{t+1} + (γλ)²δ_{t+2} + ...
δ_t = r_t + γV(s_{t+1}) - V(s_t)

"""

import torch
import torch.nn as nn
import torch.nn.functional as F
import numpy as np
from typing import List, Dict, Tuple, Optional
from dataclasses import dataclass


@dataclass
class PPOConfig:
    """PPO 配置"""
    gamma: float = 0.99           # 折扣因子
    lambda_gae: float = 0.95      # GAE 参数
    clip_epsilon: float = 0.2      # PPO clip 参数
    entropy_coef: float = 0.01     # 熵系数
    value_coef: float = 0.5        # 价值损失系数
    max_grad_norm: float = 0.5     # 梯度裁剪
    n_epochs: int = 10            # 每批数据训练次数
    batch_size: int = 64          # 批量大小
```

## 2. PPO 网络结构

### 2.1 Actor-Critic 网络

```python
class ActorCritic(nn.Module):
    """
    Actor-Critic 网络

    Actor: 策略网络，输出动作概率
    Critic: 价值网络，输出状态价值
    """

    def __init__(self,
                 state_dim: int,
                 n_actions: int,
                 hidden_dim: int = 256):
        super().__init__()

        # 共享特征提取层
        self.shared = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )

        # Actor: 策略头
        self.actor = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_actions)
        )

        # Critic: 价值头
        self.critic = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

    def forward(self, state: torch.Tensor
               ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        前向传播

        返回:
            action_logits: (batch, n_actions) 动作 logits
            value: (batch, 1) 状态价值
        """
        features = self.shared(state)
        action_logits = self.actor(features)
        value = self.critic(features)
        return action_logits, value

    def get_action(self, state: torch.Tensor,
                  deterministic: bool = False
                  ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """
        获取动作

        返回:
            action: 选择的动作
            log_prob: 动作的对数概率
            value: 状态价值
        """
        action_logits, value = self.forward(state)
        probs = F.softmax(action_logits, dim=-1)

        if deterministic:
            action = probs.argmax(dim=-1)
        else:
            dist = torch.distributions.Categorical(probs)
            action = dist.sample()

        log_prob = F.log_softmax(action_logits, dim=-1)
        log_prob = log_prob.gather(1, action.unsqueeze(1)).squeeze(1)

        return action, log_prob, value.squeeze(-1)

    def evaluate_actions(self, state: torch.Tensor,
                        action: torch.Tensor
                        ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """
        评估动作

        返回:
            log_prob: 动作对数概率
            entropy: 策略熵
            value: 状态价值
        """
        action_logits, value = self.forward(state)
        probs = F.softmax(action_logits, dim=-1)
        log_probs = F.log_softmax(action_logits, dim=-1)

        # 选中动作的对数概率
        action_log_prob = log_probs.gather(1, action.unsqueeze(1)).squeeze(1)

        # 熵
        entropy = -(probs * log_probs).sum(dim=-1)

        return action_log_prob, entropy, value.squeeze(-1)


class RecSysPPOActorCritic(nn.Module):
    """
    推荐系统专用 PPO 网络

    处理用户状态和物品推荐
    """

    def __init__(self,
                 user_feature_dim: int,
                 item_embedding_dim: int,
                 n_items: int,
                 hidden_dim: int = 256):
        super().__init__()

        self.n_items = n_items

        # 用户状态编码器
        self.user_encoder = nn.Sequential(
            nn.Linear(user_feature_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim)
        )

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items, item_embedding_dim)

        # 交互网络 (计算用户-物品分数)
        self.interaction_net = nn.Sequential(
            nn.Linear(hidden_dim + item_embedding_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        # Critic
        self.critic = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

    def compute_scores(self, user_state: torch.Tensor
                      ) -> torch.Tensor:
        """计算所有物品的推荐分数"""
        user_embed = self.user_encoder(user_state)  # (batch, hidden_dim)
        batch_size = user_state.size(0)

        # 所有物品嵌入
        all_items = torch.arange(self.n_items, device=user_state.device)
        item_embeds = self.item_embedding(all_items)  # (n_items, embed_dim)

        # 扩展用户嵌入
        user_embed_expanded = user_embed.unsqueeze(1).expand(
            -1, self.n_items, -1
        )  # (batch, n_items, hidden_dim)

        # 扩展物品嵌入
        item_embeds_expanded = item_embeds.unsqueeze(0).expand(
            batch_size, -1, -1
        )  # (batch, n_items, embed_dim)

        # 拼接并计算分数
        concat = torch.cat([user_embed_expanded, item_embeds_expanded], dim=-1)
        scores = self.interaction_net(concat).squeeze(-1)  # (batch, n_items)

        return scores

    def forward(self, user_state: torch.Tensor
               ) -> Tuple[torch.Tensor, torch.Tensor]:
        """前向传播"""
        user_embed = self.user_encoder(user_state)
        scores = self.compute_scores(user_state)
        value = self.critic(user_embed).squeeze(-1)
        return scores, value

    def get_action(self, user_state: torch.Tensor,
                  candidate_items: torch.Tensor = None,
                  deterministic: bool = False
                  ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """获取推荐动作"""
        scores, value = self.forward(user_state)

        if candidate_items is not None:
            # 只考虑候选物品
            candidate_scores = scores.gather(1, candidate_items)
            probs = F.softmax(candidate_scores, dim=-1)

            if deterministic:
                action_idx = probs.argmax(dim=-1)
                action = candidate_items.gather(1, action_idx.unsqueeze(1)).squeeze(1)
            else:
                dist = torch.distributions.Categorical(probs)
                action_idx = dist.sample()
                action = candidate_items.gather(1, action_idx.unsqueeze(1)).squeeze(1)

            log_prob = F.log_softmax(candidate_scores, dim=-1)
            log_prob = log_prob.gather(1, action_idx.unsqueeze(1)).squeeze(1)
        else:
            probs = F.softmax(scores, dim=-1)

            if deterministic:
                action = probs.argmax(dim=-1)
            else:
                dist = torch.distributions.Categorical(probs)
                action = dist.sample()

            log_prob = F.log_softmax(scores, dim=-1)
            log_prob = log_prob.gather(1, action.unsqueeze(1)).squeeze(1)

        return action, log_prob, value
```

## 3. PPO 训练器

### 3.1 完整训练流程

```python
class RolloutBuffer:
    """
    经验缓冲区

    存储一条轨迹的数据
    """

    def __init__(self):
        self.states = []
        self.actions = []
        self.rewards = []
        self.values = []
        self.log_probs = []
        self.dones = []

    def add(self, state, action, reward, value, log_prob, done):
        """添加一步经验"""
        self.states.append(state)
        self.actions.append(action)
        self.rewards.append(reward)
        self.values.append(value)
        self.log_probs.append(log_prob)
        self.dones.append(done)

    def clear(self):
        """清空"""
        self.states.clear()
        self.actions.clear()
        self.rewards.clear()
        self.values.clear()
        self.log_probs.clear()
        self.dones.clear()

    def compute_gae(self, gamma: float, lambda_gae: float,
                   last_value: float) -> Tuple[np.ndarray, np.ndarray]:
        """
        计算广义优势估计 (GAE)

        返回: (returns, advantages)
        """
        n = len(self.rewards)
        advantages = np.zeros(n)
        returns = np.zeros(n)

        gae = 0
        for t in reversed(range(n)):
            if t == n - 1:
                next_value = last_value
                next_non_terminal = 1.0 - self.dones[t]
            else:
                next_value = self.values[t + 1]
                next_non_terminal = 1.0 - self.dones[t]

            delta = (self.rewards[t] + gamma * next_value * next_non_terminal
                    - self.values[t])
            gae = delta + gamma * lambda_gae * next_non_terminal * gae
            advantages[t] = gae
            returns[t] = advantages[t] + self.values[t]

        return returns, advantages

    def get_batch(self, batch_size: int
                 ) -> List[Dict[str, torch.Tensor]]:
        """生成训练批次"""
        n = len(self.states)
        indices = np.random.permutation(n)

        batches = []
        for start in range(0, n, batch_size):
            end = min(start + batch_size, n)
            batch_indices = indices[start:end]

            batch = {
                'states': torch.tensor(
                    np.array([self.states[i] for i in batch_indices]),
                    dtype=torch.float32
                ),
                'actions': torch.tensor(
                    [self.actions[i] for i in batch_indices],
                    dtype=torch.long
                ),
                'old_log_probs': torch.tensor(
                    [self.log_probs[i] for i in batch_indices],
                    dtype=torch.float32
                ),
                'returns': torch.tensor(
                    [self.returns[i] for i in batch_indices] if hasattr(self, 'returns')
                    else [0] * len(batch_indices),
                    dtype=torch.float32
                ),
                'advantages': torch.tensor(
                    [self.advantages[i] for i in batch_indices] if hasattr(self, 'advantages')
                    else [0] * len(batch_indices),
                    dtype=torch.float32
                )
            }
            batches.append(batch)

        return batches


class PPOTrainer:
    """
    PPO 训练器
    """

    def __init__(self,
                 model: ActorCritic,
                 config: PPOConfig,
                 lr: float = 3e-4):
        self.model = model
        self.config = config

        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

        # 旧策略参数 (用于计算概率比)
        self.old_log_probs = None

    def collect_rollout(self, env, n_steps: int) -> RolloutBuffer:
        """
        收集轨迹

        env: 推荐环境
        n_steps: 收集步数
        """
        buffer = RolloutBuffer()
        state = env.reset()

        for _ in range(n_steps):
            # 获取动作
            with torch.no_grad():
                state_tensor = torch.tensor(state, dtype=torch.float32).unsqueeze(0)
                action, log_prob, value = self.model.get_action(state_tensor)

            # 执行动作
            next_state, reward, done, info = env.step(action.item())

            # 存储
            buffer.add(
                state=state,
                action=action.item(),
                reward=reward,
                value=value.item(),
                log_prob=log_prob.item(),
                done=done
            )

            state = next_state if not done else env.reset()

        # 计算最后状态的价值
        with torch.no_grad():
            state_tensor = torch.tensor(state, dtype=torch.float32).unsqueeze(0)
            _, _, last_value = self.model.get_action(state_tensor)

        # 计算 GAE
        returns, advantages = buffer.compute_gae(
            self.config.gamma,
            self.config.lambda_gae,
            last_value.item()
        )
        buffer.returns = returns
        buffer.advantages = advantages

        return buffer

    def update(self, buffer: RolloutBuffer) -> Dict[str, float]:
        """
        更新策略

        PPO 核心训练逻辑
        """
        # 准备数据
        buffer.returns = getattr(buffer, 'returns', np.zeros(len(buffer.rewards)))
        buffer.advantages = getattr(buffer, 'advantages', np.zeros(len(buffer.rewards)))

        # 优势归一化
        advantages = torch.tensor(buffer.advantages, dtype=torch.float32)
        advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)

        # 转换为张量
        states = torch.tensor(np.array(buffer.states), dtype=torch.float32)
        actions = torch.tensor(buffer.actions, dtype=torch.long)
        old_log_probs = torch.tensor(buffer.log_probs, dtype=torch.float32)
        returns = torch.tensor(buffer.returns, dtype=torch.float32)

        # 多轮更新
        total_losses = []
        policy_losses = []
        value_losses = []
        entropies = []

        n_samples = len(buffer.states)

        for _ in range(self.config.n_epochs):
            # 打乱数据
            indices = np.random.permutation(n_samples)

            for start in range(0, n_samples, self.config.batch_size):
                end = min(start + self.config.batch_size, n_samples)
                batch_indices = indices[start:end]

                batch_states = states[batch_indices]
                batch_actions = actions[batch_indices]
                batch_old_log_probs = old_log_probs[batch_indices]
                batch_returns = returns[batch_indices]
                batch_advantages = advantages[batch_indices]

                # 评估当前策略
                log_probs, entropy, values = self.model.evaluate_actions(
                    batch_states, batch_actions
                )

                # 计算概率比
                ratio = torch.exp(log_probs - batch_old_log_probs)

                # PPO Clip 损失
                surr1 = ratio * batch_advantages
                surr2 = torch.clamp(
                    ratio,
                    1 - self.config.clip_epsilon,
                    1 + self.config.clip_epsilon
                ) * batch_advantages
                policy_loss = -torch.min(surr1, surr2).mean()

                # 价值损失
                value_loss = F.mse_loss(values, batch_returns)

                # 熵损失 (鼓励探索)
                entropy_loss = -entropy.mean()

                # 总损失
                total_loss = (
                    policy_loss +
                    self.config.value_coef * value_loss +
                    self.config.entropy_coef * entropy_loss
                )

                # 反向传播
                self.optimizer.zero_grad()
                total_loss.backward()
                nn.utils.clip_grad_norm_(
                    self.model.parameters(),
                    self.config.max_grad_norm
                )
                self.optimizer.step()

                total_losses.append(total_loss.item())
                policy_losses.append(policy_loss.item())
                value_losses.append(value_loss.item())
                entropies.append(entropy.mean().item())

        return {
            'total_loss': np.mean(total_losses),
            'policy_loss': np.mean(policy_losses),
            'value_loss': np.mean(value_losses),
            'entropy': np.mean(entropies)
        }
```

```python
class RecommenderEnvironment:
    """
    推荐系统强化学习环境

    模拟用户与推荐系统的交互
    """

    def __init__(self,
                 user_profiles: Dict[int, np.ndarray],
                 item_features: np.ndarray,
                 reward_model=None):
        """
        参数:
            user_profiles: 用户特征
            item_features: 物品特征
            reward_model: 用户响应模型 (模拟点击/转化)
        """
        self.user_profiles = user_profiles
        self.item_features = item_features
        self.reward_model = reward_model

        self.n_items = len(item_features)
        self.current_user = None
        self.current_state = None
        self.history = []

    def reset(self) -> np.ndarray:
        """重置环境"""
        # 随机选择用户
        self.current_user = np.random.choice(list(self.user_profiles.keys()))
        self.current_state = self.user_profiles[self.current_user].copy()
        self.history = []

        return self.current_state

    def step(self, action: int) -> Tuple[np.ndarray, float, bool, Dict]:
        """
        执行动作

        action: 推荐的物品ID
        返回: (next_state, reward, done, info)
        """
        # 计算奖励
        reward = self._compute_reward(action)

        # 更新状态
        self.history.append(action)
        self.current_state = self._update_state(action)

        # 检查是否结束
        done = len(self.history) >= 20  # 最多推荐20个

        info = {
            'user_id': self.current_user,
            'item_id': action,
            'history_length': len(self.history)
        }

        return self.current_state, reward, done, info

    def _compute_reward(self, action: int) -> float:
        """计算奖励"""
        if self.reward_model is not None:
            # 使用用户响应模型
            reward = self.reward_model.predict(
                self.current_state,
                self.item_features[action]
            )
        else:
            # 简化: 基于余弦相似度
            user_pref = self.current_state
            item_feat = self.item_features[action]
            similarity = np.dot(user_pref, item_feat) / (
                np.linalg.norm(user_pref) * np.linalg.norm(item_feat) + 1e-10
            )

            # 模拟点击 (相似度越高，点击概率越大)
            click_prob = (similarity + 1) / 2  # 归一化到 [0, 1]
            reward = 1.0 if np.random.random() < click_prob else 0.0

        return reward

    def _update_state(self, action: int) -> np.ndarray:
        """更新用户状态"""
        # 简化: 将物品特征融入状态
        item_feat = self.item_features[action]
        new_state = 0.9 * self.current_state + 0.1 * item_feat
        return new_state


class PPORecommender:
    """
    基于 PPO 的推荐系统
    """

    def __init__(self,
                 state_dim: int,
                 n_items: int,
                 config: PPOConfig = None):
        self.config = config or PPOConfig()

        self.model = ActorCritic(state_dim, n_items)
        self.trainer = PPOTrainer(self.model, self.config)

    def train(self, env: RecommenderEnvironment,
             total_steps: int = 100000,
             rollout_steps: int = 2048):
        """训练"""
        n_updates = total_steps // rollout_steps

        for update in range(n_updates):
            # 收集轨迹
            buffer = self.trainer.collect_rollout(env, rollout_steps)

            # 更新策略
            losses = self.trainer.update(buffer)

            if update % 10 == 0:
                print(f"Update {update}/{n_updates}")
                print(f"  Policy Loss: {losses['policy_loss']:.4f}")
                print(f"  Value Loss: {losses['value_loss']:.4f}")
                print(f"  Entropy: {losses['entropy']:.4f}")

    def recommend(self, state: np.ndarray, top_k: int = 10
                 ) -> List[int]:
        """推荐"""
        with torch.no_grad():
            state_tensor = torch.tensor(state, dtype=torch.float32).unsqueeze(0)
            scores, _ = self.model.forward(state_tensor)
            probs = F.softmax(scores, dim=-1)

            top_items = torch.topk(probs, top_k, dim=-1)
            return top_items.indices[0].tolist()
```

## 5. 学习总结

### 5.1 核心要点

```
1. PPO 核心: 通过 Clip 限制策略更新幅度
2. 优势估计: 使用 GAE 计算优势函数
3. 多轮更新: 每批数据多次更新
4. 熵正则: 鼓励探索

PPO vs 其他算法:
- 稳定性好于 Policy Gradient
- 实现简单于 TRPO
- 适合连续和离散动作
```

### 5.2 超参数调优

```
参数            推荐值           说明
───────────────────────────────────────
clip_epsilon    0.1-0.3         更新限制
lambda_gae      0.9-0.99        GAE 参数
entropy_coef    0.01-0.1        探索鼓励
value_coef      0.5-1.0         价值损失权重
n_epochs        5-15            每批更新次数
batch_size      64-256          批量大小
```

### 5.3 推荐系统应用建议

```
1. 状态设计: 包含用户历史、偏好
2. 动作空间: 可以是物品ID或动作类型
3. 奖励设计: 点击+转化+长期价值
4. 离线预训练 + 在线微调
5. 结合 Bandit 进行探索
```
