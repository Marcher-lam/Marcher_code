# 会话推荐 学习文档

## 1. 会话推荐概述

### 1.1 什么是会话推荐？

```
会话推荐 (Session-based Recommendation):

- 基于用户当前会话行为进行推荐
- 不依赖长期用户画像
- 适用于匿名用户或新用户

会话 (Session):
- 一次访问期间的行为序列
- 通常有时间窗口限制
- 短期兴趣表达

应用场景:
- 电商匿名浏览
- 短视频信息流
- 新闻资讯
- 临时访问
```

### 1.2 与序列推荐的区别

```python
"""
会话推荐 vs 序列推荐

会话推荐:
- 短期行为 (几分钟到几小时)
- 通常无用户ID
- 强调即时兴趣
- 会话独立

序列推荐:
- 长期历史 (几天到几年)
- 需要用户ID
- 综合长期兴趣
- 历史跨会话
"""
```

## 2. 经典会话推荐模型

### 2.1 Item-KNN

```python
class ItemKNN:
    """
    Item-KNN 基线

    基于物品共现关系推荐
    """

    def __init__(self, k: int = 100):
        self.k = k
        self.co_occurrence = {}  # 物品共现矩阵
        self.item_norms = {}     # 物品出现次数

    def fit(self, sessions: List[List[int]]):
        """
        训练

        sessions: 会话列表，每个会话是物品序列
        """
        for session in sessions:
            items = set(session)

            for item in items:
                self.item_norms[item] = self.item_norms.get(item, 0) + 1

            for item1 in items:
                for item2 in items:
                    if item1 != item2:
                        if item1 not in self.co_occurrence:
                            self.co_occurrence[item1] = {}
                        self.co_occurrence[item1][item2] = (
                            self.co_occurrence[item1].get(item2, 0) + 1
                        )

    def predict(self, session: List[int], n: int = 20) -> List[Tuple[int, float]]:
        """
        推荐
        """
        # 会话中的唯一物品
        session_items = set(session)

        # 计算得分
        scores = {}
        for item in session_items:
            if item in self.co_occurrence:
                for neighbor, count in self.co_occurrence[item].items():
                    if neighbor not in session_items:
                        # 余弦相似度
                        sim = count / np.sqrt(
                            self.item_norms[item] * self.item_norms[neighbor]
                        )
                        scores[neighbor] = scores.get(neighbor, 0) + sim

        # 排序
        sorted_items = sorted(scores.items(), key=lambda x: x[1], reverse=True)

        return sorted_items[:n]
```

### 2.2 GRU4Rec

```python
class GRU4Rec(nn.Module):
    """
    GRU4Rec: 会话推荐经典模型

    使用 GRU 建模会话序列
    """

    def __init__(self,
                 n_items: int,
                 embed_dim: int = 128,
                 hidden_dim: int = 256,
                 n_layers: int = 1,
                 dropout: float = 0.2):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim
        self.hidden_dim = hidden_dim

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items + 1, embed_dim, padding_idx=0)

        # GRU
        self.gru = nn.GRU(
            input_size=embed_dim,
            hidden_size=hidden_dim,
            num_layers=n_layers,
            batch_first=True,
            dropout=dropout if n_layers > 1 else 0
        )

        # 输出层
        self.output_layer = nn.Sequential(
            nn.Linear(hidden_dim, embed_dim),
            nn.Tanh()
        )

        # 候选物品嵌入 (与输入共享)
        # 使用矩阵乘法计算得分

    def forward(self, item_seq: torch.Tensor,
                seq_mask: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播

        参数:
            item_seq: (batch, seq_len) 会话物品序列
            seq_mask: (batch, seq_len) 有效位置掩码

        返回:
            (batch, n_items) 各物品得分
        """
        # 嵌入
        x = self.item_embedding(item_seq)  # (batch, seq_len, embed_dim)

        # GRU
        output, hidden = self.gru(x)  # output: (batch, seq_len, hidden_dim)

        # 取最后状态
        if seq_mask is not None:
            # 获取每个序列的实际最后位置
            lengths = seq_mask.sum(dim=1).long() - 1
            batch_size = item_seq.size(0)
            last_output = output[torch.arange(batch_size), lengths]
        else:
            last_output = output[:, -1, :]

        # 输出
        session_repr = self.output_layer(last_output)  # (batch, embed_dim)

        # 计算与所有物品的得分
        all_item_embeds = self.item_embedding.weight[1:]  # 排除 padding
        scores = torch.matmul(session_repr, all_item_embeds.T)  # (batch, n_items)

        return scores

    def predict(self, item_seq: torch.Tensor,
                n: int = 20) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        预测 Top-N
        """
        scores = self.forward(item_seq)

        # 排除已在会话中的物品
        for i, seq in enumerate(item_seq):
            for item in seq:
                if item > 0:
                    scores[i, item - 1] = float('-inf')

        # Top-N
        top_scores, top_indices = torch.topk(scores, n, dim=-1)

        return top_indices, top_scores
```

## 3. 图神经网络会话推荐

### 3.1 SR-GNN

```python
class SRGNN(nn.Module):
    """
    SR-GNN: Session-based Recommendation with Graph Neural Networks

    将会话建模为图，使用 GNN 学习物品表示
    """

    def __init__(self,
                 n_items: int,
                 embed_dim: int = 128,
                 n_layers: int = 1,
                 dropout: float = 0.1):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items + 1, embed_dim, padding_idx=0)

        # GNN 层
        self.gnn_layers = nn.ModuleList([
            GNNLayer(embed_dim, dropout) for _ in range(n_layers)
        ])

        # 会话表示
        self.session_attention = nn.Sequential(
            nn.Linear(embed_dim * 2, embed_dim),
            nn.Sigmoid()
        )

        self.session_transform = nn.Linear(embed_dim * 2, embed_dim)

    def forward(self, item_seq: torch.Tensor,
                adj_matrix: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        参数:
            item_seq: (batch, seq_len) 会话序列
            adj_matrix: (batch, seq_len, seq_len) 邻接矩阵

        返回:
            (batch, n_items) 物品得分
        """
        batch_size, seq_len = item_seq.shape

        # 物品嵌入
        item_embeds = self.item_embedding(item_seq)  # (batch, seq_len, embed_dim)

        # GNN
        hidden = item_embeds
        for gnn_layer in self.gnn_layers:
            hidden = gnn_layer(hidden, adj_matrix)

        # 会话表示
        # 最后一个物品
        last_item = hidden[:, -1, :]

        # 注意力加权
        last_item_expand = last_item.unsqueeze(1).expand(-1, seq_len, -1)
        attention_input = torch.cat([hidden, last_item_expand], dim=-1)
        attention_weights = self.session_attention(attention_input)

        session_repr = (hidden * attention_weights).sum(dim=1)
        session_repr = self.session_transform(
            torch.cat([session_repr, last_item], dim=-1)
        )

        # 计算得分
        all_item_embeds = self.item_embedding.weight[1:]
        scores = torch.matmul(session_repr, all_item_embeds.T)

        return scores


class GNNLayer(nn.Module):
    """
    GNN 层
    """

    def __init__(self, embed_dim: int, dropout: float = 0.1):
        super().__init__()

        self.linear_in = nn.Linear(embed_dim, embed_dim)
        self.linear_out = nn.Linear(embed_dim, embed_dim)
        self.dropout = nn.Dropout(dropout)

    def forward(self, hidden: torch.Tensor,
                adj_matrix: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        hidden: (batch, seq_len, embed_dim)
        adj_matrix: (batch, seq_len, seq_len) 归一化邻接矩阵
        """
        # 消息传递
        message = self.linear_in(hidden)
        message = torch.matmul(adj_matrix, message)

        # 更新
        output = self.linear_out(message)
        output = F.relu(output)
        output = self.dropout(output)

        # 残差
        return hidden + output


def build_session_graph(item_seq: torch.Tensor,
                        n_items: int) -> torch.Tensor:
    """
    构建会话图

    将会话序列转换为邻接矩阵
    """
    batch_size, seq_len = item_seq.shape

    # 创建邻接矩阵
    adj = torch.zeros(batch_size, seq_len, seq_len, device=item_seq.device)

    # 添加边 (i -> i+1)
    for i in range(seq_len - 1):
        adj[:, i, i + 1] = 1

    # 入度归一化
    degree = adj.sum(dim=-1, keepdim=True).clamp(min=1)
    adj = adj / degree

    return adj
```

## 4. 注意力会话推荐

### 4.1 NARM

```python
class NARM(nn.Module):
    """
    NARM: Neural Attentive Session-based Recommendation

    结合全局和局部会话表示
    """

    def __init__(self,
                 n_items: int,
                 embed_dim: int = 128,
                 hidden_dim: int = 256,
                 n_layers: int = 1,
                 dropout: float = 0.2):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim
        self.hidden_dim = hidden_dim

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items + 1, embed_dim, padding_idx=0)

        # GRU
        self.gru = nn.GRU(
            input_size=embed_dim,
            hidden_size=hidden_dim,
            num_layers=n_layers,
            batch_first=True,
            dropout=dropout if n_layers > 1 else 0
        )

        # 注意力 (局部表示)
        self.attention = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.Tanh()
        )
        self.attention_weight = nn.Linear(hidden_dim, 1)

        # 全局表示
        self.global_transform = nn.Linear(hidden_dim, hidden_dim)

        # 局部表示
        self.local_transform = nn.Linear(hidden_dim, hidden_dim)

        # 输出
        self.output_layer = nn.Linear(hidden_dim * 2, embed_dim)

    def forward(self, item_seq: torch.Tensor,
                seq_mask: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播
        """
        batch_size, seq_len = item_seq.shape

        # 嵌入
        x = self.item_embedding(item_seq)

        # GRU
        gru_output, hidden = self.gru(x)  # gru_output: (batch, seq_len, hidden_dim)

        # 全局表示 (最后隐藏状态)
        global_repr = hidden[-1]  # (batch, hidden_dim)
        global_repr = self.global_transform(global_repr)

        # 局部表示 (注意力加权)
        attention_scores = self.attention(gru_output)  # (batch, seq_len, hidden_dim)
        attention_scores = self.attention_weight(attention_scores).squeeze(-1)  # (batch, seq_len)

        if seq_mask is not None:
            attention_scores = attention_scores.masked_fill(seq_mask == 0, float('-inf'))

        attention_weights = F.softmax(attention_scores, dim=-1)

        local_repr = torch.matmul(
            attention_weights.unsqueeze(1),  # (batch, 1, seq_len)
            gru_output  # (batch, seq_len, hidden_dim)
        ).squeeze(1)  # (batch, hidden_dim)

        local_repr = self.local_transform(local_repr)

        # 组合
        combined = torch.cat([global_repr, local_repr], dim=-1)
        session_repr = self.output_layer(combined)

        # 计算得分
        all_item_embeds = self.item_embedding.weight[1:]
        scores = torch.matmul(session_repr, all_item_embeds.T)

        return scores
```

## 5. Transformer 会话推荐

### 5.1 BERT4Rec for Session

```python
class BERT4RecSession(nn.Module):
    """
    BERT4Rec 用于会话推荐

    使用双向 Transformer 建模会话
    """

    def __init__(self,
                 n_items: int,
                 embed_dim: int = 128,
                 n_heads: int = 4,
                 n_layers: int = 2,
                 max_seq_len: int = 50,
                 dropout: float = 0.2,
                 mask_ratio: float = 0.2):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim
        self.mask_ratio = mask_ratio

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items + 2, embed_dim)  # +2 for padding and mask

        # 位置嵌入
        self.position_embedding = nn.Embedding(max_seq_len, embed_dim)

        # Transformer
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim,
            nhead=n_heads,
            dim_feedforward=embed_dim * 4,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)

        # 输出层
        self.output_layer = nn.Linear(embed_dim, n_items + 1)

        self.dropout = nn.Dropout(dropout)
        self.norm = nn.LayerNorm(embed_dim)

    def forward(self, item_seq: torch.Tensor,
                masked_seq: torch.Tensor = None,
                mask_positions: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播
        """
        batch_size, seq_len = item_seq.shape

        if masked_seq is None:
            masked_seq = item_seq

        # 嵌入
        x = self.item_embedding(masked_seq)
        positions = torch.arange(seq_len, device=item_seq.device)
        x = x + self.position_embedding(positions)

        x = self.norm(x)
        x = self.dropout(x)

        # Transformer
        x = self.transformer(x)

        # 输出
        if mask_positions is not None:
            # 只输出 mask 位置的预测
            mask_output = x[torch.arange(batch_size), mask_positions]
            logits = self.output_layer(mask_output)
        else:
            # 输出所有位置
            logits = self.output_layer(x)

        return logits

    def mask_sequence(self, item_seq: torch.Tensor
                     ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        随机 mask 序列
        """
        batch_size, seq_len = item_seq.shape

        # 生成 mask
        mask = torch.rand(batch_size, seq_len, device=item_seq.device) < self.mask_ratio

        # 不 mask padding
        mask = mask & (item_seq > 0)

        # 创建 mask 序列
        masked_seq = item_seq.clone()
        masked_seq[mask] = self.n_items + 1  # mask token

        # 记录 mask 位置
        mask_positions = mask.long().argmax(dim=-1)

        return masked_seq, mask_positions

    def recommend(self, item_seq: torch.Tensor, n: int = 20
                 ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        推荐

        预测会话的下一个物品
        """
        batch_size, seq_len = item_seq.shape

        # 在末尾添加 mask token
        if seq_len < item_seq.size(1):
            input_seq = item_seq.clone()
            input_seq[:, -1] = self.n_items + 1
        else:
            input_seq = torch.cat([
                item_seq,
                torch.full((batch_size, 1), self.n_items + 1,
                          dtype=torch.long, device=item_seq.device)
            ], dim=1)

        # 预测
        logits = self.forward(input_seq)

        # 取最后一个位置
        last_logits = logits[:, -1, :self.n_items]

        # 排除已有物品
        for i, seq in enumerate(item_seq):
            for item in seq:
                if 0 < item <= self.n_items:
                    last_logits[i, item - 1] = float('-inf')

        # Top-N
        top_scores, top_indices = torch.topk(last_logits, n, dim=-1)

        return top_indices, top_scores
```

## 6. 会话推荐训练与评估

### 6.1 训练器

```python
class SessionRecTrainer:
    """
    会话推荐训练器
    """

    def __init__(self, model: nn.Module, lr: float = 0.001):
        self.model = model
        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    def train_epoch(self, dataloader):
        """训练一个 epoch"""
        self.model.train()
        total_loss = 0

        for batch in dataloader:
            item_seq = batch['item_seq']
            target = batch['target']

            self.optimizer.zero_grad()

            # 前向传播
            scores = self.model(item_seq)

            # 交叉熵损失
            loss = F.cross_entropy(scores, target)

            # 反向传播
            loss.backward()
            self.optimizer.step()

            total_loss += loss.item()

        return total_loss / len(dataloader)

    def evaluate(self, dataloader, k_list: List[int] = [5, 10, 20]):
        """评估"""
        self.model.eval()

        metrics = {f'Recall@{k}': [] for k in k_list}
        metrics.update({f'MRR@{k}': [] for k in k_list})

        with torch.no_grad():
            for batch in dataloader:
                item_seq = batch['item_seq']
                target = batch['target']

                # 预测
                scores = self.model(item_seq)

                # 排除已有物品
                for i, seq in enumerate(item_seq):
                    for item in seq:
                        if item > 0:
                            scores[i, item - 1] = float('-inf')

                # 计算指标
                for k in k_list:
                    _, topk = torch.topk(scores, k, dim=-1)

                    # Recall
                    hit = (topk == target.unsqueeze(1)).any(dim=1).float()
                    metrics[f'Recall@{k}'].extend(hit.tolist())

                    # MRR
                    rank = (topk == target.unsqueeze(1)).float().argmax(dim=1).float()
                    rank = torch.where(hit > 0, 1.0 / (rank + 1), torch.zeros_like(rank))
                    metrics[f'MRR@{k}'].extend(rank.tolist())

        # 平均
        for key in metrics:
            metrics[key] = np.mean(metrics[key])

        return metrics
```

## 7. 学习总结

### 7.1 模型对比

```
模型        特点                    适用场景
────────────────────────────────────────────────
Item-KNN    简单快速                基线
GRU4Rec     序列建模                通用会话
NARM        注意力 + 全局/局部      长会话
SR-GNN      图结构                  复杂依赖
BERT4Rec    双向建模                预训练
```

### 7.2 最佳实践

```
1. 数据处理: 合理切分会话
2. 负采样: 使用 mini-batch 负采样
3. 评估: 使用 Recall@K 和 MRR@K
4. 冷启动: 会话推荐天然适合冷启动
5. 实时性: 模型需要快速推理
```
