# 推荐系统实时更新 学习文档

## 1. 实时更新概述

### 1.1 为什么需要实时更新？

```
用户兴趣变化:
- 短期兴趣: 当前会话、最近行为
- 长期兴趣: 累积偏好
- 上下文变化: 时间、地点、设备

离线更新的局限:
- 更新延迟: 数小时到数天
- 冷启动: 新用户、新物品
- 时效性: 错过热点、促销
```

### 1.2 实时更新层次

```
┌─────────────────────────────────────────┐
│  在线学习 (Online Learning)              │
│  - 实时样本实时更新模型                   │
│  - 毫秒级响应                            │
├─────────────────────────────────────────┤
│  近线学习 (Nearline Learning)            │
│  - 准实时处理                            │
│  - 秒到分钟级                            │
├─────────────────────────────────────────┤
│  离线学习 (Offline Learning)             │
│  - 批量训练                              │
│  - 小时到天级                            │
└─────────────────────────────────────────┘
```

## 2. 在线学习

### 2.1 FTRL (Follow-The-Regularized-Leader)

```python
import numpy as np
from typing import Dict, List, Tuple
from collections import defaultdict


class FTRLProximal:
    """
    FTRL-Proximal 算法

    Google 提出的在线学习算法，广泛用于 CTR 预估

    特点:
    - L1 正则化产生稀疏解
    - 每维特征独立学习率
    - 适合高维稀疏特征
    """

    def __init__(self, alpha: float = 0.1, beta: float = 1.0,
                 l1: float = 1.0, l2: float = 1.0):
        """
        参数:
            alpha: 学习率缩放因子
            beta: 学习率平滑因子
            l1: L1 正则化系数
            l2: L2 正则化系数
        """
        self.alpha = alpha
        self.beta = beta
        self.l1 = l1
        self.l2 = l2

        # 累积梯度平方和
        self.z = defaultdict(float)  # z_i
        self.n = defaultdict(float)  # n_i = Σg_i²

        # 当前权重缓存
        self.w = defaultdict(float)

    def _get_weight(self, feature: str) -> float:
        """
        计算特征权重

        w_i = {
            0,                              if |z_i| ≤ λ1
            -(z_i - sign(z_i)*λ1) / ((β+√n_i)/α + λ2)
        }
        """
        z_i = self.z[feature]
        n_i = self.n[feature]

        if abs(z_i) <= self.l1:
            return 0.0

        sign = 1 if z_i > 0 else -1
        denominator = (self.beta + np.sqrt(n_i)) / self.alpha + self.l2
        return -sign * (abs(z_i) - self.l1) / denominator

    def predict(self, features: Dict[str, float]) -> float:
        """
        预测

        参数:
            features: {feature_name: feature_value}
        """
        # 更新权重缓存并计算内积
        score = 0.0
        for feat, val in features.items():
            self.w[feat] = self._get_weight(feat)
            score += self.w[feat] * val

        # Sigmoid
        return 1.0 / (1.0 + np.exp(-np.clip(score, -10, 10)))

    def update(self, features: Dict[str, float], label: int):
        """
        在线更新

        参数:
            features: 特征字典
            label: 标签 (0 or 1)
        """
        # 预测
        pred = self.predict(features)

        # 梯度
        g = pred - label

        # 更新每个特征
        for feat, val in features.items():
            old_n = self.n[feat]
            old_w = self.w[feat]

            # 更新 n
            self.n[feat] += g * g * val * val

            # 更新 z
            sigma = (np.sqrt(self.n[feat]) - np.sqrt(old_n)) / self.alpha
            self.z[feat] += g * val - sigma * old_w

    def get_weights(self) -> Dict[str, float]:
        """获取所有非零权重"""
        return {k: v for k, v in self.w.items() if v != 0}


class FTRLWithFactorization:
    """
    带 FM 特征交叉的 FTRL
    """

    def __init__(self, n_features: int, embed_dim: int = 10,
                 alpha: float = 0.1, beta: float = 1.0):
        self.n_features = n_features
        self.embed_dim = embed_dim
        self.alpha = alpha
        self.beta = beta

        # 线性部分
        self.z_linear = np.zeros(n_features)
        self.n_linear = np.zeros(n_features)
        self.w_linear = np.zeros(n_features)

        # 交叉部分（嵌入）
        self.z_fm = np.zeros((n_features, embed_dim))
        self.n_fm = np.zeros((n_features, embed_dim))
        self.v = np.zeros((n_features, embed_dim))

    def predict(self, feature_indices: List[int]) -> Tuple[float, np.ndarray]:
        """
        预测

        返回:
            prediction: 预测值
            sum_vx: 嵌入的加权和（用于梯度计算）
        """
        # 线性部分
        linear = np.sum(self.w_linear[feature_indices])

        # FM 部分
        sum_vx = np.sum(self.v[feature_indices], axis=0)
        fm = 0.5 * np.sum(sum_vx ** 2 - np.sum(self.v[feature_indices] ** 2, axis=0))

        score = linear + fm
        pred = 1.0 / (1.0 + np.exp(-np.clip(score, -10, 10)))

        return pred, sum_vx

    def update(self, feature_indices: List[int], label: int):
        """更新"""
        pred, sum_vx = self.predict(feature_indices)
        g = pred - label

        # 更新线性部分
        for i in feature_indices:
            old_n = self.n_linear[i]
            self.n_linear[i] += g ** 2

            sigma = (np.sqrt(self.n_linear[i]) - np.sqrt(old_n)) / self.alpha
            self.z_linear[i] += g - sigma * self.w_linear[i]

            # 更新权重
            if abs(self.z_linear[i]) <= 1.0:  # l1
                self.w_linear[i] = 0
            else:
                sign = 1 if self.z_linear[i] > 0 else -1
                self.w_linear[i] = -sign * (abs(self.z_linear[i]) - 1.0) / \
                                   ((self.beta + np.sqrt(self.n_linear[i])) / self.alpha + 1.0)

        # 更新 FM 部分
        for i in feature_indices:
            for f in range(self.embed_dim):
                g_f = g * (sum_vx[f] - self.v[i, f])

                old_n = self.n_fm[i, f]
                self.n_fm[i, f] += g_f ** 2

                sigma = (np.sqrt(self.n_fm[i, f]) - np.sqrt(old_n)) / self.alpha
                self.z_fm[i, f] += g_f - sigma * self.v[i, f]

                self.v[i, f] = -self.z_fm[i, f] / \
                              ((self.beta + np.sqrt(self.n_fm[i, f])) / self.alpha + 1.0)
```

### 2.2 SGD 在线更新

```python
class OnlineSGD:
    """
    在线 SGD

    简单但有效的在线学习
    """

    def __init__(self, n_features: int, learning_rate: float = 0.01,
                 l2_reg: float = 0.001):
        self.weights = np.zeros(n_features)
        self.lr = learning_rate
        self.l2 = l2_reg

    def predict(self, features: np.ndarray) -> float:
        """预测"""
        score = np.dot(features, self.weights)
        return 1.0 / (1.0 + np.exp(-np.clip(score, -10, 10)))

    def update(self, features: np.ndarray, label: int):
        """更新"""
        pred = self.predict(features)
        error = pred - label

        # L2 正则化
        grad = error * features + self.l2 * self.weights

        self.weights -= self.lr * grad


class AdaptiveSGD:
    """
    自适应学习率 SGD (AdaGrad)
    """

    def __init__(self, n_features: int, learning_rate: float = 0.01,
                 epsilon: float = 1e-8):
        self.weights = np.zeros(n_features)
        self.lr = learning_rate
        self.epsilon = epsilon

        # 累积梯度平方和
        self.accumulated_grad_sq = np.zeros(n_features)

    def predict(self, features: np.ndarray) -> float:
        """预测"""
        score = np.dot(features, self.weights)
        return 1.0 / (1.0 + np.exp(-np.clip(score, -10, 10)))

    def update(self, features: np.ndarray, label: int):
        """更新"""
        pred = self.predict(features)
        grad = (pred - label) * features

        # 更新累积梯度
        self.accumulated_grad_sq += grad ** 2

        # 自适应学习率
        adaptive_lr = self.lr / (np.sqrt(self.accumulated_grad_sq) + self.epsilon)

        self.weights -= adaptive_lr * grad


class AdamOnline:
    """
    Adam 在线学习
    """

    def __init__(self, n_features: int, lr: float = 0.001,
                 beta1: float = 0.9, beta2: float = 0.999,
                 epsilon: float = 1e-8):
        self.weights = np.zeros(n_features)
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.epsilon = epsilon

        # 一阶矩和二阶矩
        self.m = np.zeros(n_features)  # 动量
        self.v = np.zeros(n_features)  # 速度
        self.t = 0

    def predict(self, features: np.ndarray) -> float:
        """预测"""
        score = np.dot(features, self.weights)
        return 1.0 / (1.0 + np.exp(-np.clip(score, -10, 10)))

    def update(self, features: np.ndarray, label: int):
        """更新"""
        self.t += 1

        pred = self.predict(features)
        grad = (pred - label) * features

        # 更新矩
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)

        # 偏差修正
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)

        # 更新权重
        self.weights -= self.lr * m_hat / (np.sqrt(v_hat) + self.epsilon)
```

## 3. 近线更新

### 3.1 流式处理架构

```python
from collections import deque
import threading
import time


class StreamProcessor:
    """
    流式数据处理
    """

    def __init__(self, batch_size: int = 1000, flush_interval: float = 5.0):
        self.batch_size = batch_size
        self.flush_interval = flush_interval

        self.buffer = deque()
        self.buffer_lock = threading.Lock()

        self.last_flush = time.time()
        self.processors = []

    def add_processor(self, processor):
        """添加处理器"""
        self.processors.append(processor)

    def push(self, event: Dict):
        """推送事件"""
        with self.buffer_lock:
            self.buffer.append(event)

            # 达到批次大小或超时则刷新
            if (len(self.buffer) >= self.batch_size or
                time.time() - self.last_flush >= self.flush_interval):
                self._flush()

    def _flush(self):
        """刷新缓冲区"""
        if not self.buffer:
            return

        batch = []
        while self.buffer:
            batch.append(self.buffer.popleft())

        self.last_flush = time.time()

        # 处理批次
        for processor in self.processors:
            processor.process_batch(batch)


class NearlineUpdater:
    """
    近线更新器

    处理实时行为数据，更新模型和特征
    """

    def __init__(self, update_interval: int = 60):
        self.update_interval = update_interval

        # 行为累积
        self.user_behaviors = defaultdict(list)
        self.item_behaviors = defaultdict(list)

        # 特征存储
        self.user_features = {}
        self.item_features = {}

        # 模型
        self.model = None

    def process_event(self, event: Dict):
        """
        处理单个事件

        事件格式:
        {
            'user_id': int,
            'item_id': int,
            'action': str,  # click, purchase, etc.
            'timestamp': int,
            'features': dict
        }
        """
        user_id = event['user_id']
        item_id = event['item_id']
        action = event['action']

        # 记录行为
        self.user_behaviors[user_id].append({
            'item_id': item_id,
            'action': action,
            'timestamp': event['timestamp']
        })

        self.item_behaviors[item_id].append({
            'user_id': user_id,
            'action': action,
            'timestamp': event['timestamp']
        })

    def process_batch(self, events: List[Dict]):
        """批量处理"""
        for event in events:
            self.process_event(event)

    def update_user_features(self):
        """更新用户特征"""
        for user_id, behaviors in self.user_behaviors.items():
            # 计算实时特征
            clicks = sum(1 for b in behaviors if b['action'] == 'click')
            purchases = sum(1 for b in behaviors if b['action'] == 'purchase')

            self.user_features[user_id] = {
                'recent_clicks': clicks,
                'recent_purchases': purchases,
                'last_active': max(b['timestamp'] for b in behaviors)
            }

        # 清空行为缓存
        self.user_behaviors.clear()

    def update_item_features(self):
        """更新物品特征"""
        for item_id, behaviors in self.item_behaviors.items():
            clicks = sum(1 for b in behaviors if b['action'] == 'click')
            purchases = sum(1 for b in behaviors if b['action'] == 'purchase')

            self.item_features[item_id] = {
                'recent_clicks': clicks,
                'recent_purchases': purchases,
                'last_exposed': max(b['timestamp'] for b in behaviors)
            }

        self.item_behaviors.clear()


class IncrementalEmbeddingUpdater:
    """
    增量嵌入更新
    """

    def __init__(self, user_embeddings: np.ndarray,
                 item_embeddings: np.ndarray,
                 learning_rate: float = 0.01):
        self.user_embeddings = user_embeddings
        self.item_embeddings = item_embeddings
        self.lr = learning_rate

    def update_from_interaction(self, user_id: int, item_id: int,
                                label: float):
        """
        从交互更新嵌入

        类似于隐式反馈的增量 SVD
        """
        user_vec = self.user_embeddings[user_id]
        item_vec = self.item_embeddings[item_id]

        # 预测
        pred = np.dot(user_vec, item_vec)
        error = label - pred

        # 更新
        user_grad = -error * item_vec
        item_grad = -error * user_vec

        self.user_embeddings[user_id] -= self.lr * user_grad
        self.item_embeddings[item_id] -= self.lr * item_grad

    def update_from_sequence(self, user_id: int, item_sequence: List[int]):
        """
        从行为序列更新

        类似于 Word2Vec 的增量更新
        """
        window = 3

        for i, center in enumerate(item_sequence):
            start = max(0, i - window)
            end = min(len(item_sequence), i + window + 1)

            for j in range(start, end):
                if j != i:
                    context = item_sequence[j]

                    # Skip-gram 风格更新
                    user_vec = self.user_embeddings[user_id]
                    center_vec = self.item_embeddings[center]
                    context_vec = self.item_embeddings[context]

                    score = np.dot(center_vec, context_vec)
                    sigmoid = 1 / (1 + np.exp(-np.clip(score, -10, 10)))

                    grad = (sigmoid - 1)  # 正样本

                    self.item_embeddings[center] -= self.lr * grad * context_vec
                    self.item_embeddings[context] -= self.lr * grad * center_vec
```

```python
class RealtimeFeatureStore:
    """
    实时特征存储
    """

    def __init__(self):
        # 使用 Redis 的接口模拟
        self.user_features = {}
        self.item_features = {}
        self.context_features = {}

        # TTL 设置
        self.user_feature_ttl = 3600  # 1小时
        self.item_feature_ttl = 86400  # 24小时

    def get_user_features(self, user_id: int) -> Dict:
        """获取用户特征"""
        return self.user_features.get(user_id, {})

    def set_user_features(self, user_id: int, features: Dict):
        """设置用户特征"""
        self.user_features[user_id] = features

    def update_user_realtime_features(self, user_id: int, event: Dict):
        """
        更新用户实时特征
        """
        if user_id not in self.user_features:
            self.user_features[user_id] = {}

        user_feat = self.user_features[user_id]

        # 更新最近行为
        if 'recent_items' not in user_feat:
            user_feat['recent_items'] = []

        user_feat['recent_items'].append(event['item_id'])
        user_feat['recent_items'] = user_feat['recent_items'][-50:]  # 保留最近50个

        # 更新行为计数
        action = event['action']
        key = f'recent_{action}_count'
        user_feat[key] = user_feat.get(key, 0) + 1

        # 更新最后活跃时间
        user_feat['last_active_time'] = event['timestamp']

    def get_item_features(self, item_id: int) -> Dict:
        """获取物品特征"""
        return self.item_features.get(item_id, {})

    def update_item_realtime_features(self, item_id: int, event: Dict):
        """更新物品实时特征"""
        if item_id not in self.item_features:
            self.item_features[item_id] = {}

        item_feat = self.item_features[item_id]

        # 更新实时统计
        action = event['action']
        key = f'recent_{action}_count'
        item_feat[key] = item_feat.get(key, 0) + 1

        # 更新最后曝光时间
        item_feat['last_expose_time'] = event['timestamp']


class FeatureAggregator:
    """
    特征聚合器

    聚合多时间窗口的特征
    """

    def __init__(self, windows: List[int] = [300, 3600, 86400]):
        """
        参数:
            windows: 时间窗口（秒）[5分钟, 1小时, 24小时]
        """
        self.windows = windows
        self.event_buffer = defaultdict(list)  # user_id -> events

    def add_event(self, user_id: int, event: Dict):
        """添加事件"""
        self.event_buffer[user_id].append(event)

    def aggregate_features(self, user_id: int, current_time: int) -> Dict:
        """聚合特征"""
        events = self.event_buffer.get(user_id, [])

        features = {}

        for window in self.windows:
            # 过滤时间窗口内的事件
            window_events = [e for e in events
                           if current_time - e['timestamp'] <= window]

            window_name = self._window_name(window)

            # 统计特征
            features[f'{window_name}_clicks'] = sum(
                1 for e in window_events if e['action'] == 'click'
            )
            features[f'{window_name}_purchases'] = sum(
                1 for e in window_events if e['action'] == 'purchase'
            )
            features[f'{window_name}_unique_items'] = len(
                set(e['item_id'] for e in window_events)
            )

        return features

    @staticmethod
    def _window_name(seconds: int) -> str:
        """窗口名称"""
        if seconds < 60:
            return f'{seconds}s'
        elif seconds < 3600:
            return f'{seconds // 60}m'
        else:
            return f'{seconds // 3600}h'

    def cleanup(self, current_time: int, max_age: int = 86400):
        """清理过期事件"""
        for user_id in list(self.event_buffer.keys()):
            self.event_buffer[user_id] = [
                e for e in self.event_buffer[user_id]
                if current_time - e['timestamp'] <= max_age
            ]
```

### 5.1 完整架构

```python
class RealtimeUpdateSystem:
    """
    实时更新系统
    """

    def __init__(self):
        # 流处理器
        self.stream_processor = StreamProcessor(batch_size=100, flush_interval=1.0)

        # 近线更新器
        self.nearline_updater = NearlineUpdater()

        # 特征存储
        self.feature_store = RealtimeFeatureStore()

        # 在线模型
        self.online_model = FTRLProximal()

        # 注册处理器
        self.stream_processor.add_processor(self)

    def process_batch(self, events: List[Dict]):
        """处理事件批次"""
        for event in events:
            # 1. 更新实时特征
            self.feature_store.update_user_realtime_features(
                event['user_id'], event
            )
            self.feature_store.update_item_realtime_features(
                event['item_id'], event
            )

            # 2. 在线模型更新
            features = self._extract_features(event)
            label = 1 if event['action'] == 'click' else 0
            self.online_model.update(features, label)

            # 3. 记录近线更新
            self.nearline_updater.process_event(event)

    def _extract_features(self, event: Dict) -> Dict[str, float]:
        """提取特征"""
        features = {}

        # 用户特征
        user_feat = self.feature_store.get_user_features(event['user_id'])
        for k, v in user_feat.items():
            if isinstance(v, (int, float)):
                features[f'user_{k}'] = float(v)

        # 物品特征
        item_feat = self.feature_store.get_item_features(event['item_id'])
        for k, v in item_feat.items():
            if isinstance(v, (int, float)):
                features[f'item_{k}'] = float(v)

        # 上下文特征
        features['hour'] = float(event['timestamp'] % 86400 // 3600)

        return features

    def predict(self, user_id: int, item_id: int) -> float:
        """预测"""
        features = {}

        user_feat = self.feature_store.get_user_features(user_id)
        for k, v in user_feat.items():
            if isinstance(v, (int, float)):
                features[f'user_{k}'] = float(v)

        item_feat = self.feature_store.get_item_features(item_id)
        for k, v in item_feat.items():
            if isinstance(v, (int, float)):
                features[f'item_{k}'] = float(v)

        return self.online_model.predict(features)

    def push_event(self, event: Dict):
        """推送事件"""
        self.stream_processor.push(event)


class RealtimeRecaller:
    """
    实时召回器

    基于用户实时行为的召回
    """

    def __init__(self, feature_store: RealtimeFeatureStore):
        self.feature_store = feature_store
        self.item_similarity = {}  # 物品相似度矩阵

    def recall(self, user_id: int, n: int = 100) -> List[Tuple[int, float]]:
        """
        实时召回

        基于用户最近行为的相似物品
        """
        user_feat = self.feature_store.get_user_features(user_id)
        recent_items = user_feat.get('recent_items', [])

        scores = defaultdict(float)
        for item_id in recent_items[-10:]:  # 最近10个
            if item_id in self.item_similarity:
                for sim_item, sim_score in self.item_similarity[item_id]:
                    if sim_item not in recent_items:
                        scores[sim_item] += sim_score

        sorted_items = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return sorted_items[:n]


class RealtimeRanker:
    """
    实时排序器
    """

    def __init__(self, online_model, feature_store: RealtimeFeatureStore):
        self.model = online_model
        self.feature_store = feature_store

    def rank(self, user_id: int, items: List[int]) -> List[Tuple[int, float]]:
        """排序"""
        user_feat = self.feature_store.get_user_features(user_id)

        scores = []
        for item_id in items:
            item_feat = self.feature_store.get_item_features(item_id)

            # 构建特征
            features = {}
            for k, v in user_feat.items():
                if isinstance(v, (int, float)):
                    features[f'user_{k}'] = float(v)
            for k, v in item_feat.items():
                if isinstance(v, (int, float)):
                    features[f'item_{k}'] = float(v)

            score = self.model.predict(features)
            scores.append((item_id, score))

        return sorted(scores, key=lambda x: x[1], reverse=True)
```

## 6. 学习总结

### 6.1 核心要点

1. **在线学习**: FTRL、SGD 等实时更新
2. **近线处理**: 流式处理、增量更新
3. **实时特征**: 滑动窗口、特征聚合
4. **系统架构**: 分层更新、实时召回排序

### 6.2 更新策略

```
更新类型        延迟        方法
──────────────────────────────────
在线学习        <100ms      FTRL、SGD
近线更新        秒~分钟     增量训练
离线更新        小时~天     全量训练
```

### 6.3 最佳实践

```
1. 冷热分离: 在线/离线模型分离
2. A/B测试: 验证实时更新效果
3. 监控告警: 监控模型稳定性
4. 回滚机制: 支持快速回滚
```
