# 推荐系统对比学习 学习文档

## 1. 对比学习基础

### 1.1 什么是对比学习？

```
核心思想:
- 学习表示使得相似样本接近，不相似样本远离
- 自监督学习范式，无需人工标注

关键概念:
- Anchor: 锚点样本
- Positive: 正样本（与锚点相似）
- Negative: 负样本（与锚点不相似）

目标: score(anchor, positive) > score(anchor, negative)
```

### 1.2 推荐系统中的应用

```
1. 用户/物品表示学习
2. 序列推荐中的行为对比
3. 跨域推荐的对齐
4. 数据增强的正样本构造
```

## 2. 对比损失函数

### 2.1 基础对比损失

```python
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from typing import List, Tuple, Dict


class ContrastiveLoss(nn.Module):
    """
    基础对比损失

    L = (1/2) * [y * d² + (1-y) * max(0, margin - d)²]

    y=1 表示正样本对，y=0 表示负样本对
    """

    def __init__(self, margin: float = 1.0):
        super().__init__()
        self.margin = margin

    def forward(self, anchor: torch.Tensor,
               other: torch.Tensor,
               label: torch.Tensor) -> torch.Tensor:
        """
        参数:
            anchor: 锚点表示 (batch, dim)
            other: 对比样本表示 (batch, dim)
            label: 1=正样本, 0=负样本 (batch,)
        """
        # 欧氏距离
        distance = F.pairwise_distance(anchor, other)

        # 对比损失
        loss_pos = label * distance ** 2
        loss_neg = (1 - label) * F.relu(self.margin - distance) ** 2

        return torch.mean(loss_pos + loss_neg) / 2


class TripletLoss(nn.Module):
    """
    三元组损失

    L = max(0, d(anchor, positive) - d(anchor, negative) + margin)
    """

    def __init__(self, margin: float = 1.0):
        super().__init__()
        self.margin = margin

    def forward(self, anchor: torch.Tensor,
               positive: torch.Tensor,
               negative: torch.Tensor) -> torch.Tensor:
        """
        参数:
            anchor: 锚点 (batch, dim)
            positive: 正样本 (batch, dim)
            negative: 负样本 (batch, dim)
        """
        pos_distance = F.pairwise_distance(anchor, positive)
        neg_distance = F.pairwise_distance(anchor, negative)

        loss = F.relu(pos_distance - neg_distance + self.margin)

        return torch.mean(loss)


class InfoNCELoss(nn.Module):
    """
    InfoNCE 损失

    L = -log(exp(sim(a, p)/τ) / Σ exp(sim(a, n)/τ))

    τ 是温度参数
    """

    def __init__(self, temperature: float = 0.1):
        super().__init__()
        self.temperature = temperature

    def forward(self, anchor: torch.Tensor,
               positive: torch.Tensor,
               negatives: torch.Tensor) -> torch.Tensor:
        """
        参数:
            anchor: 锚点 (batch, dim)
            positive: 正样本 (batch, dim)
            negatives: 负样本 (batch, n_neg, dim)
        """
        batch_size = anchor.size(0)

        # 正样本相似度
        pos_sim = F.cosine_similarity(anchor, positive) / self.temperature

        # 负样本相似度
        neg_sims = []
        for i in range(negatives.size(1)):
            neg_sim = F.cosine_similarity(anchor, negatives[:, i, :])
            neg_sims.append(neg_sim)
        neg_sims = torch.stack(neg_sims, dim=1) / self.temperature

        # 分母：正样本 + 负样本
        all_sims = torch.cat([pos_sim.unsqueeze(1), neg_sims], dim=1)

        # 交叉熵（正样本在第0位）
        labels = torch.zeros(batch_size, dtype=torch.long, device=anchor.device)

        return F.cross_entropy(all_sims, labels)


class NTXentLoss(nn.Module):
    """
    NT-Xent 损失 (SimCLR 使用)

    批内对比，每个样本作为其他样本的负样本
    """

    def __init__(self, temperature: float = 0.5):
        super().__init__()
        self.temperature = temperature

    def forward(self, z_i: torch.Tensor, z_j: torch.Tensor) -> torch.Tensor:
        """
        参数:
            z_i: 第一视图表示 (batch, dim)
            z_j: 第二视图表示 (batch, dim)
        """
        batch_size = z_i.size(0)

        # 拼接
        z = torch.cat([z_i, z_j], dim=0)  # (2*batch, dim)

        # 相似度矩阵
        sim = F.cosine_similarity(z.unsqueeze(1), z.unsqueeze(0), dim=2)

        # 缩放
        sim = sim / self.temperature

        # 创建标签
        # z_i[k] 的正样本是 z_j[k]，反之亦然
        labels = torch.cat([
            torch.arange(batch_size, 2*batch_size),
            torch.arange(0, batch_size)
        ]).to(z.device)

        # 移除对角线（自己与自己的相似度）
        mask = torch.eye(2*batch_size, dtype=torch.bool).to(z.device)
        sim = sim[~mask].view(2*batch_size, -1)

        return F.cross_entropy(sim, labels)
```

```python
class UserItemContrastive(nn.Module):
    """
    用户-物品对比学习
    """

    def __init__(self, n_users: int, n_items: int,
                 embed_dim: int = 64, temperature: float = 0.1):
        super().__init__()

        self.temperature = temperature

        # 嵌入层
        self.user_embedding = nn.Embedding(n_users, embed_dim)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        # 投影头
        self.user_projector = nn.Sequential(
            nn.Linear(embed_dim, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim)
        )
        self.item_projector = nn.Sequential(
            nn.Linear(embed_dim, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim)
        )

        self._init_weights()

    def _init_weights(self):
        nn.init.normal_(self.user_embedding.weight, std=0.01)
        nn.init.normal_(self.item_embedding.weight, std=0.01)

    def encode_user(self, user_ids: torch.Tensor) -> torch.Tensor:
        """编码用户"""
        user_embed = self.user_embedding(user_ids)
        return F.normalize(self.user_projector(user_embed), dim=-1)

    def encode_item(self, item_ids: torch.Tensor) -> torch.Tensor:
        """编码物品"""
        item_embed = self.item_embedding(item_ids)
        return F.normalize(self.item_projector(item_embed), dim=-1)

    def contrastive_loss(self, user_ids: torch.Tensor,
                        pos_item_ids: torch.Tensor,
                        neg_item_ids: torch.Tensor) -> torch.Tensor:
        """
        对比损失

        参数:
            user_ids: 用户ID (batch,)
            pos_item_ids: 正样本物品 (batch,)
            neg_item_ids: 负样本物品 (batch, n_neg)
        """
        user_embed = self.encode_user(user_ids)  # (batch, dim)
        pos_item_embed = self.encode_item(pos_item_ids)  # (batch, dim)
        neg_item_embed = self.encode_item(neg_item_ids.view(-1)).view(
            len(user_ids), -1, user_embed.size(-1)
        )  # (batch, n_neg, dim)

        # 正样本相似度
        pos_sim = F.cosine_similarity(user_embed, pos_item_embed)

        # 负样本相似度
        n_neg = neg_item_embed.size(1)
        neg_sims = []
        for i in range(n_neg):
            neg_sim = F.cosine_similarity(user_embed, neg_item_embed[:, i, :])
            neg_sims.append(neg_sim)
        neg_sims = torch.stack(neg_sims, dim=1)  # (batch, n_neg)

        # InfoNCE
        all_sims = torch.cat([
            pos_sim.unsqueeze(1) / self.temperature,
            neg_sims / self.temperature
        ], dim=1)

        labels = torch.zeros(len(user_ids), dtype=torch.long, device=user_ids.device)

        return F.cross_entropy(all_sims, labels)

    def forward(self, user_ids: torch.Tensor,
               item_ids: torch.Tensor) -> torch.Tensor:
        """预测分数"""
        user_embed = self.encode_user(user_ids)
        item_embed = self.encode_item(item_ids)
        return torch.sum(user_embed * item_embed, dim=-1)


class SequentialContrastive(nn.Module):
    """
    序列推荐对比学习

    对比不同时间步的行为
    """

    def __init__(self, n_items: int, embed_dim: int = 64,
                 n_heads: int = 4, n_layers: int = 2,
                 temperature: float = 0.1):
        super().__init__()

        self.temperature = temperature
        self.embed_dim = embed_dim

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items, embed_dim)
        self.position_embedding = nn.Embedding(50, embed_dim)

        # Transformer 编码器
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim, nhead=n_heads, batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)

        # 投影头
        self.projector = nn.Sequential(
            nn.Linear(embed_dim, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim)
        )

    def encode_sequence(self, item_ids: torch.Tensor,
                       mask: torch.Tensor = None) -> torch.Tensor:
        """编码序列"""
        batch_size, seq_len = item_ids.size()

        # 物品嵌入 + 位置嵌入
        item_embed = self.item_embedding(item_ids)
        pos_ids = torch.arange(seq_len, device=item_ids.device).unsqueeze(0)
        pos_embed = self.position_embedding(pos_ids)

        embed = item_embed + pos_embed

        # Transformer
        if mask is not None:
            embed = embed * mask.unsqueeze(-1)

        encoded = self.transformer(embed)

        # 取最后一个位置
        return encoded[:, -1, :]

    def contrastive_loss(self, seq1: torch.Tensor, seq2: torch.Tensor,
                        neg_seqs: torch.Tensor,
                        mask1: torch.Tensor = None,
                        mask2: torch.Tensor = None) -> torch.Tensor:
        """
        序列对比损失

        参数:
            seq1: 序列视图1 (batch, seq_len)
            seq2: 序列视图2 (batch, seq_len)
            neg_seqs: 负样本序列 (batch, n_neg, seq_len)
        """
        # 编码
        z1 = F.normalize(self.projector(self.encode_sequence(seq1, mask1)), dim=-1)
        z2 = F.normalize(self.projector(self.encode_sequence(seq2, mask2)), dim=-1)

        batch_size = seq1.size(0)
        n_neg = neg_seqs.size(1)

        # 编码负样本
        z_neg = []
        for i in range(n_neg):
            z = F.normalize(
                self.projector(self.encode_sequence(neg_seqs[:, i, :])), dim=-1
            )
            z_neg.append(z)
        z_neg = torch.stack(z_neg, dim=1)  # (batch, n_neg, dim)

        # 正样本相似度
        pos_sim = F.cosine_similarity(z1, z2)

        # 负样本相似度
        neg_sims = []
        for i in range(n_neg):
            neg_sim = F.cosine_similarity(z1, z_neg[:, i, :])
            neg_sims.append(neg_sim)
        neg_sims = torch.stack(neg_sims, dim=1)

        # InfoNCE
        all_sims = torch.cat([
            pos_sim.unsqueeze(1) / self.temperature,
            neg_sims / self.temperature
        ], dim=1)

        labels = torch.zeros(batch_size, dtype=torch.long, device=seq1.device)

        return F.cross_entropy(all_sims, labels)
```

### 3.2 数据增强对比

```python
class SequenceAugmentation:
    """
    序列数据增强

    用于构造对比学习的正样本对
    """

    def __init__(self, n_items: int, aug_ratio: float = 0.1):
        self.n_items = n_items
        self.aug_ratio = aug_ratio

    def item_crop(self, sequence: List[int], crop_ratio: float = 0.2) -> List[int]:
        """
        物品裁剪

        随机删除序列中的一部分物品
        """
        n_crop = int(len(sequence) * crop_ratio)
        if n_crop == 0:
            return sequence

        crop_indices = np.random.choice(
            len(sequence), n_crop, replace=False
        )
        return [s for i, s in enumerate(sequence) if i not in crop_indices]

    def item_mask(self, sequence: List[int], mask_ratio: float = 0.1,
                 mask_token: int = 0) -> List[int]:
        """
        物品遮掩

        随机将部分物品替换为 [MASK]
        """
        n_mask = int(len(sequence) * mask_ratio)
        if n_mask == 0:
            return sequence

        mask_indices = np.random.choice(
            len(sequence), n_mask, replace=False
        )

        aug_seq = sequence.copy()
        for idx in mask_indices:
            aug_seq[idx] = mask_token

        return aug_seq

    def item_reorder(self, sequence: List[int],
                    reorder_ratio: float = 0.1) -> List[int]:
        """
        物品重排

        随机打乱部分物品的顺序
        """
        n_reorder = int(len(sequence) * reorder_ratio)
        if n_reorder < 2:
            return sequence

        reorder_indices = np.random.choice(
            len(sequence), n_reorder, replace=False
        )
        reorder_indices = sorted(reorder_indices)

        # 提取并打乱
        sub_seq = [sequence[i] for i in reorder_indices]
        np.random.shuffle(sub_seq)

        aug_seq = sequence.copy()
        for i, idx in enumerate(reorder_indices):
            aug_seq[idx] = sub_seq[i]

        return aug_seq

    def item_substitute(self, sequence: List[int],
                       substitute_ratio: float = 0.1) -> List[int]:
        """
        物品替换

        随机替换部分物品
        """
        n_sub = int(len(sequence) * substitute_ratio)
        if n_sub == 0:
            return sequence

        sub_indices = np.random.choice(
            len(sequence), n_sub, replace=False
        )

        aug_seq = sequence.copy()
        for idx in sub_indices:
            aug_seq[idx] = np.random.randint(1, self.n_items)

        return aug_seq

    def random_augment(self, sequence: List[int]) -> List[int]:
        """随机选择一种增强"""
        aug_type = np.random.choice(['crop', 'mask', 'reorder', 'substitute'])

        if aug_type == 'crop':
            return self.item_crop(sequence, self.aug_ratio)
        elif aug_type == 'mask':
            return self.item_mask(sequence, self.aug_ratio)
        elif aug_type == 'reorder':
            return self.item_reorder(sequence, self.aug_ratio)
        else:
            return self.item_substitute(sequence, self.aug_ratio)


class GraphAugmentation:
    """
    图数据增强
    """

    def __init__(self, edge_dropout_ratio: float = 0.1,
                 node_dropout_ratio: float = 0.1):
        self.edge_dropout_ratio = edge_dropout_ratio
        self.node_dropout_ratio = node_dropout_ratio

    def edge_dropout(self, edge_index: torch.Tensor) -> torch.Tensor:
        """
        边删除
        """
        n_edges = edge_index.size(1)
        n_drop = int(n_edges * self.edge_dropout_ratio)

        keep_indices = np.random.choice(
            n_edges, n_edges - n_drop, replace=False
        )

        return edge_index[:, keep_indices]

    def node_dropout(self, edge_index: torch.Tensor,
                    n_nodes: int) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        节点删除
        """
        n_drop = int(n_nodes * self.node_dropout_ratio)

        drop_nodes = np.random.choice(n_nodes, n_drop, replace=False)
        keep_nodes = set(range(n_nodes)) - set(drop_nodes)

        # 创建映射
        node_map = {old: new for new, old in enumerate(keep_nodes)}

        # 过滤边
        mask = torch.tensor([
            (edge_index[0, i].item() in keep_nodes and
             edge_index[1, i].item() in keep_nodes)
            for i in range(edge_index.size(1))
        ])

        new_edge_index = edge_index[:, mask]

        # 重新映射节点ID
        for i in range(new_edge_index.size(1)):
            new_edge_index[0, i] = node_map[new_edge_index[0, i].item()]
            new_edge_index[1, i] = node_map[new_edge_index[1, i].item()]

        return new_edge_index, keep_nodes
```

### 4.1 SGL (Self-supervised Graph Learning)

```python
class SGL(nn.Module):
    """
    SGL: Self-supervised Graph Learning for Recommendation

    在图神经网络上应用对比学习
    """

    def __init__(self, n_users: int, n_items: int,
                 embed_dim: int = 64, n_layers: int = 3,
                 temperature: float = 0.1):
        super().__init__()

        self.n_users = n_users
        self.n_items = n_items
        self.embed_dim = embed_dim
        self.n_layers = n_layers
        self.temperature = temperature

        # 嵌入
        self.user_embedding = nn.Embedding(n_users, embed_dim)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        self.augmentor = GraphAugmentation()

        self._init_weights()

    def _init_weights(self):
        nn.init.normal_(self.user_embedding.weight, std=0.01)
        nn.init.normal_(self.item_embedding.weight, std=0.01)

    def lightgcn_forward(self, user_embed: torch.Tensor,
                        item_embed: torch.Tensor,
                        edge_index: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        LightGCN 前向传播
        """
        all_embed = torch.cat([user_embed, item_embed], dim=0)

        # 构建归一化邻接矩阵
        n_nodes = self.n_users + self.n_items
        edge_index_full = edge_index.clone()
        edge_index_full[1] += self.n_users  # 物品节点偏移

        # 对称边
        edge_index_sym = torch.cat([
            edge_index_full,
            torch.stack([edge_index_full[1], edge_index_full[0]], dim=0)
        ], dim=1)

        # 度归一化
        row, col = edge_index_sym
        deg = torch.zeros(n_nodes)
        deg.scatter_add_(0, row, torch.ones_like(row, dtype=torch.float))
        deg_inv_sqrt = 1.0 / torch.sqrt(deg + 1e-8)

        # 消息传递
        all_embeds = [all_embed]
        for _ in range(self.n_layers):
            # 简化的 GCN 聚合
            # 实际应使用稀疏矩阵乘法
            all_embed = self._sparse_aggregate(
                all_embed, edge_index_sym, deg_inv_sqrt
            )
            all_embeds.append(all_embed)

        # 平均
        final_embed = torch.stack(all_embeds, dim=0).mean(dim=0)

        return final_embed[:self.n_users], final_embed[self.n_users:]

    def _sparse_aggregate(self, embed: torch.Tensor,
                         edge_index: torch.Tensor,
                         deg_inv_sqrt: torch.Tensor) -> torch.Tensor:
        """稀疏聚合"""
        row, col = edge_index

        # 归一化消息
        norm = deg_inv_sqrt[row] * deg_inv_sqrt[col]

        # 聚合（简化实现）
        aggregated = torch.zeros_like(embed)
        for i in range(edge_index.size(1)):
            r, c = row[i], col[i]
            aggregated[r] += norm[i] * embed[c]

        return aggregated

    def contrastive_loss(self, user_ids: torch.Tensor,
                        edge_index: torch.Tensor) -> torch.Tensor:
        """
        对比损失

        对同一用户在不同增强视图下的表示进行对比
        """
        # 两个增强视图
        edge_index_1 = self.augmentor.edge_dropout(edge_index)
        edge_index_2 = self.augmentor.edge_dropout(edge_index)

        # 编码
        user_embed = self.user_embedding.weight
        item_embed = self.item_embedding.weight

        z1_user, z1_item = self.lightgcn_forward(user_embed, item_embed, edge_index_1)
        z2_user, z2_item = self.lightgcn_forward(user_embed, item_embed, edge_index_2)

        # 归一化
        z1_user = F.normalize(z1_user, dim=-1)
        z2_user = F.normalize(z2_user, dim=-1)

        # InfoNCE
        batch_user_embed = z1_user[user_ids]
        batch_pos_embed = z2_user[user_ids]

        # 批内负样本
        pos_sim = torch.sum(batch_user_embed * batch_pos_embed, dim=-1) / self.temperature

        # 与其他用户的相似度作为负样本
        neg_sims = torch.matmul(batch_user_embed, z2_user.T) / self.temperature

        # 标签
        labels = user_ids

        return F.cross_entropy(neg_sims, labels)

    def forward(self, user_ids: torch.Tensor,
               item_ids: torch.Tensor,
               edge_index: torch.Tensor) -> torch.Tensor:
        """预测"""
        user_embed = self.user_embedding.weight
        item_embed = self.item_embedding.weight

        user_out, item_out = self.lightgcn_forward(user_embed, item_embed, edge_index)

        user_vec = user_out[user_ids]
        item_vec = item_out[item_ids]

        return torch.sum(user_vec * item_vec, dim=-1)
```

### 5.1 负采样策略

```python
class HardNegativeSampler:
    """
    困难负样本采样

    选择与正样本相似但为负的样本
    """

    def __init__(self, item_embeddings: np.ndarray,
                 n_hard: int = 5, n_random: int = 5):
        self.item_embeddings = item_embeddings
        self.n_hard = n_hard
        self.n_random = n_random

    def sample(self, user_id: int, pos_item_ids: List[int],
              n_neg: int = 10) -> List[int]:
        """
        采样负样本

        结合困难负样本和随机负样本
        """
        n_items = len(self.item_embeddings)

        # 随机负样本
        random_negs = []
        while len(random_negs) < self.n_random:
            neg = np.random.randint(n_items)
            if neg not in pos_item_ids and neg not in random_negs:
                random_negs.append(neg)

        # 困难负样本
        # 计算与正样本的相似度
        pos_embeds = self.item_embeddings[pos_item_ids]
        pos_mean = np.mean(pos_embeds, axis=0)

        similarities = np.dot(self.item_embeddings, pos_mean)

        # 排除正样本
        for pos_id in pos_item_ids:
            similarities[pos_id] = -np.inf

        # 选择相似度最高的作为困难负样本
        hard_negs = np.argsort(similarities)[-self.n_hard:].tolist()

        return hard_negs + random_negs


class DynamicNegativeSampler:
    """
    动态负采样

    根据训练进度调整负样本难度
    """

    def __init__(self, item_embeddings: np.ndarray,
                 initial_ratio: float = 0.5):
        self.item_embeddings = item_embeddings
        self.hard_ratio = initial_ratio
        self.n_items = len(item_embeddings)

    def update_ratio(self, epoch: int, total_epochs: int):
        """
        更新困难负样本比例

        训练初期使用更多随机负样本
        后期增加困难负样本
        """
        self.hard_ratio = min(0.9, 0.5 + epoch / total_epochs * 0.4)

    def sample(self, pos_item_ids: List[int], n_neg: int = 10) -> List[int]:
        """采样"""
        n_hard = int(n_neg * self.hard_ratio)
        n_random = n_neg - n_hard

        negs = []

        # 困难负样本
        if n_hard > 0:
            pos_embeds = self.item_embeddings[pos_item_ids]
            pos_mean = np.mean(pos_embeds, axis=0)
            similarities = np.dot(self.item_embeddings, pos_mean)

            for pos_id in pos_item_ids:
                similarities[pos_id] = -np.inf

            hard_negs = np.argsort(similarities)[-n_hard*2:]
            np.random.shuffle(hard_negs)
            negs.extend(hard_negs[:n_hard].tolist())

        # 随机负样本
        while len(negs) < n_neg:
            neg = np.random.randint(self.n_items)
            if neg not in pos_item_ids and neg not in negs:
                negs.append(neg)

        return negs
```

## 6. 学习总结

### 6.1 核心要点

1. **对比学习**: 相似样本靠近，不相似样本远离
2. **损失函数**: Contrastive, Triplet, InfoNCE
3. **数据增强**: 裁剪、遮掩、重排、替换
4. **负采样**: 困难负样本提升效果

### 6.2 损失函数对比

```
损失函数        适用场景            特点
────────────────────────────────────────────
Contrastive    成对对比            简单直观
Triplet        三元组              需要挖掘
InfoNCE        批内对比            高效
NT-Xent        SimCLR              自监督
```

### 6.3 最佳实践

```
1. 温度参数: 0.05-0.2 之间调优
2. 批量大小: 越大越好（更多负样本）
3. 增强策略: 多种增强组合
4. 负样本: 结合随机和困难负样本
```
