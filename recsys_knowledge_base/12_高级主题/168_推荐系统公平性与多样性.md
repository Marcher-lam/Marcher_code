# 推荐系统公平性与多样性 学习文档

## 1. 公平性概述

### 1.1 什么是推荐公平性？

```
公平性定义:
- 用户公平: 不同用户群体获得公平的推荐质量
- 物品公平: 不同物品获得公平的曝光机会
- 多方公平: 平衡平台、用户、提供者利益

公平性挑战:
- 热门偏差: 热门物品过曝光
- 群体偏差: 特定群体被忽视
- 位置偏差: 排名靠前的物品优势
```

### 1.2 公平性类型

```python
from typing import List, Dict, Tuple
import numpy as np
from collections import defaultdict


class FairnessTypes:
    """
    公平性类型定义
    """

    @staticmethod
    def demographic_parity(predictions: np.ndarray,
                          sensitive_attribute: np.ndarray) -> float:
        """
        人口统计均等

        P(Y=1|A=0) ≈ P(Y=1|A=1)

        不同敏感属性组的正向预测率应相等
        """
        group_0 = predictions[sensitive_attribute == 0]
        group_1 = predictions[sensitive_attribute == 1]

        rate_0 = np.mean(group_0)
        rate_1 = np.mean(group_1)

        return abs(rate_0 - rate_1)

    @staticmethod
    def equalized_odds(predictions: np.ndarray,
                      labels: np.ndarray,
                      sensitive_attribute: np.ndarray) -> Dict:
        """
        机会均等

        P(Y_hat=1|Y=1, A=0) ≈ P(Y_hat=1|Y=1, A=1)
        P(Y_hat=1|Y=0, A=0) ≈ P(Y_hat=1|Y=0, A=1)

        在真实标签相同时，预测应与敏感属性无关
        """
        # Y=1 时的 TPR 差异
        mask_pos = labels == 1
        tpr_0 = np.mean(predictions[mask_pos & (sensitive_attribute == 0)])
        tpr_1 = np.mean(predictions[mask_pos & (sensitive_attribute == 1)])

        # Y=0 时的 FPR 差异
        mask_neg = labels == 0
        fpr_0 = np.mean(predictions[mask_neg & (sensitive_attribute == 0)])
        fpr_1 = np.mean(predictions[mask_neg & (sensitive_attribute == 1)])

        return {
            'tpr_diff': abs(tpr_0 - tpr_1),
            'fpr_diff': abs(fpr_0 - fpr_1)
        }

    @staticmethod
    def calibration(predictions: np.ndarray,
                   labels: np.ndarray,
                   sensitive_attribute: np.ndarray) -> float:
        """
        校准性

        P(Y=1|Y_hat=p, A) ≈ p

        预测概率应反映真实概率
        """
        # 分桶计算
        n_bins = 10
        bin_edges = np.linspace(0, 1, n_bins + 1)

        cal_errors = []

        for i in range(n_bins):
            mask_bin = (predictions >= bin_edges[i]) & (predictions < bin_edges[i+1])

            if mask_bin.sum() > 0:
                # 整体校准误差
                mean_pred = np.mean(predictions[mask_bin])
                mean_label = np.mean(labels[mask_bin])
                cal_errors.append(abs(mean_pred - mean_label))

        return np.mean(cal_errors) if cal_errors else 0


class FairnessMetrics:
    """
    公平性评估指标
    """

    @staticmethod
    def exposure_fairness(exposures: Dict[str, List[float]]) -> float:
        """
        曝光公平性

        不同物品/提供者的曝光分布差异
        """
        exposure_means = [np.mean(exp) for exp in exposures.values()]
        return np.std(exposure_means) / (np.mean(exposure_means) + 1e-10)

    @staticmethod
    def gini_coefficient(values: List[float]) -> float:
        """
        基尼系数

        度量不平等程度
        """
        values = np.sort(values)
        n = len(values)
        cumsum = np.cumsum(values)

        gini = (2 * np.sum((np.arange(1, n+1)) * values) -
               (n+1) * np.sum(values)) / (n * np.sum(values))

        return gini

    @staticmethod
    def theil_index(values: List[float]) -> float:
        """
        Theil 指数

        信息论角度的不平等度量
        """
        values = np.array(values)
        mean_val = np.mean(values)

        if mean_val == 0:
            return 0

        # 避免日志(0)
        values = values[values > 0]

        theil = np.mean((values / mean_val) * np.log(values / mean_val))

        return theil
```

```python
import torch
import torch.nn as nn


class ReweightingPreprocessor:
    """
    重加权预处理

    根据敏感属性调整样本权重
    """

    def __init__(self):
        self.sample_weights = {}

    def fit(self, features: np.ndarray,
            labels: np.ndarray,
            sensitive_attribute: np.ndarray):
        """
        计算样本权重

        权重 = 1 / (P(Y=y, A=a) / (P(Y=y) * P(A=a)))
        """
        n_samples = len(labels)

        # 计算联合概率和边缘概率
        joint_counts = defaultdict(int)
        y_counts = defaultdict(int)
        a_counts = defaultdict(int)

        for y, a in zip(labels, sensitive_attribute):
            joint_counts[(y, a)] += 1
            y_counts[y] += 1
            a_counts[a] += 1

        # 计算权重
        self.sample_weights = []

        for i in range(n_samples):
            y = labels[i]
            a = sensitive_attribute[i]

            p_ya = joint_counts[(y, a)] / n_samples
            p_y = y_counts[y] / n_samples
            p_a = a_counts[a] / n_samples

            weight = p_y * p_a / (p_ya + 1e-10)
            self.sample_weights.append(weight)

        self.sample_weights = np.array(self.sample_weights)
        self.sample_weights = self.sample_weights / np.mean(self.sample_weights)

    def transform(self, indices: np.ndarray) -> np.ndarray:
        """获取样本权重"""
        return self.sample_weights[indices]


class FairRepresentationLearning(nn.Module):
    """
    公平表示学习

    学习对敏感属性不变的表示
    """

    def __init__(self, input_dim: int, hidden_dim: int, output_dim: int,
                 n_sensitive: int = 2, lambda_fair: float = 1.0):
        super().__init__()

        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, output_dim)
        )

        # 主任务预测器
        self.predictor = nn.Linear(output_dim, 1)

        # 敏感属性预测器（对抗）
        self.adversary = nn.Sequential(
            nn.Linear(output_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, n_sensitive)
        )

        self.lambda_fair = lambda_fair

    def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """前向传播"""
        representation = self.encoder(x)
        prediction = self.predictor(representation)
        sensitive_pred = self.adversary(representation)

        return prediction, sensitive_pred

    def fair_loss(self, prediction: torch.Tensor,
                 sensitive_pred: torch.Tensor,
                 labels: torch.Tensor,
                 sensitive_labels: torch.Tensor) -> torch.Tensor:
        """
        公平损失

        L = L_task - λ * L_adversary

        主任务最小化，对抗任务最大化（让表示无法预测敏感属性）
        """
        # 主任务损失
        task_loss = nn.BCEWithLogitsLoss()(prediction.squeeze(), labels.float())

        # 对抗损失（梯度反转）
        adv_loss = nn.CrossEntropyLoss()(sensitive_pred, sensitive_labels)

        # 总损失
        total_loss = task_loss - self.lambda_fair * adv_loss

        return total_loss, task_loss, adv_loss


class GradientReversalLayer(torch.autograd.Function):
    """
    梯度反转层

    前向传播不变，反向传播反转梯度
    """

    @staticmethod
    def forward(ctx, x, lambda_):
        ctx.lambda_ = lambda_
        return x.clone()

    @staticmethod
    def backward(ctx, grad_output):
        return -ctx.lambda_ * grad_output, None


class AdversarialFairModel(nn.Module):
    """
    对抗公平模型
    """

    def __init__(self, input_dim: int, hidden_dim: int, n_sensitive: int = 2):
        super().__init__()

        # 共享编码器
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim // 2)
        )

        # 主任务头
        self.task_head = nn.Linear(hidden_dim // 2, 1)

        # 敏感属性头（带梯度反转）
        self.sensitive_head = nn.Linear(hidden_dim // 2, n_sensitive)

        self.lambda_ = 1.0

    def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        representation = self.encoder(x)

        # 主任务
        task_output = self.task_head(representation)

        # 敏感属性预测（梯度反转）
        reversed_repr = GradientReversalLayer.apply(representation, self.lambda_)
        sensitive_output = self.sensitive_head(reversed_repr)

        return task_output, sensitive_output
```

### 2.2 后处理方法

```python
class FairPostProcessor:
    """
    公平性后处理

    调整预测结果以满足公平性约束
    """

    def __init__(self):
        self.thresholds = {}

    def calibrate_thresholds(self, scores: np.ndarray,
                            labels: np.ndarray,
                            sensitive_attribute: np.ndarray,
                            target_tpr: float = 0.8):
        """
        校准阈值

        调整各组阈值以实现机会均等
        """
        unique_groups = np.unique(sensitive_attribute)

        for group in unique_groups:
            group_mask = sensitive_attribute == group
            group_scores = scores[group_mask]
            group_labels = labels[group_mask]

            # 找到使 TPR 接近目标值的阈值
            sorted_scores = np.sort(group_scores)[::-1]
            n_pos = np.sum(group_labels == 1)
            target_n = int(n_pos * target_tpr)

            threshold = sorted_scores[target_n] if target_n < len(sorted_scores) else sorted_scores[-1]
            self.thresholds[group] = threshold

    def predict(self, scores: np.ndarray,
               sensitive_attribute: np.ndarray) -> np.ndarray:
        """应用组特定阈值"""
        predictions = np.zeros_like(scores)

        for group in np.unique(sensitive_attribute):
            group_mask = sensitive_attribute == group
            predictions[group_mask] = (scores[group_mask] >= self.thresholds[group]).astype(int)

        return predictions


class ExposureAllocator:
    """
    曝光分配器

    公平地分配曝光机会
    """

    def __init__(self, total_exposure: int):
        self.total_exposure = total_exposure

    def allocate_by_proportional_fairness(self,
                                          item_scores: Dict[int, float],
                                          item_groups: Dict[int, int],
                                          min_exposure_per_group: Dict[int, int] = None) -> Dict[int, int]:
        """
        比例公平分配

        参数:
            item_scores: 物品分数
            item_groups: 物品所属组
            min_exposure_per_group: 每组最小曝光
        """
        # 按组分组
        group_items = defaultdict(list)
        for item_id, group in item_groups.items():
            group_items[group].append((item_id, item_scores[item_id]))

        # 计算每组应得曝光
        group_exposures = {}
        if min_exposure_per_group:
            for group, min_exp in min_exposure_per_group.items():
                group_exposures[group] = min_exp

        remaining_exposure = self.total_exposure - sum(group_exposures.values())

        # 按分数比例分配剩余曝光
        group_scores = {}
        for group, items in group_items.items():
            group_scores[group] = sum(score for _, score in items)

        total_score = sum(group_scores.values())

        for group, score in group_scores.items():
            proportional = int(remaining_exposure * score / (total_score + 1e-10))
            group_exposures[group] = group_exposures.get(group, 0) + proportional

        # 按组内分数分配到物品
        allocation = {}
        for group, items in group_items.items():
            items_sorted = sorted(items, key=lambda x: x[1], reverse=True)
            group_exp = group_exposures.get(group, 0)

            for i, (item_id, _) in enumerate(items_sorted):
                if i < group_exp:
                    allocation[item_id] = 1
                else:
                    allocation[item_id] = 0

        return allocation
```

### 3.1 多样性指标

```python
class DiversityMetrics:
    """
    多样性评估指标
    """

    @staticmethod
    def intra_list_diversity(recommendations: List[int],
                            item_features: np.ndarray) -> float:
        """
        列表内多样性

        ILD = (1/k(k-1)) * Σ d(i, j)
        """
        if len(recommendations) < 2:
            return 0

        k = len(recommendations)
        total_distance = 0

        for i in range(k):
            for j in range(i+1, k):
                feat_i = item_features[recommendations[i]]
                feat_j = item_features[recommendations[j]]
                distance = 1 - np.dot(feat_i, feat_j) / (
                    np.linalg.norm(feat_i) * np.linalg.norm(feat_j) + 1e-10
                )
                total_distance += distance

        return 2 * total_distance / (k * (k - 1))

    @staticmethod
    def category_coverage(recommendations: List[int],
                         item_categories: Dict[int, int]) -> float:
        """
        类别覆盖率

        推荐列表覆盖的类别比例
        """
        categories = set(item_categories[item] for item in recommendations)
        total_categories = len(set(item_categories.values()))

        return len(categories) / total_categories if total_categories > 0 else 0

    @staticmethod
    def gini_diversity(recommendations: List[int],
                      item_popularity: Dict[int, float]) -> float:
        """
        Gini 多样性

        推荐物品流行度分布的不平等程度
        （越大越不均匀，即越不多样）
        """
        pop_values = [item_popularity.get(item, 0) for item in recommendations]
        pop_values = np.sort(pop_values)
        n = len(pop_values)

        cumsum = np.cumsum(pop_values)
        gini = (2 * np.sum((np.arange(1, n+1)) * pop_values) -
               (n+1) * np.sum(pop_values)) / (n * np.sum(pop_values) + 1e-10)

        return 1 - gini  # 返回多样性（与Gini相反）

    @staticmethod
    def serendipity(recommendations: List[int],
                   user_history: List[int],
                   item_similarity: Dict[Tuple[int, int], float]) -> float:
        """
        意外性

        推荐"意外但相关"的物品
        """
        if not recommendations or not user_history:
            return 0

        serendipity_scores = []

        for rec_item in recommendations:
            # 计算与历史物品的最大相似度
            max_sim = max(
                item_similarity.get((rec_item, hist_item), 0)
                for hist_item in user_history
            )

            # 意外性 = 1 - 相似度
            serendipity_scores.append(1 - max_sim)

        return np.mean(serendipity_scores)


class NoveltyMetrics:
    """
    新颖性指标
    """

    @staticmethod
    def expected_popularity_complement(recommendations: List[int],
                                       item_popularity: Dict[int, float]) -> float:
        """
        期望流行度补数 (EPC)

        EPC = (1/k) * Σ (1 - p(i))^-log p(i)
        """
        if not recommendations:
            return 0

        total_novelty = 0
        for item in recommendations:
            p = item_popularity.get(item, 0.5)
            if p > 0:
                novelty = -np.log(p + 1e-10)
            else:
                novelty = 0
            total_novelty += novelty

        return total_novelty / len(recommendations)

    @staticmethod
    def long_tail_percentage(recommendations: List[int],
                            item_popularity: Dict[int, float],
                            threshold: float = 0.2) -> float:
        """
        长尾比例

        推荐中长尾物品的占比
        """
        sorted_pop = sorted(item_popularity.values())
        threshold_value = sorted_pop[int(len(sorted_pop) * threshold)]

        long_tail_count = sum(
            1 for item in recommendations
            if item_popularity.get(item, 0) <= threshold_value
        )

        return long_tail_count / len(recommendations) if recommendations else 0
```

### 3.2 多样性优化

```python
class MMRDiversifier:
    """
    MMR (Maximal Marginal Relevance) 多样化

    平衡相关性和多样性
    """

    def __init__(self, lambda_diversity: float = 0.5):
        self.lambda_diversity = lambda_diversity

    def diversify(self, candidates: List[Tuple[int, float]],
                 item_features: np.ndarray,
                 n: int = 10) -> List[int]:
        """
        MMR 多样化

        MMR = λ * relevance - (1-λ) * max_similarity_to_selected
        """
        selected = []
        remaining = list(candidates)

        while len(selected) < n and remaining:
            best_score = -np.inf
            best_item = None
            best_idx = 0

            for idx, (item_id, relevance) in enumerate(remaining):
                # 相关性
                rel_score = relevance

                # 多样性惩罚
                if selected:
                    max_sim = max(
                        np.dot(item_features[item_id], item_features[s]) / (
                            np.linalg.norm(item_features[item_id]) *
                            np.linalg.norm(item_features[s]) + 1e-10
                        )
                        for s in selected
                    )
                else:
                    max_sim = 0

                # MMR 分数
                mmr = self.lambda_diversity * rel_score - \
                     (1 - self.lambda_diversity) * max_sim

                if mmr > best_score:
                    best_score = mmr
                    best_item = item_id
                    best_idx = idx

            if best_item is not None:
                selected.append(best_item)
                remaining.pop(best_idx)

        return selected


class DeterminantalPointProcess:
    """
    行列式点过程 (DPP)

    基于行列式的多样化方法
    """

    def __init__(self, lambda_rel: float = 0.5):
        self.lambda_rel = lambda_rel

    def diversify(self, candidates: List[Tuple[int, float]],
                 item_features: np.ndarray,
                 n: int = 10) -> List[int]:
        """
        DPP 多样化

        最大化 det(L_Y)，其中 L 是核矩阵
        """
        n_candidates = len(candidates)
        item_ids = [c[0] for c in candidates]
        relevances = np.array([c[1] for c in candidates])

        # 构建核矩阵
        # L_ij = q_i * q_j * S_ij
        # q_i = 相关性，S_ij = 相似度
        similarities = np.dot(item_features[item_ids], item_features[item_ids].T)
        norms = np.linalg.norm(item_features[item_ids], axis=1)
        similarities = similarities / (np.outer(norms, norms) + 1e-10)

        q = np.sqrt(relevances + 1e-10)
        L = np.outer(q, q) * similarities

        # 贪婪 DPP
        selected = []
        remaining = set(range(n_candidates))

        while len(selected) < n and remaining:
            best_gain = -np.inf
            best_item = None

            for item in remaining:
                if not selected:
                    gain = L[item, item]
                else:
                    # 计算添加 item 后的行列式增益
                    L_selected = L[np.ix_(selected, selected)]
                    L_item_selected = L[item, selected]

                    try:
                        # 增益 = L[item,item] - L[item,S] @ L[S,S]^-1 @ L[S,item]
                        L_selected_inv = np.linalg.inv(L_selected)
                        gain = L[item, item] - np.dot(
                            L_item_selected,
                            np.dot(L_selected_inv, L_item_selected)
                        )
                    except np.linalg.LinAlgError:
                        gain = L[item, item]

                if gain > best_gain:
                    best_gain = gain
                    best_item = item

            if best_item is not None:
                selected.append(best_item)
                remaining.remove(best_item)

        return [item_ids[i] for i in selected]


class CategoryBalancer:
    """
    类别平衡器

    确保推荐覆盖多个类别
    """

    def __init__(self, max_per_category: int = 2):
        self.max_per_category = max_per_category

    def diversify(self, candidates: List[Tuple[int, float]],
                 item_categories: Dict[int, int],
                 n: int = 10) -> List[int]:
        """
        类别平衡多样化
        """
        selected = []
        category_counts = defaultdict(int)

        # 按相关性排序
        sorted_candidates = sorted(candidates, key=lambda x: x[1], reverse=True)

        for item_id, score in sorted_candidates:
            if len(selected) >= n:
                break

            category = item_categories.get(item_id)

            if category is None or category_counts[category] < self.max_per_category:
                selected.append(item_id)
                if category is not None:
                    category_counts[category] += 1

        return selected
```

## 4. 学习总结

### 4.1 核心要点

1. **用户公平**: 不同用户群体公平对待
2. **物品公平**: 不同物品公平曝光
3. **多样性**: 推荐结果的丰富性
4. **新颖性**: 推荐非热门物品

### 4.2 方法对比

```
方法类型        时机            特点
────────────────────────────────────────
预处理          训练前          重加权、重采样
处理中          训练时          正则化、对抗训练
后处理          预测后          阈值调整、重排序
```

### 4.3 最佳实践

```
1. 明确公平性定义和约束
2. 权衡准确性与公平性
3. 多样性与相关性平衡
4. 持续监控公平性指标
```
