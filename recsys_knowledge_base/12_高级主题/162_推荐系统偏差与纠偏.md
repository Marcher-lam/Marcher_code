# 推荐系统偏差与纠偏 学习文档

## 1. 问题定义

### 1.1 推荐系统中的偏差类型

| 偏差类型 | 描述 | 产生原因 |
|----------|------|----------|
| 选择偏差 | 用户主动选择导致 | 用户非随机交互 |
| 曝光偏差 | 物品曝光不均等 | 推荐策略偏好 |
| 位置偏差 | 位置影响点击 | 用户注意力有限 |
| 流行度偏差 | 热门物品过拟合 | 马太效应 |
| 一致性偏差 | 用户从众行为 | 社会影响 |
| 嵌入偏差 | 嵌入空间偏差 | 训练数据偏差 |

### 1.2 偏差影响

```
1. 模型学习偏差 → 预测不准
2. 推荐结果偏差 → 信息茧房
3. 评估偏差 → 指标失真
4. 长期偏差 → 系统退化
```

## 2. 位置偏差

### 2.1 问题分析

```
用户点击物品 = f(相关性, 位置)
而不是仅依赖相关性

问题: 高位置（排序靠前）的点击被高估
```

### 2.2 建模与纠正

```python
import numpy as np
from typing import List, Tuple, Dict
import torch
import torch.nn as nn


class PositionBiasModel:
    """
    位置偏差模型
    """

    def __init__(self, max_position: int = 20):
        self.max_position = max_position
        # 可学习的位置权重
        self.position_weights = np.ones(max_position)

    def estimate_position_weights(self, click_data: List[Dict]):
        """
        估计位置权重

        使用逆倾向加权（IPW）方法
        """
        position_clicks = np.zeros(self.max_position)
        position_impressions = np.zeros(self.max_position)

        for item in click_data:
            positions = item['positions']  # 展示位置
            clicks = item['clicks']  # 点击情况

            for pos, click in zip(positions, clicks):
                if pos < self.max_position:
                    position_impressions[pos] += 1
                    if click:
                        position_clicks[pos] += 1

        # 点击率作为位置权重
        for pos in range(self.max_position):
            if position_impressions[pos] > 0:
                self.position_weights[pos] = position_clicks[pos] / position_impressions[pos]

        # 归一化
        self.position_weights = self.position_weights / self.position_weights[0]

    def debias_click(self, click: int, position: int) -> float:
        """纠正点击"""
        if position < self.max_position:
            weight = self.position_weights[position]
            if weight > 0:
                return click / weight
        return float(click)


class PositionBiasNetwork(nn.Module):
    """
    神经网络建模位置偏差
    """

    def __init__(self, n_positions: int = 20, embed_dim: int = 8):
        super().__init__()

        # 位置嵌入
        self.position_embedding = nn.Embedding(n_positions, embed_dim)

        # 位置偏置网络
        self.bias_net = nn.Sequential(
            nn.Linear(embed_dim, 16),
            nn.ReLU(),
            nn.Linear(16, 1),
            nn.Sigmoid()
        )

    def forward(self, position: torch.Tensor) -> torch.Tensor:
        """计算位置偏差"""
        pos_embed = self.position_embedding(position)
        bias = self.bias_net(pos_embed)
        return bias.squeeze(-1)


class PositionDebiasedModel(nn.Module):
    """
    去位置偏差的 CTR 模型
    """

    def __init__(self, feature_dim: int, n_positions: int = 20):
        super().__init__()

        # 主模型（预测相关性）
        self.main_net = nn.Sequential(
            nn.Linear(feature_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 64),
            nn.ReLU(),
            nn.Linear(64, 1)
        )

        # 位置偏差网络
        self.position_net = PositionBiasNetwork(n_positions)

    def forward(self, features: torch.Tensor,
               position: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        前向传播

        返回:
            relevance: 相关性分数
            position_bias: 位置偏差
        """
        relevance = self.main_net(features)
        position_bias = self.position_net(position)

        # 观测到的点击概率
        observed = torch.sigmoid(relevance.squeeze(-1)) * position_bias

        return relevance, observed

    def predict_relevance(self, features: torch.Tensor) -> torch.Tensor:
        """预测纯相关性（无位置偏差）"""
        return self.main_net(features)
```

## 3. 选择偏差

### 3.1 问题分析

```
观测数据 = 用户主动选择的结果
未观测数据 = 用户未选择的（可能也喜欢）

问题: 训练数据分布 ≠ 真实偏好分布
```

### 3.2 IPS 纠正

```python
class IPSEstimator:
    """
    逆倾向分数（IPS）估计器
    """

    def __init__(self):
        # 倾向分数模型
        self.propensity_model = None

    def fit_propensity(self, features: np.ndarray, observed: np.ndarray):
        """
        训练倾向分数模型

        参数:
            features: 特征
            observed: 是否被观测到（0/1）
        """
        from sklearn.linear_model import LogisticRegression

        self.propensity_model = LogisticRegression()
        self.propensity_model.fit(features, observed)

    def get_propensity(self, features: np.ndarray) -> np.ndarray:
        """获取倾向分数"""
        if self.propensity_model is None:
            return np.ones(len(features))

        probs = self.propensity_model.predict_proba(features)[:, 1]
        # 截断防止极端值
        probs = np.clip(probs, 0.01, 0.99)
        return probs

    def ips_loss(self, predictions: np.ndarray, targets: np.ndarray,
                 propensities: np.ndarray) -> float:
        """
        IPS 加权损失

        L = (1/N) * Σ (y - ŷ)² / p(x)
        """
        weights = 1.0 / propensities
        weighted_loss = np.sum(weights * (predictions - targets) ** 2)
        return weighted_loss / len(predictions)


class DoublyRobustEstimator:
    """
    双重鲁棒（Doubly Robust）估计器

    结合 IPS 和结果建模
    """

    def __init__(self):
        self.propensity_model = None
        self.outcome_model = None

    def fit(self, features: np.ndarray, observed: np.ndarray,
            outcomes: np.ndarray):
        """
        训练两个模型
        """
        from sklearn.linear_model import LogisticRegression, Ridge

        # 倾向模型
        self.propensity_model = LogisticRegression()
        self.propensity_model.fit(features, observed)

        # 结果模型（只在观测数据上训练）
        observed_idx = observed == 1
        self.outcome_model = Ridge()
        self.outcome_model.fit(features[observed_idx], outcomes[observed_idx])

    def estimate(self, features: np.ndarray, observed: np.ndarray,
                outcomes: np.ndarray) -> float:
        """
        双重鲁棒估计

        DR = (1/N) * Σ [m(x) + (y - m(x)) * o / p(x)]
        """
        # 倾向分数
        propensity = self.propensity_model.predict_proba(features)[:, 1]
        propensity = np.clip(propensity, 0.01, 0.99)

        # 预测结果
        predicted_outcome = self.outcome_model.predict(features)

        # 双重鲁棒估计
        dr_estimate = predicted_outcome + \
                     (outcomes - predicted_outcome) * observed / propensity

        return np.mean(dr_estimate)
```

## 4. 流行度偏差

### 4.1 问题分析

```
热门物品获得更多曝光 → 更多交互 → 更高推荐
→ 马太效应 → 长尾物品被忽略
```

### 4.2 纠正方法

```python
class PopularityDebiaser:
    """
    流行度偏差纠正
    """

    def __init__(self, alpha: float = 0.5):
        """
        参数:
            alpha: 平衡参数（0=无纠正，1=完全纠正）
        """
        self.alpha = alpha
        self.item_popularity = {}

    def fit(self, interactions: List[Tuple]):
        """计算物品流行度"""
        item_counts = {}

        for user_id, item_id, _ in interactions:
            item_counts[item_id] = item_counts.get(item_id, 0) + 1

        max_count = max(item_counts.values()) if item_counts else 1

        for item_id, count in item_counts.items():
            self.item_popularity[item_id] = count / max_count

    def debias_score(self, item_id: str, original_score: float) -> float:
        """
        纠正分数

        基于流行度惩罚
        """
        if item_id not in self.item_popularity:
            return original_score

        pop = self.item_popularity[item_id]
        # 流行度惩罚
        penalty = 1 - self.alpha * pop

        return original_score * penalty

    def inverse_propensity_weighting(self, item_id: str,
                                     original_score: float) -> float:
        """逆倾向加权"""
        if item_id not in self.item_popularity:
            return original_score

        pop = self.item_popularity[item_id]
        # 倾向分数近似为流行度
        propensity = np.clip(pop, 0.01, 0.99)

        return original_score / propensity


class DiversifiedRecommender:
    """
    通过多样性缓解流行度偏差
    """

    def __init__(self, lambda_diversity: float = 0.5):
        self.lambda_diversity = lambda_diversity

    def diversify(self, candidates: List[Tuple[str, float]],
                 item_features: Dict[str, np.ndarray],
                 n: int = 10) -> List[str]:
        """
        多样化重排

        使用 MMR (Maximal Marginal Relevance)
        """
        selected = []
        remaining = list(candidates)

        while len(selected) < n and remaining:
            best_score = -np.inf
            best_item = None
            best_idx = 0

            for idx, (item_id, relevance) in enumerate(remaining):
                # 多样性分数
                if selected:
                    # 与已选物品的最大相似度
                    max_sim = max(
                        np.dot(item_features[item_id], item_features[s])
                        / (np.linalg.norm(item_features[item_id]) *
                           np.linalg.norm(item_features[s]))
                        for s in selected
                    )
                else:
                    max_sim = 0

                # MMR 分数
                mmr = (1 - self.lambda_diversity) * relevance - \
                     self.lambda_diversity * max_sim

                if mmr > best_score:
                    best_score = mmr
                    best_item = item_id
                    best_idx = idx

            if best_item:
                selected.append(best_item)
                remaining.pop(best_idx)

        return selected
```

    """
    无偏学习框架
    """

    def __init__(self, feature_dim: int):
        self.feature_dim = feature_dim

    def causal_loss(self, predictions: torch.Tensor,
                   targets: torch.Tensor,
                   treatments: torch.Tensor,
                   propensities: torch.Tensor) -> torch.Tensor:
        """
        因果损失函数

        L = Σ (t / p) * (y - ŷ)² + λ * R(θ)
        """
        # IPS 加权
        weights = treatments / propensities

        # 加权 MSE
        loss = torch.mean(weights * (predictions - targets) ** 2)

        return loss

## 6. 偏差检测与监控

```python
class BiasMonitor:
    """
    偏差监控器
    """

    def __init__(self):
        self.metrics_history = defaultdict(list)

    def compute_popularity_bias(self, recommendations: List[List[str]],
                                item_popularity: Dict[str, float]) -> float:
        """
        计算流行度偏差

        推荐 Top-K 中热门物品的占比
        """
        popular_count = 0
        total_count = 0

        for rec_list in recommendations:
            for item_id in rec_list:
                if item_id in item_popularity:
                    if item_popularity[item_id] > 0.5:  # 高流行度
                        popular_count += 1
                    total_count += 1

        return popular_count / total_count if total_count > 0 else 0

    def compute_coverage(self, recommendations: List[List[str]],
                        total_items: int) -> float:
        """计算物品覆盖率"""
        unique_items = set()
        for rec_list in recommendations:
            unique_items.update(rec_list)

        return len(unique_items) / total_items

    def compute_gini_coefficient(self, item_exposure: Dict[str, int]) -> float:
        """计算基尼系数（曝光不平等程度）"""
        exposures = np.array(list(item_exposure.values()))
        exposures = np.sort(exposures)

        n = len(exposures)
        cumsum = np.cumsum(exposures)

        # 基尼系数
        gini = (2 * np.sum((np.arange(1, n + 1)) * exposures) -
               (n + 1) * np.sum(exposures)) / (n * np.sum(exposures))

        return gini

    def log_metrics(self, recommendations: List[List[str]],
                   item_popularity: Dict[str, float],
                   total_items: int):
        """记录指标"""
        pop_bias = self.compute_popularity_bias(recommendations, item_popularity)
        coverage = self.compute_coverage(recommendations, total_items)

        self.metrics_history['popularity_bias'].append(pop_bias)
        self.metrics_history['coverage'].append(coverage)

        return {
            'popularity_bias': pop_bias,
            'coverage': coverage
        }
```

## 7. 学习总结

### 7.1 核心要点

1. **位置偏差**: 用户更关注靠前位置
2. **选择偏差**: 观测数据非随机
3. **流行度偏差**: 热门物品过拟合
4. **曝光偏差**: 未曝光物品偏好未知

### 7.2 纠正策略

| 偏差类型 | 纠正方法 |
|----------|----------|
| 位置偏差 | 位置建模、IPW |
| 选择偏差 | IPS、双重鲁棒 |
| 流行度偏差 | 惩罚、多样性 |
| 曝光偏差 | 无偏学习 |
