# 大模型与推荐系统 学习文档

## 1. 大模型在推荐中的应用

### 1.1 应用场景

```
大模型在推荐系统中的应用:
1. 特征理解: 理解用户/物品的文本描述
2. 表示学习: 生成高质量的 embedding
3. 序列建模: 捕捉用户行为序列的语义
4. 可解释性: 生成推荐理由
5. 冷启动: 利用语义理解解决冷启动
```

### 1.2 技术路线

```
传统推荐 → 大模型增强推荐 → 大模型原生推荐

1. 特征增强: 用 LLM 增强传统特征
2. 模型融合: LLM + 传统模型
3. 端到端: 直接用 LLM 预测
```

```python
class ItemDescriptionEncoder(nn.Module):
    """
    物品描述编码器
    """

    def __init__(self, vocab_size: int = 50000,
                 embed_dim: int = 256,
                 hidden_dim: int = 512,
                 output_dim: int = 128):
        super().__init__()

        # 嵌入层
        self.embedding = nn.Embedding(vocab_size, embed_dim)

        # 编码器
        self.encoder = nn.LSTM(
            embed_dim, hidden_dim // 2,
            num_layers=2,
            bidirectional=True,
            batch_first=True
        )

        # 输出层
        self.output_layer = nn.Sequential(
            nn.Linear(hidden_dim, output_dim),
            nn.ReLU()
        )

    def forward(self, input_ids: torch.Tensor,
               attention_mask: torch.Tensor = None) -> torch.Tensor:
        """
        编码物品描述

        参数:
            input_ids: (batch, seq_len)
            attention_mask: (batch, seq_len)

        返回:
            embeddings: (batch, output_dim)
        """
        # 嵌入
        embeds = self.embedding(input_ids)

        # LSTM 编码
        if attention_mask is not None:
            lengths = attention_mask.sum(dim=1).cpu()
            packed = nn.utils.rnn.pack_padded_sequence(
                embeds, lengths, batch_first=True, enforce_sorted=False
            )
            _, hidden = self.encoder(packed)
        else:
            _, hidden = self.encoder(embeds)

        # 拼接双向 LSTM 的最后隐状态
        hidden = torch.cat([hidden[0][-2], hidden[0][-1]], dim=-1)

        # 输出
        output = self.output_layer(hidden)

        return output


class MultiModalItemEncoder(nn.Module):
    """
    多模态物品编码器

    融合文本、图像、类别等多种模态
    """

    def __init__(self, text_dim: int = 256,
                 image_dim: int = 512,
                 categorical_dim: int = 64,
                 output_dim: int = 128):
        super().__init__()

        # 文本编码器
        self.text_encoder = nn.Sequential(
            nn.Linear(text_dim, output_dim),
            nn.ReLU()
        )

        # 图像编码器
        self.image_encoder = nn.Sequential(
            nn.Linear(image_dim, output_dim),
            nn.ReLU()
        )

        # 类别编码器
        self.category_encoder = nn.Sequential(
            nn.Linear(categorical_dim, output_dim),
            nn.ReLU()
        )

        # 融合层
        self.fusion = nn.Sequential(
            nn.Linear(output_dim * 3, output_dim * 2),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(output_dim * 2, output_dim)
        )

        # 注意力融合
        self.attention = nn.MultiheadAttention(output_dim, num_heads=4)

    def forward(self, text_features: torch.Tensor,
               image_features: torch.Tensor,
               category_features: torch.Tensor) -> torch.Tensor:
        """
        多模态编码
        """
        # 各模态编码
        text_embed = self.text_encoder(text_features)
        image_embed = self.image_encoder(image_features)
        category_embed = self.category_encoder(category_features)

        # 拼接
        concat = torch.cat([text_embed, image_embed, category_embed], dim=-1)

        # 融合
        fused = self.fusion(concat)

        # 注意力融合
        stacked = torch.stack([text_embed, image_embed, category_embed], dim=1)
        attended, _ = self.attention(stacked, stacked, stacked)
        attended = attended.mean(dim=1)

        # 组合
        output = fused + attended

        return output
```

```python
    """
```

```python
                'input': input_text,
                'output': output_text
            })

Predict whether the user will be interested in the recommended item.

{item['input']}

### Output:
{item['output']}
"""
            formatted.append(text)

        return formatted


class PEFTForRecommendation(nn.Module):
    """
    参数高效微调 (PEFT) 用于推荐

    LoRA 适配器
    """

    def __init__(self, base_model: nn.Module, embed_dim: int = 768,
                 lora_r: int = 8, lora_alpha: float = 16.0):
        super().__init__()
        self.base_model = base_model
        self.embed_dim = embed_dim
        self.lora_r = lora_r
        self.lora_alpha = lora_alpha

        # LoRA 适配器
        self.lora_A = nn.Linear(embed_dim, lora_r, bias=False)
        self.lora_B = nn.Linear(lora_r, embed_dim, bias=False)

        # 初始化
        nn.init.kaiming_uniform_(self.lora_A.weight)
        nn.init.zeros_(self.lora_B.weight)

        # 推荐头
        self.rec_head = nn.Linear(embed_dim, 1)

        # 冻结基础模型
        for param in self.base_model.parameters():
            param.requires_grad = False

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        使用 LoRA 适配器
        """
        # 基础模型输出
        with torch.no_grad():
            base_output = self.base_model(input_ids)
            hidden_states = base_output.last_hidden_state[:, 0, :]

        # LoRA 适配
        lora_output = self.lora_B(self.lora_A(hidden_states))
        adapted = hidden_states + self.lora_alpha / self.lora_r * lora_output

        # 推荐
        score = self.rec_head(adapted)

        return score
```

## 4. LLM 与传统模型融合

### 4.1 特征融合

```python
class LLMTraditionalFusion(nn.Module):
    """
    LLM 特征与传统特征融合
    """

    def __init__(self, llm_dim: int = 768,
                 traditional_dim: int = 256,
                 hidden_dim: int = 256,
                 output_dim: int = 1):
        super().__init__()

        # LLM 特征处理
        self.llm_projector = nn.Sequential(
            nn.Linear(llm_dim, hidden_dim),
            nn.LayerNorm(hidden_dim),
            nn.ReLU()
        )

        # 传统特征处理
        self.traditional_projector = nn.Sequential(
            nn.Linear(traditional_dim, hidden_dim),
            nn.LayerNorm(hidden_dim),
            nn.ReLU()
        )

        # 融合层
        self.fusion = nn.Sequential(
            nn.Linear(hidden_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim, output_dim)
        )

        # 门控机制
        self.gate = nn.Sequential(
            nn.Linear(hidden_dim * 2, 2),
            nn.Softmax(dim=-1)
        )

    def forward(self, llm_features: torch.Tensor,
               traditional_features: torch.Tensor) -> torch.Tensor:
        """
        融合预测
        """
        # 投影
        llm_proj = self.llm_projector(llm_features)
        trad_proj = self.traditional_projector(traditional_features)

        # 门控权重
        concat = torch.cat([llm_proj, trad_proj], dim=-1)
        gate_weights = self.gate(concat)

        # 加权融合
        weighted_llm = gate_weights[:, 0:1] * llm_proj
        weighted_trad = gate_weights[:, 1:2] * trad_proj

        fused = torch.cat([weighted_llm, weighted_trad], dim=-1)

        # 输出
        output = self.fusion(fused)

        return output


class HybridRecommenderWithLLM:
    """
    LLM 增强的混合推荐器
    """

    def __init__(self, traditional_model: nn.Module,
                 llm_encoder: nn.Module,
                 fusion_model: nn.Module):
        self.traditional_model = traditional_model
        self.llm_encoder = llm_encoder
        self.fusion_model = fusion_model

    def get_item_embedding(self, item_id: int,
                          item_features: Dict) -> torch.Tensor:
        """
        获取物品嵌入（融合）
        """
        # 传统嵌入
        trad_embed = self.traditional_model.get_item_embedding(item_id)

        # LLM 嵌入
        text = item_features.get('title', '') + ' ' + item_features.get('description', '')
        llm_embed = self.llm_encoder.encode_text([text])[0]
        llm_embed = torch.tensor(llm_embed)

        # 融合
        fused = self.fusion_model(llm_embed.unsqueeze(0), trad_embed.unsqueeze(0))

        return fused.squeeze(0)

    def recommend(self, user_id: int, candidates: List[int],
                 item_features: Dict) -> List[Tuple[int, float]]:
        """
        推荐
        """
        # 用户嵌入
        user_embed = self.traditional_model.get_user_embedding(user_id)

        # 计算分数
        scores = []
        for item_id in candidates:
            item_embed = self.get_item_embedding(item_id, item_features.get(item_id, {}))
            score = torch.dot(user_embed, item_embed).item()
            scores.append((item_id, score))

        # 排序
        scores.sort(key=lambda x: x[1], reverse=True)

        return scores
```

```python
class GenerativeRecommender(nn.Module):
    """
    生成式推荐器

    直接生成推荐列表
    """

    def __init__(self, vocab_size: int, embed_dim: int = 512,
                 n_heads: int = 8, n_layers: int = 6):
        super().__init__()

        self.item_embedding = nn.Embedding(vocab_size, embed_dim)
        self.position_embedding = nn.Embedding(100, embed_dim)

        # Transformer 解码器
        decoder_layer = nn.TransformerDecoderLayer(
            d_model=embed_dim,
            nhead=n_heads,
            dim_feedforward=embed_dim * 4,
            batch_first=True
        )
        self.decoder = nn.TransformerDecoder(decoder_layer, num_layers=n_layers)

        # 输出层
        self.output_layer = nn.Linear(embed_dim, vocab_size)

    def forward(self, user_history: torch.Tensor,
               target_length: int = 10) -> torch.Tensor:
        """
        生成推荐序列

        参数:
            user_history: 用户历史物品 (batch, history_len)
            target_length: 生成长度

        返回:
            logits: (batch, target_length, vocab_size)
        """
        batch_size = user_history.size(0)

        # 编码历史
        history_embed = self.item_embedding(user_history)
        history_embed = history_embed + self.position_embedding(
            torch.arange(user_history.size(1), device=user_history.device)
        )

        # 自回归生成
        generated = []
        current_input = torch.zeros(batch_size, 1, dtype=torch.long, device=user_history.device)

        for i in range(target_length):
            # 目标嵌入
            target_embed = self.item_embedding(current_input)
            target_embed = target_embed + self.position_embedding(
                torch.arange(current_input.size(1), device=user_history.device)
            )

            # 解码
            output = self.decoder(target_embed, history_embed)
            logits = self.output_layer(output[:, -1, :])

            generated.append(logits)

            # 贪婪解码
            next_item = logits.argmax(dim=-1)
            current_input = torch.cat([current_input, next_item.unsqueeze(1)], dim=1)

        return torch.stack(generated, dim=1)


class RetrievalAugmentedGeneration:
    """
    检索增强生成推荐
    """

    def __init__(self, retriever, generator, item_index):
        self.retriever = retriever
        self.generator = generator
        self.item_index = item_index

    def recommend(self, user_history: List[int], n_candidates: int = 100,
                 n_final: int = 10) -> List[int]:
        """
        RAG 推荐
        """
        # 1. 检索阶段
        # 基于历史检索候选
        candidates = self.retriever.retrieve(user_history, n_candidates)

        # 2. 重排序阶段
        # 使用 LLM/生成模型对候选重排
        ranked_items = self.generator.rerank(user_history, candidates, n_final)

        return ranked_items
```

## 6. 学习总结

### 6.1 核心要点

1. **文本增强**: 利用 LLM 理解物品文本
2. **提示学习**: 设计有效的提示模板
3. **模型融合**: LLM 与传统模型结合
4. **生成式**: 直接生成推荐结果

### 6.2 应用场景

```
场景            方法                优势
────────────────────────────────────────────
冷启动          文本嵌入            利用语义信息
可解释          LLM 生成            自然语言解释
长尾推荐        语义匹配            泛化能力强
多模态          多模态融合          信息丰富
```

### 6.3 最佳实践

```
1. 特征融合: LLM 特征与传统特征互补
2. 高效微调: 使用 PEFT 降低成本
3. 缓存策略: 缓存 LLM 嵌入减少调用
4. 混合架构: 检索 + 生成结合
```
