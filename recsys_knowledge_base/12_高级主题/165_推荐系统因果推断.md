# 推荐系统因果推断 学习文档

## 1. 因果推断基础

### 1.1 为什么需要因果推断？

```
传统推荐的问题:
- 关联 ≠ 因果: 相关性不等于因果关系
- 偏差影响: 选择偏差、曝光偏差
- 反事实问题: 如果推荐了其他物品会怎样？
- 长期效应: 当前推荐的长期影响

因果推断的价值:
- 无偏估计用户真实偏好
- 评估推荐策略的因果效应
- 消除混杂因素影响
```

```python
        """

    @staticmethod

        推荐系统中的因果图:

        User Features ──┬──> Treatment (Exposure)

                        └──> Outcome (Click)

        混杂因素: User Features 同时影响 Treatment 和 Outcome
        """

class ConfoundingBias:
    """
    混杂偏差
    """

    def __init__(self):
        pass

    def naive_ate_estimate(self, y_treated: np.ndarray,
                          y_control: np.ndarray) -> float:
        """
        朴素的 ATE 估计

        ATE_naive = E[Y | T=1] - E[Y | T=0]

        问题: 混杂偏差导致估计有偏
        """
        return np.mean(y_treated) - np.mean(y_control)

    def adjusted_ate_estimate(self, y_treated: np.ndarray,
                             y_control: np.ndarray,
                             treated_weights: np.ndarray,
                             control_weights: np.ndarray) -> float:
        """
        加权调整的 ATE 估计

        使用倾向分数进行逆概率加权
        """
        weighted_treated = np.average(y_treated, weights=treated_weights)
        weighted_control = np.average(y_control, weights=control_weights)

        return weighted_treated - weighted_control
```

## 2. 倾向分数方法

### 2.1 倾向分数估计

```python
from sklearn.linear_model import LogisticRegression


class PropensityScore:
    """
    倾向分数方法

    倾向分数: e(X) = P(T=1 | X)
    用户被曝光/推荐某物品的概率
    """

    def __init__(self):
        self.propensity_model = None

    def fit(self, X: np.ndarray, treatment: np.ndarray):
        """
        训练倾向分数模型

        参数:
            X: 特征矩阵 (n_samples, n_features)
            treatment: 处理变量 (n_samples,) 0/1
        """
        self.propensity_model = LogisticRegression(max_iter=1000)
        self.propensity_model.fit(X, treatment)

    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        预测倾向分数

        返回: e(X) = P(T=1 | X)
        """
        probs = self.propensity_model.predict_proba(X)[:, 1]
        # 截断防止极端值
        return np.clip(probs, 0.01, 0.99)

    def inverse_probability_weighting(self, y: np.ndarray,
                                      treatment: np.ndarray,
                                      propensity: np.ndarray) -> float:
        """
        逆概率加权 (IPW) 估计 ATE

        ATE_IPW = (1/n) * Σ [T*Y/e(X) - (1-T)*Y/(1-e(X))]
        """
        weights_treated = treatment / propensity
        weights_control = (1 - treatment) / (1 - propensity)

        weighted_y_treated = np.sum(weights_treated * y) / np.sum(weights_treated)
        weighted_y_control = np.sum(weights_control * y) / np.sum(weights_control)

        return weighted_y_treated - weighted_y_control

    def check_balance(self, X: np.ndarray, treatment: np.ndarray,
                     propensity: np.ndarray) -> Dict:
        """
        检查协变量平衡

        加权后处理组和对照组的特征分布应相似
        """
        weights_treated = treatment / propensity
        weights_control = (1 - treatment) / (1 - propensity)

        balance_stats = {}
        n_features = X.shape[1]

        for i in range(n_features):
            treated_mean = np.average(X[:, i], weights=weights_treated)
            control_mean = np.average(X[:, i], weights=weights_control)

            treated_std = np.sqrt(np.average(
                (X[:, i] - treated_mean)**2, weights=weights_treated
            ))
            control_std = np.sqrt(np.average(
                (X[:, i] - control_mean)**2, weights=weights_control
            ))

            # 标准化差异
            std_diff = (treated_mean - control_mean) / np.sqrt(
                (treated_std**2 + control_std**2) / 2
            )

            balance_stats[f'feature_{i}'] = {
                'std_diff': std_diff,
                'treated_mean': treated_mean,
                'control_mean': control_mean
            }

        return balance_stats


class DoublyRobustEstimation:
    """
    双重鲁棒估计

    结合 IPW 和结果建模，只要其中一个正确即可
    """

    def __init__(self):
        self.propensity_model = None
        self.outcome_model_treated = None
        self.outcome_model_control = None

    def fit(self, X: np.ndarray, treatment: np.ndarray, y: np.ndarray):
        """训练模型"""
        from sklearn.linear_model import LogisticRegression, Ridge

        # 倾向分数模型
        self.propensity_model = LogisticRegression(max_iter=1000)
        self.propensity_model.fit(X, treatment)

        # 结果模型（分别对处理组和对照组训练）
        treated_idx = treatment == 1
        control_idx = treatment == 0

        self.outcome_model_treated = Ridge()
        self.outcome_model_treated.fit(X[treated_idx], y[treated_idx])

        self.outcome_model_control = Ridge()
        self.outcome_model_control.fit(X[control_idx], y[control_idx])

    def estimate_ate(self, X: np.ndarray, treatment: np.ndarray,
                    y: np.ndarray) -> float:
        """
        双重鲁棒估计

        DR = (1/n) * Σ [
            m1(X) + T*(Y - m1(X))/e(X) -
            m0(X) - (1-T)*(Y - m0(X))/(1-e(X))
        ]

        其中:
        - m1(X) = E[Y | X, T=1] 预测模型
        - m0(X) = E[Y | X, T=0] 预测模型
        - e(X) = P(T=1 | X) 倾向分数
        """
        # 倾向分数
        propensity = self.propensity_model.predict_proba(X)[:, 1]
        propensity = np.clip(propensity, 0.01, 0.99)

        # 预测结果
        m1 = self.outcome_model_treated.predict(X)
        m0 = self.outcome_model_control.predict(X)

        # 双重鲁棒估计
        dr_treated = m1 + treatment * (y - m1) / propensity
        dr_control = m0 + (1 - treatment) * (y - m0) / (1 - propensity)

        ate = np.mean(dr_treated - dr_control)

        return ate
```

### 3.1 无偏推荐学习

```python
import torch
import torch.nn as nn


class UnbiasedRecommender(nn.Module):
    """
    无偏推荐模型

    使用 IPS 损失消除选择偏差
    """

    def __init__(self, n_users: int, n_items: int, embed_dim: int = 64):
        super().__init__()

        self.user_embedding = nn.Embedding(n_users, embed_dim)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        self.user_bias = nn.Embedding(n_users, 1)
        self.item_bias = nn.Embedding(n_items, 1)

    def forward(self, user_ids: torch.Tensor,
               item_ids: torch.Tensor) -> torch.Tensor:
        """预测"""
        user_embed = self.user_embedding(user_ids)
        item_embed = self.item_embedding(item_ids)

        user_bias = self.user_bias(user_ids).squeeze(-1)
        item_bias = self.item_bias(item_ids).squeeze(-1)

        dot = torch.sum(user_embed * item_embed, dim=-1)

        return dot + user_bias + item_bias


class IPSLoss(nn.Module):
    """
    IPS 加权损失
    """

    def __init__(self, clip: float = 10.0):
        super().__init__()
        self.clip = clip

    def forward(self, predictions: torch.Tensor,
               targets: torch.Tensor,
               propensities: torch.Tensor) -> torch.Tensor:
        """
        IPS 损失

        L_IPS = (1/N) * Σ (y - ŷ)² / p(x)

        参数:
            predictions: 预测值
            targets: 真实值
            propensities: 倾向分数（截断后）
        """
        # 截断倾向分数
        clipped_prop = torch.clamp(propensities, min=1.0/self.clip)

        # IPS 权重
        weights = 1.0 / clipped_prop

        # 加权 MSE
        squared_error = (predictions - targets) ** 2
        weighted_loss = weights * squared_error

        return torch.mean(weighted_loss)


class CausalLoss(nn.Module):
    """
    因果推断损失

    结合多种偏差纠正方法
    """

    def __init__(self, lambda_position: float = 0.5,
                 lambda_popularity: float = 0.5):
        super().__init__()
        self.lambda_position = lambda_position
        self.lambda_popularity = lambda_popularity

    def forward(self, predictions: torch.Tensor,
               targets: torch.Tensor,
               position_weights: torch.Tensor,
               popularity_weights: torch.Tensor) -> torch.Tensor:
        """
        综合因果损失

        L = Σ w_pos * w_pop * (y - ŷ)²
        """
        weights = position_weights * popularity_weights
        squared_error = (predictions - targets) ** 2

        return torch.mean(weights * squared_error)
```

### 3.2 反事实评估

```python
class CounterfactualEvaluation:
    """
    反事实评估

    评估"如果使用不同策略会怎样"
    """

    def __init__(self):
        pass

    def off_policy_evaluation(self, logged_data: List[Dict],
                              target_policy,
                              propensity_scores: np.ndarray) -> float:
        """
        离线策略评估 (Off-Policy Evaluation, OPE)

        使用日志数据评估新策略

        参数:
            logged_data: 日志数据 [{
                'context': x,
                'action': a,
                'reward': r,
                'propensity': p
            }, ...]
            target_policy: 目标策略 π(a|x)
            propensity_scores: 记录策略的倾向分数
        """
        estimates = []

        for item in logged_data:
            context = item['context']
            action = item['action']
            reward = item['reward']
            logged_prop = item['propensity']

            # 目标策略在该动作上的概率
            target_prop = target_policy.get_propensity(context, action)

            # 重要性采样权重
            is_weight = target_prop / logged_prop

            estimates.append(is_weight * reward)

        # IPS 估计
        return np.mean(estimates)

    def doubly_robust_ope(self, logged_data: List[Dict],
                         target_policy,
                         reward_model) -> float:
        """
        双重鲁棒 OPE

        结合 IPS 和奖励模型
        """
        estimates = []

        for item in logged_data:
            context = item['context']
            action = item['action']
            reward = item['reward']
            logged_prop = item['propensity']

            target_prop = target_policy.get_propensity(context, action)
            is_weight = target_prop / logged_prop

            # 奖励模型预测
            predicted_reward = reward_model.predict(context, action)

            # 双重鲁棒估计
            dr_estimate = (predicted_reward +
                          is_weight * (reward - predicted_reward))

            estimates.append(dr_estimate)

        return np.mean(estimates)


class PolicyEvaluator:
    """
    策略评估器
    """

    def __init__(self):
        self.ips_estimate = None
        self.dr_estimate = None

    def evaluate_policy(self, data: List[Dict],
                       new_policy,
                       reward_model=None) -> Dict:
        """
        评估新策略
        """
        ope = CounterfactualEvaluation()

        # IPS 估计
        ips_value = ope.off_policy_evaluation(
            data, new_policy,
            np.array([d['propensity'] for d in data])
        )

        result = {'ips_estimate': ips_value}

        # 双重鲁棒估计
        if reward_model:
            dr_value = ope.doubly_robust_ope(data, new_policy, reward_model)
            result['dr_estimate'] = dr_value

        return result
```

## 4. 因果推荐模型

### 4.1 去混杂推荐

```python
class DeconfoundedRecommender:
    """
    去混杂推荐

    消除用户偏好等混杂因素
    """

    def __init__(self, n_users: int, n_items: int, embed_dim: int = 64,
                 n_confounders: int = 10):
        self.n_users = n_users
        self.n_items = n_items
        self.embed_dim = embed_dim
        self.n_confounders = n_confounders

        # 用户嵌入
        self.user_embedding = np.random.normal(0, 0.1, (n_users, embed_dim))
        # 物品嵌入
        self.item_embedding = np.random.normal(0, 0.1, (n_items, embed_dim))
        # 混杂因子
        self.confounders = np.random.normal(0, 0.1, (n_confounders, embed_dim))

    def fit(self, interactions: List[Tuple[int, int, float]],
            n_epochs: int = 20, lr: float = 0.01):
        """
        训练

        使用后门准则去混杂
        """
        for epoch in range(n_epochs):
            np.random.shuffle(interactions)
            total_loss = 0

            for user_id, item_id, rating in interactions:
                # 预测（考虑混杂因子）
                user_vec = self.user_embedding[user_id]
                item_vec = self.item_embedding[item_id]

                # 混杂因子的影响
                confounder_effect = 0
                for c in self.confounders:
                    confounder_effect += (np.dot(user_vec, c) *
                                         np.dot(item_vec, c))
                confounder_effect /= self.n_confounders

                # 去混杂的预测
                pred = np.dot(user_vec, item_vec) - confounder_effect

                # 误差
                error = rating - pred
                total_loss += error ** 2

                # 更新
                grad_user = -error * item_vec
                grad_item = -error * user_vec

                self.user_embedding[user_id] -= lr * grad_user
                self.item_embedding[item_id] -= lr * grad_item

            rmse = np.sqrt(total_loss / len(interactions))
            print(f"Epoch {epoch+1}: RMSE = {rmse:.4f}")

    def predict(self, user_id: int, item_id: int) -> float:
        """预测（去混杂）"""
        user_vec = self.user_embedding[user_id]
        item_vec = self.item_embedding[item_id]

        # 混杂因子的影响
        confounder_effect = 0
        for c in self.confounders:
            confounder_effect += np.dot(user_vec, c) * np.dot(item_vec, c)
        confounder_effect /= self.n_confounders

        return np.dot(user_vec, item_vec) - confounder_effect


class InstrumentalVariableRecommender:
    """
    工具变量推荐

    使用工具变量消除内生性
    """

    def __init__(self):
        self.instrument_strength = None
        self.treatment_effect = None

    def fit(self, instrument: np.ndarray, treatment: np.ndarray,
           outcome: np.ndarray):
        """
        两阶段最小二乘法 (2SLS)

        第一阶段: T = γZ + ε
        第二阶段: Y = βT̂ + η

        参数:
            instrument: 工具变量 Z (如随机化曝光)
            treatment: 处理变量 T (如实际曝光)
            outcome: 结果变量 Y (如点击)
        """
        # 第一阶段：预测处理变量
        # T = γZ
        self.instrument_strength = np.linalg.lstsq(
            instrument.reshape(-1, 1), treatment, rcond=None
        )[0][0]

        treatment_predicted = instrument * self.instrument_strength

        # 第二阶段：估计因果效应
        # Y = βT̂
        self.treatment_effect = np.linalg.lstsq(
            treatment_predicted.reshape(-1, 1), outcome, rcond=None
        )[0][0]

    def get_causal_effect(self) -> float:
        """获取因果效应"""
        return self.treatment_effect
```

```python
class CausalDiscovery:
    """
    因果结构学习

    从数据中发现因果关系
    """

    def __init__(self):
        self.causal_graph = None

    def pc_algorithm(self, data: np.ndarray,
                    alpha: float = 0.05) -> Dict:
        """
        PC 算法

        基于条件独立性检验学习因果图
        """
        from scipy import stats

        n_vars = data.shape[1]

        # 初始化完全图
        adj_matrix = np.ones((n_vars, n_vars)) - np.eye(n_vars)

        # 阶段1: 移除条件独立的边
        for i in range(n_vars):
            for j in range(i+1, n_vars):
                # 检验无条件独立性
                corr, p_value = stats.pearsonr(data[:, i], data[:, j])

                if p_value > alpha:
                    adj_matrix[i, j] = 0
                    adj_matrix[j, i] = 0

        self.causal_graph = adj_matrix

        return {'adjacency_matrix': adj_matrix}

    def conditional_independence_test(self, x: np.ndarray, y: np.ndarray,
                                     z: np.ndarray, alpha: float = 0.05) -> bool:
        """
        条件独立性检验

        X ⊥ Y | Z ?
        """
        from sklearn.linear_model import LinearRegression

        # 偏相关检验
        # 1. X 对 Z 回归的残差
        model_x = LinearRegression().fit(z.reshape(-1, 1), x)
        res_x = x - model_x.predict(z.reshape(-1, 1))

        # 2. Y 对 Z 回归的残差
        model_y = LinearRegression().fit(z.reshape(-1, 1), y)
        res_y = y - model_y.predict(z.reshape(-1, 1))

        # 3. 残差相关性
        from scipy import stats
        corr, p_value = stats.pearsonr(res_x, res_y)

        return p_value > alpha  # 独立?
```

## 6. 因果推断应用

### 6.1 推荐策略评估

```python
class CausalRecommendationEvaluator:
    """
    因果推荐评估器
    """

    def __init__(self):
        self.ips_estimator = PropensityScore()
        self.dr_estimator = DoublyRobustEstimation()

    def evaluate_recommendation_effect(self,
                                       features: np.ndarray,
                                       exposures: np.ndarray,
                                       clicks: np.ndarray,
                                       propensities: np.ndarray) -> Dict:
        """
        评估推荐策略的因果效应

        参数:
            features: 用户-物品特征
            exposures: 是否曝光
            clicks: 是否点击
            propensities: 曝光倾向分数
        """
        # IPS 估计
        self.ips_estimator.fit(features, exposures)
        estimated_prop = self.ips_estimator.predict(features)

        ate_ips = self.ips_estimator.inverse_probability_weighting(
            clicks, exposures, estimated_prop
        )

        # 双重鲁棒估计
        self.dr_estimator.fit(features, exposures, clicks)
        ate_dr = self.dr_estimator.estimate_ate(features, exposures, clicks)

        return {
            'ate_ips': ate_ips,
            'ate_dr': ate_dr,
            'naive_ate': np.mean(clicks[exposures==1]) - np.mean(clicks[exposures==0])
        }
```

## 7. 学习总结

### 7.1 核心要点

1. **因果 vs 关联**: 区分相关性和因果性
2. **倾向分数**: 估计处理概率，进行加权
3. **双重鲁棒**: 结合 IPW 和结果建模
4. **反事实评估**: 评估未实施策略的效果

### 7.2 方法对比

```
方法            优势                劣势
──────────────────────────────────────────
IPS            简单直观            高方差
双重鲁棒       鲁棒性强            需要两个模型
工具变量       处理未观测混杂      难找合适IV
回归调整       低方差              模型依赖
```

### 7.3 最佳实践

```
1. 倾向分数截断: 避免极端权重
2. 平衡性检验: 确保加权后平衡
3. 敏感性分析: 检验假设稳健性
4. 多方法验证: 使用多种方法交叉验证
```
