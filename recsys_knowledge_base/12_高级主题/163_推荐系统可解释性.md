# 推荐系统可解释性 学习文档

## 1. 问题定义

### 1.1 什么是可解释推荐？

可解释推荐不仅给出推荐结果，还提供推荐理由，帮助用户理解为什么被推荐这些物品。

### 1.2 为什么需要可解释性？

| 目的 | 说明 |
|------|------|
| 透明性 | 用户了解推荐原因 |
| 信任度 | 增强用户信任 |
| 决策辅助 | 帮助用户决策 |
| 调试 | 帮助开发者优化 |

### 1.3 可解释性分类

```
1. 事后解释 (Post-hoc): 推荐后生成解释
2. 内在解释 (Intrinsic): 模型本身可解释
3. 全局解释: 整体模型行为
4. 局部解释: 单个推荐解释
```

## 2. 解释类型

### 2.1 基于内容的解释

```python
from typing import List, Dict, Tuple
import numpy as np


class ContentBasedExplainer:
    """
    基于内容的解释生成器
    """

    def __init__(self, item_features: Dict[str, Dict]):
        """
        参数:
            item_features: {item_id: {feature: value}}
        """
        self.item_features = item_features

    def explain(self, user_profile: Dict[str, float],
               item_id: str, top_k: int = 3) -> List[Tuple[str, float]]:
        """
        生成解释

        参数:
            user_profile: 用户画像 {feature: preference}
            item_id: 推荐物品
            top_k: 返回 top_k 个解释特征

        返回:
            [(feature_name, contribution), ...]
        """
        if item_id not in self.item_features:
            return []

        item_features = self.item_features[item_id]
        contributions = {}

        for feature, value in item_features.items():
            if feature in user_profile:
                # 特征匹配度 × 用户偏好
                contributions[feature] = value * user_profile.get(feature, 0)

        # 排序返回
        sorted_contributions = sorted(
            contributions.items(),
            key=lambda x: abs(x[1]),
            reverse=True
        )[:top_k]

        return sorted_contributions

    def format_explanation(self, item_id: str,
                          contributions: List[Tuple[str, float]]) -> str:
        """格式化解释文本"""
        if not contributions:
            return f"为您推荐了 {item_id}"

        top_feature = contributions[0][0]
        return f"因为您喜欢 {top_feature} 相关的内容，为您推荐了 {item_id}"


class TagBasedExplainer:
    """
    基于标签的解释
    """

    def __init__(self, item_tags: Dict[str, List[str]]):
        self.item_tags = item_tags

    def explain(self, user_interest_tags: List[str],
               item_id: str) -> Tuple[List[str], str]:
        """
        基于标签生成解释

        返回:
            (匹配的标签列表, 解释文本)
        """
        if item_id not in self.item_tags:
            return [], f"为您推荐"

        item_tags = set(self.item_tags[item_id])
        user_tags = set(user_interest_tags)

        matched_tags = list(item_tags & user_tags)

        if matched_tags:
            text = f"因为您对 {', '.join(matched_tags[:3])} 感兴趣"
        else:
            text = "为您推荐"

        return matched_tags, text
```

### 2.2 基于协同过滤的解释

```python
class CFExplainer:
    """
    协同过滤解释器
    """

    def __init__(self, user_items: Dict[int, List[int]],
                 item_users: Dict[int, List[int]]):
        self.user_items = user_items
        self.item_users = item_users

    def explain_user_based(self, user_id: int, item_id: int,
                          similar_users: List[Tuple[int, float]]) -> str:
        """
        UserCF 解释

        参数:
            user_id: 目标用户
            item_id: 推荐物品
            similar_users: 相似用户列表 [(user_id, similarity), ...]
        """
        # 找到喜欢该物品的相似用户
        item_fans = set(self.item_users.get(item_id, []))

        for sim_user, sim_score in similar_users[:5]:
            if sim_user in item_fans:
                return f"和您兴趣相似的用户也喜欢这个"

        return "根据您的兴趣推荐"

    def explain_item_based(self, user_id: int, item_id: int,
                          user_history: List[int],
                          similar_items: List[Tuple[int, float]]) -> str:
        """
        ItemCF 解释

        参数:
            user_history: 用户历史行为
            similar_items: 相似物品 [(item_id, similarity), ...]
        """
        history_set = set(user_history)

        for sim_item, sim_score in similar_items[:5]:
            if sim_item in history_set:
                return f"因为您浏览过相似的商品"

        return "根据您的行为推荐"
```

## 3. 注意力解释

### 3.1 基于注意力权重的解释

```python
import torch
import torch.nn as nn


class AttentionExplainer:
    """
    基于注意力权重的解释
    """

    def __init__(self, model):
        self.model = model
        self.attention_weights = None

        # 注册钩子
        self._register_hooks()

    def _register_hooks(self):
        """注册钩子获取注意力权重"""
        def hook(module, input, output):
            if isinstance(output, tuple) and len(output) > 1:
                self.attention_weights = output[1]

        # 找到注意力层并注册
        for name, module in self.model.named_modules():
            if 'attention' in name.lower():
                module.register_forward_hook(hook)

    def explain(self, features: Dict, item_id: int) -> Dict[str, float]:
        """
        基于注意力权重生成解释

        返回:
            {feature_name: importance_weight}
        """
        # 前向传播
        with torch.no_grad():
            _ = self.model(features)

        if self.attention_weights is None:
            return {}

        # 获取注意力权重
        weights = self.attention_weights.cpu().numpy()

        # 映射到特征名
        feature_importance = {}
        feature_names = features.get('feature_names', [])

        for i, name in enumerate(feature_names):
            if i < len(weights):
                feature_importance[name] = float(weights[i])

        return feature_importance
```

## 4. SHAP 解释

```python
class SHAPExplainer:
    """
    基于 SHAP 的解释器
    """

    def __init__(self, model, background_data):
        self.model = model
        self.background_data = background_data

    def explain_instance(self, instance: np.ndarray,
                        feature_names: List[str]) -> Dict[str, float]:
        """
        使用 SHAP 解释单个实例

        参数:
            instance: 单个样本特征
            feature_names: 特征名列表

        返回:
            {feature_name: shap_value}
        """
        try:
            import shap

            # 创建解释器
            explainer = shap.KernelExplainer(
                self._predict_proba,
                self.background_data
            )

            # 计算 SHAP 值
            shap_values = explainer.shap_values(instance)

            # 映射到特征名
            explanations = {}
            for i, name in enumerate(feature_names):
                if i < len(shap_values[0]):
                    explanations[name] = float(shap_values[0][i])

            return explanations

        except ImportError:
            return {}

    def _predict_proba(self, X):
        """预测概率包装"""
        return self.model.predict_proba(X)


class FeatureImportanceExplainer:
    """
    基于特征重要性的解释
    """

    def __init__(self, model):
        self.model = model

    def get_global_importance(self, feature_names: List[str]) -> Dict[str, float]:
        """获取全局特征重要性"""
        if hasattr(self.model, 'feature_importances_'):
            importances = self.model.feature_importances_
            return dict(zip(feature_names, importances.tolist()))
        return {}

    def get_local_importance(self, features: np.ndarray,
                            feature_names: List[str]) -> Dict[str, float]:
        """获取局部特征重要性（基于置换）"""
        base_pred = self.model.predict_proba(features)[0, 1]
        importances = {}

        for i, name in enumerate(feature_names):
            # 置换特征
            permuted = features.copy()
            permuted[0, i] = np.random.permutation(features[:, i])[0]
            perm_pred = self.model.predict_proba(permuted)[0, 1]

            importances[name] = abs(base_pred - perm_pred)

        return importances
```

## 5. 自然语言解释

### 5.1 模板生成

```python
class TemplateExplainer:
    """
    基于模板的自然语言解释
    """

    def __init__(self):
        self.templates = {
            'similar_user': "和您兴趣相似的用户也{action}了这个",
            'similar_item': "因为您{action}过相似的商品",
            'popular': "这是热门{category}",
            'recent': "根据您最近{action}的{item}推荐",
            'tag': "因为您对{tags}感兴趣",
            'price': "符合您的价格偏好",
        }

    def generate(self, explanation_type: str, **kwargs) -> str:
        """生成解释"""
        template = self.templates.get(explanation_type, "为您推荐")

        try:
            return template.format(**kwargs)
        except KeyError:
            return template


class ExplanationGenerator:
    """
    综合解释生成器
    """

    def __init__(self):
        self.template_explainer = TemplateExplainer()
        self.explanation_strategies = []

    def add_strategy(self, strategy, priority: int = 0):
        """添加解释策略"""
        self.explanation_strategies.append((strategy, priority))
        self.explanation_strategies.sort(key=lambda x: x[1], reverse=True)

    def generate(self, context: Dict) -> str:
        """生成最佳解释"""
        for strategy, _ in self.explanation_strategies:
            explanation = strategy(context)
            if explanation:
                return explanation

        return "为您推荐"


# 使用示例
def create_explanation_generator():
    """创建解释生成器"""
    generator = ExplanationGenerator()

    # 添加策略（按优先级）
    generator.add_strategy(lambda ctx: ctx.get('tag_explanation'), priority=10)
    generator.add_strategy(lambda ctx: ctx.get('cf_explanation'), priority=8)
    generator.add_strategy(lambda ctx: ctx.get('content_explanation'), priority=5)

    return generator
```

## 6. 评估可解释性

```python
class ExplainabilityEvaluator:
    """
    可解释性评估
    """

    @staticmethod
    def user_study_score(explanations: List[str], ratings: List[int]) -> float:
        """用户研究评分"""
        return np.mean(ratings)

    @staticmethod
    def explanation_coverage(features: List[str],
                            explanation: List[str]) -> float:
        """解释覆盖率"""
        explained = set(explanation)
        total = set(features)
        return len(explained & total) / len(total) if total else 0

    @staticmethod
    def faithfulness_score(model, features: np.ndarray,
                          explanation_features: List[int]) -> float:
        """
        忠实度评估

        移除解释中最重要的特征后，预测变化多少
        """
        original_pred = model.predict_proba(features)[0, 1]

        # 移除解释特征
        modified = features.copy()
        for idx in explanation_features:
            modified[0, idx] = 0

        modified_pred = model.predict_proba(modified)[0, 1]

        return abs(original_pred - modified_pred)
```

## 7. 学习总结

### 7.1 核心要点

1. **内容解释**: 基于物品特征
2. **协同解释**: 基于相似用户/物品
3. **注意力解释**: 基于模型权重
4. **SHAP解释**: 基于特征贡献

### 7.2 最佳实践

```
1. 简洁明了: 一句话解释
2. 真实可信: 解释要准确
3. 用户视角: 使用用户能理解的术语
4. 个性化: 解释要因人而异
```
