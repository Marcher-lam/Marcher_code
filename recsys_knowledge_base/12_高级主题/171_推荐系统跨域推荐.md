# 推荐系统跨域推荐 学习文档

## 1. 跨域推荐概述

### 1.1 什么是跨域推荐？

```
跨域推荐 (Cross-Domain Recommendation):
- 利用源域 (Source Domain) 数据辅助目标域 (Target Domain)
- 解决数据稀疏、冷启动问题
- 知识迁移与共享

典型场景:
- 源域: 电商购买 → 目标域: 内容推荐
- 源域: 电影评分 → 目标域: 书籍推荐
- 源域: App行为 → 目标域: 广告点击
```

## 2. 跨域方法

### 2.1 基于映射的方法

```python
import torch
import torch.nn as nn


class DomainMappingModel(nn.Module):
    """
    域映射模型

    学习将源域表示映射到目标域
    """

    def __init__(self, source_dim: int, target_dim: int,
                 hidden_dim: int = 128):
        super().__init__()

        # 源域编码器
        self.source_encoder = nn.Sequential(
            nn.Linear(source_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim)
        )

        # 域映射
        self.domain_mapper = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, target_dim)
        )

        # 目标域预测器
        self.target_predictor = nn.Sequential(
            nn.Linear(target_dim, target_dim // 2),
            nn.ReLU(),
            nn.Linear(target_dim // 2, 1)
        )

    def encode_source(self, source_features: torch.Tensor) -> torch.Tensor:
        """编码源域特征"""
        return self.source_encoder(source_features)

    def map_to_target(self, source_repr: torch.Tensor) -> torch.Tensor:
        """映射到目标域"""
        return self.domain_mapper(source_repr)

    def forward(self, source_features: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        source_repr = self.encode_source(source_features)
        target_repr = self.map_to_target(source_repr)
        return self.target_predictor(target_repr)


class CMF(nn.Module):
    """
    Collective Matrix Factorization (CMF)

    联合矩阵分解，共享用户表示
    """

    def __init__(self, n_users: int, n_source_items: int,
                 n_target_items: int, embed_dim: int = 64):
        super().__init__()

        # 共享用户嵌入
        self.user_embedding = nn.Embedding(n_users, embed_dim)

        # 源域物品嵌入
        self.source_item_embedding = nn.Embedding(n_source_items, embed_dim)

        # 目标域物品嵌入
        self.target_item_embedding = nn.Embedding(n_target_items, embed_dim)

        # 偏置
        self.user_bias = nn.Embedding(n_users, 1)
        self.source_item_bias = nn.Embedding(n_source_items, 1)
        self.target_item_bias = nn.Embedding(n_target_items, 1)

    def forward_source(self, user_ids: torch.Tensor,
                      item_ids: torch.Tensor) -> torch.Tensor:
        """源域预测"""
        user_embed = self.user_embedding(user_ids)
        item_embed = self.source_item_embedding(item_ids)

        user_bias = self.user_bias(user_ids)
        item_bias = self.source_item_bias(item_ids)

        pred = torch.sum(user_embed * item_embed, dim=-1, keepdim=True)
        return pred + user_bias + item_bias

    def forward_target(self, user_ids: torch.Tensor,
                      item_ids: torch.Tensor) -> torch.Tensor:
        """目标域预测"""
        user_embed = self.user_embedding(user_ids)
        item_embed = self.target_item_embedding(item_ids)

        user_bias = self.user_bias(user_ids)
        item_bias = self.target_item_bias(item_ids)

        pred = torch.sum(user_embed * item_embed, dim=-1, keepdim=True)
        return pred + user_bias + item_bias
```

### 2.2 基于迁移学习的方法

```python
class TransferLearningRecommender(nn.Module):
    """
    基于迁移学习的跨域推荐
    """

    def __init__(self, source_n_items: int, target_n_items: int,
                 embed_dim: int = 64, hidden_dim: int = 128):
        super().__init__()

        # 源域模型
        self.source_item_embed = nn.Embedding(source_n_items, embed_dim)
        self.source_encoder = nn.Sequential(
            nn.Linear(embed_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

        # 目标域模型
        self.target_item_embed = nn.Embedding(target_n_items, embed_dim)
        self.target_encoder = nn.Sequential(
            nn.Linear(embed_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

        # 共享层
        self.shared_layer = nn.Sequential(
            nn.Linear(embed_dim, embed_dim),
            nn.ReLU()
        )

        # 迁移层
        self.transfer_layer = nn.Linear(embed_dim, embed_dim)

    def pretrain_source(self, user_history: List[int]) -> torch.Tensor:
        """源域预训练"""
        items = torch.tensor(user_history)
        item_embeds = self.source_item_embed(items)
        return self.source_encoder(item_embeds.mean(dim=0))

    def transfer_to_target(self, source_repr: torch.Tensor) -> torch.Tensor:
        """迁移到目标域"""
        return self.transfer_layer(source_repr)

    def forward(self, user_history: List[int],
               target_item: int,
               domain: str = 'target') -> torch.Tensor:
        """前向传播"""
        if domain == 'source':
            # 源域预测
            source_repr = self.pretrain_source(user_history)
            target_embed = self.source_item_embed(torch.tensor([target_item]))
            return torch.sum(source_repr * target_embed, dim=-1)
        else:
            # 目标域预测
            user_repr = self.pretrain_source(user_history)
            transferred = self.transfer_to_target(user_repr)

            target_embed = self.target_item_embed(torch.tensor([target_item]))
            target_embed = self.target_encoder(target_embed)

            return torch.sum(transferred * target_embed.squeeze(0), dim=-1)


class CDAETransfer(nn.Module):
    """
    基于 Denoising Autoencoder 的跨域迁移
    """

    def __init__(self, n_source_items: int, n_target_items: int,
                 hidden_dim: int = 128, noise_ratio: float = 0.3):
        super().__init__()

        self.noise_ratio = noise_ratio

        # 源域编码器
        self.source_encoder = nn.Sequential(
            nn.Linear(n_source_items, hidden_dim),
            nn.ReLU()
        )

        # 源域解码器
        self.source_decoder = nn.Sequential(
            nn.Linear(hidden_dim, n_source_items),
            nn.Sigmoid()
        )

        # 目标域编码器
        self.target_encoder = nn.Sequential(
            nn.Linear(n_target_items, hidden_dim),
            nn.ReLU()
        )

        # 目标域解码器
        self.target_decoder = nn.Sequential(
            nn.Linear(hidden_dim, n_target_items),
            nn.Sigmoid()
        )

        # 迁移矩阵
        self.transfer_matrix = nn.Linear(hidden_dim, hidden_dim, bias=False)

    def add_noise(self, x: torch.Tensor) -> torch.Tensor:
        """添加噪声"""
        mask = torch.rand_like(x) > self.noise_ratio
        return x * mask.float()

    def encode_source(self, x: torch.Tensor) -> torch.Tensor:
        """编码源域"""
        noisy_x = self.add_noise(x)
        return self.source_encoder(noisy_x)

    def decode_source(self, h: torch.Tensor) -> torch.Tensor:
        """解码源域"""
        return self.source_decoder(h)

    def transfer(self, source_h: torch.Tensor) -> torch.Tensor:
        """迁移"""
        return self.transfer_matrix(source_h)

    def decode_target(self, h: torch.Tensor) -> torch.Tensor:
        """解码目标域"""
        return self.target_decoder(h)
```

### 2.3 域对抗方法

```python
class DomainAdversarialNetwork(nn.Module):
    """
    域对抗网络 (DANN)

    学习域不变表示
    """

    def __init__(self, feature_dim: int, hidden_dim: int = 128,
                 n_domains: int = 2):
        super().__init__()

        # 特征提取器（共享）
        self.feature_extractor = nn.Sequential(
            nn.Linear(feature_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )

        # 推荐预测器
        self.predictor = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1)
        )

        # 域分类器
        self.domain_classifier = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, n_domains)
        )

    def forward(self, features: torch.Tensor,
               alpha: float = 1.0) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        前向传播

        参数:
            features: 输入特征
            alpha: 梯度反转系数
        """
        # 提取特征
        shared_features = self.feature_extractor(features)

        # 推荐预测
        pred = self.predictor(shared_features)

        # 域分类（梯度反转）
        reverse_features = GradientReversalFunction.apply(shared_features, alpha)
        domain_pred = self.domain_classifier(reverse_features)

        return pred, domain_pred


class GradientReversalFunction(torch.autograd.Function):
    """梯度反转函数"""

    @staticmethod
    def forward(ctx, x, alpha):
        ctx.alpha = alpha
        return x.view_as(x)

    @staticmethod
    def backward(ctx, grad_output):
        return grad_output.neg() * ctx.alpha, None
```

## 4. 学习总结

### 4.1 核心要点

1. **知识迁移**: 源域知识辅助目标域
2. **用户重叠**: 重叠用户是桥梁
3. **域不变性**: 学习域不变表示
4. **负迁移**: 避免源域噪声

### 4.2 方法对比

```
方法            优势                劣势
──────────────────────────────────────────
CMF             简单有效            需要用户重叠
域映射          灵活                需要映射学习
迁移学习        知识复用            需要预训练
域对抗          域不变性            训练不稳定
```

### 4.3 最佳实践

```
1. 用户重叠: 确保有足够的重叠用户
2. 特征对齐: 源域和目标域特征对齐
3. 渐进迁移: 逐步迁移知识
4. 效果评估: 在目标域验证效果
```
