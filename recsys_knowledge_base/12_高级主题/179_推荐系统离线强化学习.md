# 推荐系统离线强化学习 学习文档

## 1. 离线强化学习概述

### 1.1 为什么需要离线强化学习？

```
在线 RL 的问题:
- 需要与环境交互
- 探索可能影响用户体验
- 成本高、风险大

离线 RL (Offline RL) 优势:
- 只用历史数据训练
- 不需要在线交互
- 安全、低成本

应用场景:
- 推荐系统 (大量历史日志)
- 广告系统
- 医疗决策
- 机器人

挑战:
- 分布偏移 (Distribution Shift)
- 过估计 (Overestimation)
- 策略偏离历史数据太远
```

### 1.2 离线 RL vs 在线 RL

```python
"""
离线 RL vs 在线 RL

在线 RL:
- 策略 π 与环境交互收集数据
- 数据分布随策略变化
- 可以探索新的状态-动作对

离线 RL:
- 只有固定的历史数据 D = {(s, a, r, s')}
- 数据由行为策略 β 产生
- 不能探索，只能利用已有数据

核心问题:
- 当学习策略 π 与行为策略 β 差异大时
- Q 值估计可能不准确
- 导致策略性能下降
"""

from typing import List, Dict, Tuple, Optional
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from collections import defaultdict


class OfflineRLDataset:
    """
    离线 RL 数据集

    从推荐系统日志构建
    """

    def __init__(self, max_size: int = 1000000):
        self.max_size = max_size
        self.buffer = []

    def add(self, state: np.ndarray, action: int,
            reward: float, next_state: np.ndarray, done: bool):
        """添加一条经验"""
        if len(self.buffer) >= self.max_size:
            self.buffer.pop(0)
        self.buffer.append((state, action, reward, next_state, done))

    def add_from_logs(self, logs: List[Dict]):
        """
        从推荐日志构建数据集

        logs: [
            {
                'user_features': ...,
                'recommended_items': [...],
                'clicked_item': int,
                'reward': float,
                'next_user_features': ...
            },
            ...
        ]
        """
        for log in logs:
            state = self._extract_state(log['user_features'])
            action = log['clicked_item']  # 或推荐列表位置
            reward = log['reward']
            next_state = self._extract_state(log.get('next_user_features', {}))
            done = log.get('session_end', False)

            self.add(state, action, reward, next_state, done)

    def _extract_state(self, features: Dict) -> np.ndarray:
        """提取状态向量"""
        # 实际实现需要根据特征设计
        return np.array(list(features.values()))

    def sample(self, batch_size: int) -> Dict[str, torch.Tensor]:
        """采样"""
        indices = np.random.choice(len(self.buffer), batch_size, replace=False)

        states, actions, rewards, next_states, dones = [], [], [], [], []

        for idx in indices:
            s, a, r, ns, d = self.buffer[idx]
            states.append(s)
            actions.append(a)
            rewards.append(r)
            next_states.append(ns)
            dones.append(d)

        return {
            'states': torch.tensor(np.array(states), dtype=torch.float32),
            'actions': torch.tensor(actions, dtype=torch.long),
            'rewards': torch.tensor(rewards, dtype=torch.float32),
            'next_states': torch.tensor(np.array(next_states), dtype=torch.float32),
            'dones': torch.tensor(dones, dtype=torch.float32)
        }
```

## 2. BCQ 算法

### 2.1 BCQ 原理与实现

```python
class BCQ(nn.Module):
    """
    Batch-Constrained Q-learning (BCQ)

    核心思想:
    - 限制动作选择在行为策略的动作范围内
    - 使用生成模型建模行为策略
    - 只选择高概率的动作

    公式:
    π(s) = argmax_{a ∈ A_β(s)} Q(s, a)

    其中 A_β(s) 是行为策略在状态 s 下可能采取的动作集合
    """

    def __init__(self,
                 state_dim: int,
                 n_actions: int,
                 hidden_dim: int = 256,
                 n_quantiles: int = 10,
                 n_candidates: int = 100):
        super().__init__()

        self.n_actions = n_actions
        self.n_candidates = n_candidates

        # Q 网络
        self.q_network = nn.Sequential(
            nn.Linear(state_dim + 1, hidden_dim),  # state + action (离散用embedding)
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        # 动作嵌入 (用于离散动作)
        self.action_embedding = nn.Embedding(n_actions, hidden_dim)

        # 生成模型: 从状态生成可能的动作
        # 使用 VAE 或简单的条件生成网络
        self.generator = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_actions)  # 输出动作 logits
        )

        # 扰动网络: 对生成的动作进行小幅调整
        self.perturbation = nn.Sequential(
            nn.Linear(state_dim + hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1),
            nn.Tanh()  # 输出扰动量
        )

    def generate_actions(self, state: torch.Tensor) -> torch.Tensor:
        """
        生成候选动作

        返回: (batch, n_candidates) 候选动作索引
        """
        logits = self.generator(state)  # (batch, n_actions)

        # 采样多个候选动作
        probs = F.softmax(logits, dim=-1)

        # 对于离散动作，选择概率最高的 n_candidates 个
        top_probs, top_actions = torch.topk(probs, self.n_candidates, dim=-1)

        return top_actions

    def select_action(self, state: torch.Tensor) -> torch.Tensor:
        """
        选择动作

        从候选动作中选择 Q 值最高的
        """
        # 生成候选动作
        candidate_actions = self.generate_actions(state)  # (batch, n_candidates)

        batch_size = state.size(0)

        # 计算每个候选动作的 Q 值
        q_values = []
        for i in range(self.n_candidates):
            actions = candidate_actions[:, i]
            # 构建输入
            action_embed = self.action_embedding(actions)
            # 这里简化，实际应该用 Q 网络
            q = self._compute_q(state, actions)
            q_values.append(q)

        q_values = torch.stack(q_values, dim=1)  # (batch, n_candidates)

        # 选择 Q 值最高的动作
        best_idx = q_values.argmax(dim=1)
        best_actions = candidate_actions[torch.arange(batch_size), best_idx]

        return best_actions

    def _compute_q(self, state: torch.Tensor, action: torch.Tensor) -> torch.Tensor:
        """计算 Q 值"""
        # 简化实现
        action_embed = self.action_embedding(action)
        return self.q_network(state).squeeze(-1)


class BCQTrainer:
    """
    BCQ 训练器
    """

    def __init__(self, model: BCQ, lr: float = 1e-3,
                 gamma: float = 0.99, tau: float = 0.005):
        self.model = model
        self.gamma = gamma
        self.tau = tau

        # 目标网络
        self.target_q = nn.Sequential(
            nn.Linear(model.q_network[0].in_features, 256),
            nn.ReLU(),
            nn.Linear(256, 256),
            nn.ReLU(),
            nn.Linear(256, 1)
        )

        # 复制权重
        self.target_q.load_state_dict(model.q_network.state_dict())

        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    def train_step(self, batch: Dict[str, torch.Tensor]) -> float:
        """训练一步"""
        states = batch['states']
        actions = batch['actions']
        rewards = batch['rewards']
        next_states = batch['next_states']
        dones = batch['dones']

        # 当前 Q 值
        current_q = self.model._compute_q(states, actions)

        # 目标 Q 值
        with torch.no_grad():
            # 使用生成模型采样下一个动作
            next_actions = self.model.select_action(next_states)
            next_q = self.model._compute_q(next_states, next_actions)

            target_q = rewards + (1 - dones) * self.gamma * next_q

        # Q 损失
        q_loss = F.mse_loss(current_q, target_q)

        # 生成模型损失 (模仿行为策略)
        generator_logits = self.model.generator(states)
        imitation_loss = F.cross_entropy(generator_logits, actions)

        # 总损失
        total_loss = q_loss + imitation_loss

        # 反向传播
        self.optimizer.zero_grad()
        total_loss.backward()
        self.optimizer.step()

        # 软更新目标网络
        self._soft_update()

        return total_loss.item()

    def _soft_update(self):
        """软更新目标网络"""
        for target_param, param in zip(
            self.target_q.parameters(),
            self.model.q_network.parameters()
        ):
            target_param.data.copy_(
                self.tau * param.data + (1 - self.tau) * target_param.data
            )
```

## 3. CQL 算法

### 3.1 Conservative Q-Learning

```python
class CQL(nn.Module):
    """
    Conservative Q-Learning (CQL)

    核心思想:
    - 对 Q 值进行保守估计
    - 惩罚对未见动作的高 Q 值估计
    - 最小化对数据集中动作的 Q 值

    损失函数:
    L_CQL = α * (E_{s~D, a~π}[Q(s,a)] - E_{s~D, a~β}[Q(s,a)])

    结合标准 TD 学习
    """

    def __init__(self,
                 state_dim: int,
                 n_actions: int,
                 hidden_dim: int = 256,
                 cql_alpha: float = 1.0):
        super().__init__()

        self.n_actions = n_actions
        self.cql_alpha = cql_alpha

        # Q 网络
        self.q1 = nn.Sequential(
            nn.Linear(state_dim + n_actions, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        self.q2 = nn.Sequential(
            nn.Linear(state_dim + n_actions, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        # 动作嵌入 (离散动作)
        self.action_onehot = nn.functional.one_hot

    def forward(self, state: torch.Tensor,
               action: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """计算 Q 值"""
        # 离散动作转 one-hot
        action_onehot = F.one_hot(action, self.n_actions).float()
        state_action = torch.cat([state, action_onehot], dim=-1)

        q1 = self.q1(state_action)
        q2 = self.q2(state_action)

        return q1.squeeze(-1), q2.squeeze(-1)

    def get_all_q_values(self, state: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """获取所有动作的 Q 值"""
        batch_size = state.size(0)

        # 扩展状态
        state_expanded = state.unsqueeze(1).expand(-1, self.n_actions, -1)
        actions = torch.arange(self.n_actions, device=state.device)
        action_onehot = F.one_hot(actions, self.n_actions).float()
        action_onehot = action_onehot.unsqueeze(0).expand(batch_size, -1, -1)

        state_action = torch.cat([state_expanded, action_onehot], dim=-1)

        q1 = self.q1(state_action).squeeze(-1)
        q2 = self.q2(state_action).squeeze(-1)

        return q1, q2


class CQLTrainer:
    """
    CQL 训练器
    """

    def __init__(self, model: CQL, lr: float = 1e-3,
                 gamma: float = 0.99, cql_alpha: float = 1.0):
        self.model = model
        self.gamma = gamma
        self.cql_alpha = cql_alpha

        # 目标网络
        self.target_q1 = self._copy_network(model.q1)
        self.target_q2 = self._copy_network(model.q2)

        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    def _copy_network(self, network: nn.Module) -> nn.Module:
        """复制网络"""
        import copy
        return copy.deepcopy(network)

    def train_step(self, batch: Dict[str, torch.Tensor]) -> Dict[str, float]:
        """训练一步"""
        states = batch['states']
        actions = batch['actions']
        rewards = batch['rewards']
        next_states = batch['next_states']
        dones = batch['dones']

        batch_size = states.size(0)

        # 当前 Q 值
        current_q1, current_q2 = self.model(states, actions)

        # 目标 Q 值
        with torch.no_grad():
            # 获取所有下一状态的 Q 值
            next_q1, next_q2 = self.model.get_all_q_values(next_states)

            # 使用 Double Q-learning
            # 从 Q1 选择动作
            next_actions = next_q1.argmax(dim=1)

            # 从目标网络获取 Q 值
            next_q1_target, next_q2_target = self.model.get_all_q_values(next_states)

            # 使用选择动作的 Q 值
            next_q1_target = next_q1_target.gather(1, next_actions.unsqueeze(1)).squeeze(1)
            next_q2_target = next_q2_target.gather(1, next_actions.unsqueeze(1)).squeeze(1)

            next_q = torch.min(next_q1_target, next_q2_target)

            target_q = rewards + (1 - dones) * self.gamma * next_q

        # TD 损失
        td_loss1 = F.mse_loss(current_q1, target_q)
        td_loss2 = F.mse_loss(current_q2, target_q)
        td_loss = td_loss1 + td_loss2

        # CQL 正则化损失
        # 获取当前状态所有动作的 Q 值
        all_q1, all_q2 = self.model.get_all_q_values(states)

        # 数据集中动作的 Q 值
        dataset_q1 = all_q1.gather(1, actions.unsqueeze(1)).squeeze(1)
        dataset_q2 = all_q2.gather(1, actions.unsqueeze(1)).squeeze(1)

        # 随机动作 (均匀分布) 的期望 Q 值
        random_q1 = all_q1.mean(dim=1)
        random_q2 = all_q2.mean(dim=1)

        # CQL 损失: 最大化数据集动作 Q 值，最小化随机动作 Q 值
        cql_loss1 = (random_q1 - dataset_q1).mean()
        cql_loss2 = (random_q2 - dataset_q2).mean()
        cql_loss = cql_loss1 + cql_loss2

        # 总损失
        total_loss = td_loss + self.cql_alpha * cql_loss

        # 反向传播
        self.optimizer.zero_grad()
        total_loss.backward()
        self.optimizer.step()

        return {
            'total_loss': total_loss.item(),
            'td_loss': td_loss.item(),
            'cql_loss': cql_loss.item()
        }


class CQLRecommender:
    """
    基于 CQL 的推荐系统
    """

    def __init__(self, model: CQL, n_items: int):
        self.model = model
        self.n_items = n_items

    def recommend(self, state: np.ndarray, top_k: int = 10,
                 exclude_items: set = None) -> List[int]:
        """
        生成推荐

        state: 用户状态向量
        """
        self.model.eval()

        with torch.no_grad():
            state_tensor = torch.tensor(state, dtype=torch.float32).unsqueeze(0)

            # 获取所有物品的 Q 值
            q1, q2 = self.model.get_all_q_values(state_tensor)

            # 使用最小 Q 值 (保守)
            q_values = torch.min(q1, q2).squeeze(0).numpy()

        # 排除已交互物品
        if exclude_items:
            for item in exclude_items:
                if 0 <= item < len(q_values):
                    q_values[item] = float('-inf')

        # 排序
        top_items = np.argsort(q_values)[::-1][:top_k]

        return top_items.tolist()
```

## 4. 其他离线 RL 算法

### 4.1 IQL (Implicit Q-Learning)

```python
class IQL(nn.Module):
    """
    Implicit Q-Learning (IQL)

    核心思想:
    - 不需要从策略采样来估计 Q 值
    - 使用值函数的期望回归
    - 通过 Expectile Regression 提取最优值

    优点:
    - 完全离线
    - 不需要行为策略采样
    - 稳定性好
    """

    def __init__(self,
                 state_dim: int,
                 n_actions: int,
                 hidden_dim: int = 256,
                 expectile: float = 0.7,
                 temperature: float = 3.0):
        super().__init__()

        self.n_actions = n_actions
        self.expectile = expectile
        self.temperature = temperature

        # 值函数 V(s)
        self.v_network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        # Q 网络 Q(s, a)
        self.q_network = nn.Sequential(
            nn.Linear(state_dim + n_actions, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        # 目标网络
        self.target_q = self._copy_network(self.q_network)

    def _copy_network(self, network: nn.Module) -> nn.Module:
        import copy
        return copy.deepcopy(network)

    def expectile_loss(self, diff: torch.Tensor) -> torch.Tensor:
        """
        Expectile Regression 损失

        L^τ(u) = |τ - 1(u < 0)| * u²
        """
        weight = torch.where(diff > 0, self.expectile, 1 - self.expectile)
        return (weight * (diff ** 2)).mean()

    def train_step(self, batch: Dict[str, torch.Tensor],
                   optimizer: torch.optim.Optimizer) -> Dict[str, float]:
        """训练一步"""
        states = batch['states']
        actions = batch['actions']
        rewards = batch['rewards']
        next_states = batch['next_states']
        dones = batch['dones']

        # 动作转 one-hot
        actions_onehot = F.one_hot(actions, self.n_actions).float()

        # 1. 更新 V 网络
        with torch.no_grad():
            # 目标 Q 值
            state_action = torch.cat([states, actions_onehot], dim=-1)
            target_q = self.target_q(state_action).squeeze(-1)

        v_pred = self.v_network(states).squeeze(-1)
        v_loss = self.expectile_loss(target_q - v_pred)

        # 2. 更新 Q 网络
        with torch.no_grad():
            next_v = self.v_network(next_states).squeeze(-1)
            target_q = rewards + (1 - dones) * 0.99 * next_v

        current_q = self.q_network(state_action).squeeze(-1)
        q_loss = F.mse_loss(current_q, target_q)

        # 3. 提取策略 (advantage weighted)
        # 这里简化，实际需要更复杂的策略提取

        # 总损失
        total_loss = v_loss + q_loss

        optimizer.zero_grad()
        total_loss.backward()
        optimizer.step()

        return {
            'v_loss': v_loss.item(),
            'q_loss': q_loss.item(),
            'total_loss': total_loss.item()
        }
```

## 5. 推荐系统应用

### 5.1 离线 RL 推荐流程

```python
class OfflineRLRecSystem:
    """
    离线 RL 推荐系统
    """

    def __init__(self, algorithm: str = 'cql'):
        self.algorithm = algorithm

        # 初始化模型
        if algorithm == 'cql':
            self.model = CQL(state_dim=64, n_actions=10000)
            self.trainer = CQLTrainer(self.model)
        elif algorithm == 'iql':
            self.model = IQL(state_dim=64, n_actions=10000)

        # 数据集
        self.dataset = OfflineRLDataset()

    def load_historical_data(self, logs: List[Dict]):
        """加载历史数据"""
        self.dataset.add_from_logs(logs)

    def train(self, n_epochs: int = 100, batch_size: int = 256):
        """训练"""
        for epoch in range(n_epochs):
            batch = self.dataset.sample(batch_size)
            losses = self.trainer.train_step(batch)

            if epoch % 10 == 0:
                print(f"Epoch {epoch}: {losses}")

    def recommend(self, user_features: Dict, top_k: int = 10
                 ) -> List[int]:
        """推荐"""
        state = self._extract_state(user_features)
        return self.model.recommend(state, top_k)

    def _extract_state(self, features: Dict) -> np.ndarray:
        """提取状态"""
        return np.array(list(features.values()))
```

## 6. 学习总结

### 6.1 算法对比

```
算法        核心思想                优点                缺点
──────────────────────────────────────────────────────────────
BCQ         约束动作在数据范围内    简单直接            需要生成模型
CQL         保守Q值估计            理论保证            超参数敏感
IQL         隐式Q学习              不需要策略采样      实现复杂
```

### 6.2 最佳实践

```
1. 数据质量: 确保历史数据覆盖度足够
2. 状态表示: 好的状态表示是关键
3. 奖励设计: 奖励函数要反映业务目标
4. 算法选择: CQL 适合大多数场景
5. 评估: 离线评估 + 小规模在线验证
```
