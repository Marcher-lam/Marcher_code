# 社交推荐 学习文档

## 1. 社交推荐概述

### 1.1 什么是社交推荐？

```
社交推荐 (Social Recommendation):

- 利用社交网络信息增强推荐
- 用户之间的社交关系作为辅助信息
- "朋友喜欢的东西你可能也喜欢"

社交信息类型:
1. 显式关系: 关注、好友、粉丝
2. 隐式关系: 共同兴趣、相似行为
3. 社交互动: 评论、转发、点赞
4. 社交组群: 群组、圈子、社区

优势:
- 缓解数据稀疏
- 提供信任背书
- 发现潜在兴趣
- 增强可解释性
```

### 1.2 社交网络表示

```python
from typing import List, Dict, Tuple, Optional, Set
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from collections import defaultdict


class SocialNetwork:
    """
    社交网络
    """

    def __init__(self):
        # 邻接表
        self.following: Dict[int, Set[int]] = defaultdict(set)  # 关注
        self.followers: Dict[int, Set[int]] = defaultdict(set)  # 粉丝

        # 互动记录
        self.interactions: Dict[Tuple[int, int], List[dict]] = defaultdict(list)

    def add_follow(self, user_id: int, follow_id: int):
        """添加关注关系"""
        self.following[user_id].add(follow_id)
        self.followers[follow_id].add(user_id)

    def add_interaction(self, user_id: int, target_id: int,
                       interaction_type: str, item_id: int = None):
        """添加互动"""
        self.interactions[(user_id, target_id)].append({
            'type': interaction_type,
            'item_id': item_id,
            'timestamp': None
        })

    def get_friends(self, user_id: int) -> Set[int]:
        """获取好友 (互相关注)"""
        following = self.following[user_id]
        friends = {f for f in following if user_id in self.following[f]}
        return friends

    def get_following(self, user_id: int) -> Set[int]:
        """获取关注"""
        return self.following[user_id]

    def get_followers(self, user_id: int) -> Set[int]:
        """获取粉丝"""
        return self.followers[user_id]

    def get_adjacency_matrix(self, n_users: int) -> np.ndarray:
        """获取邻接矩阵"""
        adj = np.zeros((n_users, n_users))
        for user_id, following in self.following.items():
            for follow_id in following:
                adj[user_id, follow_id] = 1
        return adj
```

## 2. 社交正则化方法

### 2.1 社交正则化

```python
class SocialRegularization(nn.Module):
    """
    社交正则化

    约束朋友之间的嵌入相似
    """

    def __init__(self, lambda_social: float = 0.1):
        super().__init__()
        self.lambda_social = lambda_social

    def forward(self, user_embeds: torch.Tensor,
                social_pairs: torch.Tensor) -> torch.Tensor:
        """
        计算社交正则化损失

        user_embeds: (n_users, embed_dim)
        social_pairs: (n_pairs, 2) 社交关系对 (user, friend)
        """
        user_idx = social_pairs[:, 0]
        friend_idx = social_pairs[:, 1]

        user_emb = user_embeds[user_idx]
        friend_emb = user_embeds[friend_idx]

        # L2 距离损失
        social_loss = torch.norm(user_emb - friend_emb, p=2, dim=-1).mean()

        return self.lambda_social * social_loss


class SocialMF(nn.Module):
    """
    SocialMF: 社交正则化矩阵分解

    在传统 MF 基础上加入社交正则化
    """

    def __init__(self, n_users: int, n_items: int,
                 embed_dim: int = 64, lambda_social: float = 0.1):
        super().__init__()

        # 用户和物品嵌入
        self.user_embedding = nn.Embedding(n_users, embed_dim)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        # 偏置
        self.user_bias = nn.Embedding(n_users, 1)
        self.item_bias = nn.Embedding(n_items, 1)
        self.global_bias = nn.Parameter(torch.zeros(1))

        # 社交正则化
        self.social_reg = SocialRegularization(lambda_social)

        # 初始化
        nn.init.normal_(self.user_embedding.weight, std=0.01)
        nn.init.normal_(self.item_embedding.weight, std=0.01)

    def forward(self, user_ids: torch.Tensor,
                item_ids: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        user_emb = self.user_embedding(user_ids)
        item_emb = self.item_embedding(item_ids)

        user_b = self.user_bias(user_ids)
        item_b = self.item_bias(item_ids)

        # 内积 + 偏置
        score = (user_emb * item_emb).sum(dim=-1, keepdim=True)
        score = score + user_b + item_b + self.global_bias

        return score.squeeze(-1)

    def get_loss(self, user_ids: torch.Tensor, item_ids: torch.Tensor,
                 labels: torch.Tensor, social_pairs: torch.Tensor) -> torch.Tensor:
        """计算总损失"""
        # 预测
        pred = self.forward(user_ids, item_ids)

        # 预测损失
        pred_loss = nn.BCEWithLogitsLoss()(pred, labels.float())

        # 社交正则化
        social_loss = self.social_reg(self.user_embedding.weight, social_pairs)

        return pred_loss + social_loss
```

## 3. 社交图神经网络

### 3.1 Social GNN

```python
class SocialGNN(nn.Module):
    """
    Social GNN: 社交图神经网络

    通过 GNN 在社交图上传播用户表示
    """

    def __init__(self, n_users: int, n_items: int,
                 embed_dim: int = 64, n_layers: int = 2,
                 dropout: float = 0.1):
        super().__init__()

        self.n_users = n_users
        self.n_items = n_items
        self.embed_dim = embed_dim
        self.n_layers = n_layers

        # 嵌入
        self.user_embedding = nn.Embedding(n_users, embed_dim)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        # GNN 层
        self.gnn_layers = nn.ModuleList([
            SocialGNNLayer(embed_dim, dropout) for _ in range(n_layers)
        ])

    def forward(self, user_ids: torch.Tensor,
                item_ids: torch.Tensor,
                social_adj: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        social_adj: (n_users, n_users) 社交邻接矩阵
        """
        # 初始用户嵌入
        user_embed = self.user_embedding.weight.clone()

        # GNN 传播
        all_user_embeds = [user_embed]

        for gnn_layer in self.gnn_layers:
            user_embed = gnn_layer(user_embed, social_adj)
            all_user_embeds.append(user_embed)

        # 聚合所有层
        user_embed = torch.stack(all_user_embeds, dim=0).mean(dim=0)

        # 获取目标用户和物品
        target_user_embed = user_embed[user_ids]
        item_embed = self.item_embedding(item_ids)

        # 预测
        score = (target_user_embed * item_embed).sum(dim=-1)

        return score

    def get_all_scores(self, social_adj: torch.Tensor) -> torch.Tensor:
        """获取所有用户-物品得分"""
        # 传播用户嵌入
        user_embed = self.user_embedding.weight.clone()

        for gnn_layer in self.gnn_layers:
            user_embed = gnn_layer(user_embed, social_adj)

        # 计算所有得分
        item_embed = self.item_embedding.weight
        scores = torch.matmul(user_embed, item_embed.T)

        return scores


class SocialGNNLayer(nn.Module):
    """
    社交 GNN 层
    """

    def __init__(self, embed_dim: int, dropout: float = 0.1):
        super().__init__()

        self.linear = nn.Linear(embed_dim, embed_dim)
        self.norm = nn.LayerNorm(embed_dim)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor, adj: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        x: (n_users, embed_dim)
        adj: (n_users, n_users) 归一化邻接矩阵
        """
        # 邻居聚合
        neighbor_embed = torch.matmul(adj, x)

        # 转换
        output = self.linear(neighbor_embed)
        output = self.norm(output)
        output = self.dropout(output)

        # 残差
        return x + output
```

### 3.2 DiffNet

```python
class DiffNet(nn.Module):
    """
    DiffNet: 深度兴趣扩散网络

    模拟社交影响在用户间的扩散
    """

    def __init__(self, n_users: int, n_items: int,
                 embed_dim: int = 64, n_layers: int = 2,
                 hidden_dim: int = 128):
        super().__init__()

        self.n_users = n_users
        self.n_items = n_items
        self.embed_dim = embed_dim
        self.n_layers = n_layers

        # 嵌入
        self.user_embedding = nn.Embedding(n_users, embed_dim)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        # 兴趣扩散层
        self.diffusion_layers = nn.ModuleList([
            DiffusionLayer(embed_dim) for _ in range(n_layers)
        ])

        # 预测网络
        self.predictor = nn.Sequential(
            nn.Linear(embed_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

    def forward(self, user_ids: torch.Tensor,
                item_ids: torch.Tensor,
                social_adj: torch.Tensor,
                item_features: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播
        """
        # 用户嵌入
        user_embed = self.user_embedding.weight

        # 兴趣扩散
        for diffusion_layer in self.diffusion_layers:
            user_embed = diffusion_layer(user_embed, social_adj)

        # 获取目标用户和物品
        target_user_embed = user_embed[user_ids]
        item_embed = self.item_embedding(item_ids)

        # 预测
        combined = torch.cat([target_user_embed, item_embed], dim=-1)
        score = self.predictor(combined)

        return score.squeeze(-1)


class DiffusionLayer(nn.Module):
    """
    兴趣扩散层
    """

    def __init__(self, embed_dim: int):
        super().__init__()

        self.linear_self = nn.Linear(embed_dim, embed_dim)
        self.linear_neighbor = nn.Linear(embed_dim, embed_dim)
        self.gate = nn.Linear(embed_dim * 2, embed_dim)

    def forward(self, x: torch.Tensor, adj: torch.Tensor) -> torch.Tensor:
        """
        前向传播
        """
        # 自身特征
        self_feature = self.linear_self(x)

        # 邻居聚合
        neighbor_feature = torch.matmul(adj, x)
        neighbor_feature = self.linear_neighbor(neighbor_feature)

        # 门控融合
        gate_input = torch.cat([self_feature, neighbor_feature], dim=-1)
        gate = torch.sigmoid(self.gate(gate_input))

        output = gate * self_feature + (1 - gate) * neighbor_feature

        return output
```

## 4. 社交信任模型

### 4.1 TrustWalker

```python
class TrustWalker:
    """
    TrustWalker: 基于信任的随机游走

    在社交网络上进行带信任权重的随机游走
    """

    def __init__(self, social_network: SocialNetwork,
                 trust_decay: float = 0.9,
                 max_walk_length: int = 3):
        self.social_network = social_network
        self.trust_decay = trust_decay
        self.max_walk_length = max_walk_length

        # 信任矩阵 (可以预计算)
        self.trust_matrix = None

    def compute_trust(self, user_id: int, target_id: int) -> float:
        """
        计算用户到目标的信任度
        """
        # 简化: 使用最短路径
        distance = self._shortest_path_length(user_id, target_id)

        if distance is None:
            return 0.0

        return self.trust_decay ** distance

    def _shortest_path_length(self, start: int, end: int) -> Optional[int]:
        """BFS 找最短路径"""
        if start == end:
            return 0

        from collections import deque

        visited = {start}
        queue = deque([(start, 0)])

        while queue:
            current, dist = queue.popleft()

            if dist >= self.max_walk_length:
                continue

            for neighbor in self.social_network.get_following(current):
                if neighbor == end:
                    return dist + 1

                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, dist + 1))

        return None

    def recommend(self, user_id: int, n: int = 10) -> List[Tuple[int, float]]:
        """
        基于 trust 的推荐

        聚合信任朋友的喜好
        """
        recommendations = defaultdict(float)

        # 获取直接朋友
        friends = self.social_network.get_following(user_id)

        for friend_id in friends:
            trust = self.compute_trust(user_id, friend_id)

            # 获取朋友的喜好 (简化)
            friend_items = self._get_user_items(friend_id)

            for item_id, score in friend_items.items():
                recommendations[item_id] += trust * score

        # 排序
        sorted_items = sorted(recommendations.items(),
                             key=lambda x: x[1], reverse=True)

        return sorted_items[:n]

    def _get_user_items(self, user_id: int) -> Dict[int, float]:
        """获取用户的物品偏好"""
        # 实际实现需要访问用户行为数据
        return {}
```

## 5. 社交推荐实战

### 5.1 完整训练流程

```python
class SocialRecTrainer:
    """
    社交推荐训练器
    """

    def __init__(self, model: nn.Module,
                 lr: float = 0.001,
                 weight_decay: float = 1e-5):
        self.model = model
        self.optimizer = torch.optim.Adam(
            model.parameters(),
            lr=lr,
            weight_decay=weight_decay
        )

    def train_epoch(self, train_loader, social_adj: torch.Tensor):
        """训练一个 epoch"""
        self.model.train()
        total_loss = 0

        for batch in train_loader:
            user_ids = batch['user_id']
            item_ids = batch['item_id']
            labels = batch['label']

            self.optimizer.zero_grad()

            # 前向传播
            if isinstance(self.model, SocialGNN):
                pred = self.model(user_ids, item_ids, social_adj)
            else:
                pred = self.model(user_ids, item_ids)

            # 损失
            loss = nn.BCEWithLogitsLoss()(pred, labels.float())

            # 反向传播
            loss.backward()
            self.optimizer.step()

            total_loss += loss.item()

        return total_loss / len(train_loader)

    def evaluate(self, test_loader, social_adj: torch.Tensor,
                 k_list: List[int] = [10, 20, 50]):
        """评估"""
        self.model.eval()

        metrics = {f'Hit@{k}': [] for k in k_list}
        metrics.update({f'NDCG@{k}': [] for k in k_list})

        with torch.no_grad():
            for batch in test_loader:
                user_ids = batch['user_id']
                item_ids = batch['item_id']
                labels = batch['label']

                # 预测
                if isinstance(self.model, SocialGNN):
                    pred = torch.sigmoid(self.model(user_ids, item_ids, social_adj))
                else:
                    pred = torch.sigmoid(self.model(user_ids, item_ids))

                # 计算指标 (简化)
                for k in k_list:
                    hit = (pred > 0.5) == (labels == 1)
                    metrics[f'Hit@{k}'].extend(hit.float().tolist())

        # 平均
        for key in metrics:
            metrics[key] = np.mean(metrics[key])

        return metrics


def normalize_adj(adj: np.ndarray) -> torch.Tensor:
    """
    归一化邻接矩阵

    D^{-1/2} A D^{-1/2}
    """
    # 添加自环
    adj = adj + np.eye(adj.shape[0])

    # 度矩阵
    degree = adj.sum(axis=1)
    degree_inv_sqrt = np.power(degree, -0.5)
    degree_inv_sqrt[np.isinf(degree_inv_sqrt)] = 0.

    # 归一化
    adj_normalized = degree_inv_sqrt[:, np.newaxis] * adj * degree_inv_sqrt[np.newaxis, :]

    return torch.tensor(adj_normalized, dtype=torch.float32)
```
