# 推荐系统强化学习进阶 学习文档

## 1. 强化学习与推荐系统

### 1.1 为什么用强化学习？

```
传统推荐 vs 强化学习推荐:

传统方法 (静态):
- 预测 CTR/CVR
- 单步优化
- 独立同分布假设

强化学习 (动态):
- 优化长期收益
- 多步决策
- 考虑状态转移
- 探索与利用平衡

典型应用:
- 信息流推荐 (用户停留时长)
- 电商推荐 (长期 GMV)
- 视频推荐 (用户留存)
```

class UserState:
    """
    用户状态表示
    """

    def __init__(self, history_len: int = 50, embed_dim: int = 64):
        self.history_len = history_len
        self.embed_dim = embed_dim

        # 用户行为历史
        self.item_history: List[int] = []
        self.action_history: List[str] = []  # click, skip, like, etc.
        self.reward_history: List[float] = []

        # 用户画像
        self.user_features: Dict = {}

    def update(self, item_id: int, action: str, reward: float):
        """更新状态"""
        self.item_history.append(item_id)
        self.action_history.append(action)
        self.reward_history.append(reward)

        # 保持固定长度
        if len(self.item_history) > self.history_len:
            self.item_history = self.item_history[-self.history_len:]
            self.action_history = self.action_history[-self.history_len:]
            self.reward_history = self.reward_history[-self.history_len:]

    def get_state_vector(self, item_embeddings: np.ndarray) -> np.ndarray:
        """获取状态向量"""
        if not self.item_history:
            return np.zeros(self.embed_dim)

        # 平均池化历史物品
        history_embeds = item_embeddings[self.item_history]
        return history_embeds.mean(axis=0)

## 2. DQN for Recommendations

### 2.1 DQN 架构

```python
class DQNRec(nn.Module):
    """
    DQN for Recommendations

    Q(s, a) = 期望累积奖励
    """

    def __init__(self,
                 state_dim: int,
                 n_items: int,
                 embed_dim: int = 64,
                 hidden_dim: int = 128):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim

        # 状态编码器
        self.state_encoder = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

        # 物品嵌入
        self.item_embedding = nn.Embedding(n_items, embed_dim)

        # Q 值网络
        self.q_network = nn.Sequential(
            nn.Linear(embed_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1)
        )

    def encode_state(self, state: torch.Tensor) -> torch.Tensor:
        """编码状态"""
        return self.state_encoder(state)

    def get_q_values(self, state: torch.Tensor,
                     item_ids: torch.Tensor = None) -> torch.Tensor:
        """
        获取 Q 值

        如果 item_ids 给定: 返回 (batch,) 这些物品的 Q 值
        否则: 返回 (batch, n_items) 所有物品的 Q 值
        """
        state_embed = self.encode_state(state)  # (batch, embed_dim)

        if item_ids is not None:
            # 特定物品
            item_embed = self.item_embedding(item_ids)  # (batch, embed_dim)
            combined = torch.cat([state_embed, item_embed], dim=-1)
            q_values = self.q_network(combined).squeeze(-1)
            return q_values
        else:
            # 所有物品
            batch_size = state.size(0)
            all_items = torch.arange(self.n_items, device=state.device)

            item_embed = self.item_embedding(all_items)  # (n_items, embed_dim)
            item_embed = item_embed.unsqueeze(0).expand(batch_size, -1, -1)

            state_embed = state_embed.unsqueeze(1).expand(-1, self.n_items, -1)

            combined = torch.cat([state_embed, item_embed], dim=-1)
            q_values = self.q_network(combined).squeeze(-1)

            return q_values

    def forward(self, state: torch.Tensor,
                item_ids: torch.Tensor = None) -> torch.Tensor:
        """前向传播"""
        return self.get_q_values(state, item_ids)


class ReplayBuffer:
    """
    经验回放缓冲区
    """

    def __init__(self, capacity: int = 100000):
        self.capacity = capacity
        self.buffer = []
        self.position = 0

    def push(self, state: np.ndarray, action: int, reward: float,
             next_state: np.ndarray, done: bool):
        """添加经验"""
        if len(self.buffer) < self.capacity:
            self.buffer.append(None)

        self.buffer[self.position] = (state, action, reward, next_state, done)
        self.position = (self.position + 1) % self.capacity

    def sample(self, batch_size: int) -> Dict[str, torch.Tensor]:
        """采样"""
        indices = np.random.choice(len(self.buffer), batch_size)

        states, actions, rewards, next_states, dones = [], [], [], [], []

        for idx in indices:
            s, a, r, ns, d = self.buffer[idx]
            states.append(s)
            actions.append(a)
            rewards.append(r)
            next_states.append(ns)
            dones.append(d)

        return {
            'states': torch.tensor(np.array(states), dtype=torch.float32),
            'actions': torch.tensor(actions, dtype=torch.long),
            'rewards': torch.tensor(rewards, dtype=torch.float32),
            'next_states': torch.tensor(np.array(next_states), dtype=torch.float32),
            'dones': torch.tensor(dones, dtype=torch.float32)
        }

    def __len__(self):
        return len(self.buffer)
```

### 2.2 DQN 训练

```python
class DQNTrainer:
    """
    DQN 训练器
    """

    def __init__(self,
                 model: DQNRec,
                 n_items: int,
                 lr: float = 1e-4,
                 gamma: float = 0.99,
                 epsilon_start: float = 1.0,
                 epsilon_end: float = 0.1,
                 epsilon_decay: int = 10000,
                 target_update_freq: int = 1000):
        self.model = model
        self.n_items = n_items

        # 目标网络
        self.target_model = DQNRec(
            state_dim=model.state_encoder[0].in_features,
            n_items=n_items,
            embed_dim=model.embed_dim
        )
        self.target_model.load_state_dict(model.state_dict())

        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)

        self.gamma = gamma
        self.epsilon = epsilon_start
        self.epsilon_end = epsilon_end
        self.epsilon_decay = epsilon_decay
        self.target_update_freq = target_update_freq

        self.replay_buffer = ReplayBuffer()
        self.step_count = 0

    def select_action(self, state: np.ndarray,
                      candidate_items: List[int],
                      explore: bool = True) -> int:
        """
        选择动作 (ε-greedy)
        """
        if explore and np.random.random() < self.epsilon:
            # 探索
            return np.random.choice(candidate_items)
        else:
            # 利用
            with torch.no_grad():
                state_tensor = torch.tensor(state, dtype=torch.float32).unsqueeze(0)
                item_tensor = torch.tensor(candidate_items)

                q_values = self.model(state_tensor, item_tensor)
                best_idx = q_values.argmax().item()
                return candidate_items[best_idx]

    def update_epsilon(self):
        """更新探索率"""
        self.epsilon = self.epsilon_end + (
            self.epsilon - self.epsilon_end
        ) * np.exp(-1.0 * self.step_count / self.epsilon_decay)

    def train_step(self, batch_size: int = 32) -> float:
        """训练一步"""
        if len(self.replay_buffer) < batch_size:
            return 0.0

        # 采样
        batch = self.replay_buffer.sample(batch_size)

        states = batch['states']
        actions = batch['actions']
        rewards = batch['rewards']
        next_states = batch['next_states']
        dones = batch['dones']

        # 当前 Q 值
        current_q = self.model(states, actions)

        # 目标 Q 值
        with torch.no_grad():
            # Double DQN
            next_q_values = self.model(next_states)
            next_actions = next_q_values.argmax(dim=1)

            target_next_q = self.target_model(next_states, next_actions)

            target_q = rewards + (1 - dones) * self.gamma * target_next_q

        # Loss
        loss = F.smooth_l1_loss(current_q, target_q)

        # 反向传播
        self.optimizer.zero_grad()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(self.model.parameters(), 10)
        self.optimizer.step()

        # 更新目标网络
        self.step_count += 1
        if self.step_count % self.target_update_freq == 0:
            self.target_model.load_state_dict(self.model.state_dict())

        self.update_epsilon()

        return loss.item()
```

## 3. Policy Gradient 方法

### 3.1 REINFORCE

```python
class REINFORCERec(nn.Module):
    """
    REINFORCE for Recommendations

    直接优化策略 π(a|s)
    """

    def __init__(self, state_dim: int, n_items: int,
                 embed_dim: int = 64, hidden_dim: int = 128):
        super().__init__()

        self.n_items = n_items

        # 状态编码器
        self.state_encoder = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

        # 策略网络 (输出所有物品的概率)
        self.policy_head = nn.Sequential(
            nn.Linear(embed_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_items)
        )

        # 价值网络 (基线)
        self.value_head = nn.Sequential(
            nn.Linear(embed_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

    def forward(self, state: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        前向传播

        返回:
            action_probs: (batch, n_items) 动作概率
            state_value: (batch, 1) 状态价值
        """
        state_embed = self.state_encoder(state)

        logits = self.policy_head(state_embed)
        action_probs = F.softmax(logits, dim=-1)

        state_value = self.value_head(state_embed)

        return action_probs, state_value

    def select_action(self, state: torch.Tensor,
                      candidate_items: List[int] = None) -> Tuple[int, torch.Tensor]:
        """
        采样动作
        """
        action_probs, _ = self.forward(state.unsqueeze(0))

        if candidate_items:
            # 只考虑候选物品
            mask = torch.zeros(self.n_items)
            mask[candidate_items] = 1
            action_probs = action_probs * mask
            action_probs = action_probs / action_probs.sum()

        dist = torch.distributions.Categorical(action_probs)
        action = dist.sample()
        log_prob = dist.log_prob(action)

        return action.item(), log_prob


class REINFORCETrainer:
    """
    REINFORCE 训练器
    """

    def __init__(self, model: REINFORCERec, lr: float = 1e-4,
                 gamma: float = 0.99, value_loss_coef: float = 0.5):
        self.model = model
        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)
        self.gamma = gamma
        self.value_loss_coef = value_loss_coef

        # 存储轨迹
        self.log_probs = []
        self.values = []
        self.rewards = []

    def store_transition(self, log_prob: torch.Tensor, value: torch.Tensor, reward: float):
        """存储转移"""
        self.log_probs.append(log_prob)
        self.values.append(value)
        self.rewards.append(reward)

    def compute_returns(self) -> torch.Tensor:
        """计算回报"""
        returns = []
        R = 0

        for r in reversed(self.rewards):
            R = r + self.gamma * R
            returns.insert(0, R)

        returns = torch.tensor(returns)
        returns = (returns - returns.mean()) / (returns.std() + 1e-8)

        return returns

    def update(self) -> float:
        """更新策略"""
        returns = self.compute_returns()

        log_probs = torch.stack(self.log_probs)
        values = torch.stack(self.values).squeeze()

        # 策略损失 (带基线)
        advantage = returns - values.detach()
        policy_loss = -(log_probs * advantage).mean()

        # 价值损失
        value_loss = F.mse_loss(values, returns)

        # 总损失
        loss = policy_loss + self.value_loss_coef * value_loss

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        # 清空轨迹
        self.log_probs = []
        self.values = []
        self.rewards = []

        return loss.item()
```

### 4.1 A2C for Recommendations

```python
class A2CRec(nn.Module):
    """
    Actor-Critic for Recommendations
    """

    def __init__(self, state_dim: int, n_items: int,
                 embed_dim: int = 64, hidden_dim: int = 128):
        super().__init__()

        self.n_items = n_items

        # 共享编码器
        self.shared_encoder = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, embed_dim)
        )

        # Actor (策略)
        self.actor = nn.Sequential(
            nn.Linear(embed_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_items)
        )

        # Critic (价值)
        self.critic = nn.Sequential(
            nn.Linear(embed_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

        # 物品嵌入 (用于动作嵌入)
        self.item_embedding = nn.Embedding(n_items, embed_dim)

    def forward(self, state: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """前向传播"""
        shared = self.shared_encoder(state)

        # Actor: 动作概率
        actor_logits = self.actor(shared)
        action_probs = F.softmax(actor_logits, dim=-1)

        # Critic: 状态价值
        value = self.critic(shared)

        return action_probs, value

    def evaluate_action(self, state: torch.Tensor, action: torch.Tensor
                       ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """评估动作"""
        action_probs, value = self.forward(state)

        dist = torch.distributions.Categorical(action_probs)
        log_prob = dist.log_prob(action)
        entropy = dist.entropy()

        return log_prob, value, entropy


class A2CTrainer:
    """
    A2C 训练器
    """

    def __init__(self, model: A2CRec, lr: float = 1e-4,
                 gamma: float = 0.99, entropy_coef: float = 0.01,
                 value_loss_coef: float = 0.5, max_grad_norm: float = 0.5):
        self.model = model
        self.optimizer = torch.optim.Adam(model.parameters(), lr=lr)
        self.gamma = gamma
        self.entropy_coef = entropy_coef
        self.value_loss_coef = value_loss_coef
        self.max_grad_norm = max_grad_norm

    def compute_gae(self, rewards: List[float], values: List[torch.Tensor],
                    dones: List[bool], next_value: torch.Tensor,
                    gae_lambda: float = 0.95) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        计算广义优势估计 (GAE)
        """
        advantages = []
        returns = []

        gae = 0

        for t in reversed(range(len(rewards))):
            if t == len(rewards) - 1:
                next_val = next_value
            else:
                next_val = values[t + 1]

            delta = rewards[t] + self.gamma * next_val * (1 - dones[t]) - values[t]
            gae = delta + self.gamma * gae_lambda * (1 - dones[t]) * gae

            advantages.insert(0, gae)
            returns.insert(0, gae + values[t])

        return torch.stack(advantages), torch.stack(returns)

    def update(self, rollout: Dict) -> float:
        """
        更新
        """
        states = rollout['states']
        actions = rollout['actions']
        rewards = rollout['rewards']
        dones = rollout['dones']

        # 获取当前值
        with torch.no_grad():
            _, next_value = self.model(states[-1].unsqueeze(0))

        # 计算优势
        _, values = self.model(states)
        values = values.squeeze()

        advantages, returns = self.compute_gae(
            rewards, values, dones, next_value.squeeze()
        )

        # 计算损失
        log_probs, value_preds, entropies = self.model.evaluate_action(
            states, actions
        )

        # 策略损失
        policy_loss = -(log_probs * advantages.detach()).mean()

        # 价值损失
        value_loss = F.mse_loss(value_preds.squeeze(), returns)

        # 熵损失 (鼓励探索)
        entropy_loss = -entropies.mean()

        # 总损失
        loss = (
            policy_loss +
            self.value_loss_coef * value_loss +
            self.entropy_coef * entropy_loss
        )

        self.optimizer.zero_grad()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(self.model.parameters(), self.max_grad_norm)
        self.optimizer.step()

        return loss.item()
```

        生成推荐

        state = self.get_user_state(user_id)

            # 随机探索

                candidate_items, min(n_recommend, len(candidate_items)), replace=False

                state_tensor = torch.tensor(state_vector, dtype=torch.float32).unsqueeze(0)

## 6. 学习总结

### 6.1 方法对比

```
方法        优点                缺点                适用场景
─────────────────────────────────────────────────────────────
DQN         稳定、样本效率高    高估Q值             离散动作空间
REINFORCE   简单直接           高方差              简单场景
A2C         效率高、稳定       需要同步            并行环境
PPO         稳定、效果好       实现复杂            通用场景
```

### 6.2 RL 在推荐中的挑战

```
1. 奖励设计: 如何定义好的奖励函数
2. 样本效率: 用户交互数据有限
3. 状态表示: 如何有效表示用户状态
4. 探索成本: 探索可能影响用户体验
5. 离线评估: 难以评估策略效果
6. 安全性: 避免推荐有害内容
```

### 6.3 最佳实践

```
1. 从模仿学习开始，再用RL优化
2. 使用离线RL处理历史数据
3. 结合Bandit方法进行探索
4. 使用基线减少方差
5. 注意探索的安全性
```
