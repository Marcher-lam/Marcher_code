# 多臂老虎机 (Multi-Armed Bandit) 推荐 学习文档

## 1. 问题定义

### 1.1 什么是多臂老虎机？

多臂老虎机（MAB）是一种序贯决策问题：在 K 个选项（手臂）中，每次选择一个，获得一个随机奖励，目标是最大化累计奖励。

### 1.2 核心问题

```
探索 vs 利用（Exploration vs Exploitation）:
- 探索: 尝试未知选项，获取信息
- 利用: 选择当前最优选项，获取奖励
```

### 1.3 在推荐中的应用

| 应用场景 | 描述 |
|----------|------|
| 冷启动 | 快速学习新物品/用户偏好 |
| 在线学习 | 实时调整推荐策略 |
| 氛围推荐 | 根据上下文调整 |
| 排序优化 | 动态调整排序策略 |

## 2. 经典算法

### 2.1 ε-Greedy

```python
import numpy as np
from typing import List, Tuple, Optional
from collections import defaultdict


class EpsilonGreedy:
    """
    ε-Greedy 算法

    以 ε 的概率随机探索，1-ε 的概率选择当前最优
    """

    def __init__(self, n_arms: int, epsilon: float = 0.1):
        """
        参数:
            n_arms: 手臂数量
            epsilon: 探索概率
        """
        self.n_arms = n_arms
        self.epsilon = epsilon

        self.counts = np.zeros(n_arms)  # 每个手臂的尝试次数
        self.values = np.zeros(n_arms)  # 每个手臂的平均奖励

    def select_arm(self) -> int:
        """选择手臂"""
        if np.random.random() < self.epsilon:
            # 探索: 随机选择
            return np.random.randint(self.n_arms)
        else:
            # 利用: 选择当前最优
            return np.argmax(self.values)

    def update(self, arm: int, reward: float):
        """更新统计信息"""
        self.counts[arm] += 1
        n = self.counts[arm]

        # 增量更新均值
        self.values[arm] = self.values[arm] + (reward - self.values[arm]) / n


class DecayingEpsilonGreedy(EpsilonGreedy):
    """
    衰减 ε-Greedy

    ε 随时间衰减，早期多探索，后期多利用
    """

    def __init__(self, n_arms: int, initial_epsilon: float = 1.0,
                 decay_rate: float = 0.99, min_epsilon: float = 0.01):
        super().__init__(n_arms, initial_epsilon)
        self.decay_rate = decay_rate
        self.min_epsilon = min_epsilon
        self.initial_epsilon = initial_epsilon

    def select_arm(self) -> int:
        arm = super().select_arm()
        # 衰减 ε
        self.epsilon = max(self.min_epsilon, self.epsilon * self.decay_rate)
        return arm
```

### 2.2 UCB (Upper Confidence Bound)

```python
class UCB:
    """
    UCB 算法

    选择上置信界最大的手臂，自动平衡探索和利用
    """

    def __init__(self, n_arms: int, c: float = 2.0):
        """
        参数:
            n_arms: 手臂数量
            c: 探索系数
        """
        self.n_arms = n_arms
        self.c = c

        self.counts = np.zeros(n_arms)
        self.values = np.zeros(n_arms)
        self.total_count = 0

    def select_arm(self) -> int:
        """选择手臂"""
        # 未尝试过的手臂优先
        for arm in range(self.n_arms):
            if self.counts[arm] == 0:
                return arm

        # 计算每个手臂的 UCB 值
        ucb_values = np.zeros(self.n_arms)

        for arm in range(self.n_arms):
            # 平均奖励
            avg_reward = self.values[arm]
            # 探索项（置信区间宽度）
            exploration = np.sqrt(
                self.c * np.log(self.total_count) / self.counts[arm]
            )
            ucb_values[arm] = avg_reward + exploration

        return np.argmax(ucb_values)

    def update(self, arm: int, reward: float):
        """更新"""
        self.counts[arm] += 1
        self.total_count += 1
        n = self.counts[arm]
        self.values[arm] = self.values[arm] + (reward - self.values[arm]) / n


class UCB1(UCB):
    """
    UCB1 算法（c=2 的特例）
    """

    def __init__(self, n_arms: int):
        super().__init__(n_arms, c=2.0)


class UCBTuned(UCB):
    """
    UCB-Tuned 算法

    使用方差调整置信区间
    """

    def __init__(self, n_arms: int):
        super().__init__(n_arms, c=2.0)
        self.squared_rewards = np.zeros(n_arms)  # 平方奖励和

    def select_arm(self) -> int:
        # 未尝试过的手臂优先
        for arm in range(self.n_arms):
            if self.counts[arm] == 0:
                return arm

        ucb_values = np.zeros(self.n_arms)

        for arm in range(self.n_arms):
            avg_reward = self.values[arm]
            n = self.counts[arm]

            # 计算方差
            variance = self.squared_rewards[arm] / n - avg_reward ** 2

            # 调整的探索项
            v = variance + np.sqrt(2 * np.log(self.total_count) / n)
            exploration = np.sqrt(np.log(self.total_count) / n * min(0.25, v))

            ucb_values[arm] = avg_reward + exploration

        return np.argmax(ucb_values)

    def update(self, arm: int, reward: float):
        self.counts[arm] += 1
        self.total_count += 1
        n = self.counts[arm]

        self.squared_rewards[arm] += reward ** 2
        self.values[arm] = self.values[arm] + (reward - self.values[arm]) / n
```

### 2.3 Thompson Sampling

```python
class ThompsonSampling:
    """
    Thompson Sampling（汤普森采样）

    基于贝叶斯推断的概率匹配算法
    """

    def __init__(self, n_arms: int):
        self.n_arms = n_arms

        # Beta 分布参数（伯努利奖励）
        self.alpha = np.ones(n_arms)  # 成功次数
        self.beta = np.ones(n_arms)   # 失败次数

    def select_arm(self) -> int:
        """从后验分布中采样，选择采样值最大的"""
        samples = np.array([
            np.random.beta(self.alpha[i], self.beta[i])
            for i in range(self.n_arms)
        ])
        return np.argmax(samples)

    def update(self, arm: int, reward: float):
        """
        更新后验分布

        参数:
            reward: 0 或 1（伯努利）
        """
        if reward > 0.5:
            self.alpha[arm] += 1
        else:
            self.beta[arm] += 1


class ThompsonSamplingGaussian:
    """
    高斯奖励的 Thompson Sampling
    """

    def __init__(self, n_arms: int, mu_0: float = 0, sigma_0: float = 1,
                 sigma: float = 1):
        """
        参数:
            n_arms: 手臂数量
            mu_0: 先验均值
            sigma_0: 先验标准差
            sigma: 奖励标准差
        """
        self.n_arms = n_arms
        self.sigma = sigma

        # 后验参数
        self.mu = np.ones(n_arms) * mu_0
        self.sigma_posterior = np.ones(n_arms) * sigma_0

        # 统计信息
        self.sum_rewards = np.zeros(n_arms)
        self.counts = np.zeros(n_arms)

    def select_arm(self) -> int:
        """从后验分布采样"""
        samples = np.array([
            np.random.normal(self.mu[i], self.sigma_posterior[i])
            for i in range(self.n_arms)
        ])
        return np.argmax(samples)

    def update(self, arm: int, reward: float):
        """更新后验"""
        self.counts[arm] += 1
        self.sum_rewards[arm] += reward
        n = self.counts[arm]

        # 更新后验参数
        prior_precision = 1 / (self.sigma_posterior[arm] ** 2)
        likelihood_precision = n / (self.sigma ** 2)

        posterior_precision = prior_precision + likelihood_precision
        self.sigma_posterior[arm] = np.sqrt(1 / posterior_precision)

        # 更新均值
        self.mu[arm] = (prior_precision * self.mu[arm] +
                       likelihood_precision * self.sum_rewards[arm] / n) / posterior_precision
```

## 3. 上下文老虎机

### 3.1 LinUCB

```python
class LinUCB:
    """
    LinUCB 算法

    线性上下文老虎机
    """

    def __init__(self, n_arms: int, context_dim: int, alpha: float = 1.0):
        """
        参数:
            n_arms: 手臂数量
            context_dim: 上下文维度
            alpha: 探索参数
        """
        self.n_arms = n_arms
        self.context_dim = context_dim
        self.alpha = alpha

        # 每个手臂的参数
        self.A = [np.eye(context_dim) for _ in range(n_arms)]  # A = X^T X + I
        self.b = [np.zeros(context_dim) for _ in range(n_arms)]  # b = X^T y

    def select_arm(self, context: np.ndarray) -> int:
        """
        根据上下文选择手臂

        参数:
            context: (context_dim,) 上下文特征
        """
        ucb_values = np.zeros(self.n_arms)

        for arm in range(self.n_arms):
            # 计算参数
            A_inv = np.linalg.inv(self.A[arm])
            theta = A_inv @ self.b[arm]

            # 预测值
            pred = context @ theta

            # 置信区间宽度
            uncertainty = self.alpha * np.sqrt(context @ A_inv @ context)

            # UCB 值
            ucb_values[arm] = pred + uncertainty

        return np.argmax(ucb_values)

    def update(self, arm: int, context: np.ndarray, reward: float):
        """更新参数"""
        self.A[arm] += np.outer(context, context)
        self.b[arm] += reward * context


class LinUCBHybrid:
    """
    LinUCB Hybrid

    考虑手臂和上下文的交叉特征
    """

    def __init__(self, n_arms: int, context_dim: int,
                 shared_dim: int, alpha: float = 1.0):
        self.n_arms = n_arms
        self.context_dim = context_dim
        self.shared_dim = shared_dim
        self.alpha = alpha

        # 共享参数
        self.A0 = np.eye(shared_dim)
        self.b0 = np.zeros(shared_dim)

        # 手臂特定参数
        self.A = [np.eye(context_dim) for _ in range(n_arms)]
        self.B = [np.zeros((context_dim, shared_dim)) for _ in range(n_arms)]
        self.b = [np.zeros(context_dim) for _ in range(n_arms)]

    def select_arm(self, context: np.ndarray, shared_features: np.ndarray) -> int:
        """选择手臂"""
        ucb_values = np.zeros(self.n_arms)

        for arm in range(self.n_arms):
            # 计算参数
            A_inv = np.linalg.inv(self.A[arm])
            A0_inv = np.linalg.inv(self.A0)

            theta = A_inv @ (self.b[arm] - self.B[arm] @ A0_inv @ self.b0)
            beta = A0_inv @ (self.b0 - sum(self.B[i].T @ A_inv @ self.b[i]
                                           for i in range(self.n_arms)))

            # 预测
            pred = shared_features @ beta + context @ theta

            # 置信区间
            # 简化版本
            uncertainty = self.alpha * np.sqrt(
                context @ A_inv @ context +
                shared_features @ A0_inv @ shared_features
            )

            ucb_values[arm] = pred + uncertainty

        return np.argmax(ucb_values)
```

## 4. 推荐系统应用

### 4.1 新闻推荐

```python
class NewsBanditRecommender:
    """
    新闻推荐多臂老虎机
    """

    def __init__(self, n_categories: int, context_dim: int):
        self.bandit = LinUCB(n_categories, context_dim)

        # 新闻缓存
        self.news_pool = {i: [] for i in range(n_categories)}

    def add_news(self, category: int, news_id: str, features: np.ndarray):
        """添加新闻到池"""
        self.news_pool[category].append({
            'id': news_id,
            'features': features
        })

    def recommend(self, user_context: np.ndarray, n: int = 10) -> List[str]:
        """推荐新闻"""
        # 使用 LinUCB 选择类别
        category = self.bandit.select_arm(user_context)

        # 从该类别中选择新闻
        news_list = self.news_pool[category]

        if len(news_list) <= n:
            return [news['id'] for news in news_list]
        else:
            # 随机选择（可以改进为基于特征的相似度）
            indices = np.random.choice(len(news_list), n, replace=False)
            return [news_list[i]['id'] for i in indices]

    def update(self, category: int, user_context: np.ndarray,
               clicked: bool):
        """更新反馈"""
        self.bandit.update(category, user_context, float(clicked))
```

### 4.2 广告投放优化

```python
class AdBanditOptimizer:
    """
    广告投放多臂老虎机优化
    """

    def __init__(self, n_ads: int, context_dim: int):
        self.n_ads = n_ads
        self.bandit = ThompsonSampling(n_ads)
        self.context_bandit = LinUCB(n_ads, context_dim)

    def select_ad(self, user_features: np.ndarray,
                 exploration_mode: str = 'thompson') -> int:
        """选择广告"""
        if exploration_mode == 'thompson':
            return self.bandit.select_arm()
        else:
            return self.context_bandit.select_arm(user_features)

    def update_feedback(self, ad_id: int, user_features: np.ndarray,
                       reward: float):
        """更新反馈"""
        # Thompson Sampling 更新
        self.bandit.update(ad_id, reward)

        # LinUCB 更新
        self.context_bandit.update(ad_id, user_features, reward)
```

## 5. 非平稳环境

### 5.1 滑动窗口 UCB

```python
class SlidingWindowUCB:
    """
    滑动窗口 UCB

    适用于非平稳环境，只考虑最近的奖励
    """

    def __init__(self, n_arms: int, window_size: int = 100, c: float = 2.0):
        self.n_arms = n_arms
        self.window_size = window_size
        self.c = c

        # 存储最近的历史
        self.history = [[] for _ in range(n_arms)]
        self.total_count = 0

    def select_arm(self) -> int:
        # 优先尝试未尝试的手臂
        for arm in range(self.n_arms):
            if len(self.history[arm]) == 0:
                return arm

        ucb_values = np.zeros(self.n_arms)

        for arm in range(self.n_arms):
            rewards = self.history[arm]
            if len(rewards) == 0:
                ucb_values[arm] = float('inf')
            else:
                avg_reward = np.mean(rewards)
                exploration = np.sqrt(
                    self.c * np.log(min(self.total_count, self.window_size)) / len(rewards)
                )
                ucb_values[arm] = avg_reward + exploration

        return np.argmax(ucb_values)

    def update(self, arm: int, reward: float):
        self.history[arm].append(reward)
        self.total_count += 1

        # 保持窗口大小
        if len(self.history[arm]) > self.window_size:
            self.history[arm].pop(0)
```

## 6. 评估与比较

### 6.1 遗憾分析

```python
def simulate_bandit(bandit, true_rewards: np.ndarray,
                   n_rounds: int) -> Tuple[List[float], float]:
    """
    模拟老虎机算法

    参数:
        bandit: 老虎机算法
        true_rewards: 每个手臂的真实奖励
        n_rounds: 模拟轮数

    返回:
        (累计奖励列表, 总遗憾)
    """
    optimal_arm = np.argmax(true_rewards)
    optimal_reward = true_rewards[optimal_arm]

    cumulative_rewards = []
    total_reward = 0
    total_regret = 0

    for _ in range(n_rounds):
        arm = bandit.select_arm()

        # 模拟奖励
        reward = np.random.binomial(1, true_rewards[arm])
        bandit.update(arm, reward)

        total_reward += reward
        total_regret += optimal_reward - true_rewards[arm]

        cumulative_rewards.append(total_reward)

    return cumulative_rewards, total_regret


def compare_algorithms(n_arms: int, true_rewards: np.ndarray,
                       n_rounds: int = 1000):
    """比较不同算法"""
    algorithms = {
        'ε-Greedy': EpsilonGreedy(n_arms, epsilon=0.1),
        'UCB': UCB(n_arms),
        'Thompson Sampling': ThompsonSampling(n_arms),
    }

    results = {}
    for name, bandit in algorithms.items():
        # 重新创建算法实例
        if 'ε-Greedy' in name:
            bandit = EpsilonGreedy(n_arms, epsilon=0.1)
        elif 'UCB' in name:
            bandit = UCB(n_arms)
        else:
            bandit = ThompsonSampling(n_arms)

        rewards, regret = simulate_bandit(bandit, true_rewards, n_rounds)
        results[name] = {
            'cumulative_rewards': rewards,
            'regret': regret
        }

    return results
```

## 7. 学习总结

### 7.1 算法对比

| 算法 | 复杂度 | 探索方式 | 适用场景 |
|------|--------|----------|----------|
| ε-Greedy | O(1) | 随机 | 简单场景 |
| UCB | O(K) | 置信区间 | 需要理论保证 |
| Thompson Sampling | O(K) | 概率匹配 | 不确定性量化 |
| LinUCB | O(Kd²) | 上下文 | 个性化推荐 |

### 7.2 选择建议

```
1. 无上下文: Thompson Sampling（效果好）
2. 有上下文: LinUCB
3. 非平稳: 滑动窗口或衰减
4. 简单场景: ε-Greedy
```
