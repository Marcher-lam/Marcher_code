# 多模态推荐系统 学习文档

## 1. 多模态推荐概述

### 1.1 什么是多模态推荐？

```
多模态推荐 (Multimodal Recommendation):

- 利用多种模态信息进行推荐
- 物品包含: 文本、图像、视频、音频
- 用户行为: 点击、浏览、评论等

优势:
1. 更丰富的物品表示
2. 更好的冷启动处理
3. 更准确的相似度计算
4. 跨模态语义理解

应用场景:
- 电商: 商品图片 + 描述文本
- 短视频: 视频内容 + 标题 + 音频
- 音乐: 音频 + 封面 + 歌词
- 新闻: 标题 + 正文 + 图片
```

    多模态物品表示
    """

    def __init__(self, item_id: int):

        self.modalities: Dict[str, np.ndarray] = {}

    def add_modality(self, modality: str, features: np.ndarray):

        self.modalities[modality] = features

    def get_modality(self, modality: str) -> Optional[np.ndarray]:
        """获取模态特征"""
        return self.modalities.get(modality)

## 2. 文本模态处理

### 2.1 文本编码器

```python
class TextEncoder(nn.Module):
    """
    文本编码器

    将文本转换为向量表示
    """

    def __init__(self,
                 vocab_size: int = 50000,
                 embed_dim: int = 256,
                 hidden_dim: int = 512,
                 output_dim: int = 128,
                 n_layers: int = 2,
                 n_heads: int = 4,
                 pretrained: bool = False):
        super().__init__()

        self.embed_dim = embed_dim
        self.output_dim = output_dim

        # 词嵌入
        self.embedding = nn.Embedding(vocab_size, embed_dim)

        # Transformer 编码器
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim,
            nhead=n_heads,
            dim_feedforward=hidden_dim,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)

        # 输出层
        self.output_layer = nn.Sequential(
            nn.Linear(embed_dim, output_dim),
            nn.LayerNorm(output_dim)
        )

    def forward(self, text_ids: torch.Tensor,
                attention_mask: torch.Tensor = None) -> torch.Tensor:
        """
        前向传播

        参数:
            text_ids: (batch, seq_len) 文本 token ids
            attention_mask: (batch, seq_len) 注意力掩码

        返回:
            (batch, output_dim) 文本表示
        """
        # 嵌入
        x = self.embedding(text_ids)  # (batch, seq_len, embed_dim)

        # Transformer
        if attention_mask is not None:
            # 转换为 Transformer 期望的格式
            src_key_padding_mask = (attention_mask == 0)
            x = self.transformer(x, src_key_padding_mask=src_key_padding_mask)
        else:
            x = self.transformer(x)

        # 池化 (使用第一个 token 或平均池化)
        # 这里使用平均池化
        if attention_mask is not None:
            mask = attention_mask.unsqueeze(-1)
            x = (x * mask).sum(dim=1) / mask.sum(dim=1).clamp(min=1)
        else:
            x = x.mean(dim=1)

        # 输出
        return self.output_layer(x)


class BERTTextEncoder(nn.Module):
    """
    基于 BERT 的文本编码器
    """

    def __init__(self, model_name: str = 'bert-base-uncased',
                 output_dim: int = 128,
                 freeze_bert: bool = False):
        super().__init__()

        from transformers import BertModel, BertTokenizer

        self.bert = BertModel.from_pretrained(model_name)
        self.tokenizer = BertTokenizer.from_pretrained(model_name)

        if freeze_bert:
            for param in self.bert.parameters():
                param.requires_grad = False

        # 投影层
        self.projection = nn.Sequential(
            nn.Linear(768, output_dim),
            nn.LayerNorm(output_dim)
        )

    def forward(self, text_ids: torch.Tensor,
                attention_mask: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        outputs = self.bert(text_ids, attention_mask=attention_mask)

        # 使用 [CLS] token 的表示
        cls_output = outputs.last_hidden_state[:, 0, :]

        return self.projection(cls_output)
```

class ImageEncoder(nn.Module):
    """
    图像编码器

    将图像转换为向量表示
    """

    def __init__(self,
                 output_dim: int = 128,
                 pretrained: bool = True,
                 backbone: str = 'resnet50'):
        super().__init__()

        import torchvision.models as models

        # 选择骨干网络
        if backbone == 'resnet50':
            if pretrained:
                self.backbone = models.resnet50(pretrained=True)
            else:
                self.backbone = models.resnet50()
            feature_dim = 2048
        elif backbone == 'resnet18':
            if pretrained:
                self.backbone = models.resnet18(pretrained=True)
            else:
                self.backbone = models.resnet18()
            feature_dim = 512

        # 移除最后的全连接层
        self.backbone = nn.Sequential(*list(self.backbone.children())[:-1])

        # 投影层
        self.projection = nn.Sequential(
            nn.Flatten(),
            nn.Linear(feature_dim, output_dim),
            nn.LayerNorm(output_dim)
        )

    def forward(self, images: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        参数:
            images: (batch, 3, H, W) 图像张量

        返回:
            (batch, output_dim) 图像表示
        """
        features = self.backbone(images)
        return self.projection(features)

    """

    def __init__(self,

                 output_dim: int = 128,
                 freeze: bool = False):
        super().__init__()

        from transformers import ViTModel, ViTImageProcessor

        self.vit = ViTModel.from_pretrained(model_name)
        self.processor = ViTImageProcessor.from_pretrained(model_name)

        if freeze:
            for param in self.vit.parameters():
                param.requires_grad = False

        self.projection = nn.Sequential(
            nn.Linear(768, output_dim),
            nn.LayerNorm(output_dim)
        )

    def forward(self, pixel_values: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        outputs = self.vit(pixel_values)

        # 使用 [CLS] token
        cls_output = outputs.last_hidden_state[:, 0, :]

        return self.projection(cls_output)

## 4. 多模态融合

### 4.1 早期融合

```python
class EarlyFusion(nn.Module):
    """
    早期融合

    在特征层直接拼接
    """

    def __init__(self,
                 text_dim: int = 128,
                 image_dim: int = 128,
                 output_dim: int = 256,
                 fusion_type: str = 'concat'):
        super().__init__()

        self.fusion_type = fusion_type

        if fusion_type == 'concat':
            fusion_input_dim = text_dim + image_dim
        elif fusion_type == 'add':
            assert text_dim == image_dim
            fusion_input_dim = text_dim
        elif fusion_type == 'gate':
            fusion_input_dim = text_dim  # 门控融合后维度

        self.fusion = nn.Sequential(
            nn.Linear(fusion_input_dim, output_dim),
            nn.ReLU(),
            nn.LayerNorm(output_dim)
        )

        # 门控网络 (如果使用门控融合)
        if fusion_type == 'gate':
            self.gate = nn.Sequential(
                nn.Linear(text_dim + image_dim, 128),
                nn.ReLU(),
                nn.Linear(128, 2),
                nn.Softmax(dim=-1)
            )

    def forward(self, text_features: torch.Tensor,
                image_features: torch.Tensor) -> torch.Tensor:
        """
        前向传播
        """
        if self.fusion_type == 'concat':
            fused = torch.cat([text_features, image_features], dim=-1)
        elif self.fusion_type == 'add':
            fused = text_features + image_features
        elif self.fusion_type == 'gate':
            gate_input = torch.cat([text_features, image_features], dim=-1)
            gate_weights = self.gate(gate_input)
            fused = (gate_weights[:, 0:1] * text_features +
                    gate_weights[:, 1:2] * image_features)

        return self.fusion(fused)
```

### 4.2 注意力融合

```python
class CrossModalAttention(nn.Module):
    """
    跨模态注意力

    使用一个模态查询另一个模态
    """

    def __init__(self, dim: int = 128, n_heads: int = 4, dropout: float = 0.1):
        super().__init__()

        self.attention = nn.MultiheadAttention(
            embed_dim=dim,
            num_heads=n_heads,
            dropout=dropout,
            batch_first=True
        )

        self.norm = nn.LayerNorm(dim)

    def forward(self, query: torch.Tensor,
                key_value: torch.Tensor) -> torch.Tensor:
        """
        前向传播

        query: (batch, dim) 查询模态
        key_value: (batch, dim) 被查询模态
        """
        # 扩展维度
        query = query.unsqueeze(1)  # (batch, 1, dim)
        key_value = key_value.unsqueeze(1)  # (batch, 1, dim)

        # 注意力
        attended, _ = self.attention(query, key_value, key_value)

        # 残差 + 归一化
        output = self.norm(query + attended)

        return output.squeeze(1)


class MultimodalAttentionFusion(nn.Module):
    """
    多模态注意力融合

    使用双向跨模态注意力
    """

    def __init__(self, dim: int = 128, n_heads: int = 4, dropout: float = 0.1):
        super().__init__()

        # 文本 -> 图像 注意力
        self.text_to_image = CrossModalAttention(dim, n_heads, dropout)

        # 图像 -> 文本 注意力
        self.image_to_text = CrossModalAttention(dim, n_heads, dropout)

        # 融合层
        self.fusion = nn.Sequential(
            nn.Linear(dim * 2, dim),
            nn.LayerNorm(dim)
        )

    def forward(self, text_features: torch.Tensor,
                image_features: torch.Tensor) -> torch.Tensor:
        """
        前向传播
        """
        # 双向注意力
        text_attended = self.text_to_image(text_features, image_features)
        image_attended = self.image_to_text(image_features, text_features)

        # 融合
        fused = torch.cat([text_attended, image_attended], dim=-1)
        fused = self.fusion(fused)

        return fused
```

### 4.3 多模态推荐模型

```python
class MultimodalRecommender(nn.Module):
    """
    多模态推荐模型
    """

    def __init__(self,
                 n_users: int,
                 n_items: int,
                 text_dim: int = 128,
                 image_dim: int = 128,
                 fusion_dim: int = 128,
                 hidden_dim: int = 256):
        super().__init__()

        # 用户嵌入
        self.user_embedding = nn.Embedding(n_users, fusion_dim)

        # 物品编码器
        self.text_encoder = TextEncoder(output_dim=text_dim)
        self.image_encoder = ImageEncoder(output_dim=image_dim)

        # 多模态融合
        self.fusion = MultimodalAttentionFusion(
            dim=max(text_dim, image_dim),
            n_heads=4
        )

        # 对齐层 (如果维度不同)
        self.text_proj = nn.Linear(text_dim, max(text_dim, image_dim))
        self.image_proj = nn.Linear(image_dim, max(text_dim, image_dim))

        # 预测网络
        self.predictor = nn.Sequential(
            nn.Linear(fusion_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1)
        )

    def encode_item(self, text_ids: torch.Tensor,
                    attention_mask: torch.Tensor,
                    images: torch.Tensor) -> torch.Tensor:
        """编码物品"""
        # 编码各模态
        text_features = self.text_encoder(text_ids, attention_mask)
        image_features = self.image_encoder(images)

        # 对齐维度
        text_features = self.text_proj(text_features)
        image_features = self.image_proj(image_features)

        # 融合
        item_features = self.fusion(text_features, image_features)

        return item_features

    def forward(self, user_ids: torch.Tensor,
                item_text_ids: torch.Tensor,
                item_attention_mask: torch.Tensor,
                item_images: torch.Tensor) -> torch.Tensor:
        """
        前向传播
        """
        # 用户嵌入
        user_embed = self.user_embedding(user_ids)

        # 物品编码
        item_embed = self.encode_item(
            item_text_ids, item_attention_mask, item_images
        )

        # 预测
        combined = torch.cat([user_embed, item_embed], dim=-1)
        output = self.predictor(combined)

        return output


class MultimodalRecall(nn.Module):
    """
    多模态召回模型
    """

    def __init__(self,
                 n_items: int,
                 text_dim: int = 128,
                 image_dim: int = 128,
                 embed_dim: int = 128):
        super().__init__()

        self.n_items = n_items
        self.embed_dim = embed_dim

        # 编码器
        self.text_encoder = TextEncoder(output_dim=text_dim)
        self.image_encoder = ImageEncoder(output_dim=image_dim)

        # 融合
        self.item_fusion = EarlyFusion(
            text_dim=text_dim,
            image_dim=image_dim,
            output_dim=embed_dim,
            fusion_type='concat'
        )

        # 用户编码器 (行为序列)
        self.user_encoder = nn.TransformerEncoder(
            nn.TransformerEncoderLayer(
                d_model=embed_dim,
                nhead=4,
                batch_first=True
            ),
            num_layers=2
        )

    def encode_items(self, text_features: torch.Tensor,
                     image_features: torch.Tensor) -> torch.Tensor:
        """
        编码所有物品 (离线计算)
        """
        return self.item_fusion(text_features, image_features)

    def encode_user(self, history_item_embeds: torch.Tensor,
                    history_mask: torch.Tensor = None) -> torch.Tensor:
        """
        编码用户 (在线计算)
        """
        if history_mask is not None:
            src_key_padding_mask = (history_mask == 0)
            user_embed = self.user_encoder(
                history_item_embeds,
                src_key_padding_mask=src_key_padding_mask
            )
        else:
            user_embed = self.user_encoder(history_item_embeds)

        # 取最后位置的输出
        return user_embed[:, -1, :]

    def recall(self, user_embed: torch.Tensor,
               all_item_embeds: torch.Tensor,
               top_k: int = 100) -> torch.Tensor:
        """
        召回 Top-K 物品
        """
        # 计算相似度
        scores = torch.matmul(user_embed, all_item_embeds.T)

        # Top-K
        _, indices = torch.topk(scores, top_k, dim=-1)

        return indices
```

## 5. 跨模态预训练

### 5.1 CLIP 风格预训练

```python
class CLIPStylePretraining(nn.Module):
    """
    CLIP 风格的图文预训练
    """

    def __init__(self,
                 text_encoder: nn.Module,
                 image_encoder: nn.Module,
                 embed_dim: int = 256,
                 temperature: float = 0.07):
        super().__init__()

        self.text_encoder = text_encoder
        self.image_encoder = image_encoder

        # 投影层
        self.text_projection = nn.Linear(128, embed_dim)
        self.image_projection = nn.Linear(128, embed_dim)

        self.temperature = nn.Parameter(torch.tensor(temperature))

    def forward(self, text_ids: torch.Tensor,
                attention_mask: torch.Tensor,
                images: torch.Tensor) -> Dict[str, torch.Tensor]:
        """
        前向传播

        返回对比学习损失
        """
        # 编码
        text_features = self.text_encoder(text_ids, attention_mask)
        image_features = self.image_encoder(images)

        # 投影
        text_embeds = self.text_projection(text_features)
        image_embeds = self.image_projection(image_features)

        # 归一化
        text_embeds = F.normalize(text_embeds, dim=-1)
        image_embeds = F.normalize(image_embeds, dim=-1)

        # 计算相似度矩阵
        logits = torch.matmul(text_embeds, image_embeds.T) / self.temperature

        # 对比损失
        batch_size = text_embeds.shape[0]
        labels = torch.arange(batch_size, device=text_embeds.device)

        loss_i2t = F.cross_entropy(logits, labels)
        loss_t2i = F.cross_entropy(logits.T, labels)

        loss = (loss_i2t + loss_t2i) / 2

        return {
            'loss': loss,
            'logits': logits,
            'text_embeds': text_embeds,
            'image_embeds': image_embeds
        }
```

## 6. 学习总结

### 6.1 多模态方法对比

```
融合策略    优点                缺点
───────────────────────────────────────────
早期融合    简单直接            信息损失
晚期融合    各模态独立          缺乏跨模态交互
注意力融合  自适应融合          计算复杂
预训练      效果好              需要大量数据
```

### 6.2 最佳实践

```
1. 模态选择: 根据场景选择有信息量的模态
2. 预训练: 使用预训练模型提取特征
3. 对齐: 确保各模态特征空间对齐
4. 融合: 根据任务选择合适的融合策略
5. 冷启动: 多模态特别适合处理冷启动
```
