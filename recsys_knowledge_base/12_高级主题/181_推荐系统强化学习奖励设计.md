# 推荐系统强化学习奖励设计 学习文档

## 1. 奖励设计概述

### 1.1 奖励函数的重要性

```
强化学习中，奖励函数定义了优化目标:

"奖励设计 = 定义推荐系统的目标"

推荐系统常见目标:
- 点击率 (CTR)
- 转化率 (CVR)
- 用户停留时长
- GMV (商品交易总额)
- 用户满意度
- 长期留存

挑战:
- 多目标平衡
- 延迟奖励 (长期效果)
- 稀疏奖励 (转化很少)
- 奖励塑造 (Reward Shaping)
```

### 1.2 奖励类型

```python
"""
奖励类型:

1. 即时奖励 (Immediate Reward)
   - 点击、点赞、收藏
   - 立即可观测

2. 延迟奖励 (Delayed Reward)
   - 转化、复购、留存
   - 需要时间才能观测

3. 内在奖励 (Intrinsic Reward)
   - 探索奖励
   - 好奇心驱动

4. 形状奖励 (Shaped Reward)
   - 人为设计的中间奖励
   - 引导学习
"""

from typing import List, Dict, Tuple, Optional, Callable
import numpy as np
from dataclasses import dataclass
from enum import Enum


class RewardType(Enum):
    """奖励类型"""
    CLICK = "click"
    PURCHASE = "purchase"
    LIKE = "like"
    SHARE = "share"
    DWELL_TIME = "dwell_time"
    WATCH_TIME = "watch_time"
    RETENTION = "retention"


@dataclass
class RewardConfig:
    """奖励配置"""
    reward_type: RewardType
    weight: float = 1.0
    delay_steps: int = 0  # 延迟步数


class BaseRewardFunction:
    """
    奖励函数基类
    """

    def __init__(self, config: RewardConfig = None):
        self.config = config or RewardConfig(RewardType.CLICK)

    def compute(self, context: Dict) -> float:
        """
        计算奖励

        context: 包含用户、物品、行为等信息
        """
        raise NotImplementedError

    def __call__(self, context: Dict) -> float:
        return self.compute(context)
```

## 2. 基础奖励函数

### 2.1 点击奖励

```python
class ClickReward(BaseRewardFunction):
    """
    点击奖励

    最基础的奖励类型
    """

    def __init__(self, click_weight: float = 1.0):
        super().__init__(RewardConfig(RewardType.CLICK, click_weight))
        self.click_weight = click_weight

    def compute(self, context: Dict) -> float:
        """
        计算点击奖励

        context:
            - is_clicked: bool
        """
        is_clicked = context.get('is_clicked', False)
        return self.click_weight if is_clicked else 0.0


class WeightedActionReward(BaseRewardFunction):
    """
    加权行为奖励

    不同行为赋予不同权重
    """

    def __init__(self,
                 click_weight: float = 1.0,
                 like_weight: float = 2.0,
                 share_weight: float = 5.0,
                 purchase_weight: float = 10.0):
        super().__init__()

        self.action_weights = {
            'click': click_weight,
            'like': like_weight,
            'share': share_weight,
            'purchase': purchase_weight,
            'add_to_cart': 3.0,
            'favorite': 2.5,
            'comment': 3.0
        }

    def compute(self, context: Dict) -> float:
        """计算加权奖励"""
        total_reward = 0.0

        for action, weight in self.action_weights.items():
            if context.get(action, False):
                total_reward += weight

        return total_reward

    def add_action(self, action: str, weight: float):
        """添加新的行为类型"""
        self.action_weights[action] = weight


class DwellTimeReward(BaseRewardFunction):
    """
    停留时长奖励

    基于用户在内容上停留的时间
    """

    def __init__(self,
                 min_threshold: float = 5.0,
                 max_threshold: float = 300.0,
                 reward_scale: float = 1.0):
        super().__init__(RewardConfig(RewardType.DWELL_TIME))
        self.min_threshold = min_threshold  # 最小有效时长 (秒)
        self.max_threshold = max_threshold  # 最大时长
        self.reward_scale = reward_scale

    def compute(self, context: Dict) -> float:
        """
        计算时长奖励

        使用对数函数避免长尾影响
        """
        dwell_time = context.get('dwell_time', 0.0)

        if dwell_time < self.min_threshold:
            return 0.0

        # 对数缩放
        normalized_time = min(dwell_time, self.max_threshold)
        reward = np.log1p(normalized_time - self.min_threshold) * self.reward_scale

        return reward


class WatchTimeReward(BaseRewardFunction):
    """
    观看时长奖励 (视频)

    考虑完播率
    """

    def __init__(self,
                 completion_bonus: float = 2.0,
                 time_weight: float = 0.01):
        super().__init__(RewardConfig(RewardType.WATCH_TIME))
        self.completion_bonus = completion_bonus
        self.time_weight = time_weight

    def compute(self, context: Dict) -> float:
        """
        计算观看奖励

        context:
            - watch_time: 观看时长
            - video_duration: 视频总时长
        """
        watch_time = context.get('watch_time', 0)
        video_duration = context.get('video_duration', 1)

        # 时长奖励
        time_reward = watch_time * self.time_weight

        # 完播奖励
        completion_rate = watch_time / video_duration if video_duration > 0 else 0
        completion_reward = self.completion_bonus if completion_rate > 0.9 else 0

        return time_reward + completion_reward
```

### 2.2 转化奖励

```python
class ConversionReward(BaseRewardFunction):
    """
    转化奖励

    处理延迟转化
    """

    def __init__(self,
                 conversion_weight: float = 10.0,
                 attribution_window: int = 7 * 24 * 3600):  # 7天归因窗口
        super().__init__(RewardConfig(RewardType.PURCHASE, conversion_weight))
        self.conversion_weight = conversion_weight
        self.attribution_window = attribution_window

    def compute(self, context: Dict) -> float:
        """
        计算转化奖励

        context:
            - is_converted: bool
            - conversion_value: float (转化金额)
        """
        if not context.get('is_converted', False):
            return 0.0

        conversion_value = context.get('conversion_value', 1.0)

        # 可以基于金额调整奖励
        return self.conversion_weight * np.log1p(conversion_value)


class GMVReward(BaseRewardFunction):
    """
    GMV (商品交易总额) 奖励

    综合考虑转化率和客单价
    """

    def __init__(self,
                 gmv_weight: float = 1.0,
                 log_transform: bool = True):
        super().__init__()
        self.gmv_weight = gmv_weight
        self.log_transform = log_transform

    def compute(self, context: Dict) -> float:
        """
        计算 GMV 奖励

        context:
            - is_purchased: bool
            - price: float
        """
        if not context.get('is_purchased', False):
            return 0.0

        price = context.get('price', 0)

        if self.log_transform:
            # 对数变换避免大额订单主导
            return self.gmv_weight * np.log1p(price)
        else:
            return self.gmv_weight * price
```

## 3. 多目标奖励

### 3.1 组合奖励

```python
class MultiObjectiveReward(BaseRewardFunction):
    """
    多目标组合奖励

    平衡多个业务目标
    """

    def __init__(self,
                 objectives: Dict[str, Tuple[Callable, float]]):
        """
        参数:
            objectives: {
                'click': (click_reward_fn, 1.0),
                'dwell_time': (dwell_reward_fn, 0.5),
                'purchase': (purchase_reward_fn, 5.0)
            }
        """
        super().__init__()
        self.objectives = objectives

    def compute(self, context: Dict) -> float:
        """计算组合奖励"""
        total_reward = 0.0

        for name, (reward_fn, weight) in self.objectives.items():
            reward = reward_fn(context)
            total_reward += weight * reward

        return total_reward

    @classmethod
    def create_default(cls) -> 'MultiObjectiveReward':
        """创建默认的多目标奖励"""
        click_reward = ClickReward()
        dwell_reward = DwellTimeReward()
        conversion_reward = ConversionReward()

        objectives = {
            'click': (click_reward, 1.0),
            'dwell_time': (dwell_reward, 0.01),
            'conversion': (conversion_reward, 10.0)
        }

        return cls(objectives)


class WeightedSumReward(BaseRewardFunction):
    """
    加权求和奖励

    简单的线性组合
    """

    def __init__(self, weights: Dict[str, float]):
        super().__init__()
        self.weights = weights

    def compute(self, context: Dict) -> float:
        """计算加权求和"""
        total = 0.0

        for key, weight in self.weights.items():
            value = context.get(key, 0)
            if isinstance(value, bool):
                value = 1.0 if value else 0.0
            total += weight * value

        return total
```

### 3.2 动态权重奖励

```python
class DynamicWeightReward(BaseRewardFunction):
    """
    动态权重奖励

    根据用户/场景调整权重
    """

    def __init__(self, base_weights: Dict[str, float]):
        super().__init__()
        self.base_weights = base_weights
        self.current_weights = base_weights.copy()

    def compute(self, context: Dict) -> float:
        """计算动态权重奖励"""
        # 根据用户类型调整
        user_type = context.get('user_type', 'normal')

        if user_type == 'new_user':
            # 新用户更关注点击和时长
            self.current_weights = {
                'click': 2.0,
                'dwell_time': 0.02,
                'conversion': 1.0
            }
        elif user_type == 'high_value':
            # 高价值用户更关注转化
            self.current_weights = {
                'click': 1.0,
                'dwell_time': 0.01,
                'conversion': 15.0
            }
        else:
            self.current_weights = self.base_weights.copy()

        # 计算奖励
        total = 0.0
        for key, weight in self.current_weights.items():
            value = context.get(key, 0)
            if isinstance(value, bool):
                value = 1.0 if value else 0.0
            total += weight * value

        return total
```

## 4. 延迟奖励处理

### 4.1 延迟奖励归因

```python
class DelayedRewardHandler:
    """
    延迟奖励处理器

    处理转化的延迟归因
    """

    def __init__(self, max_delay: int = 7 * 24 * 3600):
        self.max_delay = max_delay

        # 存储待归因的转化
        # {user_id: [(timestamp, item_id, reward), ...]}
        self.pending_conversions = {}

        # 存储历史曝光
        # {user_id: [(timestamp, item_id, state, action), ...]}
        self.exposure_history = {}

    def record_exposure(self, user_id: int, item_id: int,
                       state: np.ndarray, action: int, timestamp: float):
        """记录曝光"""
        if user_id not in self.exposure_history:
            self.exposure_history[user_id] = []

        self.exposure_history[user_id].append({
            'timestamp': timestamp,
            'item_id': item_id,
            'state': state,
            'action': action
        })

        # 清理过期记录
        self._cleanup_history(user_id, timestamp)

    def record_conversion(self, user_id: int, item_id: int,
                         reward: float, timestamp: float):
        """记录转化"""
        if user_id not in self.pending_conversions:
            self.pending_conversions[user_id] = []

        self.pending_conversions[user_id].append({
            'timestamp': timestamp,
            'item_id': item_id,
            'reward': reward
        })

    def get_delayed_rewards(self, user_id: int,
                           current_time: float) -> List[Dict]:
        """
        获取延迟奖励

        将转化归因到历史曝光
        """
        delayed_rewards = []

        if user_id not in self.pending_conversions:
            return delayed_rewards

        conversions = self.pending_conversions[user_id]
        exposures = self.exposure_history.get(user_id, [])

        for conv in conversions:
            conv_time = conv['timestamp']
            conv_item = conv['item_id']

            # 找到对应的曝光 (最后点击归因)
            for exposure in reversed(exposures):
                time_diff = conv_time - exposure['timestamp']

                if time_diff > self.max_delay or time_diff < 0:
                    continue

                if exposure['item_id'] == conv_item:
                    delayed_rewards.append({
                        'state': exposure['state'],
                        'action': exposure['action'],
                        'reward': conv['reward'],
                        'delay': time_diff
                    })
                    break

        # 清理已处理的转化
        self.pending_conversions[user_id] = []

        return delayed_rewards

    def _cleanup_history(self, user_id: int, current_time: float):
        """清理过期历史"""
        if user_id not in self.exposure_history:
            return

        self.exposure_history[user_id] = [
            h for h in self.exposure_history[user_id]
            if current_time - h['timestamp'] <= self.max_delay
        ]
```

### 4.2 奖励衰减

```python
class RewardDecay:
    """
    奖励衰减

    处理延迟奖励的时效性
    """

    @staticmethod
    def exponential_decay(reward: float, delay: float,
                         decay_rate: float = 0.1) -> float:
        """
        指数衰减

        reward_discounted = reward * exp(-decay_rate * delay)
        """
        return reward * np.exp(-decay_rate * delay)

    @staticmethod
    def linear_decay(reward: float, delay: float,
                    max_delay: float) -> float:
        """
        线性衰减

        reward_discounted = reward * (1 - delay / max_delay)
        """
        if delay >= max_delay:
            return 0.0
        return reward * (1 - delay / max_delay)

    @staticmethod
    def step_decay(reward: float, delay: float,
                  thresholds: List[float]) -> float:
        """
        阶梯衰减

        thresholds: [(delay_threshold, discount_factor), ...]
        """
        discount = 1.0

        for threshold, factor in sorted(thresholds):
            if delay >= threshold:
                discount = factor

        return reward * discount
```

## 5. 内在奖励与探索

### 5.1 探索奖励

```python
class ExplorationReward(BaseRewardFunction):
    """
    探索奖励 (内在奖励)

    鼓励探索新的状态-动作对
    """

    def __init__(self, bonus_scale: float = 0.1):
        super().__init__()
        self.bonus_scale = bonus_scale
        self.visit_counts = {}  # (state_hash, action) -> count

    def compute(self, context: Dict) -> float:
        """
        计算探索奖励

        基于状态-动作对的访问次数
        """
        state = context.get('state')
        action = context.get('action')

        state_hash = self._hash_state(state)
        key = (state_hash, action)

        # 访问次数
        count = self.visit_counts.get(key, 0)
        self.visit_counts[key] = count + 1

        # 探索奖励 (访问越少，奖励越高)
        exploration_bonus = self.bonus_scale / np.sqrt(count + 1)

        return exploration_bonus

    def _hash_state(self, state: np.ndarray) -> int:
        """状态哈希"""
        if state is None:
            return 0
        return hash(state.tobytes())


class CuriosityReward(BaseRewardFunction):
    """
    好奇心奖励

    基于预测误差的内在奖励
    """

    def __init__(self, state_dim: int, action_dim: int,
                 hidden_dim: int = 128, eta: float = 0.01):
        super().__init__()
        self.eta = eta

        # 前向动力学模型: 预测下一个状态
        self.forward_model = nn.Sequential(
            nn.Linear(state_dim + action_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, state_dim)
        )

        # 逆向动力学模型: 预测动作
        self.inverse_model = nn.Sequential(
            nn.Linear(state_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim)
        )

    def compute(self, context: Dict) -> float:
        """
        计算好奇心奖励

        基于预测误差
        """
        state = context.get('state')
        action = context.get('action')
        next_state = context.get('next_state')

        if state is None or next_state is None:
            return 0.0

        # 预测下一个状态
        action_onehot = F.one_hot(torch.tensor(action),
                                  self.action_dim).float()
        state_tensor = torch.tensor(state, dtype=torch.float32)
        input_tensor = torch.cat([state_tensor, action_onehot])

        predicted_next_state = self.forward_model(input_tensor)

        # 预测误差作为奖励
        error = F.mse_loss(predicted_next_state,
                          torch.tensor(next_state, dtype=torch.float32))

        return self.eta * error.item()
```

## 6. 奖励归一化

```python
class RewardNormalizer:
    """
    奖励归一化

    使奖励分布在合理范围
    """

    def __init__(self, method: str = 'running_mean_std'):
        self.method = method
        self.mean = 0.0
        self.std = 1.0
        self.count = 0
        self.min_reward = float('inf')
        self.max_reward = float('-inf')

    def update(self, reward: float):
        """更新统计量"""
        self.count += 1
        self.min_reward = min(self.min_reward, reward)
        self.max_reward = max(self.max_reward, reward)

        # 在线更新均值和方差
        delta = reward - self.mean
        self.mean += delta / self.count
        delta2 = reward - self.mean
        # 简化，实际应该用 Welford 算法

    def normalize(self, reward: float) -> float:
        """归一化奖励"""
        if self.method == 'running_mean_std':
            return (reward - self.mean) / (self.std + 1e-8)
        elif self.method == 'min_max':
            if self.max_reward == self.min_reward:
                return 0.0
            return (reward - self.min_reward) / (self.max_reward - self.min_reward)
        elif self.method == 'clip':
            return np.clip(reward, -10, 10)
        else:
            return reward
```

## 7. 学习总结

### 7.1 奖励设计原则

```
1. 明确目标: 奖励要反映业务目标
2. 稀疏vs稠密: 平衡信息量和简洁性
3. 即时vs延迟: 处理好延迟奖励
4. 归一化: 使奖励在合理范围
5. 避免奖励黑客: 防止策略钻空子
```

### 7.2 常见奖励设计

```
场景          推荐奖励设计
───────────────────────────────────────
电商          CTR + CVR + GMV权重
短视频        完播率 + 时长 + 互动
信息流        点击 + 停留时长
广告          CTR × 出价
内容社区      互动 + 留存
```

### 7.3 最佳实践

```
1. 从简单奖励开始
2. 逐步添加复杂目标
3. 监控奖励分布
4. A/B 测试验证效果
5. 定期审查奖励设计
```
