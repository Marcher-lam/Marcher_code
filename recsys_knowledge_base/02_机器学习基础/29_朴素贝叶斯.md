# 朴素贝叶斯 学习文档

## 1. 算法基础认知

### 1.1 什么是朴素贝叶斯？

```
朴素贝叶斯 (Naive Bayes):

- 基于贝叶斯定理的概率分类器
- "朴素"指特征条件独立性假设
- 适合文本分类、垃圾邮件过滤等

在推荐系统中的应用:
- 文本分类 (新闻分类、情感分析)
- 标签推荐
- 用户意图识别
- 内容过滤
```

### 1.2 贝叶斯定理回顾

```python
"""
贝叶斯定理:

P(A|B) = P(B|A) * P(A) / P(B)

在分类中:
P(类别|特征) = P(特征|类别) * P(类别) / P(特征)

由于 P(特征) 对所有类别相同，可以简化为:
P(类别|特征) ∝ P(特征|类别) * P(类别)
"""
```

## 2. 核心原理

### 2.1 多项式朴素贝叶斯

```python
class MultinomialNaiveBayes:
    """
    多项式朴素贝叶斯

    适用于离散特征（如词频）
    常用于文本分类

    P(特征_i|类别) = (N_特征_i,类别 + α) / (N_类别 + α * n_特征)
    """

    def __init__(self, alpha: float = 1.0):
        """
        参数:
            alpha: Laplace 平滑参数
        """
        self.alpha = alpha
        self.class_log_prior = None    # log P(类别)
        self.feature_log_prob = None   # log P(特征|类别)
        self.classes = None

    def fit(self, X: np.ndarray, y: np.ndarray):
        """
        训练

        X: (n_samples, n_features) 特征矩阵（词频）
        y: (n_samples,) 标签
        """
        self.classes = np.unique(y)
        n_classes = len(self.classes)
        n_features = X.shape[1]

        # 计算类别先验概率
        class_counts = np.array([np.sum(y == c) for c in self.classes])
        self.class_log_prior = np.log(class_counts / class_counts.sum())

        # 计算特征条件概率
        self.feature_log_prob = np.zeros((n_classes, n_features))

        for i, c in enumerate(self.classes):
            # 获取该类别的所有样本
            X_c = X[y == c]

            # 每个特征在该类别中出现的次数
            feature_counts = X_c.sum(axis=0) + self.alpha

            # 该类别所有特征的总数
            total_count = feature_counts.sum()

            # log 概率
            self.feature_log_prob[i] = np.log(feature_counts / total_count)

        return self

    def predict_log_proba(self, X: np.ndarray) -> np.ndarray:
        """
        预测对数概率

        返回: (n_samples, n_classes)
        """
        # log P(类别|特征) ∝ log P(类别) + log P(特征|类别)
        # = log P(类别) + sum(log P(特征_i|类别))
        return X @ self.feature_log_prob.T + self.class_log_prior

    def predict_proba(self, X: np.ndarray) -> np.ndarray:
        """预测概率"""
        log_proba = self.predict_log_proba(X)

        # Log-Sum-Exp 技巧避免数值溢出
        log_proba_max = log_proba.max(axis=1, keepdims=True)
        log_proba_normalized = log_proba - log_proba_max
        proba = np.exp(log_proba_normalized)
        proba = proba / proba.sum(axis=1, keepdims=True)

        return proba

    def predict(self, X: np.ndarray) -> np.ndarray:
        """预测类别"""
        log_proba = self.predict_log_proba(X)
        return self.classes[np.argmax(log_proba, axis=1)]


class BernoulliNaiveBayes:
    """
    伯努利朴素贝叶斯

    适用于二值特征（词是否出现）
    适合短文本分类

    P(特征_i|类别) = P(特征_i=1|类别)^x_i * (1-P(特征_i=1|类别))^(1-x_i)
    """

    def __init__(self, alpha: float = 1.0, binarize: float = 0.0):
        self.alpha = alpha
        self.binarize = binarize
        self.class_log_prior = None
        self.feature_log_prob = None
        self.feature_log_neg_prob = None
        self.classes = None

    def fit(self, X: np.ndarray, y: np.ndarray):
        """训练"""
        # 二值化
        if self.binarize is not None:
            X = (X > self.binarize).astype(int)

        self.classes = np.unique(y)
        n_classes = len(self.classes)
        n_features = X.shape[1]

        # 类别先验
        class_counts = np.array([np.sum(y == c) for c in self.classes])
        self.class_log_prior = np.log(class_counts / class_counts.sum())

        # 特征条件概率
        self.feature_log_prob = np.zeros((n_classes, n_features))
        self.feature_log_neg_prob = np.zeros((n_classes, n_features))

        for i, c in enumerate(self.classes):
            X_c = X[y == c]
            n_c = len(X_c)

            # P(特征=1|类别)
            feature_counts = X_c.sum(axis=0) + self.alpha
            prob_1 = feature_counts / (n_c + 2 * self.alpha)

            self.feature_log_prob[i] = np.log(prob_1)
            self.feature_log_neg_prob[i] = np.log(1 - prob_1)

        return self

    def predict_log_proba(self, X: np.ndarray) -> np.ndarray:
        """预测对数概率"""
        if self.binarize is not None:
            X = (X > self.binarize).astype(int)

        # P(特征|类别) = prod(P(x_i|类别)^x_i * (1-P(x_i|类别))^(1-x_i))
        # log P = sum(x_i * log P(x_i=1|类别) + (1-x_i) * log P(x_i=0|类别))

        log_proba = np.zeros((X.shape[0], len(self.classes)))

        for i in range(len(self.classes)):
            log_proba[:, i] = (
                self.class_log_prior[i] +
                (X * self.feature_log_prob[i] +
                 (1 - X) * self.feature_log_neg_prob[i]).sum(axis=1)
            )

        return log_proba

    def predict(self, X: np.ndarray) -> np.ndarray:
        """预测"""
        log_proba = self.predict_log_proba(X)
        return self.classes[np.argmax(log_proba, axis=1)]
```

### 2.2 高斯朴素贝叶斯

```python
class GaussianNaiveBayes:
    """
    高斯朴素贝叶斯

    适用于连续特征
    假设特征服从高斯分布

    P(特征|类别) = 1/sqrt(2πσ²) * exp(-(x-μ)²/(2σ²))
    """

    def __init__(self):
        self.classes = None
        self.class_prior = None
        self.mu = None      # 均值
        self.sigma = None   # 标准差

    def fit(self, X: np.ndarray, y: np.ndarray):
        """训练"""
        self.classes = np.unique(y)
        n_classes = len(self.classes)
        n_features = X.shape[1]

        # 类别先验
        class_counts = np.array([np.sum(y == c) for c in self.classes])
        self.class_prior = class_counts / class_counts.sum()

        # 每个类别每个特征的均值和标准差
        self.mu = np.zeros((n_classes, n_features))
        self.sigma = np.zeros((n_classes, n_features))

        for i, c in enumerate(self.classes):
            X_c = X[y == c]
            self.mu[i] = X_c.mean(axis=0)
            self.sigma[i] = X_c.std(axis=0) + 1e-9  # 避免除零

        return self

    def _gaussian_pdf(self, X: np.ndarray, mu: np.ndarray,
                      sigma: np.ndarray) -> np.ndarray:
        """高斯概率密度函数"""
        return np.exp(-0.5 * ((X - mu) / sigma) ** 2) / (sigma * np.sqrt(2 * np.pi))

    def predict_proba(self, X: np.ndarray) -> np.ndarray:
        """预测概率"""
        n_samples = X.shape[0]
        n_classes = len(self.classes)

        likelihood = np.zeros((n_samples, n_classes))

        for i in range(n_classes):
            # P(特征|类别) = prod(P(特征_j|类别))
            feature_probs = self._gaussian_pdf(X, self.mu[i], self.sigma[i])
            likelihood[:, i] = np.prod(feature_probs, axis=1) * self.class_prior[i]

        # 归一化
        proba = likelihood / likelihood.sum(axis=1, keepdims=True)

        return proba

    def predict(self, X: np.ndarray) -> np.ndarray:
        """预测"""
        proba = self.predict_proba(X)
        return self.classes[np.argmax(proba, axis=1)]
```

## 3. 文本分类应用

### 3.1 文本预处理

```python
import re
from collections import Counter


class TextPreprocessor:
    """
    文本预处理器
    """

    def __init__(self, max_features: int = 10000,
                 min_df: int = 2, max_df: float = 0.95):
        self.max_features = max_features
        self.min_df = min_df
        self.max_df = max_df
        self.vocabulary = {}
        self.idf = None

    def tokenize(self, text: str) -> List[str]:
        """分词"""
        # 简单英文分词
        text = text.lower()
        tokens = re.findall(r'\b\w+\b', text)
        return tokens

    def fit(self, texts: List[str]):
        """构建词汇表"""
        # 统计词频
        doc_freq = Counter()
        n_docs = len(texts)

        for text in texts:
            tokens = set(self.tokenize(text))  # 每个文档只算一次
            doc_freq.update(tokens)

        # 过滤
        max_count = int(self.max_df * n_docs)
        vocab_words = [
            word for word, count in doc_freq.items()
            if self.min_df <= count <= max_count
        ]

        # 限制特征数量
        vocab_words = sorted(vocab_words, key=lambda w: doc_freq[w], reverse=True)
        vocab_words = vocab_words[:self.max_features]

        self.vocabulary = {word: idx for idx, word in enumerate(vocab_words)}

        # 计算 IDF
        self.idf = np.zeros(len(self.vocabulary))
        for word, idx in self.vocabulary.items():
            self.idf[idx] = np.log(n_docs / (1 + doc_freq[word]))

        return self

    def transform(self, texts: List[str], use_tfidf: bool = False) -> np.ndarray:
        """
        转换为特征矩阵

        返回: (n_samples, n_features) 词频或TF-IDF矩阵
        """
        n_samples = len(texts)
        n_features = len(self.vocabulary)

        # 词频矩阵
        X = np.zeros((n_samples, n_features))

        for i, text in enumerate(texts):
            tokens = self.tokenize(text)
            for token in tokens:
                if token in self.vocabulary:
                    X[i, self.vocabulary[token]] += 1

        if use_tfidf:
            # TF-IDF
            tf = X / (X.sum(axis=1, keepdims=True) + 1e-10)
            X = tf * self.idf

        return X

    def fit_transform(self, texts: List[str], use_tfidf: bool = False) -> np.ndarray:
        """拟合并转换"""
        self.fit(texts)
        return self.transform(texts, use_tfidf)


class TextClassifier:
    """
    文本分类器
    """

    def __init__(self, alpha: float = 1.0, use_tfidf: bool = False):
        self.preprocessor = TextPreprocessor()
        self.classifier = MultinomialNaiveBayes(alpha=alpha)
        self.use_tfidf = use_tfidf

    def fit(self, texts: List[str], labels: List[str]):
        """训练"""
        X = self.preprocessor.fit_transform(texts, self.use_tfidf)
        self.classifier.fit(X, np.array(labels))
        return self

    def predict(self, texts: List[str]) -> List[str]:
        """预测"""
        X = self.preprocessor.transform(texts, self.use_tfidf)
        return self.classifier.predict(X).tolist()

    def predict_proba(self, texts: List[str]) -> np.ndarray:
        """预测概率"""
        X = self.preprocessor.transform(texts, self.use_tfidf)
        return self.classifier.predict_proba(X)
```

```python
class ContentCategoryClassifier:
    """
    内容分类器

    用于给物品自动打标签/分类
    """

    def __init__(self, categories: List[str]):
        self.categories = categories
        self.classifier = TextClassifier(alpha=1.0, use_tfidf=True)

    def train(self, texts: List[str], labels: List[str]):
        """训练"""
        self.classifier.fit(texts, labels)

    def categorize(self, text: str, top_k: int = 3) -> List[Tuple[str, float]]:
        """
        对内容分类

        返回 top-k 类别及概率
        """
        proba = self.classifier.predict_proba([text])[0]

        top_indices = np.argsort(proba)[::-1][:top_k]
        results = [
            (self.classifier.classifier.classes[i], proba[i])
            for i in top_indices
        ]

        return results


class SentimentAnalyzer:
    """
    情感分析器

    分析用户评论的情感倾向
    """

    def __init__(self):
        self.classifier = TextClassifier(alpha=0.5)
        self.labels = ['negative', 'neutral', 'positive']

    def train(self, texts: List[str], labels: List[str]):
        """训练"""
        self.classifier.fit(texts, labels)

    def analyze(self, text: str) -> Dict[str, float]:
        """
        分析情感

        返回各情感的概率
        """
        proba = self.classifier.predict_proba([text])[0]

        return {
            label: float(prob)
            for label, prob in zip(self.classifier.classifier.classes, proba)
        }

    def get_sentiment_score(self, text: str) -> float:
        """
        获取情感分数

        -1 (负面) 到 1 (正面)
        """
        proba = self.classifier.predict_proba([text])[0]
        classes = self.classifier.classifier.classes

        score = 0
        for label, prob in zip(classes, proba):
            if label == 'positive':
                score += prob
            elif label == 'negative':
                score -= prob

        return score
```

```python
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB, BernoulliNB, GaussianNB
from sklearn.pipeline import Pipeline
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, confusion_matrix


def create_text_classifier():
    """创建文本分类管道"""
    pipeline = Pipeline([
        ('tfidf', TfidfVectorizer(max_features=10000, stop_words='english')),
        ('clf', MultinomialNB(alpha=1.0))
    ])
    return pipeline


def train_and_evaluate(texts, labels):
    """训练并评估"""
    # 分割数据
    X_train, X_test, y_train, y_test = train_test_split(
        texts, labels, test_size=0.2, random_state=42
    )

    # 创建分类器
    clf = create_text_classifier()

    # 训练
    clf.fit(X_train, y_train)

    # 预测
    y_pred = clf.predict(X_test)

    # 评估
    print("Classification Report:")
    print(classification_report(y_test, y_pred))

    return clf


# 示例用法
```

## 6. 学习总结

### 6.1 核心要点

```
1. 贝叶斯定理: P(类别|特征) ∝ P(特征|类别) * P(类别)
2. 朴素假设: 特征条件独立
3. 拉普拉斯平滑: 避免零概率
4. 三种变体: 多项式、伯努利、高斯
```

### 6.2 变体对比

```
变体        适用特征        典型应用
────────────────────────────────────────
多项式      离散计数        文本分类(词频)
伯努利      二值特征        短文本、垃圾邮件
高斯        连续特征        数值分类
```

### 6.3 优缺点

```
优点:
- 训练速度快
- 需要数据少
- 对高维数据表现好
- 可解释性强

缺点:
- 特征独立假设过强
- 对特征相关性敏感
- 需要合适的平滑
```

### 6.4 最佳实践

```
1. 文本任务使用 TF-IDF + 多项式NB
2. 调整 alpha 平滑参数
3. 特征选择移除噪声
4. 注意特征之间的相关性
```
